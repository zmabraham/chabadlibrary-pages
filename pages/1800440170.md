לד"ה בהעלותך הראשון [שבלקו"ת]
לד"ה בהעלותך הראשון*
הנה ענין אברהם יובן ממ"ש ע"פ אלקי אברהם כו' זה שמי לעלם כי ע"י שהי' בבחי' מרכבה נמשך גילוי זה שמי כו' ע"ד לעשות לך שם עמ"ש מזה ע"פ אנכי ה' אלקיך אשר הוצאתיך כו' עמ"ש בד"ה ויצוום אל בנ"י ועכ"ז ושמי הוי' לא נודעתי להם לאברהם כו' זה נמשך ע"י יצ"מ ומ"ת והיינו ע"י משה ואהרן שהם חו"ב וזהו ויצוום לשון התקשרות שחיברם שיהיו בבחי' תרין ריעין כו' אהרן שודמ"ט מעלה כנ"י מלמטלמ"ע ומשה מלמעלה למטה, ופי' ושמי הוי' אא"ס המלובש בחכמה משא"כ להאבות הגילוי מחג"ת, בחי' באל שדי כו', וא"כ י"ל ע"י אהבה דאברהם נמשך הגילוי מבחי' וארא כו' באל שדי שאמר לעולמו די והוא ע"י הפרסא שבין אצי' לבי"ע. וע' בת"א פ' וארא בד"ה וידבר אלקי' כו' וארא כו' שזהו"ע עולם חסד יבנה, אבל אה"ר דאהרן שיהי' גילוי ש' הוי' היינו מבחי' חכמה כו', וע' במא"א א' קל"ו אברהם נקרא החסד וכן ח"פ אל גימטריא אברהם עכ"ל אפ"ל כי ח"פ אל שיהי' חסד אל בז' רקיעים והארץ כענין הפי' באחד כו'. ועו"ש סעי' קל"ט אהרן נשמתו מיסוד אבא כו' וסעי' ק"מ כי כה"ג באבא כו' והיינו כי פי' כהן גדול שהוא כהן ומשמש לגדול העולמים, גדול הוי' ומהולל מאד כו' וע' זח"ב ר"פ יתרו ס"ז אית מלך לעילא ברזא דקה"ק וכהן דמשמש תחותי', ועמ"ש בענין יומם יצוה ה' חסדו דהוא חסד שביסוד אבא וזהו בחי' אהרן, עזח"ג פ' אמור דק"ג ע"א ובפ' בלק דקצ"א ע"ב, וא"ת איך יתכן שאהרן יהי' גבוה מאברהם, אך התירוץ כי אברהם הגיע לבחי' שלו מצ"ע באתעדל"ת אבל בחי' כה"ג זהו שנמשך מלמעלה ע"י מ"ת כו' וא"כ זהו כח עליון ע"ד אלקיכם כהן הוא כו', והיינו אהרן ממשיך אה"ר הבאה מבחי' ראייה שהוא חסד דאבא וזה ניתן הכח לאהרן שימשיך מבחי' ראייה והיינו ע"י כי חו"ב תרין ריעין כו' משא"כ אברהם שהשיג באתעדל"ת לא יכול להמשיך מבחי' חכמה ראי' כ"א מבחי' חסד ולמטה זהו אהבה הבאה מבחי' התבוננות שמע ישראל ואהבת כו' ואין דומה שמיעה לראי' כו'. וע' בד"ה רני ושמחי השני שמחה בצפרא זהו בחי' אהרן כו' ועמ"ש ע"פ וכל העם רואים הפרש בין רו"ש דאברהם ובין רו"ש הנמשך ע"י תומ"צ שזה בחי' אהרן ממשיך מבחי' ורב חסד בד"ה ועשית בגדי קדש, ולבא לבחי' ראי' צ"ל ע"י היתה לי דמעתי לחם כו' מן העינים עמ"ש ע"פ אז תפקחנה עיני עורים. וע"ד אין מוסרין רזי תורה אלא למי שלבו דואג בקרבו. והנה כדי שיבא לבחי' ראי' דאהרן זהוע"י התורה כשמן הטוב כו' שבבחי' זקן אהרן, הלכות ממשיך ההילוך כו'. 
ב) בביאור משה ידבר אברהם המשיך מוחין דיניקה ולכן הגילוי רק למעלה אבל משה המשיך מוחין דגדלות לכן הגילוי גם למטה וזהו וארא כו' באל שדי מלשון ברכות שדים יניקה כו' אך ושמי כו' בחי' יחיד כו' סד"ה וארא, וזהו ההפרש בין אהבה דאברהם שממשיך בחי' אחד ויניקה אבל אה"ר דאהרן ממשיך בחי' יחיד ע"י משה ידבר כו' וזהו אתה כהן לעולם על דברתי דברת יוד שיהי' הגילוי בהדיבור מבחי' יוד כו'. והנה הנר מקבל האור ושרשו מהשמש עזח"ג ר"פ בהעלותך וע"י אהבה דאברהם מקבלים הנרות שהם הנשמות מבחי' שמש ומגן אבל אהרן בהעלותך את הנרות לקבל מבחי' שלמעלה משמש ומגן כמ"ש סד"ה וארא כו' הנ"ל סעי' א', כי ז' נרות המנורה איתא ברבות ר"פ
