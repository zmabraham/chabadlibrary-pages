# 

ועיין מ"ש בתו' ספ"ג). (נת' ב' פי' אלו איך אם אין יראה תתא' עול מצות אין חכ' כי מצות חצוני' האברי' דז"א ומצות המלך ותו' פנימי' וממט' למעלה צ"ל תחלה הכלי חצוני' עי"ז יגיע להפנימי', בשגם ששרשם גבוה יותר מגלגלתא ואח"כ כשיזכה לתו' ח"ע יוכל להגיע ליר"ע מזלייהו חזי בחי' יחידה שלמעלה מחכ' כי יחידה א"א לה להשיג גילוי אא"ס רק ע"י אמצעות התו' דהוי' בחכ' כמ"ש במ"א במצו' העבודה בת"ת, וע' מ"ש בשע"ק דמזל שבנשמו' למטה מהספי' כי הוא בחי' לבושי' להספי' גופ' ע"כ א"א להיות מזלייהו דנשמות חזי גילוי אא"ס רק ע"י הספי' והיינו ע"י התו' הוי' בחכ' חי' דספי' ודו"ק). 
ד) וכ"ז הוא סדר השתלשלות אבל בשעת מ"ת שעדיין לא נצטוו על המצות הי' גילוי הכתר שלא כסדר המדריגות ע"י יר"ת אלא ע"י ספה"ע, ואף שעדיין לא נצטוו על ספה"ע מ"מ ע"י השבע' שבועו' עצמן שהם בחי' יצ"מ ממ"ט שערי טומאה למ"ט שערי קדושה שהוא בחי' אתכפי' בבחי' פנימי' המדות להפכן מן הקצה אל הקצה וכמ"ש במדר' ע"פ מהחל חרמש בקמה א"ת בקמה אלא בקומה בחי' [קומה] הוא מדות דקליפה בחי' יש וקומה שלימה להפכן להיות נכללין במדות דקדושה ע"י ספה"ע היום יום גי' ז"ן עי' בכוונת בהמ"ז גבי כי הוא אל זן וגימטרי' אל הוי' ביצירה כי ביצי' הוא שרש ומקור המדו' דקליפה וע"י התהפכות מדות דקליפה במדות דקדושה בבחי' פנימי' הנה עי"ז מאליו וממילא נתגלה שער החמשי' בחי' כתר וירד הוי' על ה"ס שנתגלה בחי' הכתר ממילא ומתחלק לשנים חצי לז"א וחצי' לנוק' שהפנימי' לז"א בחי' אין שבחכ' (נ"ל דהיינו מו"ס שבכתר שרש התו' כנ"ל בשם אגה"ק) וחיצוני' כתר בחי' קרקפתא לנוק', וזהו ועשית חג שבועות מסת נדבת ידך אשר תתן ולא כתי' נדבת לבך, שנדבת הלב היינו בחי' ז"א שנק' לב לפי שמקבל הארת ל"ב נ"ח בבחי' פנימי' אבל בשבועות הארת הכתר מתגלה גם בבחי' חצוני' וזהו נדבת ידך החיצוני' היד, כאשר יברכך ה' אלקיך הארת הכתר וכדלעיל דדוקא חיצוני' האברי' מתקשר בגלגלת' והוא בחי' מל' מצות המלך נדבת ידך כו', ומתן תו' כולל ב' בחי' תומ"צ וזהו ב' הכתרי' חצי לז"א בחי' תו' וחצי לנוק' בחי' מצות לפי דשרש המצות גבוה בגלגלתא ע"כ הזכיר נדבת ידך, וז"ש רז"ל בשעה שהקדימו ישראל נעשה לנשמע יצתה ב"ק וא' מי גלה רז זה לבניי, פי' הבינו ישראל מעצמן שהארה שהי' אז במ"ת גילוי הכתר מלמעלה למטה אינו קביעות הלכה לדורות בלתי ע"י קדימת עול מצות תחלה נעשה ואח"כ נשמע כו', ולכן כתי' בשתי הלחם דשבועות חמץ תאפינה כי בכל הקרבנות נאמר כי כל שאור וכל דבש כו' לפי ששאור הוא בחי' התנשאות ויכול להיות אחיזה ויניקה לחיצוני' משא"כ בשבועות הארת הכתר בחי' התנשאו' המתנשא כו' כתי' חמץ תאפינה שאין לחצוני' שם שום אחיזה ויניקה כלל. (בשעת מ"ת שלא ע"י קדימת יר"ת ועול מצות רק ע"י ספה"ע כעין ז' שבועות יציאה ממ"ט שערי טומאה למ"ט שערי קדושה נתגלה בשבועות שער הנו"ן בחי' כתר, ומתחלק חצי' לז"א וחצי' לנוק', לכן כתי' נדבת ידך כו' חצוני' האברים מבחי' גלגלתא, וזהו כאשר יברכך כו', ואל דורות הקדימו נעשה לנשמע כו' כדלעיל בפ"ג אם אין יראה כו' אלא תחלה יר"ת ואח"כ חכ' ואח"כ ע"י מזלייהו חזי לעיל פ"א).
