# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר יח. > פֶּרֶק ב

וְהִנֵּה שֹׁרֶשׁ הַהַמְשָׁכָה שֶׁנִּמְשָׁךְ בִּסְפִירַת הַמַּלְכוּת מִכֶּתֶר מַלְכוּת בְּרֹאשׁ הַשָּׁנָה שֶׁהוּא הָאוֹר וְהַחַיּוּת מִבְּחִינַת חֶסֶד דְּעַתִּיק שֶׁנִּמְשָׁךְ בִּבְחִינַת מַלְכוּת, הֲרֵי הוּא בְּחִינַת אוֹר כְּלָלִי וְלֹא נִתְחַלֵּק לִפְרָטִים כְּלָל וְאִי אֶפְשָׁר שֶׁיִּתְקַבֵּל אֶל הַמְקַבְּלִים, וּבִכְדֵי שֶׁיִּתְקַבֵּל הָאוֹר בַּמְקַבְּלִים זֶהוּ עַל יְדֵי בְּחִינַת ה' גְבוּרוֹת דְּעַתִּיק, שֶׁהֵם הַמְחַלְּקִים אֶת הַחֶסֶד לַחֲלָקִים, שֶׁעַל יְדֵי זֶה יִתְקַיֵּים עִיקַּר אֲמִיתַּת הַחֶסֶד שֶׁעַל יְדֵי זֶה יוּכְלוּ הַנִּבְרָאִים לְקַבֵּל אֶת הַחֶסֶד,
וּכְעִנְיַן גְּבוּרוֹת גְּשָׁמִים שֶׁיּוֹרְדִין בִּגְבוּרָה שֶׁמִּתְחַלְּקִים טִיפִּין טִיפִּין וְזֶהוּ תַּכְלִית וַאֲמִיתִּית הַחֶסֶד שֶׁעַל יְדֵי זֶה דַוְקָא וְהִרְוָה אֶת הָאָרֶץ וְהוֹלִידָהּ וְהִצְמִיחָהּ,
וְכֵן בְּכֹחוֹת הַנֶּפֶשׁ בָּאָדָם, דְּהַקּוֹל הַיּוֹצֵא מֵהֶבֶל הַלֵּב הוּא קוֹל פָּשׁוּט וּמִתְחַלֵּק לְאוֹתִיּוֹת בְּה' מוֹצְאוֹת הַפֶּה אחה"ע מֵהַגָּרוֹן גיכ"ק מֵהַחֵיךְ כו' וְהַיְינוּ עַל יְדֵי ה' גְבוּרוֹת מנצפ"ך מְקוֹר הַדִּבּוּר, וְכִדְאִיתָא בְּהַקְדָּמַת הַזֹּהַר בְּעִנְיַן כּוֹס יְשׁוּעוֹת אֶשָּׂא, כּוֹס שֶׁל בְּרָכָה אִצְטְרִיךְ לְמֶהֱוֵי עַל חֲמֵשׁ אֶצְבָּעִין, וּמְבוֹאָר שָׁם שֶׁהֵן בְּחִינַת ה' גְבוּרוֹת, וְהֵם דַּוְקָא נִקְרְאוּ יְשׁוּעוֹת וּכְאָמְרוֹ וְאִינּוּן חֲמֵשׁ אִיקְרִין יְשׁוּעוֹת, וְעַיֵּן שָׁם בְּפֵירוּשׁ הָרְמָ''ז [רַבִּי מֹשֶׁה זַכּוּת] שֶׁהֵן ה' גְבוּרוֹת הַנִּמְשָׁכוֹת מִיסוֹד דְּעַתִּיק.
[cup]וְזֶהוּ[/cup] שֶׁנִּקְרְאוּ הַה' גְבוּרוֹת ה' אֶצְבָּעוֹת, דִּכְמוֹ עַל דֶּרֶךְ מָשָׁל הַיָּד הִנֵּה קוֹדֶם הִתְחַלְּקוּתָהּ לְה' אֶצְבָּעוֹת הֲרֵי הַכֹּחַ כָּלוּל בָּהּ בְּיַחַד בְּלִי הִתְחַלְּקוּת, וּכְשֶׁמִּתְחַלֶּקֶת לְה' אֶצְבָּעוֹת מִתְחַלֵּק כֹּחַ הַהוּא שֶׁיִּהְיֶה לְכָל אֶצְבַּע כֹּחַ פְּרָטִי מְצוּמְצָם בִּפְנֵי עַצְמוֹ זוֹ זֶרֶת וְזוֹ קְמִיצָה כו' וְכִדְאִיתָא בִּגְמָרָא (כְּתוּבּוֹת ה. א וְשָׁם) דְּכָל חֲדָא וַחֲדָא לְמִילְּתֵיהּ עֲבִידָא וְהִתְחַלְּקוּת זוֹ הִיא בְּחִינַת גְּבוּרוֹת
אַךְ ה' גְבוּרוֹת אֵלּוּ הֵן אֲמִיתִּית הַחֶסֶד, שֶׁהֲרֵי הַיָּד אִם לֹא הָיָה בָהּ חִלּוּק הָאֶצְבָּעוֹת לֹא הָיָה יָכוֹל לַעֲשׂוֹת בָּהּ מְלָאכָה וּפְעוּלָּה וְלֹא לִכְתּוֹב שֶׁכָּל פְּעוּלּוֹת אֵלּוּ נַעֲשׂוֹת דַּוְקָא עַל יְדֵי הָאֶצְבָּעוֹת, נִמְצָא שֶׁהֲגַם שֶׁהַיָּד עַצְמָהּ הִיא כְּלָלוּת הַכֹּחַ מִכָּל מָקוֹם אִם לֹא עַל יְדֵי ה' גְבוּרוֹת דְּחִילּוּק הָאֶצְבָּעוֹת לֹא הָיָה יָכוֹל לִהְיוֹת פְּעוּלָּה וְהַשְׁפָּעָה מֵהַכֹּחַ שֶׁבַּיָּד, וְאִם כֵּן הֲרֵי ה' גְבוּרוֹת אֵלּוּ הֵן שְׁלֵמוּת פְּעוּלַּת הַחֶסֶד וּכְעִנְיַן גְּבוּרוֹת גְּשָׁמִים הַנַ"ל.
[cup]וְזֶהוּ[/cup] עִנְיַן מַה שֶּׁנִּקְרְאוּ גְּבוּרוֹת הַנַ"ל בְּשֵׁם יְשׁוּעוֹת, דְּזֶהוּ גַם כֵּן שֶׁכּוֹס שֶׁל בְּרָכָה צָרִיךְ לִהְיוֹת עַל ה' אֶצְבָּעִין דַּוְקָא. וּמִכָּל שֶׁכֵּן בִּבְחִינַת עַתִּיק שֶׁהַה' גְבוּרוֹת דְּשָׁם הֵן וַדַּאי כְּלוּלוֹת וּמְיוּחָדוֹת בְּהַה' חֲסָדִים, דְּהָא לֵית שְׂמָאלָא בְּהַאי עַתִּיקָא, וְלָכֵן מֵהֶן מִיתּוּק כָּל הַגְּבוּרוֹת שֶׁלְּמַטָּה, וּכְיָדוּעַ מֵעִנְיַן אֵין הַדִּינִין נִמְתָּקִין אֶלָּא בְּשָׁרְשָׁן, וּכְמוֹ שֶׁכָּתוּב בִּפְרִי עֵץ חַיִּים בְּכַוָּונַת הַשּׁוֹפָר, וְאַף עַל פִּי כֵן נִקְרָא ה' גְבוּרוֹת שֶׁבָּהֶן וְעַל יָדָן נִמְשָׁךְ שְׁלֵימוּת פְּעוּלַּת הַה' חֲסָדִים שֶׁתִּתְחַלֵּק הַשֶּׁפַע בִּכְדֵי שֶׁתּוּכַל לְהִתְקַבֵּל אֶל הַנִּבְרָאִים, וְגַם עַל יְדֵי זֶה נִמְשָׁךְ הַחֶסֶד לְכָל אֶחָד וְאֶחָד מֵהַנִּבְרָאִים בִּפְרָט.
[cup]דְּהִנֵּה[/cup] בְּחִינַת הַחֶסֶד שֶׁנִּמְשָׁךְ בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים מִבְּחִינַת ה' חֲסָדִים דְּעַתִּיק הוּא בְּחִינַת הַמְשָׁכַת חֶסֶד כְּלָלִי לִכְלָלוּת הָעוֹלָם, וְהַה' גְבוּרוֹת דְּעַתִּיק שֶׁנִּמְשָׁכִים בִּנְעִילָה הוּא שֶׁמִּתְחַלֵּק בְּחִינַת הַחֶסֶד לְכָל נִבְרָא בִּפְרָט כַּמָּה מִן הַחֶסֶד יוּמְשַׁךְ לְכָל אֶחָד וְאֶחָד, וְגַם עַל יְדֵי זֶה נִמְשָׁךְ הַחוֹתָם שֶׁלֹּא תוּמְשַׁךְ הַשֶּׁפַע וְתִתְפַּשֵּׁט יוֹתֵר מִדַּאי לְמָקוֹם שֶׁאֵין צָרִיךְ, וְהַיְינוּ שֶׁלֹּא תִהְיֶה יְנִיקַת הַחִיצוֹנִים, דְּעַל יְדֵי שֶׁנִּמְשָׁךְ הַחֶסֶד לְכָל אֶחָד וְאֶחָד עַל יְדֵי בְּחִינַת הַה' גְבוּרוֹת, שֶׁבָּאָה הַשְׁפָּעַת הַחֶסֶד בִּבְחִינַת הַגְבָּלָה, מִמֵּילָא לֹא תִהְיֶה מִזֶּה יְנִיקַת הַחִיצוֹנִים, דְּדַוְקָא כְּשֶׁהָאוֹר הוּא בִּבְחִינַת בְּלִי גְבוּל אָז יִתְפַּשֵּׁט בְּיוֹתֵר וְיָכוֹל לִהְיוֹת שֶׁיּוּמְשַׁךְ גַּם לְמָקוֹם שֶׁאֵינוֹ רָאוּי, אֲבָל כְּשֶׁהוּא בִּבְחִינַת הַגְבָּלָה אֵינוֹ נִמְשָׁךְ לְמָקוֹם שֶׁאֵינוֹ רָאוּי.
[cup]וְזֶהוּ[/cup] גַם כֵּן עִנְיַן הַה' גְבוּרוֹת, דִּבְכָל אֶחָד וְאֶחָד לֹא תִהְיֶה יְנִיקַת הַחִיצוֹנִים מֵהַשְׁפָּעַת הַחֶסֶד שֶׁנִּמְשָׁךְ אֵלָיו. וְזֶהוּ עִנְיַן הַחוֹתָם דִּנְעִילָה שֶׁהַיְינוּ מַה שֶּׁנֶּחְתָּם וְנֶחֱלָק הַשְׁפָּעַת הַחֶסֶד בִּפְרָט לְכָל אֶחָד וְאֶחָד.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]ה' גְבוּרוֹת דְּעַתִּיק מְחַלְּקוֹת הַשֶּׁפַע שֶׁנִּמְשָׁךְ בְּמַלְכוּת, גְּבוּרוֹת גְּשָׁמִים, ה' מוֹצָאוֹת מְחַלְּקִים הַקּוֹל, ה' אֶצְבָּעוֹת מְחַלְּקִים כֹּחַ הַכְּלָלִי וְעַל יָדָן הֵן הַפְּעוּלּוֹת, כּוֹס יְשׁוּעוֹת דְּיָתְבָא עַל ה' אֶצְבָּעִין, הַחוֹתָם בִּכְדֵי שֶׁהַשֶּׁפַע יַגִּיעַ לְמָקוֹם הָרָאוּי וּלְכָל אֶחָד בִּפְרָט.[kitzur_end]
