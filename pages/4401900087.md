מַאֲמָר כד.
וְהִנֵּה אוֹר הַחֶסֶד וְהַחַיּוּת בְּהַהַשְׁפָּעָה שֶׁנִּקְצְבָה לוֹ לְאָדָם בְּרֹאשׁ הַשָּׁנָה בְּכָל מִינֵי טוֹב בְּבָנֵי חַיֵּי וּמְזוֹנֵי הוּא מֵאִתּוֹ יִתְבָּרֵךְ, וְהַיְינוּ דְּגוּף וְעֶצֶם הַשֶּׁפַע הוּא מֵעַתִּיקָא קַדִּישָׁא שֶׁהוּא פְּנִימִיּוּת וְעַצְמִיּוּת אֵין סוֹף, וּבָא עַל יְדֵי הִתְעוֹרְרוּת הָעֶשֶׂר סְפִירוֹת דַּאֲצִילוּת דִּלְשָׁם מַגַּעַת הָעֲבוֹדָה דְנִשְׁמוֹת יִשְׂרָאֵל בַּתּוֹרָה וּמִצְוֹת.
[cup]דְּהִנֵּה[/cup] הָאוֹר וְהַשֶּׁפַע שֶׁנִּקְצָב לוֹ לְהָאָדָם בְּרֹאשׁ הַשָּׁנָה וְנֶחְתָּם בְּיוֹם הַכִּפּוּרִים הוּא לְפִי עֲבוֹדָתוֹ בַּתּוֹרָה וּמִצְוֹת בְּשָׁנָה הֶעָבְרָה, וּכְמוֹ שֶׁכָּתוּב כִּי חֹק לְיִשְׂרָאֵל הוּא מִשְׁפָּט לֵאלֹקֵי יַעֲקֹב וּפֵירוּשׁוֹ כִּי הַחֹק הַנִּקְצָב לְיִשְׂרָאֵל בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים תָּלוּי בְּמִשְׁפָּט, וְהַמִּשְׁפָּט הוּא בָּזֶה כַּמָּה אוֹר וְגִילּוּי אֱלֹקוּת, שֶׁהוּא עִנְיַן אֱלֹקֵי יַעֲקֹב, הִמְשִׁיךְ בְּשָׁנָה הֶעָבְרָה,
וְכֵן תָּלוּי בְּאוֹפֶן הַתְּשׁוּבָה בַּעֲבוֹדָה דְחֹדֶשׁ אֱלוּל שֶׁהוּא חֹדֶשׁ הָרַחֲמִים שֶׁמְּאִירִים אָז י"ג מִדּוֹת הָרַחֲמִים וְהוּא זְמַן עֵת רָצוֹן וְנָקָל יוֹתֵר עִנְיַן הַתְּשׁוּבָה, וְאַחַר כַּךְ בִּימֵי הַסְּלִיחוֹת שֶׁהֵם זְמַנֵּי עֵת רָצוֹן עוֹד יוֹתֵר מֵחֹדֶשׁ אֱלוּל, וּכְמַאֲמָר הַיָּדוּעַ מֵרַבֵּינוּ בְּשֵׁם הַבַּעַל שֵׁם טוֹב נִשְׁמָתוֹ עֵדֶן דְּבִימֵי אֱלוּל אָז מְאִירִים י"ג מִדּוֹת הָרַחֲמִים לְעוֹרֵר דִּין וְחֶשְׁבּוֹן,
וְהַמָּשָׁל בָּזֶה סוֹחֵר גָּדוֹל שֶׁמּוֹכֵר סְחוֹרָתוֹ בְּהַקָּפָה הִנֵּה בְּסוֹף הַשָּׁנָה מְעוֹרֵר אֶת הַקּוֹנִים לַעֲשׂוֹת חֶשְׁבּוֹן כַּמָּה נִמְכַּר וְכַמָּה שִׁלְּמוּ וְכַמָּה עוֹד חַיָּיבִים לְשַׁלֵּם וּמְעוֹרֵר לְשַׁלֵּם אֶת הַמַּגִּיעַ, וְהַסּוֹחֵר עַצְמוֹ בְּשִׂמְחָה גְדוֹלָה שֶׁמָּכַר סְחוֹרָה הַרְבֵּה וּבְמִקָּחִים טוֹבִים וְהָיְתָה לוֹ פַּרְנָסָה מְרוּבָּה, אֲבָל עִם זֶה תּוֹבֵעַ אֶת קוֹנָיו לְשַׁלֵּם הַמַּגִּיעַ מֵהֶם וּמַזְכִּירָם אֲשֶׁר לְקִיחַת סְחוֹרָה בְּהַקָּפָה בְּשָׁנָה הַבָּאָה תָּלוּי בִּזְרִיזוּת פִּרְעוֹן הַחוֹבוֹת מֵהָעָבָר וְקוֹבֵעַ זְמַן בְּמֶשֶׁךְ כִּשְׁלֹשָׁה שָׁבוּעוֹת, שִׂמְחָתוֹ וּצְהִילַת פְּנֵי הַסּוֹחֵר מוֹעִיל לַקּוֹנִים שֶׁעֲמֵלִים בְּעָמָל רַב וּבִיגִיעָה עֲצוּמָה לְשַׁלֵּם חוֹבוֹתֵיהֶם, וּבִכְלוֹת מוֹעֵד הַקָּצוּב בָּא הַסּוֹחֵר הַגָּדוֹל לַעֲשׂוֹת חֶשְׁבּוֹן מִכָּל הַחוֹבוֹת וּכְשֶׁרוֹאֶה עֲמַל הַקּוֹנִים וִיגִיעָתָם הָעֲצוּמָה לִפְרוֹעַ חוֹבוֹתֵיהֶם וּמַכִּיר שֶׁבֶּאֱמֶת חָפְצוּ לִפְרוֹעַ אֲבָל עֲנִיִּים הֵמָּה וְאֵין יָדָם מַשֶּׂגֶת, אֲזַי עוֹשֶׂה חֶסֶד עִמָּהֶם וְסוֹלֵחַ וּמוֹחֵל לָהֶם מַה שֶּׁאִי אֶפְשָׁר לָהֶם לְשַׁלֵּם וּמַגְבִּיל לָהֶם אֵיזֶה זְמַן לְהָכִין עַצְמָם לְהַהַקָּפָה דִסְחוֹרָה חֲדָשָׁה,
וְהֵם יְמֵי הַסְּלִיחוֹת, דְּכַאֲשֶׁר הָעֲבוֹדָה דִימֵי אֱלוּל הִיא עֲבוֹדָה שְׁלֵמָה בִּתְשׁוּבָה מֵעוּמְקָא דְלִיבָּא הִנֵּה בִּימֵי הַסְּלִיחוֹת מִתְקַבֶּלֶת תְּשׁוּבָתוֹ בְּקַבָּלָה טוֹבָה עַל לְהַבָּא, אֲשֶׁר כְּכָל הַתְּשׁוּבָה דֶאֱלוּל וִימֵי הַסְּלִיחוֹת וְקַבָּלַת עוֹל מַלְכוּתוֹ יִתְבָּרֵךְ בְּרֹאשׁ הַשָּׁנָה הִנֵּה לְפִי עֶרֶךְ זֶה נִקְצָב לוֹ הַחֶסֶד בְּבָנֵי חַיֵּי וּמְזוֹנֵי בְּרֹאשׁ הַשָּׁנָה וְנֶחְתָּם בְּיוֹם הַכִּפּוּרִים,
אֲבָל חֶסֶד זֶה הוּא עֲדַיִין חֶסֶד פָּשׁוּט וְגַם שָׁרְשֵׁי הַחֶסֶד דַּעֲשִׂיָּה הַכָּלוּל בְּמַלְכוּת דַּאֲצִילוּת הוּא רוּחָנִי עֲדַיִין, וּבִכְדֵי שֶׁיָּבֹא בְּגַשְׁמִיּוּת הָעוֹלָם הַזֶּה הוּא עַל יְדֵי הִתְלַבְּשׁוּת בְּכֵלִים וּלְבוּשִׁים דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה וְאַחַר כַּךְ בְּהַמַּזָּלוֹת דְּכָל אוֹפַנֵּי הַהִתְלַבְּשׁוּת הָאֵלּוּ הוּא מֵחֶסֶד ה' עָלֵינוּ לְהַשְׁפִּיעַ לָנוּ שֶׁפַע חַיּוּת כְּפִי עֶרְכֵּנוּ וּמַהוּתֵנוּ, מַה שֶּׁיִּהְיֶה טוֹב לָנוּ שֶׁנּוּכַל לְקַבֵּל שֶׁבִּשְׁבִיל זֶה מִתְצַמְצֵם וּמִתְלַבֵּשׁ הַחֶסֶד בְּכָל הַהִתְלַבְּשׁוּת הַנַ"ל,
וּכְמוֹ עַל דֶּרֶךְ מָשָׁל אָב הָרוֹצֶה לְשַׁעֲשֵׁעַ אֶת בְּנוֹ, הֲרֵי הוּא מְצַמְצֵם אֶת עַצְמוֹ וּמִתְלַבֵּשׁ בְּעִנְיָנִים קְטַנִּים מְאֹד, שֶׁהֵן לְפִי עֶרֶךְ הַקַּטְנוּת שֶׁל הַבֵּן, כְּמוֹ לְשַׂחֵק עִמּוֹ בֶּאֱגוֹזִים וְעוֹד בִּקְלִיפּוֹת הָאֱגוֹזִים, כִּי הַבֵּן מִצַּד קַטְנוּת שֶׁלּוֹ אֵינוֹ מַכִּיר בְּהַטּוֹב שֶׁבְּהַפְּרִי שֶׁבְּתוֹכָהּ כְּלָל, רַק הַקְּלִיפָּה תִּהְיֶה לוֹ לְטוֹב וּלְחֶסֶד בִּהְיוֹתוֹ יָכוֹל לְגַלְגְּלָהּ וְכַדּוֹמֶה, וּבְדוּגְמָא כָזֹאת מִפְּנֵי חֶסֶד ה' עָלֵינוּ מִתְצַמְצֵם הָאוֹר הָאֱלֹקִי וּמִתְלַבֵּשׁ בְּכַמָּה לְבוּשִׁים עַד הַלְּבוּשִׁים הַגַּסִּים דְּמַזָּלוֹת הַנַ"ל, בִּכְדֵי שֶׁיִּתְהַוֶּוה בְּחִינַת חֶסֶד גַּשְׁמִי לְפִי עֶרֶךְ הַנִּבְרָאִים מַה שֶּׁטּוֹב לָהֶם,
וְהַכַּוָּונָה פְנִימִית בָּזֶה הִיא אֲשֶׁר נִשְׁמוֹת יִשְׂרָאֵל עַל יְדֵי עֲבוֹדָתָם בַּתּוֹרָה וּמִצְוֹת יְבָרְרוּ אֶת נִיצוֹץ הַטּוֹב שֶׁבְּדָבָר הַגַּשְׁמִי וִיזַכְּכוּהוּ וְיַעֲלוּהוּ לְשָׁרְשׁוֹ וּמְקוֹרוֹ, אֲשֶׁר עַל יְדֵי זֶה נַעֲשֶׂה בֵּירוּר כְּלָלִי בִּכְלָלוּת הָעוֹלָם הַזֶּה אֲשֶׁר הוּא תַּכְלִית כָּל הָעוֹלָמוֹת וְכָל הַהִשְׁתַּלְשְׁלוּת, וּמֵהַאי טַעְמָא הִנֵּה רוֹב סוּגֵי הַתּוֹרָה וְהַמִּצְוָה נִתְלַבְּשׁוּ בִּדְבָרִים גַּשְׁמִים,
דְּהִנֵּה עִיקַּר עִנְיַן לִימּוּד הַתּוֹרָה הוּא שֶׁיִּהְיֶה מוּבָן וּמוּשָּׂג בְּשֵׂכֶל הַטִּבְעִי וְהָאֱנוֹשִׁי, וְגַם אֲפִילוּ פְּנִימִיּוּת הַתּוֹרָה בַּהֲבָנַת וְהַשָּׂגַת אֱלֹקוּת בְּעִנְיַן יִחוּדוֹ וְאַחְדּוּתוֹ יִתְבָּרֵךְ וְכֵן בְּעִנְיַן גְּדוּלָּתוֹ בְּמָה רַבּוּ מַעֲשֶׂיךָ ה' וּבְמַה גָּדְלוּ מַעֲשֶׂיךָ ה' וּבְרִיבּוּי הָעוֹלָמוֹת וּפַרְצוּפִים הָעֶלְיוֹנִים, הִנֵּה הָעִיקָּר הוּא שֶׁיְּבָאֵר וְיַסְבִּיר הָעִנְיָנִים לְעַצְמוֹ עַד אֲשֶׁר יַשִּׂיגֵם בְּהַשָּׂגָה שִׂכְלִית בְּשֵׂכֶל אֱנוֹשִׁי,
וְכֵן בְּעִנְיַן הַמִּצְוֹת הֲרֵי גַם הַמִּצְוֹת דְּחוֹבַת הַמּוֹחַ בִּידִיעַת אֱלֹקוּת, כְּמַאֲמָר שֶׁיֵּשׁ שָׁם מָצוּי רִאשׁוֹן מַמְצִיא כָּל נִמְצָא וְכָל הַנִּמְצָאִים בַּשָּׁמַיִם וּבָאָרֶץ וּמַה שֶּׁבֵּינֵיהֶם הֵם מֵאֲמִיתִּית הִמָּצְאוֹ, וְחוֹבוֹת הַלֵּב בְּאַהֲבָה וְיִרְאָה הֲרֵי מֵעִיקָּרֵי הַמִּצְוָה שֶׁתִּהְיֶה הַשָּׂגַת אֱלֹקוּת נִתְפֶּסֶת בְּמוֹחוֹ הַגַּשְׁמִי וְהָאַהֲבָה וְיִרְאָה יִהְיוּ נִרְגָּשִׁים בִּבְשַׂר הַלֵּב מַמָּשׁ, דְּכָל הָעֲבוֹדוֹת הָאֵלּוּ הֵם דַּוְקָא בָּעוֹלָם הַזֶּה שֶׁהוּא עוֹלָם גַּשְׁמִי,
וּמִטַּעַם זֶה נִתְּנוּ הַהַשְׁפָּעוֹת בְּגַשְׁמִיּוּת עוֹלָם הַזֶּה בִּכְדֵי לַעֲבוֹד בָּהֶן אֶת ה' כְּפִי דִינֵי הַתּוֹרָה וְהַמִּצְוָה.

[kitzur_begin]הַשֶּׁפַע בְּחֶסֶד הַנִּקְצָב הוּא מֵעַתִּיקָא, אֲבָל בָּא עַל יְדֵי הִתְעוֹרְרוּת הָעֶשֶׂר סְפִירוֹת דַּאֲצִילוּת דִּלְשָׁם מַגִּיעַ עֲבוֹדַת הַנִּבְרָאִים וּמַחְסוֹרָם, דִּלְפִי עֶרֶךְ הָעֲבוֹדָה בְּשָׁנָה הֶעָבְרָה כֵּן הִיא קִצְבַת שָׁנָה הַבָּאָה, וּתְלוּיָה בִּתְשׁוּבָה דֶאֱלוּל וַעֲבוֹדָה דִימֵי הַסְּלִיחוֹת וְהַקַּבָּלַת עֹל דְּרֹאשׁ הַשָּׁנָה. הַמָּשָׁל מֵהַסּוֹחֵר וְקוֹנָיו, הַכַּוָּונָה פְנִימִית בְּשֶׁפַע הַגַּשְׁמִי הִיא בִּשְׁבִיל קִיּוּם הַתּוֹרָה וּמִצְוֹת.[kitzur_end]
