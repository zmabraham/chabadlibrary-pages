בשליחותי בתורת שליח הולכה ליד אשתי פב"פ הנזכרת ותתגרש בו ותהי' מותרת לכל אדם. והב"ד דבאבינאוויץ יעשו שטר מעשה ב"ד מזה. וזהו הנוסח בפנינו ב"ד ח"מ מינה הבעל את איש פב"פ שבסיביר שיהי' שליח הולכה על גט ליד אשתי פב"פ הנזכרת ותתגרש בו ותהי מותרת לכל אדם. ומה שנעשה בפנינו כתבנו וחתמנו. ראובן דיין. שמעון דיין. לוי דיין. וגם הבעל יחתום ע"ז בזה"ל כל הכתוב למעלה הכל אמת וחתימתם אמת. אני פב"פ. וישלחו אח"כ לקבלה היינו או הסופר שכ' הגט. או אחד מן העדים החתומים על הגט. וכך תאמר הערט צו ב"ד פאר אייך בין איך ממנה פב"ב שהוא במקום פ' ער זאל זאיין אשליח במקומי ובחריקאי לקבל לי גט מיד בעלי פב"ב בכל מקום שיתננו לו ותהא ידו כידי ועשייתו כעשייתי ומיד כשיגיע הגט מיד בעלי לידו אהי' מגורשת ממנו ומותרת לכל אדם והגם שכבר קבלתי גט ע"י שליח הולכה אך אולי יועיל קבלת גט זה שעל ידי שליח קבלה שם יותר ע"פ הדין ע"כ אני ממנה שליח קבלה. והב"ד של מקום האשה יעשו שטר ממינוי זה. וזהו לשון השטר בפנינו ב"ד ח"מ בכך בשבת כך ימים לחדש פ' שנת פ' לבריאת עולם למנין שאנו מנין כאן מתא פ' מנתה פב"פ איש פב"פ להיות שליח לקבל לה גיטה מיד בעלה פב"פ. וכך אמרה לפנינו ב"ד הח"מ הערט צו ב"ד פאר אייך בין איך ממנה את פב"פ שהוא במקום פ' ער זאל זאיין אשליח במקומי ובחריקאי לקבל לי גט מיד בעלי פב"ב בכל מקום שיתננו לו ותהא ידו כידי ועשייתו כעשייתי. ומיד כשיגיע הגט מיד בעלי לידו אהי' מגורשת ממנו ומותרת לכל אדם והגם שכבר קבלתי גט מבעלי ע"י שליח הולכה אך אולי יועיל קבלת גט זה שע"י שליח לקבלה שם יותר ע"כ אני ממנה שליח קבלה. ומה שנעשה בפנינו כתבנו וחתמנו לזכות ולראי' על מינוי שליחות קבלת גיטה ושריר וקיים. ראובן דיין. שמעון דיין. לוי דיין. וישלחו זה ע"י בי דואר לבאבינאוויטש למקום הבעל ע"י מכתב סטראחאוויע על שם הב"ד שבמקום ההוא. ואזי הב"ד יקראו לבעל ולשליח לקבלה ויצווה הבעל לכתוב שם בבאבינאוויץ גט כהלכתו בכל דקדוקיו ופרטיו. ואח"כ ישבו שלשה דיינים שלא יהיו קרובים לא לבעל ולא לאשה ולא לשליח ויראו הגט ויקראוהו קודם נתינה ויחזירוהו לבעלה ויבטל כל מודעי כדינם ויתנו הבעל ביד שליח קבלה הנזכר וכה יאמר בלשון הזה התקבל וזכה בגט זה לאשתי פב"פ והרי היא מגורשת בו ממני ומותרת לכל אדם וצריך שיאמר שני התיבות דוקא התקבל וזכה. בכדי שיהי' שייך הענין דזכייה נוסף על ענין שליחות הקבלה. כ"כ בתשו' רש"י שבספר מכתב מאליהו שער שישי סס"י ב'. ואח"כ יכתבו הדיינים שבבאבינאוויטש שטר מכל זה. וזהו נוסחו בכך וכך בשבת כך לירח פ' שנת כך לבריאת עולם למנין שאנו מנין כאן בבאבינאוויטש אנחנו ב"ד ח"מ במותב תלתא כחדא הוינא ובא לפנינו פב"פ וגרש את אשתו פב"פ ונתן גיטה לפב"פ שלוחה לקבלה ואמר לו התקבל וזכה בגט זה לאשתי פב"פ והרי היא מגורשת בו ממני ומותרת לכל אדם וקודם מסירת הגט בטל כל מודעי דהוה לי' ואין שום תנאי בגירושין אלא הם גרושין גמורים וקרינו הגט קודם נתינתו ואחר נתינתו והי' כשר. וזה לבד הגט שנשלח ע"י פאצט לסיביר ליתנו לה ע"י שליח הולכה דשם. ושליח קבלה הנזכר הוא איש הדר כאן בבאבינאוויטש. והאשה במקומה מינתה אותו בפני ב"ד דשם שיהי' שליח קבלה שלא בפני השליח וכמבואר במעב"ד שהגיע לנו ממקום הנ"ל והשליח דכאן נתרצה בפנינו להיות שליח. ולהיות שאין אנו מכירים ח"י מעב"ד ההוא לכן אנו כותבים המעשה כמו שהוא ושולחים המעב"ד הנ"ל לחזרה לב"ד ההוא ע"י הפאצט וכשיקויים שם המעב"ד יתירוה לינשא וכתבנו שטר זה לראי' ולזכות בידה והכל שריר וקיים. ראובן דיין. שמעון דיין. לוי בעלי לידו אהיה מגורשת ממנו ומותרת לכל אדם. כן הנוסח אמינוי ש"ק שלא בפניו. הועתק סוף ספר ג"פ סי' רע"ו רע"ז מהמרדכי. ואם אין שם במקום הבעל איש שמכרת האשה אותו אזי ההכרח שתמנה אפילו מי שאין מכרת וכשר ומ"מ צריך שתמנה איש מיוחד וניכר והוא הסופר שיכתוב הגט שם בציווי הבעל הוא יהיה שלוחה ג"כ לקבל הגט אח"כ מיד הבעל. וע' בש"ע רסי' קכ"ג. והב"ד של מקום האשה יעשה שטר ממינוי זה וזהו לשון השטר בפנינו ב"ד ח"מ
