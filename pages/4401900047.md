# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ז. > פֶּרֶק ב

וְהַנִּמְשָׁל מִכָּל זֶה יוּבַן לַמַּשְׂכִּיל לְמַעְלָה, בִּהְיוֹת שֶׁבְּחִינַת פְּנִימִית שֶׁבְּרָצוֹן הָעֶלְיוֹן הוּא הַמֵּאִיר לְיִשְׂרָאֵל דַּוְקָא כְּמוֹ שֶׁכָּתוּב (בַּמִּדְבָּר ו. כה) יָאֵר ה' פָּנָיו אֵלֶיךָ, אֵלֶיךָ דַוְקָא, וְהַיְינוּ עַל יְדֵי הַמְשָׁכַת בְּחִינַת רָצוֹן הָעֶלְיוֹן הַמֵּאִיר בְּתוֹרָה וּמִצְוֹת דַּוְקָא, דְּהִנֵּה כָּל עִיקַּר חָכְמַת הַתּוֹרָה נִמְשְׁכָה מִבְּחִינַת פְּנִימִית רָצוֹן הָעֶלְיוֹן, שֶׁהֲרֵי כָּל עִנְיַן הַתּוֹרָה הוּא הַחָכְמָה אֵיךְ שֶׁיִּהְיֶה בְּחִינַת הַמְשָׁכַת עַצְמוּת אוֹר אֵין סוֹף בָּרוּךְ הוּא, בְּחִינַת אוֹר הַסּוֹבֵב (שֶׁהוּא בְּחִינַת רָצוֹן הָעֶלְיוֹן) מִן הַהֶעְלֵם לְגִילּוּי בָּעוֹלָמוֹת, דְּהַיְינוּ בְּאֵיזֶה אוֹפֶן תִּתְעוֹרֵר הַמְשָׁכַת אוֹר אֵין סוֹף בָּרוּךְ הוּא,
וְהַיְינוּ מַה שֶּׁגָּזְרָה חָכְמָתוֹ יִתְבָּרֵךְ בְּחָכְמָה שֶׁבַּתּוֹרָה בְּעִנְיַן רְמַ"ח מִצְוֹת עֲשֵׂה, שֶׁבָּהֶם דַּוְקָא יִהְיֶה גִּלּוּי בְּחִינַת אוֹר אֵין סוֹף לְמַטָּה, כַּיָּדוּעַ שֶׁרְמַ"ח מִצְוֹת עֲשֵׂה הֵן רְמַ"ח הַמְשָׁכוֹת מֵאוֹר אֵין סוֹף בְּחִינַת רָצוֹן הָעֶלְיוֹן לְהָאִיר אוֹרוֹת בְּכֵלִים מִכֵּלִים שׁוֹנִים, כְּמוֹ עַל יְדֵי תְּפִילִּין יִהְיֶה נִמְשָׁךְ בְּחִינַת רָצוֹן הָעֶלְיוֹן בְּד' מוֹחִין חָכְמָה וּבִינָה וְדַעַת שֶׁנֶּחֱלָק לְחֶסֶד וּגְבוּרָה עַל יְדֵי הַד' פַּרְשִׁיּוֹת, וְהַמְשָׁכָה זוֹ הִיא בַּאֲצִילוּת וּבָעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה עַד שֶׁנִּמְשָׁךְ וּמֵאִיר גַּם בְּהָאָדָם לְמַטָּה הָעוֹשֶׂה אֶת הַמִּצְוָה וּמֵנִיחַ תְּפִילִּין, וּבְמִצְוַת צִיצִית יִהְיֶה גִּילּוּי אוֹר אֵין סוֹף בְּל"ב נְתִיבוֹת חָכְמָה, וּבְמִצְוַת צְדָקָה יָאִיר אוֹר אֵין סוֹף בָּרוּךְ הוּא בְּמִדַּת הַחֶסֶד, וּכְהַאי גַּוְונָא בִּשְׁאָר הַמִּצְוֹת כְּמוֹ עַל יְדֵי סוּכָּה בְּמַקִּיפִים דְּאִימָּא וּבְאֶתְרוֹג בִּבְחִינַת מַלְכוּת וְכַיּוֹצֵא,
וְכָל זֶה הוּא רַק עַל פִּי הַחָכְמָה שֶׁבַּתּוֹרָה שֶׁגָּזְרָה לַעֲשׂוֹת הַסּוּכָּה וְהַתְּפִילִּין וְהָאֶתְרוֹג בְּאוֹתוֹ אוֹפֶן שֶׁעַל יְדֵי זֶה יִהְיֶה דַוְקָא הַמְשָׁכַת גִּילּוּי אוֹר אֵין סוֹף וְלֹא עַל אוֹפֶן אַחֵר זוּלָתוֹ,
וּכְמוֹ כֵן גָּזְרָה חָכְמָה שֶׁבַּתּוֹרָה אוֹפַנִּים פְּרָטִים בִּשְׁסָ"ה לֹא תַעֲשֶׂה, שֶׁהֵן כַּמָּה וְכַמָּה דְבָרִים פְּרָטִים שֶׁמּוֹנְעִים וּמַסְתִּירִים לִהְיוֹת גִּילּוּי אוֹר אֵין סוֹף בָּרוּךְ הוּא, וְצָרִיךְ לְהַעֲבִיר אֶת הַמּוֹנֵעַ מִכָּל צַד, וְהֵן הָאוֹפַנִּים פְּרָטִים שֶׁבְּמִצְוֹת לֹא תַעֲשֶׂה כְּמוֹ לֹא תִלְבַּשׁ שַׁעַטְנֵז (דְּבָרִים כב. יא) וְלֹא תֹאכַל חֵלֶב וְכַדּוֹמֶה בְּכָל הַשְּׁסָ"ה לֹא תַעֲשֶׂה, שֶׁעַל יְדֵי הַכְּפִיָּה בְּהֶעְדֵּר עֲשִׂיַּית הַדָּבָר וּבִזְהִירוּת בָּזֶה, הֲרֵי הוּא מֵסִיר אֶת הַקְּלִיפּוֹת וְסִטְרָא אָחֳרָא הַמּוֹנְעִים וּמַסְתִּירִים אֶת הַגִּילּוּיִם דְּאוֹר אֵין סוֹף בָּרוּךְ הוּא.

[kitzur_begin]פְּנִימִית הָרָצוֹן נִמְשָׁךְ לְיִשְׂרָאֵל וְהוּא עַל יְדֵי תּוֹרָה וּמִצְוֹת, דְּכָל עִנְיַן הַתּוֹרָה הוּא הַמְשָׁכַת אוֹר הַסּוֹבֵב בָּעוֹלָמוֹת, רְמַ"ח מִצְוֹת עֲשֵׂה הֵם הַמְשָׁכוֹת בַּכֵּלִים, תְּפִילִּין וְצִיצִית, צְדָקָה סוּכָּה אֶתְרוֹג, וּמִצְוֹת לֹא תַעֲשֶׂה לְהָסִיר הַקְּלִיפּוֹת וְסִטְרָא אָחֳרָא הַמּוֹנְעִים גִּילּוּי אוֹר אֵין סוֹף.[kitzur_end]
