# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר כב. > פֶּרֶק א

מַאֲמָר כב.
וְהִנֵּה עִם הֱיוֹת דְּדַעַת רַבָּנָן שֶׁמְּזוֹנוֹתָיו שֶׁל אָדָם קְצוּבִין לוֹ מֵרֹאשׁ הַשָּׁנָה, וְנִתְבָּאֵר לְעֵיל שֶׁכָּל צָרְכֵי הָאָדָם וּבְנֵי בֵיתוֹ בְּבָנֵי חַיֵּי וּמְזוֹנֵי הֵם בִּכְלַל הַמְּזוֹנוֹת בְּכָל זֶה גַּם לְדַעַת רַבָּנָן הִנֵּה יֵשׁ דִּין בְּכָל יוֹם, וְלָכֵן אָנוּ מִתְפַּלְלִים וּמְבַקְשִׁים בְּכָל יוֹם רְפָאֵנוּ וּבָרֵךְ עָלֵינוּ, וַהֲגַם דְּבִגְמָרָא (שָׁם) אָמְרוּ כְּמַאן מְצַלִּינָן הַאִידְנָא אַקְּצִירֵי וּמְרִיעֵי כְּמַאן כְּרַבִּי יוֹסֵי, מִכָּל מָקוֹם הֲלֹא גַּם לְרַבָּנָן מִי לָא מְצַלִּינָן וּבָרֵךְ עָלֵינוּ כְּמוֹ שֶׁכָּתוּב בְּתוֹסָפוֹת שָׁם,
וְהָעִנְיָן הוּא דַּהֲגַם שֶׁהַחֶסֶד נִקְצַב בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים מִכָּל מָקוֹם הוּא עֲדַיִין בִּבְחִינַת מַלְכוּת דַּאֲצִילוּת כְּמוֹ שֶׁהִיא בַּאֲצִילוּת, דְּמַלְכוּת דַּאֲצִילוּת כְּמוֹ שֶׁהִיא בַּאֲצִילוּת הֲרֵי הִיא מוּפְלָא בְּעֶרֶךְ מִבְּרִיאָה-יְצִירָה-עֲשִׂיָּה, וּבְחִינַת רַגְלֶיהָ יוֹרְדוֹת מָוֶת דְּמַלְכוּת דַּאֲצִילוּת הַיּוֹרֶדֶת לִבְרִיאָה-יְצִירָה-עֲשִׂיָּה כְּמוֹ שֶׁכָּתוּב (מִשְׁלֵי לא. טו) וַתָּקָם בְּעוֹד לַיְלָה וַתִּתֵּן טֶרֶף לְבֵיתָהּ דְּקָאֵי עַל בְּחִינַת חִיצוֹנִיּוּת הַמַּלְכוּת דַּאֲצִילוּת הַיּוֹרֶדֶת לְהַחֲיוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה, הִנֵּה בְּרִיאָה-יְצִירָה-עֲשִׂיָּה לְגַבֵּי אֲצִילוּת נִקְרֵאת מָוֶת דִּכְתִיב (דְּבָרִים ל. טו) רְאֵה נָתַתִּי לְפָנֶיךָ הַיּוֹם אֶת הַחַיִּים וְאֶת הַטּוֹב וְאֶת הַמָּוֶת וְאֶת הָרָע, דְּחַיִּים וָטוֹב מָוֶת וָרָע הֵם תּוֹאֲרֵי הָרוּחָנִי וְהַגַּשְׁמִי. דְּהָרוּחָנִי הוּא חַיִּים וָטוֹב וְהַגַּשְׁמִי הוּא מָוֶת וָרָע. וְזֶהוּ רַגְלֶיהָ יוֹרְדוֹת מָוֶת דִּבְרִיאָה לְגַבֵּי אֲצִילוּת אֵינָהּ בְּעֶרֶךְ כְּלָל,
לָכֵן הִנֵּה הֲגַם שֶׁהַחֶסֶד בְּבָנֵי חַיֵּי וּמְזוֹנֵי כְּבָר הוּקְצַב בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים אֲבָל הוּא עֲדַיִין בְּמַלְכוּת דַּאֲצִילוּת, וְלִהְיוֹתוֹ עֲדַיִין בְּמַלְכוּת דַּאֲצִילוּת הִנֵּה כְּדֵי שֶׁיּוּמְשַׁךְ לְמַטָּה הוּא עַל יְדֵי הִשְׁתַּלְשְׁלוּת וְהִתְלַבְּשׁוּת בִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה דְּבַר יוֹם בְּיוֹמוֹ, וְיֵשׁ עַל זֶה דִּין וְעִיּוּן אִם רָאוּי וּכְדַאי הָאָדָם שֶׁיּוּמְשַׁךְ וְיִתְלַבֵּשׁ הַחֶסֶד בַּעֲשִׂיָּה בְּגַשְׁמִיּוּת לְטוֹב לוֹ, שֶׁיּוֹסִיף בְּלִימּוּד הַתּוֹרָה בִּמְנוּחָה וּבְקִיּוּם הַמִּצְוֹת בְּהִידּוּר וּכְמוֹ שֶׁכָּתוּב זֶה אֵלִי וְאַנְוֵהוּ וְאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (שַׁבָּת קלג. ב) הִתְנָאֶה לְפָנָיו בְּמִצְוֹת וְכֵן בְּמַעֲשֵׂה הַצְּדָקָה בְּיָד רְחָבָה, אוֹ חַס וְשָׁלוֹם שֶׁהַהַשְׁפָּעָה בְּגַשְׁמִיּוּת תִּהְיֶה לְרָעָתוֹ וּכְמוֹ שֶׁכָּתוּב וּמְשַׁלֵּם לְשֹׂנְאוֹ לְהַאֲבִידוֹ מִשָּׂכָר הָרוּחָנִי בָּעוֹלָם הַבָּא, וּכְמוֹ שֶׁהוּא בְּכַמָּה פּוֹשְׁעֵי יִשְׂרָאֵל רַחֲמָנָא לִצְּלָן אֲשֶׁר הַגְּמוּל בְּעַד הַטּוֹב שֶׁעוֹשִׂים מְשַׁלְּמִים לָהֶם בְּטוּב עוֹלָם זֶה שֶׁמַּצְלִיחִים בְּבָנֵי חַיֵּי וּמְזוֹנֵי, וְהוּא מַה שֶּׁדֶּרֶךְ רְשָׁעִים צָלֵחָה אֲבָל הוּא רַק בָּעוֹלָם הַזֶּה וְנֶעֱנָשִׁים בְּעוֹנָשִׁים מָרִים וְקָשִׁים לָעוֹלָם הַבָּא,
וְלִפְעָמִים הִנֵּה הַפְּסַק דִּין אֲשֶׁר הַטּוֹב שֶׁנִּקְצַב לוֹ יִשָּׁאֵר לְמַעְלָה בָּעוֹלָמוֹת הָרוּחָנִים וִיקַבְּלֶנּוּ לָעוֹלָם הַבָּא,
(וַהֲגַם דְּלִכְאוֹרָה מֵאַחַר שֶׁהַחֶסֶד הַנִּקְצָב לוֹ כְּבָר כּוֹלֵל בְּחִינַת חֶסֶד דַּעֲשִׂיָּה וְהַהַשְׁפָּעוֹת שֶׁמִּזֶּה, וְאִם כֵּן לִכְאוֹרָה בְּהֶכְרַח שֶׁיּוּמְשַׁךְ בַּעֲשִׂיָּה דַוְקָא אִם כֵּן אֵיךְ הוּא נִשְׁאָר לְמַעְלָה בְּרוּחָנִיּוּת, אַךְ זֶה יוּבַן מִמַּה שֶׁיִּתְבָּאֵר לְקַמָּן בְּפֶרֶק ב' – דִּבְכָל זֶה הִנֵּה רַבָּנָן וְרַבִּי יוֹסֵי כֻּלָּם מַסְכִּימִים)
וְרַק הַנַּפְקָא מִינָּהּ בֵּין רַבָּנָן לְרַבִּי יוֹסֵי הוּא, דִּלְרַבָּנָן כְּבָר נֶאֱמַר וְנִגְזַר בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים מַה יּוּמְשַׁךְ מֵהַחֶסֶד, וַאֲשֶׁר לְפִי זֶה הַדִּין שֶׁאַחַר כַּךְ בְּכָל יוֹם שֶׁהוּא אִם יְקוּיַּים כְּפִי שֶׁנֶּאֱמַר וְנִגְזַר מִכְּבָר אֵינוֹ תְּחִלַּת דִּין, וְהוּא עַל דֶּרֶךְ שֶׁמְּבוֹאָר בִּגְמָרָא (שָׁם) דִּלְרַבִּי יוֹסֵי דְּאָמַר אָדָם נִידּוֹן בְּכָל יוֹם הָא דִכְתִיב (אִיּוֹב ז. יח) לִרְגָעִים תִּבְחָנֶנּוּ, דְּמַשְׁמָע כָּל שָׁעָה הוּא עִיּוּנֵי בְּעָלְמָא וְאֵינוֹ דִּין מַמָּשׁ, כַּךְ לְרַבָּנָן דְּאָמְרִי אָדָם נִגְמַר גְּזַר דִּינוֹ בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים הָא דִכְתִיב (מְלָכִים-א ח. נט) לַעֲשׂוֹת מִשְׁפַּט כו' דְּבַר יוֹם בְּיוֹמוֹ הוּא גַם כֵּן עִיּוּן, אֶלָּא שֶׁאֵינוֹ עִיּוּנֵי בְּעָלְמָא אֶלָּא נִקְרָא פְּקִידָה שֶׁהוּא יוֹתֵר מֵעִיּוּן וְהוּא מַה שֶּׁכָּתוּב (אִיּוֹב ז. יח) וַתִּפְקְדֶנּוּ לִבְקָרִים כו', וְלָכֵן צְרִיכִים לְהִתְפַּלֵּל וּלְבַקֵּשׁ בְּכָל יוֹם גַּם לְרַבָּנָן.
[cup]וְרַבִּי[/cup] יוֹסֵי סֹבֵר שֶׁבְּכָל יוֹם הוּא דִין גָּמוּר, דְּרַבִּי יוֹסֵי מְפָרֵשׁ פְּקִידָה הַנֶּאֱמַר בַּפָּסוּק וַתִּפְקְדֶנּוּ לִבְקָרִים שֶׁהוּא מִשְׁפָּט גָּמוּר, וְרַבָּנָן מְפָרְשִׁים מִשְׁפָּט הַנֶּאֱמַר בַּפָּסוּק לַעֲשׂוֹת מִשְׁפַּט כו' שֶׁהוּא עִנְיַן פְּקִידָה, וְלִשְׁנֵיהֶם שְׁנֵי הַכְּתוּבִים הָאֵלּוּ מְבָאֲרִים זֶה אֶת זֶה מִשּׁוּם דִּשְׁנֵיהֶם נֶאֶמְרוּ עַל הַדִּין שֶׁבְּכָל יוֹם,
מַה שֶּׁאֵין כֵּן מַה שֶּׁנֶּאֱמַר (תְּהִלִּים פא. ה) כִּי חֹק לְיִשְׂרָאֵל הוּא מִשְׁפָּט לֵאלֹקֵי יַעֲקֹב קָאֵי עַל הַדִּין דְּרֹאשׁ הַשָּׁנָה, וְהַטַּעַם הוּא לְפִי שֶׁבְּהַחֶסֶד דַּאֲצִילוּת אֵין בּוֹ שׁוּם תּוֹאַר עֲדַיִין מֵחֶסֶד דַּעֲשִׂיָּה וּמֵהַהַשְׁפָּעוֹת הַנִּמְשָׁכוֹת לְמַטָּה לָכֵן בִּכְדֵי שֶׁיִּצְטַיֵּיר בְּהַחֶסֶד צִיּוּר הַשְׁפָּעוֹת דַּעֲשִׂיָּה שֶׁזֶּה בּוֹ הִתְחַדְּשׁוּת, יֵשׁ עַל זֶה דִּין גָּמוּר אִם יִצְטַיֵּיר וּבְאֵיזֶה אוֹפֶן יִצְטַיֵּיר אוֹ שֶׁיִּשָּׁאֵר רַק בָּעוֹלָם הָעֶלְיוֹן בְּרוּחָנִיּוּת וִיקַבֵּל אוֹתוֹ לָעוֹלָם הַבָּא.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]הַחֶסֶד שֶׁנִּקְצָב לְאָדָם בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים הוּא בְּמַלְכוּת דַּאֲצִילוּת, מַלְכוּת דַּאֲצִילוּת מוּפְלָא בְּעֶרֶךְ מִבְּרִיאָה-יְצִירָה-עֲשִׂיָּה, חַיִּים וָטוֹב מָוֶת וָרָע, וְלָכֵן יֵשׁ דִּין בְּכָל יוֹם אִם רָאוּי הָאָדָם לְמַה שֶּׁנִּקְצַב לוֹ, וְהַדִּין לְדַעַת רַבָּנָן וְרַבִּי יוֹסֵי הוּא עִיּוּן וְרַק שֶׁרַבִּי יוֹסֵי סוֹבֵר דְּעִיּוּן זֶה הוּא דִין גָּמוּר.[kitzur_end]
