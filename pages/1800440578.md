שהכתיב באחד והקרי באחת והיינו כי בבחי' לאסתכלא ביקרא דמלכא יש ב' בחי' הא' ההסתכלות ביחו"ע איך שהוא ית' אחד ממש וכולא קמי' כלא חשיב ואי לזאת הוא תמיד בבחי' ביטול באור א"ס ב"ה ממש בבחי' אין כו' והב' מבחי' יחו"ת בשכמל"ו איך שהוא ית' מלך העולם וכל חיות העולם שנראה ליש מורגש אינו אלא מבחי' מלך וה"ע בחי' ביטול היש לאין דהיינו שהוא יש ואעפ"כ בטל כו' וזהו ענין באחד ובאחת כי באחד הוא ענין יחו"ע כי שם הביטול הוא בבחי' אחד ממש איהו וחיוהי חד כו' באחת הוא בחי' יחו"ת שיהי' בבחי' יש ושיהי' היש בטל כי אחת לשון נקבה בחי' נוק' המקבל מאחד כו' (בד"ה כי תשמע בקול) והנה ב' בחי' הסתכלות הנ"ל יש להן שורש למעלה שיש למעלה ג"כ ב' השגות כמ"ש כי אל דיעות הוי' שכולל ב' דיעות, שלפעמים נאמר עיני ה' אל צדיקים וכתיב עין ה' אל יראיו, עיני ה' היינו ההסתכלות כמו שלמעלה היש ולמטה כלא חשיב והוא ההסתכלות מיחו"ע ועד"ז ה"ע ציץ דכה"ג להיות עינים לראות כו', ובחי' עין ה' הוא ההשגחה מיחו"ת כו' וזה ענין ציצית לשון נוק' שלא כל אדם יכול לבא לבחי' ההסתכלות דיחו"ע אבל לבחי' הסתכלות דיחו"ת יוכל כל א' לבא, אך בציצית הי' ג"כ לבן ותכלת שהם ג"כ ב' בחי' הסתכלות הנ"ל, וזהו למען תזכרו שיהי' בבחי' זכרים ע"ד הארת הציץ שהוא לשון זכר כו'. 
ד) ונחזור למש"ל סוף אות ב' (בד"ה בשעה שהקדימו) והנה מבואר במ"א בפי' עושי דברו לשמוע בקול דברו שצ"ל עושי דברו היינו להמשיך הרצון להיות מלך שמו נק' כי בתחלת הבריאה נמשך הרצון ע"ז בחסד חנם ועכשיו באתעדל"ת תליא וכמ"ש ג"כ בד"ה יביאו לבוש מלכות דפי' יביאו להמשיך בחי' זו ע"י אתעדל"ת כו' והיינו ע"י מעשה המצות וזהו שאנו מברכים לפני' בא"י מלך העולם אשר קדשנו כו' שעל ידי עול מצות ממשיכים להיות מלך העולם כו' וזהו עושי דברו כי הלא דברי כאש בבחי' הסתלקות למקורו וגיבורי כח עושים וממשיכים בחי' דבר הוי' שהוא בחי' מל' כו' וה"ע יביאו לבוש מלכות שבחי' מל' נק' לבוש וה"ע השלמה כנ"ל, אך ארז"ל למה קדמה פ' שמע כו' שיקבל עליו עומ"ש תחלה ואח"כ יקבל עליו עול מצות ועמ"ש מזה בד"ה מנורת זהב ובד"ה יונתי גבי קומי לך כו' יפתי, וקבלת עומ"ש הוא ק"ש שמע ישראל כו' ואהבת והיינו שצריך להמשיך להיות עוטה אור כשלמה ולא בחי' שלמה ולבוש לבד וזהו ע"י ואהבת ב"פ אור כו', וזהו ענין ועשו להם ציצית על כנפי בגדיהם וכתיב על ארבע כנפות כסותך, (בחיי ס"פ שלח) והנה הזכיר במצוה זו מלת כנפיים והי' יכול לומר על ארבע צידי או רבעי בגדיהם כדי לרמוז לענין כנפי החיות הנושאות את הכסא כו' שהרי במצות ציצית יש שייכות לכסה"כ כמשארז"ל
