# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ט. > פֶּרֶק ג

אָמְנָם כָּל עִנְיַן הַדִּין וְהַחֶשְׁבּוֹן הוּא רַק מִצַּד סֵדֶר הִשְׁתַּלְשְׁלוּת, דְּכַךְ הִיא הַמִּדָּה בְּהִשְׁתַּלְשְׁלוּת שֶׁמּוּכְרָח שֶׁיִּהְיֶה בְּדִין וְחֶשְׁבּוֹן מְדוּיָּק מִצַּד אַרְבָּעָה טְעָמִים, הָא' לִהְיוֹתוֹ אוֹר פְּנִימִי, ב' מִצַּד עִנְיַן הַסֵּדֶר, ג' לִהְיוֹתוֹ אוֹר בִּכְלִי, ד' מִצַּד הַהִשְׁתַּלְשְׁלוּת לְדַרְגָּא שֶׁלְּמַטָּה מִמֶּנּוּ.
[cup]כָּל[/cup] אוֹר אֵיזֶה שֶׁהוּא בֵּין אוֹר הַמּוֹחִין בֵּין אוֹר הַמִּדּוֹת הֲרֵי יֵשׁ לוֹ טִבְעוֹ הָעַצְמִי בְּמַה שֶּׁהוּא, דְּאוֹר הַמּוֹחִין הוּא בְּטֶבַע הַהִתְיַישְּׁבוּת וְהַמְּנוּחָה וְאוֹר הַמִּדּוֹת בְּטֶבַע הַהִתְרַגְּשׁוּת וְהַהִתְפַּעֲלוּת, כֵּן הוּא בְּאוֹר הַפְּנִימִי שֶׁיֵּשׁ לוֹ טִבְעוֹ הָעַצְמִי בְּמַה שֶּׁהוּא מְחוּשְׁבָּן וּמְדוּיָּק,
טֶבַע הַסֵּדֶר הוּא טֶבַע הַחֶשְׁבּוֹן וְהַדִּיּוּק בְּכַמּוּת וּבְאֵיכוּת.
[cup]וְאוֹרוֹת[/cup] הַמִּתְלַבְּשִׁים בַּכֵּלִים הִנֵּה אַף גַּם כַּאֲשֶׁר כְּלָלוּת הַכֵּלִים מַתְאִימִים אֶל הָאוֹרוֹת, כְּמוֹ חוֹמֶר הַמּוֹחַ שֶׁהוּא בִּכְלָלוּתוֹ כְּלִי אֶל הַשְׁרָאַת הַשֵּׂכֶל וְחוֹמֶר הַלֵּב שֶׁהוּא בִּכְלָלוּתוֹ כְּלִי אֶל מִשְׁכַּן הַמִּדּוֹת, בְּכָל זֶה הֲרֵי בְּהֶכְרַח שֶׁתִּהְיֶה מְדִידָה בְּאוֹר הַשֵּׂכֶל בְּאוֹפֶן הַשְׁרָאָתוֹ בְּמוֹחַ שֶׁבָּרֹאשׁ, וְכֵן בְּעִנְיַן הַמִּדּוֹת, דְּאִם לֹא כֵן הֲרֵי הַכֵּלִים דְּחוֹמֶר הַמּוֹחַ וְחוֹמֶר הַלֵּב לֹא יָכִילוּ אֶת הָאוֹר הַמְרוּבֶּה וּמִתְקַלְקְלִים רַחֲמָנָא לִצְּלָן, וּכְמוֹ שֶׁאָנוּ רוֹאִים בְּמוּחָשׁ רַחֲמָנָא לִצְּלָן בְּכַמָּה חוֹלִים בְּחוֹלִי הַמּוֹחַ וְחוֹלִי הַלֵּב שֶׁבָּא בְּסִבַּת רִיבּוּי הָאוֹר וּמִיעוּט הַכֵּלִים,
וְאוֹר שֶׁעִנְיָנוֹ לְהִשְׁתַּלְשֵׁל לְדַרְגָּא שֶׁלְּמַטָּה מִמֶּנּוּ, הֲרֵי בְּהֶכְרַח שֶׁהָאוֹר יִהְיֶה מְחוּשְׁבָּן וּמְדוּיָּק לְפִי גּוֹדֶל וְרוֹחַב הַכְּלִי, דְּלוּלֵא זֹאת הִנֵּה לֹא זוֹ בִּלְבָד שֶׁהַכְּלִי לֹא יְקַבֵּל אֶת הָאוֹר, אֶלָּא עוֹד זֹאת שֶׁהָאוֹר מְאַבֵּד אֶת הַכְּלִי וּמְשַׁבְּרוֹ, כְּמוֹ שֶׁאָנוּ רוֹאִים בְּמוּחָשׁ בְּהַשְׁפָּעַת הַשֵּׂכֶל מֵהָרַב לְתַלְמִידוֹ, דְּכַאֲשֶׁר הַהַשְׁפָּעָה אֵינָהּ לְפִי עֶרֶךְ כְּלֵי שֵׂכֶל הַתַּלְמִיד, הִנֵּה לֹא זוֹ בִּלְבָד שֶׁהַתַּלְמִיד אֵינוֹ מְקַבֵּל אֶת הַשֵּׂכֶל הַנִּשְׁפָּע לוֹ, אֶלָּא עוֹד זֹאת שֶׁכְּלֵי שִׂכְלוֹ מִתְבַּלְבְּלִים, וְטַעַם הַדָּבָר הוּא דְּכָל אוֹר הַמִּשְׁתַּלְשֵׁל מִדַּרְגָּא לְדַרְגָּא שֶׁלְּמַטָּה הֵימֶנּוּ צָרִיךְ לִהְיוֹת לְפִי עֶרֶךְ הַדַּרְגָּא הַהִיא,
וְהַדּוּגְמָא מִזֶּה יוּבַן לְמַעְלָה בִּסְפִירוֹת הָעֶלְיוֹנוֹת הַמִּשְׁתַּלְשְׁלוֹת, אֲבָל מִצַּד אוֹר אֵין סוֹף שֶׁלְּמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת שֶׁנִּמְשָׁךְ בִּבְחִינַת מַקִּיף וְאֵינוֹ מִתְלַבֵּשׁ בַּכֵּלִים, הֲרֵי אֵינוֹ נוֹגֵעַ אִם הוּא רָאוּי לְגִילּוּי הָאוֹר וְהַחַיּוּת, אוֹ שֶׁאֵינוֹ רָאוּי, מֵאַחַר שֶׁאֵינוֹ צָרִיךְ לַכֵּלִים כְּלָל, הֲרֵי נִמְשָׁךְ גַּם לְמָקוֹם שֶׁאֵינוֹ רָאוּי (הַיְינוּ שֶׁאֵינוֹ כְלִי) כְּמוֹ לְמָקוֹם הָרָאוּי. וְזֶהוּ שֶׁגַּם הַקְּלִיפּוֹת וְהַסִּטְרָא אָחֳרָא וְעוֹבְרֵי רְצוֹנוֹ יְכוֹלִים לְקַבֵּל מִשָּׁם אַף שֶׁאֵינָם כֵּלִים הָרְאוּיִם כְּלָל, לְפִי שֶׁאֵינוֹ צָרִיךְ לַכֵּלִים כְּלָל, אֵינוֹ שַׁיָּיךְ בּוֹ עִנְיַן הַסֵּדֶר וְנִמְשָׁךְ מִשָּׁם רִיבּוּי שֶׁפַע וְחַיּוּת בְּיוֹתֵר לִהְיוֹת שֶׁהוּא בִּבְחִינַת בְּלִי גְבוּל,
דְּהָאוֹר דְּסֵדֶר הִשְׁתַּלְשְׁלוּת מֵאַחַר שֶׁהוּא אוֹר פְּנִימִי וּמִתְלַבֵּשׁ בִּכְלִי הֲרֵי הוּא מוּגְבָּל לְפִי אוֹפֶן הַכֵּלִים וְשַׁיָּיךְ בּוֹ עִנְיַן הַסֵּדֶר וְעִנְיַן הַהִשְׁתַּלְשְׁלוּת, אֲבָל הָאוֹר שֶׁלְּמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת מֵאַחַר שֶׁהוּא אוֹר מַקִּיף אֵינוֹ מוּגְבָּל וְלָזֹאת נִמְשָׁךְ מִשָּׁם רִיבּוּי שֶׁפַע,
וְזֶהוּ שֶׁכָּתוּב (עוֹבַדְיָה א. ד) אִם תַּגְבִּיהַּ כַּנֶּשֶׁר, כְּמוֹ דֶּרֶךְ הַנֶּשֶׁר בַּשָּׁמַיִם (מִשְׁלֵי ל. יט) כְּמוֹ כֵן הַקְּלִיפּוֹת וְהַסִּטְרָא אָחֳרָא מַגְבִּיהִים אֶת עַצְמָם לְקַבֵּל מִבְּחִינַת הַמַּקִּיף עֶלְיוֹן לְפִי שֶׁמִּשָּׁם נִמְשָׁךְ בְּלִי דִין וְחֶשְׁבּוֹן הַיְינוּ גַּם שֶׁאֵינָם בְּטֵלִים כְּלָל לֵאלֹקוּת, וְאַדְרַבָּא עוֹבְרִים רְצוֹנוֹ יִתְבָּרֵךְ לְגַמְרֵי, וְנִמְשָׁךְ מִשָּׁם הַשֶּׁפַע בְּרִיבּוּי בְּיוֹתֵר. וְזֶהוּ הַטַּעַם בְּהַצְלָחַת הָעַכּוּ"ם בְּרִיבּוּי עוֹשֶׁר וְכָבוֹד בָּעוֹלָם הַזֶּה, וְגַם הֵמָּה שׁוֹלְטִים עַל יִשְׂרָאֵל, וְהַיְינוּ מִפְּנֵי שֶׁשֹּׁרֶשׁ קַבָּלָתָם הוּא מִבְּחִינָה שֶׁלְּמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת, דְּהַשְׁפָּעָה לְיִשְׂרָאֵל הִיא עַל יְדֵי תּוֹרָה וּמִצְוֹת וּבָאָה בְּסֵדֶר הִשְׁתַּלְשְׁלוּת, וְהַהַשְׁפָּעָה לְעַכּוּ"ם הִיא מִלְמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת וְהִיא בְּלִי חֶשְׁבּוֹן וּגְבוּל.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]מִסְּפִירַת הַחָכְמָה וּלְמַטָּה בָּא בַחֶשְׁבּוֹן מִד' טְעָמִים. א) אוֹר פְּנִימִי. ב) סֵדֶר. ג) מִתְלַבֵּשׁ בִּכְלִי. ד) מִשְׁתַּלְשֵׁל לְמַטָּה מִמַּדְרֵיגָתוֹ, לְמַעְלָה מִן הַחָכְמָה הוּא לְמַעְלָה מֵהִשְׁתַּלְשְׁלוּת אֵינוֹ בָּא בַחֶשְׁבּוֹן, וּמֵהַאי טַעֲמָא הַסִּטְרָא אָחֳרָא מַגְבִּיהַּ עַצְמָהּ לְקַבֵּל מִשָּׁם לִהְיוֹתָהּ בְּלִי דִיּוּק וּבְרִיבּוּי.[kitzur_end]
