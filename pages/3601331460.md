בהמדרי' שאחרי' כו' משא"כ כתר נק' נקודה דלא נעיץ מפני שאינו דבוק כו' כנ"ל, ומן החכ' מצטייר בבינה שבא יותר בבחי' התגלות כו'. ובהקדמת הזהר ד"ב ע"א כ' בשעתא דסדכ"ס בעא לאתגלייא (פי' רצה לגלות מלכותו, מק"מ) עבד ברישא נקודה חדא (אבא) צייר בה כל ציורים (ע"י צייר אימא וזו"נ) חקק בה כל גליפין כו', דחקיקה היא יותר מציור, וכמ"ש הפרד"ס בענין רשימה חקיקה כו' הובא בלקו"ת בהוספות ד"ה להבין מ"ש באוצ"ח כו'. וכ"ז הוא בבחי' המשכת האור דכתר ממדרי' למדרי' עד בחי' מל', וההתהוות היא מבחי' מל' כו'. וי"ל דזהו דאחר אומרו בראשית ברא אלקים את השמים כו', דמרומז בזה כל הע"ס דאצי' וכמ"ש בתוספתא דזהר בראשית דל"א ע"ב, נא' אח"כ והארץ היתה תהו כו', שלא הי' עדיין התהוות העולמות, עד ויאמר אלקים יהי אור דמבחי' דבור העליון דבחי' מל' נתהווה האור וכן כל הנבראים כו'.
[cup]והנה[/cup] בחי' מל' ג"כ כלולה מע"ס ובחי' כתר מל' הוא בחי' הרצון למלוכה כו', וכמו עד"מ המלך שמכתירים אותו בכתר מל' הוא שמעוררים בו הרצון למלוכה כו', ועם היות שיש בו ענין ההתנשאות מצ"ע ה"ז בהעלם בעצמו, וההכתרה היא התגלות ההתנשאות שבא בבחי' רצון למלוך כו', ואין זה עדיין בבחי' התנשאות על עם ממש, ובאמת הוא ההתגלות דההתנשאות עצמו שהי' תחלה בהעלם בעצמו וע"י ההכתרה ה"ז יוצא ובא בהתגלות כו', ויש בזה ג"כ הרצון למלוך כו'. ובפרטיות הוא ב' ענינים שנעשה ע"י ההכתרה, הא' התגלות בחי' התנשאות עצמו שנעשה מוגבה ומרומם בעצמו בהתגלות כו', והב' הרצון למלוך בבחי' התנשאות על עם כו', וגם הרצון הזה אינו עדיין בבחי' התנשאות על עם שזהו דוקא אח"כ שהוא בבחי' מושל ושולט כו', רק שהוא רצון גלוי להיות מתנשא על עם כו'. והדוגמא מזה יובן במלכותא דרקיע בחי' מל' דאצי' דכתר מל' הוא בחי' רצון גלוי למלוכה כו', ואין זה עדיין בבחי' התנשאות על עולמות נבראים ונפרדים כו', והיינו דמבחי' זו אינו נמשך עדיין להיות התהוות העולמות
בפו"מ שזהו מבחי' מל' שבמל' שזהו בחי' ההתנשאות על עולמות בבחי' מקיף להיות התהוות העולמות כו', אבל בחי' הרצון למלוכה הוא רק הרצון להיות מתנשא כבי' כו', וכמו בחי' כתר שממשיכים בר"ה שהוא הרצון למלוכה כנודע זהו בחי' המלך הקדוש כו', וגם שזהו התגלות בחי' התנשאות עצמו כו'. וי"ל דכמו שבחי' כתר דאצי' כלול מבחי' עתיק ואריך כו', כמו"כ בחי' כתר מל' יש בזה ב' ענינים ההתגלות דהתנשאות עצמו כבי', וגילוי הרצון להיות בבחי' התנשאות על עם כו', ובחי' ההתגלות דהתנשאות עצמו ובחי' הרצון למלוכה עם היות שאין זה עדיין בחי' מלוכה ממש על עם כנ"ל, הוא הסיבה להיות בחי' התנשאות על עם כו', דבחי' התנשאות עצמי כמו שהוא בהעלם בעצמו יכול להיות שלא יבוא לידי התגלות, וכמ"ש במ"א שכל כח עצמי כמו שהוא נטוע בעצם הנפש יכול להיות שלא יתגלה כו' וכמ"ש מזה בד"ה שמח תשמח הנ"ל, אבל כאשר בא בהתגלות ה"ז סיבה לגילוים שיבואו ממנו כו', וע"כ בחי' התגלות ההתנשאות עצמי דבחי' כתר מל' ה"ז בבחי' סיבה להיות בחי' התנשאות על עם שהוא ההתנשאות על
העולמות כו' (וכ"ש דבחי' רצון הגלוי למלוכה הוא בבחי' סיבה כו'). וגם דהנה בחי' ההתנשאות על העולמות (שזהו מקור התהוות העולמות כו' כנ"ל) זהו מה שבחי' מל' דמל' דאצי' נעשה בחי' כתר לברי' כו' (והו"ע כח הפועל הכללי שנת"ל ד"ה אנכי אנכי הא' שזהו
