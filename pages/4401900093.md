מַאֲמָר כו.
וְזֶהוּ וּמַעְיָן מִבֵּית ה' יֵצֵא וְהִשְׁקָה אֶת נַחַל הַשִּׁטִּים דְּהִנֵּה יֵשׁ בִּקְדוּשָּׁה גַּם כֵּן בְּחִינַת שְׁטוּת וּכְמוֹ שֶׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (כְּתוּבּוֹת יז. א) אַהַנְיָא לֵיהּ שְׁטוּתֵיהּ לְסַבָּא, וּפֵירוּשׁ וְעִנְיַן שְׁטוּת זוֹ הוּא עִנְיָן שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת, וְעַל דֶּרֶךְ מַה שֶּׁכָּתוּב (מְלָכִים-ב ט. יא) מַדּוּעַ בָּא הַמְּשׁוּגָּע הַזֶּה שֶׁקָּרְאוּ לַנָּבִיא מְשׁוּגָּע, וְהַיְינוּ כִּי עִנְיָנוֹ הוּא הֵיפֶךְ עֵץ הַדַּעַת טוֹב וָרָע וְהַיְינוּ הֶעְדֵּר הַהֶרְגֵּשׁ שֶׁמִּצַּד הַחוּשִׁים הַטִּבְעִיִּים שֶׁזֶּה עִנְיַן עֵץ הַדַּעַת, וּכְיָדוּעַ שֶׁעַל יְדֵי חֵטְא עֵץ הַדַּעַת נַעֲשָׂה עִנְיַן הַהֶרְגֵּשׁ, וּכְמוֹ שֶׁכָּתוּב (בְּרֵאשִׁית ג. ז) וַיֵּדְעוּ כִּי עֵרוּמִּים הֵם, וְכִי מִתְּחִלָּה לֹא יָדְעוּ שֶׁהֵם עֵרוּמִּים, אֶלָּא שֶׁמִתְּחִלָּה לֹא הִרְגִּישׁוּ זֶה וְאַחַר כַּךְ הִרְגִּישׁוּ שֶׁהֵם עֲרוּמִּים, וּכְמוֹ שֶׁכָּתוּב תְּחִלָּה (בְּרֵאשִׁית ב. כה) וַיִּהְיוּ שְׁנֵיהֶם עֲרוּמִּים כו' וְלֹא יִתְבֹּשָׁשׁוּ, וְהַיְינוּ שֶׁלֹּא הִרְגִּישׁוּ זֹאת לְחִסָּרוֹן וְאַחַר כַּךְ וַיֵּדְעוּ כו' שֶׁהִרְגִּישׁוּ,
וּכְמוֹ שֶׁכָּתוּב בְּמוֹרֵה נְבוּכִים (חֵלֶק א פֶּרֶק ב) שֶׁהִקְשָׁה לוֹ חָכָם אֶחָד אֵיךְ עַל יְדֵי הַחֵטְא נִתּוֹסְפָה לוֹ מַעֲלַת הַיְדִיעָה וְהַהַשְׂכָּלָה, כְּמוֹ שֶׁכָּתוּב (שָׁם ג. ז) וַתִּפָּקַחְנָה עֵינֵי שְׁנֵיהֶם, וְתֵירֵץ הוּא זִכְרוֹנוֹ לִבְרָכָה שֶׁיְּדִיעָה זוֹ אֵינָהּ מַעֲלָה כְּלָל דִּתְחִלָּה הָיְתָה יְדִיעָתוֹ בִּידִיעוֹת הַמּוּשְׂכָּלוֹת וְלֹא הָיָה לוֹ כֹּחַ לְהִשְׁתַּמֵּשׁ בִּמְפוּרְסָמוֹת (הַיְינוּ בְּנָאֶה וּמְגוּנֶּה) בְּשׁוּם פָּנִים וְאַחַר כַּךְ נֶחְסְרָה מֵאִתּוֹ יְדִיעַת הַמּוּשְׂכָּלוֹת וּבָא לִידֵי דַעַת הַמְפוּרְסָמוֹת הַיְינוּ הֶרְגֵּשׁ הַטֶּבַע בְּנָאֶה וּמְגוּנֶּה שֶׁזֶּה יְרִידָה גְדוֹלָה מִמַּעֲלָתוֹ.
[cup]וּבְזֹהַר[/cup] (חֵלֶק ג רסא. סוֹף עַמּוּד ב) אִיתָא וַתִּפָּקַחְנָה עֵינֵי שְׁנֵיהֶם בְּעִיפּוּשָׁא דְהַאי עָלְמָא מַה דְלָא הֲוָה קוֹדֶם דַּהֲווּ מַשְׁגְּחִין וּפַקְחִין לְעֵילָּא,
וְכֵן הַנְּבִיאִים בְּעֵת שֶׁהִגִּיעָה עֲלֵיהֶם הַנְּבוּאָה הָיוּ בִּבְחִינַת הַפְשָׁטַת הַגַּשְׁמִיּוּת שֶׁבָּאוּ לְמַעֲלַת וּמַדְרֵיגַת אָדָם הָרִאשׁוֹן כְּמוֹ שֶׁהָיָה קוֹדֶם הַחֵטְא, וְהַיְינוּ שֶׁנִּתְבַּטְּלוּ אֶצְלָם הַחוּשִׁים הַטִּבְעִיִּים וְלֹא הִרְגִּישׁוּ בְּעִנְיָנִים הַגַּשְׁמִים. וְזֶהוּ עִנְיַן הַפְשָׁטַת הַבְּגָדִים שֶׁהָיָה בִּנְבִיאִים בְּעֵת נְבוּאָתָם וּכְמוֹ שֶׁכָּתוּב בְּשָׁאוּל (שְׁמוּאֵל-א יט. כד) וַיַּפְשֵׁט גַּם הוּא אֶת בְּגָדָיו וַיִּתְנַבֵּא כו' וַיִּפּוֹל עָרוֹם כו', וּכְמוֹ אָדָם הָרִאשׁוֹן קוֹדֶם הַחֵטְא שֶׁלֹּא הִרְגִּישׁ כְּלָל צוֹרֶךְ בִּלְבוּשִׁים כְּמוֹ כֵן הַנְּבִיאִים בְּעֵת נְבוּאָתָם לֹא יָדְעוּ וְלֹא הִכִּירוּ צוֹרֶךְ לִלְבוּשִׁים כְּלָל מִפְּנֵי הֶעְדֵּר הֶרְגֵּשׁ הַטִּבְעִי (וְעִנְיַן הַפְשָׁטַת הַבְּגָדִים הוּא עִנְיַן הַפְשָׁטַת הַחוּשִׁים הַטִּבְעִיִּים),
וְזֶהוּ שֶׁקָּרְאוּ לַנָּבִיא בְּשֵׁם מְשׁוּגָּע מִפְּנֵי בִּיטּוּל הַחוּשִׁים הַטִּבְעִיִּים דְּעֵץ הַדַּעַת, וְעַל דֶּרֶךְ זֶה הוּא גַם כֵּן עִנְיַן אַהַנְיָא לֵיהּ שְׁטוּתֵיהּ לְסַבָּא, וְכֵן מַה שֶּׁאָמַר עֲקַבְיָא בֶּן מַהֲלַלְאֵל (עֵדֻיּוֹת פֶּרֶק ה מִשְׁנָה ו) מוּטָב לִי לְהִקָּרֵא שׁוֹטֶה כָּל יָמַי וְלֹא לֵיעָשׂוֹת שָׁעָה אַחַת רָשָׁע כו' שֶׁזֶּה עִנְיַן שְׁטוּת דִּקְדוּשָּׁה שֶׁהוּא הַהֵיפֶךְ מֵהַשְּׁטוּת דִּלְעוּמַּת זֶה, דְּהַשְּׁטוּת דִּלְעוּמַּת זֶה הוּא מַה שֶּׁעוֹשֶׂה הֵיפֶךְ הַדַּעַת וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל בְּכָל אוֹפַנֵּי הַשְּׁטוּתִים שֶׁמִּצַּד הַיֵּצֶר הָרָע שֶׁמַּשְׁטֶה וּמַטְעֶה אֶת הָאָדָם וּמַטֵּהוּ מִדֶּרֶךְ הָאֱמֶת, וְהַסִּיבָּה לָזֶה הוּא מִצַּד הַדַּעַת וְהַהֶרְגֵּשׁ דְּעֵץ הַדַּעַת טוֹב וָרָע שֶׁמְּכַסֶּה עַל הָאֱמֶת.

[kitzur_begin]הַהַשְׂכָּלָה הַבָּאָה בְּסִבַּת חֵטְא עֵץ הַדַּעַת הִיא יְרִידָה מֵהַמּוּשְׂכָּלוֹת הָאֱלֹקִית לְעִיפּוּשָׁא דְהַאי עָלְמָא, הַפְשָׁטַת הַבְּגָדִים בִּנְבִיאִים וְאַהַנְיָא לֵי[הּ] שְׁטוּתֵיהּ לְסַבָּא הוּא הַיְצִיאָה מֵהֶרְגֵּשׁ הַחוּשִׁים הַטִּבְעִיִּים, וְהִיא הָעֲבוֹדָה שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת שֶׁנִּקְרֵאת שְׁטוּת דִּקְדוּשָּׁה.[kitzur_end]
