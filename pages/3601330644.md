בפנימי' יותר מהראי' כו', וכאשר ההשגה היא רק בבחי' מציאות לבד הרי מ"מ הראי' שהיא במהות הדבר ה"ז למע' מהשגה כו', אך כאשר גם ההשגה תהי' במהות הדבר תהי' למע' מבחי' הראי' כו', וזהו עלי עין למע' מבחי' עין כו', וזה יהי' בתחה"מ דאין דומה לך כו', וכמ"ש ונחי' לפניו לפניו ממש שתהי' השגה בבחי' מהוע"צ כו', כי הנשמות יהיו בבחי' עצמי' וההשגה תהי' עצמי' כו', והוא שיתעלו הנשמות ויהיו בבחי' העליונה שבהממוצע דהיינו בחי' עצמות ח"ס כו' (ובשרשו הוא בחי' העלם העצמי כו', והיינו מה שהעצמו' נושא הכל בעצמו כו'), ותהי' ההשגה במהוע"צ א"ס כו', והיינו אחר שיתעלו בימוהמ"ש בבחי' פנימי' ועצמי' ח"ס שתהי' ההשגה בבחי' השגת המהות דבחי' פנימיות ח"ס, יתעלו בתחה"מ שתהי' ההשגה בבחי' מהוע"צ א"ס כו', וזהו אין דומה לך לך ממש כו'. וידוע דהשגה הוא בחי' יש וכמ"ש להנחיל אוהבי יש, הנה לעתיד תהי' בבחי' יש האמיתי דבחי' עצמות א"ס כו' (וי"ל דראי' היא בחי' אין אמיתי והשגה בחי' יש האמיתי כו') (וגם בחי' ראי' חושי' שנת"ל י"ל שתהי' בתחה"מ דוקא
כו'), וכ"ז הוא ע"י העבודה דעכשיו בבחינת ההעלם דמבשרי אחזה כו' בחי' השגה חושי' כנ"ל, עי"ז יהי' לעתיד בתחה"מ השגת המהות ממש כו'. וזהו שח"ס הוא עיקר הממוצע שעי"ז יכול להיות חבור הנאצלים והנשמות עם המאציל ב"ה שיכולים לבא לבחי' העליונה שבהממוצע, ועי"ז באים לבחי' העצמו' בבחי' השגת המהות כו'.
[cup]וע"פ[/cup] הנ"ל יובן מ"ש וה' אמר המכסה אני מאברהם, דהכוונה שיהי' התגלות בחי' הכיסוי וההעלם כו', וזהו אשר אני עושה, דהנה עשי' ע"ה שזה גילוי היינו רק התגלות הדבר, אבל בעצם ה"ז בבחי' העלם לגמרי שהרי בעשי' אין בזה שום גילוי והוא בחי' העלם עצמי כו', ושרש הדבר הוא מבחי' הסתימות העצמי דעצמי' ח"ס ובחי' העלם העצמי דא"ס כו' כנ"ל, ועז"א המכסה אני שיתגלה בחי' כיסוי הזה כו'. דהנה אברהם הוא בחי' גילוי הקו כנודע ממ"ש מי העיר ממזרח א"ת העיר אלא האיר כו', ופי' המכסה אני מאברהם היינו שבהקו יהי' גילוי בחי' ההעלם העצמי דא"ס כו', וזהו ג"כ ענין סוד ה' ליראיו, דפי' סוד ה' היינו בחי' העלם העצמי כו' (ועמ"ש בדרוש יו"ט של ר"ה רס"ו בענין ואמת ה' כו'), שזהו באמת סוד סתום וחתום בעצם כו', ואברהם שהוא בחי' הגילוי כנ"ל שהוא בחי' אור הקו כו' יתגלה בו בחי' סוד ה' הנ"ל וזהו בבחי' פנימי' הקו כו', דמ"ש מי העיר ממזרח זהו בחי' פנימי' הקו כו', וכנודע דחיצוני' הקו שרשו מבחי' מל' דא"ס ופנימי' הקו הוא מבחי' ת"ת הנעלם שז"ע
מזרח כו' וכמ"ש במ"א. וזהו ג"כ מש"א במד"ר האלקי' נתן את העולם לאברהם שניתן לו ההעלם כו', וא"ז מ"ש והארץ נתן לבנ"א דהיינו לברר בירורים מלמטה למע' כו', ומש"א האלקי' נתן את העולם היינו בחי' העלם העצמי כו', וע"י הבירורים דמלמטלמ"ע ג"כ מגיעים לבחי' העלם העצמי כו' כנ"ל, ובאברהם הי' זה ע"י הנסיונות כו', וכמשנת"ל ספרכ"ב בענין לך לך מארצך מבחי' החומרי' ובכח או"ח כו', ועי"ז המכסה אני שנתגלה לו בחי' העלם העצמי כו'. וז"ש אח"כ ואברהם היו יהי' לגוי גדול ועצום שע"י התגלות בחי' העלם העצמי תהי' ההתפשטות בבחי' ריבוי עד אין שיעור כו', וכידוע בענין אנת הוא חד ולא בחושבן כו', וכמשנת"ל (ד"ה שיר המעלות) שע"י העומק והפנימי' יותר באה ההתגלות ביותר כו', ומכ"ש שע"י התגלות העצמי יהי' הגילוי בריבוי עד אין קץ ושיעור כו',
