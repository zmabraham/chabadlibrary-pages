# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ד. > פֶּרֶק א

מַאֲמָר ד.
וְאַחַר כָּל זֶה יוּבַן דְּמַה שֶּׁהַיֵּצֶר הָרָע מְסִיתוֹ שֶׁגַּם אִם יַעֲשֶׂה מַעֲשֶׂה הָרָע אֵינוֹ נִפְרָד מֵאַחְדּוּתוֹ יִתְבָּרֵךְ, הוּא רַק רוּחַ שְׁטוּת הַמְכַסֶּה עַל הָאֱמֶת. וְזֶהוּ מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (סוֹטָה ג. א) עַל פָּסוּק כִּי תִשְׂטֶה אִשְׁתּוֹ אֵין אָדָם עוֹבֵר עֲבֵירָה אֶלָּא אִם כֵּן נִכְנַס בּוֹ רוּחַ שְׁטוּת, דַּאֲפִילוּ אִשָּׁה הַמְנָאֶפֶת שֶׁדַּעְתָּהּ קַלָּה הָיְתָה מוֹשֶׁלֶת בְּרוּחַ תַּאֲוָתָהּ לוּלֵי רוּחַ שְׁטוּת שֶׁבָּהּ הַמְכַסֶּה וּמַסְתִּיר וּמַעְלִים אֶת הָאַהֲבָה מְסוּתֶּרֶת שֶׁבְּנַפְשָׁהּ הָאֱלֹקִית לְדָבְקָה בֶּאֱמוּנַת ה' וְיִחוּדוֹ וְאַחְדּוּתוֹ וְלֹא לִיפָּרֵד חַס וְשָׁלוֹם מֵאַחְדּוּתוֹ אֲפִילוּ נוֹטְלִים אֶת נַפְשָׁהּ מִמֶּנָּה לַעֲבוֹד עֲבוֹדָה זָרָה חַס וְשָׁלוֹם (וּכְמוֹ שֶׁנִּתְבָּאֵר מַאֲמָר ב פֶּרֶק א שֶׁיֵּשׁ בְּטֶבַע בְּנִשְׁמוֹת יִשְׂרָאֵל שֶׁאֵינָם יְכוֹלִים לִהְיוֹת נִפְרָדִים מֵאֱלֹקוּת וְהַיְינוּ מִצַּד הָאַהֲבָה מְסוּתֶּרֶת שֶׁבָּהֶם, וְלָכֵן גַּם הַקַּל שֶׁבְּקַלִּים מוֹסֵר נַפְשׁוֹ עַל קִידּוּשׁ הַשֵּׁם כו' כַּנַ"ל) וַאֲפִילוּ בְּהִשְׁתַּחֲוָאָה לְבַדָּהּ בְּלִי שׁוּם אֱמוּנָה בְלֵב כְּלָל, וְכָל שֶׁכֵּן לִכְבּוֹשׁ הַיֵּצֶר וְתַאֲוַת הַנִּיאוּף שֶׁהֵם יִסּוּרִים קַלִּים מִמִּיתָה ה' יִשְׁמְרֵנוּ, וְהַהֶפְרֵשׁ שֶׁאֶצְלָהּ בֵּין אִיסּוּר נִיאוּף לְאִיסּוּר הִשְׁתַּחֲוָאָה לַעֲבוֹדָה זָרָה הוּא גַם כֵּן שְׁטוּת דִּקְלִיפָּה הַמַּלְבֶּשֶׁת לְנֶפֶשׁ הָאֱלֹקִית,
וְהַיְינוּ שֶׁנִּדְמֶה לָהּ דְּבָזֶה אֵינֶנָּה נַעֲשָׂה נִפְרָד מִיִּחוּדוֹ יִתְבָּרֵךְ וְעוֹדֶנָּה בְּיַהֲדוּתָהּ, אֲבָל בֶּאֱמֶת לַאֲמִתּוֹ אֲפִילוּ עֲבֵירָה קַלָּה הֲרֵי הָעוֹבְרָהּ עוֹבֵר עַל רָצוֹן הָעֶלְיוֹן בָּרוּךְ הוּא, וְהוּא בְּתַכְלִית הַפֵּירוּד מִיִּחוּדוֹ וְאַחְדּוּתוֹ יִתְבָּרֵךְ יוֹתֵר מִסִּטְרָא אָחֳרָא וּקְלִיפָּה הַנִּקְרֵאת אֱלֹקִים אֲחֵרִים וַעֲבוֹדָה זָרָה מַמָּשׁ, וְיוֹתֵר מִכָּל הַדְּבָרִים הַנִּשְׁפָּעִים מִמֶּנָּה בָּעוֹלָם הַזֶּה, שֶׁהֵם בְּהֵמוֹת טְמֵאוֹת וְחַיּוֹת וְעוֹפוֹת טְמֵאִים וּשְׁקָצִים וּרְמָשִׂים,
וּכְמַאֲמָר (סַנְהֶדְרִין לח. א) יַתּוּשׁ קְדָמָךְ, פֵּירוּשׁ דְּאַף יַתּוּשׁ שֶׁמַּכְנִיס וְאֵינוֹ מוֹצִיא שֶׁהִיא קְלִיפָּה הַיּוֹתֵר תַּחְתּוֹנָה וּרְחוֹקָה מִבְּחִינַת הַקְּדוּשָּׁה הַמַּשְׁפַּעַת בְּתַכְלִית הָרִיחוּק, קוֹדֶמֶת לְאִישׁ הַחוֹטֵא בְּהִשְׁתַּלְשְׁלוּת וִירִידַת הַחַיּוּת מֵרָצוֹן הָעֶלְיוֹן בָּרוּךְ הוּא (שֶׁהַחַיּוּת שֶׁל הָאִישׁ הַחוֹטֵא הוּא בִּבְחִינַת הֶסְתֵּר פָּנִים וּבִבְחִינַת רִיחוּק הַרְבֵּה יוֹתֵר מֵהַחַיּוּת הַנִּשְׁפָּע לְהַיַּתּוּשׁ) וְכָל שֶׁכֵּן שְׁאָר הַבַּעֲלֵי חַיִּים הַטְּמֵאִים וַאֲפִילוּ חַיּוֹת רָעוֹת, שֶׁכּוּלָּם אֵינָם מְשַׁנִּים תַּפְקִידָם וּפְקוּדָּתוֹ יִתְבָּרֵךְ שָׁמְרָה רוּחָם, וְאַף עַל גַּב דְּאִיהוּ לָא חָזֵי, כְּלוֹמַר בְּלִי הַרְגָּשָׁתָם, וּכְמוֹ שֶׁכָּתוּב (בְּרֵאשִׁית ט. ב) וּמוֹרַאֲכֶם וְחִתְּכֶם יִהְיֶה עַל כָּל חַיַּת הָאָרֶץ וגו' וְכִדְפֵירְשׁוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (שַׁבָּת קנא. ב) אֵין חַיָּה רָעָה שׁוֹלֶטֶת בָּאָדָם עַד שֶׁנִּדְמֶה לָהּ כַּבְּהֵמָה, וְהַצַּדִּיקִים שֶׁאֵין צֶלֶם אֱלֹקִים מִסְתַּלֵּק מֵעַל פְּנֵיהֶם כָּל חַיּוֹת רָעוֹת אִתְכַּפְיָין קַמַּיְיהוּ, כְּמוֹ שֶׁכָּתוּב בְּזֹהַר גַּבֵּי דָּנִיֵּאל בְּגוֹב אֲרָיוֹת,
וְאִם כֵּן הַחוֹטֵא וְעוֹבֵר רְצוֹנוֹ יִתְבָּרֵךְ אֲפִילוּ בַּעֲבֵירָה קַלָּה, בְּשַׁעַת מַעֲשֶׂה הוּא בְּתַכְלִית הָרִיחוּק מִקְּדוּשָּׁה הָעֶלְיוֹנָה שֶׁהִיא יִחוּדוֹ וְאַחְדּוּתוֹ יִתְבָּרֵךְ, יוֹתֵר מִכָּל בַּעֲלֵי חַיִּים הַטְּמֵאִים וּשְׁקָצִים וּרְמָשִׂים הַמּוּשְׁפָּעִים מִסִּטְרָא אָחֳרָא וּקְלִיפַּת עֲבוֹדָה זָרָה.
[cup]וּלְבַד[/cup] כָּל הַנַ"ל בִּבְחִינַת הַפֵּירוּד וְהָרִיחוּק מְאֹד שֶׁנַּעֲשֶׂה עַל יְדֵי מַעֲשֵׂה הָעֲבֵירָה רַחֲמָנָא לִצְּלָן, הִנֵּה עַל יְדֵי זֶה גַם כֵּן נִפְגָּם נַפְשׁוֹ וְנִפְסָק מֵאֱלֹקוּת וּכְמוֹ שֶׁכָּתוּב בְּאִגֶּרֶת הַתְּשׁוּבָה (פֶּרֶק ה) בְּפֵירוּשׁ יַעֲקֹב חֶבֶל נַחֲלָתוֹ, שֶׁהוּא חֶבֶל הַמְקַשֵּׁר הַנְּשָׁמָה לֵאלֹקוּת, וּכְמוֹ חֶבֶל שֶׁרֹאשׁוֹ אֶחָד קָשׁוּר לְמַעְלָה וְקָצֵהוּ לְמַטָּה, וְהוּא הַמְשָׁכַת הַנְּשָׁמָה מִשָּׁרְשָׁהּ מֵהֵ"א תַתָּאָה דְּשֵׁם הוי', דְּמִשָּׁם נִמְשֶׁכֶת וְיוֹרֶדֶת לָעוֹלָם הַזֶּה לְהִתְלַבֵּשׁ בְּגוּף הָאָדָם כו'.
[cup]וּכְמוֹ[/cup] עַל דֶּרֶךְ מָשָׁל מֵחֶבֶל עָב שָׁזוּר מִתַּרְיַ"ג חֲבָלִים דַּקִּים, כָּכָה חֶבֶל הַהַמְשָׁכָה הַנַ"ל כָּלוּל מִתַּרְיַ"ג מִצְוֹת וּכְשֶׁעוֹבֵר חַס וְשָׁלוֹם עַל אַחַת מֵהֵנָּה נִפְסָק חֶבֶל הַדַּק וּמִמֵּילָא נַעֲשָׂה חֲלִישׁוּת גַּם בִּכְלָלוּת הַחֶבֶל, וּבְחַיְּיבֵי כְרִיתוֹת רַחֲמָנָא לִצְּלָן נִפְסָק הַחֶבֶל לְגַמְרֵי וְנִכְרָת נַפְשׁוֹ רַחֲמָנָא לִצְּלָן, וְנִמְצָא מַה שֶּׁעַל יְדֵי הֲסָתָה הַנַ"ל נִכְשָׁל בְּמַעֲשֵׂה הַחֵטְא הוּא רַק מִצַּד הָרוּחַ שְׁטוּת שֶׁמְּכַסֶּה וּמַעֲבִירוֹ עַל דַּעְתּוֹ, אֲבָל אִם לֹא הָיָה נִפְתֶּה לְפִתּוּי הַיֵּצֶר, וְהָיָה רוֹאֶה אֶת הָאֱמֶת כְּמוֹ שֶׁהוּא, שֶׁעַל יְדֵי מַעֲשֵׂה הַחֵטְא יִפָּרֵד מֵאַחְדּוּתוֹ יִתְבָּרֵךְ וְיִתְרַחֵק בְּתַכְלִית הָרִיחוּק בְּעִמְקֵי שְׁאוֹל לְמַטָּה מַּטָּה תַּחַת טוּמְאַת הַסִּטְרָא אָחֳרָא וּקְלִיפַּת עֲבוֹדָה זָרָה, ה' יִשְׁמְרֵנוּ, וְיִפָּגֵם נַפְשׁוֹ רַחֲמָנָא לִצְּלָן כַּנַ"ל, הִנֵּה מִצַּד הָאַהֲבָה מְסוּתֶּרֶת שֶׁבּוֹ שֶׁאֵינוֹ יָכוֹל לִהְיוֹת נִפְרָד מֵאֱלֹקוּת לֹא הָיָה נִכְשָׁל כְּלָל בְּמַעֲשֵׂה הַחֵטְא בְּשׁוּם אוֹפֶן.

[kitzur_begin]הַחוֹטֵא גָּרוּעַ מִיַּתּוּשׁ גַּם מִבַּעֲלֵי חַיִּים הַטְּמֵאִים וְחַיּוֹת רָעוֹת שֶׁשּׁוֹמְרִים פְּקוּדָּתוֹ יִתְבָּרֵךְ גַּם נֶגֶד טִבְעָם, נוֹסָף לָזֶה שֶׁהוּא נִפְרָד מֵאַחְדּוּתוֹ יִתְבָּרֵךְ וּפוֹגֵם בְּנִימֵי נִשְׁמָתוֹ בִּפְסִיקַת חוּט הַחֶבֶל וּלְסִבַּת הָרוּחַ שְׁטוּת אֵין הָאָדָם מַרְגִּישׁ בָּזֶה.[kitzur_end]
