# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר כד. > פֶּרֶק ב

וְהִנֵּה מֵהַמְבוֹאָר בַּאֲרוּכָה מוּבָן הֵיטֵב דְּהָעִיקָּר הוּא הָאֱלֹקוּת וְהַמַּזָּלוֹת הֵם רַק לְבוּשִׁים וּשְׁלוּחֵי הַשֶּׁפַע לְבָד וְאֵין בָּהֶם מַמָּשׁ מִצַּד עַצְמָם, אֲבָל הַמִּשְׁתַּחֲוֶה וְעוֹבֵד אוֹתָם, הֲרֵי זֶה שֶׁמְּחַשֵּׁב אוֹתָם לְדָבָר עִיקָּרִי בְּהַהַשְׁפָּעָה שֶׁהֵן הֵמָּה הַמַּשְׁפִּיעִים מִצַּד עַצְמָם, הֲרֵי זוֹ כְּפִירָה רַחֲמָנָא לִצְּלָן בְּהַשְׁגָּחַת אֱלֹקוּת וּבְהַשְׁפָּעַת הַחַיּוּת מֵאִתּוֹ יִתְבָּרֵךְ, וּמְחַשֵּׁב אֶת הַמַּזָּלוֹת לְמַשְׁפִּיעִים דְּמִשּׁוּם זֶה מִשְׁתַּחֲוֶה לָהֶם הֲרֵי זוֹ עֲבוֹדָה זָרָה מַמָּשׁ,
וְכָל זֶה הוּא מִפְּנֵי שֶׁהַמַּזָּלוֹת הֵם בִּבְחִינַת לְבוּשׁ שַׂק, שֶׁמַּסְתִּירִים מְאֹד עַל הָאוֹר הָאֱלֹקִי שֶׁאֵינוֹ מֵאִיר בְּגִילּוּי כְּלָל וּמִתְלַבֵּשׁ עַל יְדֵי הַהַשְׁפָּעָה בִּלְבוּשֵׁי הַטֶּבַע, מִשּׁוּם זֶה יֵשׁ לָהֶם מָקוֹם לְדַמּוֹת שֶׁהַהַשְׁפָּעָה הִיא מֵהַמַּזָּלוֹת, שֶׁאִם לֹא הָיָה הָאוֹר הָאֱלֹקִי מוּסְתָּר וְנֶעְלָם כָּל כַּךְ, הָיוּ הַכֹּל יוֹדְעִים שֶׁהַהַשְׁפָּעָה מֵאֵת ה' הִיא, רַק מִפּנֵי רִיבּוּי הַהַעְלָמוֹת וְהַהֶסְתֵּרִים הַנַּעֲשִׂים עַל יְדֵי לְבוּשֵׁי הַמַּזָּלוֹת הִנֵּה מִשּׁוּם זֶה טָעוּ הַטּוֹעִים לַחְשׁוֹב גַּם אֶת הַלְּבוּשִׁים לִמְצִיאוּת דְּבַר מָה,
אָמְנָם בִּשְׁבִיל הַנְהָגַת הָעוֹלָם כְּפִי רְצוֹנוֹ יִתְבָּרֵךְ שֶׁיִּהְיֶה גַּשְׁמִי וְהַהִתְהַוּוּת וְהַחַיּוּת יִהְיֶה בְּדֶרֶךְ הֶעְלֵם וְהֶסְתֵּר הַבּוֹרֵא מֵהַנִּבְרָא, הָיָה צָרִיךְ לִהְיוֹת הִתְלַבְּשׁוּת הַנַ"ל בִּשְׁבִיל זֶה גּוּפָא שֶׁיִּתְלַבֵּשׁ הַחֶסֶד בִּלְבוּשֵׁי הַטֶּבַע וְיִהְיֶה לְפִי עֶרֶךְ הַנִּבְרָאִים שֶׁלְּמַטָּה, דְּאִם הָיָה הָאֱלֹקוּת מֵאִיר בְּגִילּוּי שֶׁלֹּא הָיָה מוּסְתָּר בִּלְבוּשִׁים הַנַ"ל, הָיְתָה הַהַשְׁפָּעָה לְמַעְלָה מֵהַטֶּבַע וְלֹא הָיוּ נִבְרָאִים שֶׁבְּטֶבַע יְכוֹלִים לְקַבֵּל הַהַשְׁפָּעָה דְּהַיְינוּ טוֹב כָּזֶה שֶׁאֵינוֹ לְפִי עֶרְכָּם, וּמִשּׁוּם זֶה צָרִיךְ לִהְיוֹת הַהִתְלַבְּשׁוּת בִּלְבוּשִׁים הַנַ"ל, רַק צְרִיכִים לֵידַע שֶׁזֶּהוּ הַחֶסֶד עֶלְיוֹן הַנִּקְצָב לָנוּ מֵאֵת ה' דַּוְקָא הוּא אֲשֶׁר נִתְלַבֵּשׁ בִּלְבוּשִׁים הַנַ"ל אֲשֶׁר זֶה עַצְמוֹ הוּא מֵחֶסֶד ה' עָלֵינוּ,
וּמִמּוֹצָא דָבָר אָנוּ יוֹדְעִים דְּכָל עִנְיַן הַהִתְלַבְּשׁוּת בְּהַמַּזָּלוֹת וְשָׂרִים הָעֶלְיוֹנִים שַׁיָּיךְ רַק בְּעוֹלָם הָעֲשִׂיָּה, כִּי הַהִתְלַבְּשׁוּת הִיא גַם כֵּן בְּסֵדֶר וְהַדְרָגָה דִּתְחִלָּה מִתְלַבֵּשׁ הָאוֹר בִּלְבוּשֵׁי עוֹלָם הַבְּרִיאָה וְנִתְבָּאֵר לְעֵיל (מַאֲמָר כג פֶּרֶק ב) דְּשָׁם עֲדַיִין הַחֶסֶד בִּבְחִינַת רוּחָנִי לְגַבֵּי לְמַטָּה, לְפִי שֶׁשָּׁם מֵאִיר עֲדַיִין הַגִּילּוּי אוֹר דַּאֲצִילוּת וּמִמֵּילָא אֵינוֹ בִּבְחִינַת מְצִיאוּת עֲדַיִין כַּנַ"ל, וּמִתְלַבֵּשׁ אַחַר כַּךְ בִּלְבוּשֵׁי הַיְצִירָה וַעֲשִׂיָּה, וּבְעוֹלָם הָעֲשִׂיָּה שֶׁהוּא הָעוֹלָם שֶׁלְּמַטָּה מִכּוּלָּם, שָׁם מוּסְתָּר הָאוֹר יוֹתֵר אָז הוּא בְּעֶרֶךְ שֶׁיִּתְלַבֵּשׁ בִּלְבוּשׁ שַׂק דְּהַמַּזָּלוֹת, דִּלְמַעְלָה מִזֶּה כְּמוֹ בְּעוֹלָם יְצִירָה שֶׁאֵין הוּא מוּסְתָּר כָּל כַּךְ אִי אֶפְשָׁר שֶׁיִּתְלַבֵּשׁ בִּלְבוּשִׁים אֵלּוּ, כִּי אִם כַּאֲשֶׁר יוֹרֵד הָאוֹר וּמִתְלַבֵּשׁ בִּלְבוּשֵׁי עֲשִׂיָּה שֶׁהוּא מוּסְתָּר יוֹתֵר אָז יָכוֹל לְהִתְלַבֵּשׁ גַּם כֵּן בִּלְבוּשִׁים אֵלּוּ.
[cup]וְזֶהוּ[/cup] דְּאִיתָא בִּפְרִי עֵץ חַיִּים דְּאֵל אד' שֶׁהוּא שֵׁם דַּעֲשִׂיָּה הוּא בְּגִימַטְרִיָּא צַו וְאֵין צַו אֶלָּא כו', לְפִי שֶׁכַּאֲשֶׁר נִתְלַבֵּשׁ הָאוֹר הָאֱלֹקִי בִּלְבוּשֵׁי עֲשִׂיָּה יָכוֹל לְהִתְלַבֵּשׁ בִּלְבוּשׁ שַׂק דְּמַזָּלוֹת עֶלְיוֹנִים שֶׁמַּעְלִימִים וּמַסְתִּירִים מְאֹד עַד שֶׁיָּכוֹל לְדַמּוֹת שֶׁהֵם הַמַּשְׁפִּיעִים מִכֹּחַ עַצְמָם מִפְּנֵי שֶׁאֵין הָאֱלֹקוּת נִרְאֶה וְנִגְלֶה בְּגִילּוּי כְּלָל כַּנַ"ל, וְלָכֵן אֵל אד' בְּגִימַטְרִיָּא צַו וְאֵין צַו אֶלָּא כו', דַּהֲגַם דְּאֵל אד' הֵם שֵׁמוֹת קְדוֹשִׁים אַךְ לְפִי שֶׁעַל יְדֵי זֶה בָּאָה הַהִתְלַבְּשׁוּת בְּהַמַּזָּלוֹת שֶׁהֵם מַסְתִּירִים לְגַמְרֵי עַד שֶׁנּוֹתֵן מָקוֹם לְחַשְּׁבָם וּלְדַמּוֹת שֶׁהֵם עַצְמָם הַמַּשְׁפִּיעִים וְלַעֲבוֹד אוֹתָם מִשּׁוּם זֶה, לָכֵן הַשֵּׁם שֶׁעַל יְדֵי זֶה בָּאָה הַהִתְלַבְּשׁוּת הַזֹּאת הוּא בְּגִימַטְרִיָּא צַו כו', אָמְנָם כְּבָר נִתְבָּאֵר לְעֵיל (פֶּרֶק א) שֶׁבֶּאֱמֶת הִתְלַבְּשׁוּת זוֹ הִיא מֵחַסְדֵי ה' עָלֵינוּ, לְפִי שֶׁעַל יְדֵי הִתְלַבְּשׁוּת זוֹ בָּא הַחֶסֶד הָעֶלְיוֹן בִּבְחִינַת טוֹב וְחֶסֶד גַּשְׁמִי. שֶׁבָּזֶה תָּלוּי כָּל עִנְיַן הָעֲבוֹדָה לְהַשְׁלִים הַכַּוָּונָה עֶלְיוֹנָה בְּעִנְיַן הַבְּרִיאָה.

[kitzur_begin]הַמַּזָּלוֹת הֵם רַק שְׁלוּחִים לְגַשֵּׁם הַשֶּׁפַע הַבָּא מֵאִתּוֹ יִתְבָּרֵךְ וְהַמִּשְׁתַּחֲוֶה לָהֶם עוֹבֵד עֲבוֹדָה זָרָה. בִּכְדֵי שֶׁיִּהְיֶה הָעוֹלָם וְהַנְהָגָתוֹ כְּפִי רְצוֹנוֹ יִתְבָּרֵךְ בְּגַשְׁמִיּוּת, הוּא רַק עַל יְדֵי הֶסְתֵּר וְהֶעְלֵם הָאוֹר דְּשֵׁם דַּעֲשִׂיָּה שֶׁהוּא בְּגִימַטְרִיָּא צַו, וְלִהְיוֹתוֹ הֶעְלֵם גָּמוּר עַד שֶׁטּוֹעִים לַחְשׁוֹב אֶת הַמַּזָּלוֹת לִמְצִיאוּת לָכֵן אֵין צַו אֶלָּא עֲבוֹדָה זָרָה.[kitzur_end]
