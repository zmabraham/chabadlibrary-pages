# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר יא. > פֶּרֶק ב

וְהִנֵּה טַעַם וְסִבַּת הַדָּבָר מַה שֶּׁעַל יְדֵי הַצְּעָקָה בְּקוֹל גָּדוֹל מְעוֹרְרִים אֶת בְּחִינַת פְּנִימִית וְעַצְמוּת אוֹר אֵין סוֹף בָּרוּךְ הוּא, דְּלִכְאוֹרָה אֵינוֹ מוּבָן תּוֹכֶן עִנְיָן זֶה, דַּהֲלֹא לְגַבֵּי הַקָּדוֹשׁ בָּרוּךְ הוּא הַכֹּל שָׁוִין וְאֵיזֶה הֶבְדֵּל אִם מִתְפַּלֵּל בְּלַחַשׁ אוֹ בְקוֹל וּמַהוּ הַיִּתְרוֹן דְּקוֹל גָּדוֹל הֲלֹא הָעִיקָּר הִיא הַתְּפִלָּה.
[cup]אַךְ[/cup] הָעִנְיָן הוּא דְּהִיא הַנּוֹתֶנֶת, שֶׁלִּהְיוֹת שֶׁהָעִיקָּר הִיא הַתְּפִלָּה וּתְפִלָּה עִנְיָנָהּ אִיחוּד וְהִתְקַשְּׁרוּת, וְהוּא שֶׁעַל יְדֵי הַתְּפִלָּה הִיא הִתְקַשְּׁרוּת הַנְּשָׁמָה בֵּאלֹקוּת, הִנֵּה מִטַּעַם זֶה עַצְמוֹ צְרִיכָה לִהְיוֹת תְּפִלָּתוֹ בְּקוֹל גָּדוֹל דְּזֶה מוֹרֶה עַל הִתְעוֹרְרוּת לְבָבוֹ בְּהִתְקַשְּׁרוּתוֹ לַה' בֶּאֱמֶת, דֶּאֱמֶת הוּא מַה שֶּׁמִּתְּחִלָּתוֹ וְעַד סוֹפוֹ הוּא בְּקַו אֶחָד יָשָׁר. דְּהִנֵּה כָּל הִתְגַּלּוּת הוּא רַק חֵלֶק יָדוּעַ מִמַּה שֶּׁבִּפְנִימִיּוּתוֹ הַנֶּעֱלָם, וְעִנְיַן הָאֱמֶת הוּא מַה שֶּׁהַפְּנִימִיּוּת הַנֶּעֱלֶמֶת מִתְגַּלָּה בְּכָל מַהוּתוֹ כְּמוֹ שֶׁהוּא בְּמַהוּתוֹ הָעַצְמִי, וְלִהְיוֹת שֶׁהַהִתְגַּלּוּת בָּאָה בְּהִתְרַגְּשׁוּת עוֹמֶק הַלֵּב וּפְנִימִיּוּתוֹ לָכֵן בָּאָה בְּקוֹל גָּדוֹל דַּוְקָא,
וְאִיתָא בְּזֹהַר דְּרוּחַ אַיְיתִי רוּחַ וְאַמְשִׁיךְ רוּחַ, דְּלִהְיוֹת שֶׁזֶּהוּ כָּל הַפְּנִימִיּוּת שֶׁלּוֹ לָכֵן הִנֵּה הוּא קוֹרֵא וּמַמְשִׁיךְ בְּחִינַת פְּנִימִית וְעַצְמוּת אוֹר אֵין סוֹף בָּרוּךְ הוּא, וְהַיְינוּ דְּלִהְיוֹת דְּנִשְׁמוֹת יִשְׂרָאֵל שָׁרְשָׁם בִּבְחִינַת הַפְּנִימִית דַּוְקָא, הִנֵּה מִצַּד זֶה יֵשׁ בְּכֹחָם לְגַלּוֹת עוֹמֶק פְּנִימִית נְקוּדַּת לְבָבָם שֶׁתִּהְיֶה הַהִתְגַּלּוּת כְּמוֹ שֶׁהִיא בִּפְנִימִית הַנֶּעֱלֶמֶת, וּמִשּׁוּם זֶה יְכוֹלִים לְעוֹרֵר אֶת פְּנִימִית וְעַצְמוּת אוֹר אֵין סוֹף בָּרוּךְ הוּא הַיְינוּ לְקַבֵּל הַהַשְׁפָּעָה מִפְּנִימִית הַמַּקִּיף, אֲבָל מִבְּחִינַת חִיצוֹנִיּוּת הַמַּקִּיף אֵינָם יְכוֹלִים לְקַבֵּל, מִפְּנֵי שֶׁאֵינָם שַׁיָּיכִים לִבְחִינַת חִיצוֹנִית,
וְדַוְקָא עַכּוּ"ם יְכוֹלִים לְקַבֵּל מִבְּחִינַת חִיצוֹנִיּוּת הַמַּקִּיף לְפִי שֶׁכָּל עִיקַּר יְנִיקָתָם הוּא מִבְּחִינַת אֲחוֹרַיִים וְחִיצוֹנִיּוּת דַּוְקָא, אֲבָל יִשְׂרָאֵל עִיקָּרָם וְשָׁרְשָׁם מִבְּחִינַת הַפְּנִימִית וּמִשָּׁם הֵם מְקַבְּלִים הַשְׁפָּעָתָם וְאֵינָם מְקַבְּלִים מִבְּחִינַת חִיצוֹנִיּוּת, וְלָכֵן כְּשֶׁחוֹטְאִים מְקַבְּלִים עוֹנְשָׁם בְּגָלוּת וְהֵן נִכְנָעִים כו' עַד שֶׁיָּשׁוּבוּ אֶל ה' וְיִצְעֲקוּ בִּצְעָקָה פְנִימִית מֵעוּמְקָא דְלִיבָּא, אָז מְעוֹרְרִים בְּחִינַת רַחֲמִים רַבִּים מִמְּקוֹר הָרַחֲמִים בְּחִינַת פְּנִימִית הַמַּקִּיף כו',
וְנִמְצָא מַה שֶּׁיִּתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר שָׁלוֹם יִהְיֶה לו כִּי בִּשְׁרִירוּת לִבּוֹ יֵלֵךְ, וְהַיְינוּ שֶׁיְּקַבֵּל מֵחִיצוֹנִיּוּת הַמַּקִּיף כְּמוֹ הַקְּלִיפּוֹת וְסִטְרָא אָחֳרָא שֶׁמַּגְבִּיהִים אֶת עַצְמָם לְקַבֵּל מִשָּׁם וּכְמוֹ עַכּוּ"ם לְמַטָּה, שֶׁנִּמְשָׁךְ רִיבּוּי הַשְׁפָּעָה מִשָּׁם בְּלִי חֶשְׁבּוֹן, כֵּן יְדַמֶּה בְנַפְשׁוֹ לִהְיוֹת כְּמוֹהֶם, הוּא שֶׁקֶר גָּמוּר וְאֵינוֹ אֶלָּא רַק מַה שֶּׁהוּא מַטְעֶה אֶת עַצְמוֹ, עֶר נַארְט זִיךְ אַלֵיין אָפּ ווָאס עֶר רֶעכֶענְט אַז עֶר ווֶעט אוֹיךְ מְקַבֵּל זַיְין דִי הַשְׁפָּעָה פוּן חִיצוֹנִיּוּת הַמַּקִּיף, כִּי לְיִשְׂרָאֵל אִי אֶפְשָׁר לְקַבֵּל הַשְׁפָּעָה מִבְּחִינַת חִיצוֹנִיּוּת הַמַּקִּיף כִּי אִם מִבְּחִינַת הַפְּנִימִית דְּסֵדֶר הִשְׁתַּלְשְׁלוּת (וּבְשָׁרְשׁוֹ מִבְּחִינַת פְּנִימִית הַמַּקִּיף כַּנַ"ל) וְשָׁם הַהַשְׁפָּעָה הִיא בִּבְחִינַת דִּין וְחֶשְׁבּוֹן, וּמִמֵּילָא בְּהֶכְרַח שֶׁיּוּפְסַק,
וְכֵן אָנוּ רוֹאִים בְּמוּחָשׁ אֲשֶׁר כַּמָּה רְשָׁעִים שֶׁמַּצְלִיחִים בַּתְּחִלָּה וְאַחַר כַּךְ נִפְסֶקֶת הַצְלָחָתָם וְעָשְׁרָם כִּי הַכֹּל לְפִי הַחֶשְׁבּוֹן, וּמַה שֶּׁמַּצְלִיחִים בַּתְּחִלָּה הוּא לְב' טְעָמִים הָא' שֶׁזֶּהוּ עַל יְדֵי אֵיזֶה עֲשִׂיָּה טוֹבָה מֵאוֹתָם שֶׁאָדָם אוֹכֵל פֵּירוֹתֵיהֶן בָּעוֹלָם הַזֶּה וְהַקֶּרֶן קַיֶּימֶת לָעוֹלָם הַבָּא, וְהַב' דְּהִנֵּה נִתְבָּאֵר לְעֵיל (מַאֲמָר ז פֶּרֶק ד) דְּכָל זְמַן שֶׁלֹּא נֶאֱבַד חֵלֶק הַטּוֹב שֶׁלּוֹ הֲרֵי הַקְּלִיפּוֹת יוֹנְקוֹת עַל יְדֵי תּוֹסֶפֶת חַיּוּת מִן הַקְּדוּשָּׁה וְהוּא נוֹטֵל חֵלֶק בָּרֹאשׁ כַּנַ"ל וּמִזֶּה נִמְשָׁךְ לוֹ רִיבּוּי הַשְׁפָּעָה בְּגַשְׁמִיּוּת, וְאַחַר כַּךְ נִפְסֶקֶת הַהַשְׁפָּעָה וְנַעֲשֶׂה עָנִי וּבַעַל יִסּוּרִים וּמְרוּחָק מִן הַבְּרִיּוֹת, כִּי רוֹב עֲשִׁירִים הֵם בַּעֲלֵי מִדּוֹת לֹא טוֹבוֹת אוֹ גַם רָעוֹת בְּאַהֲבַת עַצְמוֹ וְגַסּוּת הָרוּחַ, וְכָל זְמַן שֶׁהוּא בְעָשְׁרוֹ הִנֵּה רַבִּים אוֹהֲבֵי עָשִׁיר, וְאַף שֶׁהָרוֹב מֵהֶם יוֹדְעִים כִּי אֶת עָשְׁרָם יְכַבְּדוּ וְלֹא אוֹתָם, אֲבָל עִם זֶה הִנֵּה כָּכָה הֵם מוּטְעִים בְּמַהוּתָם עַד אֲשֶׁר יָעִיזוּ בְּהַוָּותָם, אֲבָל כַּאֲשֶׁר אוֹבְדִים אֶת עָשְׁרָם וְנַעֲשִׂים עֲנִיִּים אָז הֵם מַכִּירִים בְּמַהוּתָם הַמָּלֵא מִמּוּמֵי בְּנֵי אָדָם וְחֶסְרוֹנוֹתֵיהֶם,
לִהְיוֹת כִּי הַהַשְׁפָּעָה לְיִשְׂרָאֵל הוּא מִבְּחִינַת הַפְּנִימִית שֶׁבָּא בְּדִין וְחֶשְׁבּוֹן, אִם לֹא שֶׁהוּא מֵאֵלֶּה שֶׁנֶּאֱמַר עֲלֵיהֶם (דְּבָרִים ז. י) וּמְשַׁלֵּם לְשֹׂנְאָיו עַל  פָּנָיו לְהַאֲבִידוֹ כו', וְהַיְינוּ שֶׁאֵין לוֹ שׁוּם זְכוּת כְּלָל רַחֲמָנָא לִצְּלָן, אֲבָל אִם יֵשׁ לוֹ אֵיזֶה זְכוּת וּבִכְלַל יִשְׂרָאֵל יֵחָשֵׁב אִי אֶפְשָׁר לוֹ שֶׁיְּקַבֵּל הַשְׁפָּעָתוֹ בְּלִי חֶשְׁבּוֹן כִּי אִם הַכֹּל הוּא בְחֶשְׁבּוֹן דַּוְקָא מִצַּד בְּחִינַת הַפְּנִימִית דְּשָׁם הוּא שָׁרְשׁוֹ וּמְקוֹרוֹ.

[kitzur_begin]תְּפִלָּה מְקַשֶּׁרֶת הַנְּשָׁמָה בֵּאלֹקוּת, כְּשֶׁהַהִתְגַּלּוּת הִיא כְּמוֹ הַפְּנִימִית הַנֶּעֱלֶמֶת הוּא אֱמֶת, וּבָאָה בְּצַעֲקַת קוֹל גָּדוֹל וּמְעוֹרֶרֶת אֶת הָעַצְמוּת, יִשְׂרָאֵל מְקַבְּלִים רַק מִן הַפְּנִימִית וְגַם אֵלּוּ שֶׁדַּרְכָם צָלֵחָה שֶׁלֹּא עַל פִּי תוֹרָה וְהַטַּעַם א) שָׂכָר ב) שֶׁפַע הַקְּלִיפּוֹת, הִנֵּה לְבַסּוֹף נִפְסָק.[kitzur_end]
