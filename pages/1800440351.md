שהוצרך להיות בחי' צמצום האור בעצמות א"ס שיתכן לומר במח' הפשוטה אנא אמלוך כנ"ל (עיין מזה בד"ה הבאים ישרש דפ"א* סע"א ורע"ב, ועמ"ש בד"ה כי כארץ תוציא צמחה פ"ג, ובד"ה תקעו בחדש דרוש הראשון פ"א בענין פי' בריש הורמנותא דמלכא, אך לכאורה זה שיהי' היש דבר נפרד נמשך מבחי' כבוד תתאה שהוא הדבור משא"כ מצד כבוד עילאה שם הוא הביטול כנ"ל בענין עלמא דאתכסיא וצ"ל מ"מ המחשבה היא שורש להדבור שיהי' יש ודבר נפרד) וזוהי בחי' ירידה גדולה מעצמותו מאחר שבאמת אין עוד מלבדו ועל מי ימלוך, רק שע"י הצמצום יוכל להיות התהוות בחי' נפרד מן העצמות להיות נק' מלך כו' ולפ"ז צריך לעורר ולהמשיך מעצמות א"ס ב"ה שירד מעצמותו להיות עולה במח' ורצון הפשוט אנא אמלוך כו' ואז יהי' מקור להתהוות כל העולמות כו', (ע' בענין עושי דברו והדר לשמוע בד"ה בשעה שהקדימו ובד"ה תקעו בחדש דרוש הראשון גבי והנה הדרך להמשכת רצון כו' להיות גילוי מלכותו כו' ועמ"ש סד"ה אתם נצבים בענין למען הקים אותך לו לעם, ובד"ה תקעו השני רפ"ב, בד"ה יו"ט של ר"ה פ"ב, בד"ה והי' ביום ההוא יתקע פ"ב, בד"ה הנך יפה רעייתי פ"א). 
ג) ולזאת הנה אנו אומרים בכה"מ, פי' ברוך לשון המשכה שיומשך ויתברך בחי' כבוד ה' שהוא בחי' המלוכה הנ"ל כמ"ש לכבודי בראתיו כו' ממקומו פי' ממקור חוצבו בעצמות, והענין הוא כי מקור מקום כבוד ה' שהוא בחי' אנא אמלוך הנ"ל שעלה במח' ורצה"פ הנה הוא מיוחד וכלול כבר בהעלם בעצמות אור א"ס ממש רק שיצא מן ההעלם לגילוי במח' זו כו' והטעם הוא כי הרי גם בחי' המח' הזאת היא באה מן ההעלם לגילוי ג"כ שהרי עד"מ באדם הנה כח מחשבתו הוא כח היולי להיות בכחו לחשוב כל מה שיצטרך אבל אינה עדיין בבחי' גילוי מחשבה ממש שהרי היא כלולה עדיין בעצמות הנפש ונק' עצם כח שכלו ומחשבתו ומיוחדת בעצמות בתכלית רק כשצריך לחשוב איזה ענין הנה אז נפרדת מן העצמות ובא לידי גילוי מן ההעלם העצמי כידוע, וככה יובן למעלה בעצמות א"ס ב"ה שבחי' כח מחשבתו כלולה ומיוחדת בעצמותו ממש כי הוא המדע כו' ונק' עצם כח חכמתו ומחשבתו כו' רק שבאה לידי גילוי מח' ורצון פשוט מבחי' העלם עצמות כח מחשבתו המיוחדת בו, והוא ענין הנ"ל שכאשר עלה ברצונו ומחשבתו הפשוטה אנא אמלוך וכיוצא, ונמצא הרי יש מקור מחצב בהעלם עצמות כח מחשבתו ית' לגילוי מחשבה זו דאנא אמלוך ולזה אומרים בכה"מ ממקור מקומה של בחי' גילוי מחשבה זו מן ההעלם העצמות (וזהו ג"כ פי' לכבודי בראתיו כדי שיבוא לגילוי מח' שנק' בריאה יש מאין הוא לכבודי מפני מקור מלוכה זו שנק' כבוד כמו שהוא בעצמות שנק' מל' דא"ס ובל' חז"ל הי' הוא ושמו בלבד כידוע וגם זה נק' בשם מח' בלשון הזהר אבל נק' מח' אחת כלומר כללית והוא בחי' מקיף כללי כידוע, משא"כ מח' זו הבאה בגילוי אנא אמלוך שזה רק פרטי לפי פרטי ההשתל' דאבי"ע כו' וד"ל). ובזה יובן מה שאנו אומרים בב"ש יחיד חי העולמים מלך ובישתבח מסיימין הברכה מלך יחיד חי העולמים. כי הנה בחי' ירידת מח' הזאת דאנא אמלוך מעצמות כח המחשבה הכלולה ומיוחדת בתכלית בעצמות א"ס כנ"ל הוצרך להיות בחי' ריבוי צמצומים מאד שמצמצם א"ס א"ע בכמה מיני צמצומים וזהו רק מפני רב טובו וחסדו כדי להאיר אורו למטה
