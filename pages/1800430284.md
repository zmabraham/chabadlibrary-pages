שארז"ל גדול תלמוד שמביא לידי מעשה שהוא בחי' קעומ"ש בכלל כנ"ל וד"ל. והגם שידוע בכ"מ שכל עיקר התו' הוא בבחי' המשכות אלקו' מלמע' למטה, אך להיות שהתלבשות החיות דאוא"ס בעולמו' הוא ע"י התו' כמ"ש בד"ה ביום השמע"צ כו' והמשכה זו בא בבחי' רו"ש כנ"ל, וא"כ גם בה יש ב' בחי' הללו דרו"ש והוא בחי' עיון התו' (כמו שפרש"י במס' סוכה לעמוד על הדבר להבינו על בורי') דהיינו בהתקשרות שכלו לעמוד על המכוון הפנימי דתו' דמחכ' נפקת כנ"ל שהוא בחי' ביטול דכח מ"ה כנ"ל והוא בבחי' רצוא והעלאה הנ"ל והב' בחי' אותיו' התו' הוא מה שבא בבחי' המשכה מלמע' למטה כו' (וכמשי"ת), ולכן ע"י שבתו' זו יש בחי' זו דרצוא לכן מזה נמשך ג"כ הכח באדם שיהי' בבחי' רצוא (ובד"ה וכל העם רואים כו' נתבאר שבחי' לפיד אש זהו עיקר התו' שז"ש רז"ל שניתנה באש שחורה ע"ג אש לבנה, וכן מ"ש הלא כה דברי כאש כו' ובחי' המצות שבתו' זהו בחי' מים בחי' שוב הנ"ל וכמשי"ת) שזהו ענין המס"נ בנשה"א ונה"ב (והנה בתו' יש ג"כ תרי ממאה תושב"כ ותושבע"פ כמשי"ת, ויש לומר עי"ז יש הכח להיות הרצוא בנשה"א ונה"ב וכמ"ש בתו"א בד"ה להבין ענין הברכות כו' בענין תושב"כ ותושבע"פ שזהו ענין יום ליום יביע אומר ולילה ללילה יחוה דעת כו' ובאגה"ק ד"ה והמשכילים כו' שנק' ג"כ עה"ח ועה"ד כי התושבע"פ גורמת בחי' הבירור דעה"ד טו"ר שזהו בחי' הנה"ב כנ"ל וד"ל). והנה כ"ז הוא פרשה ראשונה דק"ש שמע ישראל כו' בחי' עומ"ש בכלל שהכל הוא בבחי' ההעלאה למע' בבחי' הביטול והתכללות באחד ע"י קעומ"ש כו' שנק' בחי' הרצוא, וכל ענין פרשה זו גם במצות ת"ת ודברת בם הוא בבחי' הרצוא כנ"ל, אבל פ' והא"ש כו' הוא בחי' עומ"צ שהוא בבחי' השוב וההמשכה מלמע' למטה שזהו כל עיקר ענין המצות להיות גילוי אוא"ס ב"ה בהתלבשות בענינים גשמיים כמו בל"ב חוטים דציצית של צמר ובקלף גשמי דתפילין וסכך דסוכה וכה"ג, ושרש הענין הוא לפי שידוע בזהר דרמ"ח פקודין אינון רמ"ח אברי' דמל' (וכמ"ש בד"ה משה ידבר ובלקו"ת פ' במדבר בד"ה להבין ענין המצות ובד"ה וידעת היום ובד"ה כי תשמע בקול), וכמו אור הנפש שמתלבש באברי גוף האדם שראשית גלוי אור הנפש הוא במוח ומשם מתפשטת לכל האברים ע"י גידי המוח הנמשכים בכל אבר ואבר שהמה גידי ההרגשה שתהי' אור הנפש שבאה בהרגשת החיות בכל האברים (וידוע דבחי' פנימי' החיות הנפש מאיר במוח ולב כו' וכמ"ש בבה"ז פ' תולדות בד"ה מי ימלל גבורות כו' ורק בחי' חיצוניות החיות נמשך מן המוח והלב אל כל האברים דרך גידים ועורקים דקים הנק' מיתרים ועצבים כידוע וע"י אמצעות הגידים הללו יש קישור וחיבור לאברים עם עצם פנימי' חיות הנפש שבמוח כו' לפעול כל רצונה ופעולתה ע"י האברים, ונמצא גם שהאברי' אינם מקבלי' רק בחי' חיצוני' אור וחיות מן הנפש מ"מ פועלת הנפש כל בחי' פנימי' רצונה ע"י פעולת האברי', וככה יובן למעלה בבחי' רמ"ח מ"ע שנק' רמ"ח אברי' דמלכא כבי' היינו שהם בבחי' כלים חיצוני' מלובשים בעניני' גשמי' אבל בחי' פנימי' אוא"ס מתלבש בהן ע"י החכ' שבראש שהוא התורה) לפעול בגלוי כל בחי' רצה"ע של המאציל ב"ה, והיינו בכל מצוה שמלובש בה רצה"ע בבחי' הלבשה חיצוני', ויש לה חיבור וקשר עם בחי' פנימי' אוא"ס (שבחכ' דתו') וכנ"ל במשל האברים למטה, וזהו שעיקר המצות הוא בחי'
