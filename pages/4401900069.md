# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר טו. > פֶּרֶק ג

וְזֶהוּ מַעֲלַת עוֹלָם הַזֶּה הַגַּשְׁמִי לְגַבֵּי עוֹלָמוֹת הָעֶלְיוֹנִים הָרוּחָנִים, וּמַעֲלַת יְרִידַת הַנְּשָׁמָה בְּהִתְלַבְּשׁוּתָהּ בְּגוּף גַּשְׁמִי וְנֶפֶשׁ הַטִּבְעִי לְגַבֵּי מַעֲלַת עֲמִידָתָהּ בְּגַן עֵדֶן בְּתַכְלִית הַיִּחוּד וְהַבִּיטּוּל לֵאלֹקוּת, וּכְמוֹ שֶׁכָּתַב רַבֵּינוּ נִשְׁמָתוֹ עֵדֶן (סֵפֶר שֶׁל בֵּינוֹנִים פֶּרֶק ל"ו) שֶׁתַּכְלִית בְּרִיאַת עוֹלָם הַזֶּה הוּא שֶׁנִּתְאַוָּה הַקָּדוֹשׁ בָּרוּךְ הוּא לִהְיוֹת לוֹ דִּירָה בַּתַּחְתּוֹנִים, דְּזֶהוּ עִנְיַן הִשְׁתַּלְשְׁלוּת הָעוֹלָמוֹת וִירִידָתָם מִמַּדְרֵיגָה לְמַדְרֵיגָה עַל יְדֵי רִיבּוּי לְבוּשִׁים הַמַּסְתִּירִים הָאוֹר וְהַחַיּוּת שֶׁמִּמֶּנּוּ יִתְבָּרֵךְ, עַד שֶׁנִּבְרָא עוֹלָם הַזֶּה הַגַּשְׁמִי וְהַחוּמְרִי מַמָּשׁ, וְהוּא הַתַּחְתּוֹן בְּמַדְרֵיגָה שֶׁאֵין תַּחְתּוֹן לְמַטָּה מִמֶּנּוּ בְּעִנְיַן הֶסְתֵּר אוֹרוֹ יִתְבָּרֵךְ וְחֹשֶׁךְ כָּפוּל וּמְכוּפָּל, עַד שֶׁהוּא מָלֵא קְלִיפּוֹת וְסִטְרָא אָחֳרָא שֶׁהֵן נֶגֶד ה' מַמָּשׁ לוֹמַר אֲנִי וְאַפְסִי עוֹד, וְהַיְינוּ דְּתַכְלִית כָּל הַהִשְׁתַּלְשְׁלוּת שֶׁל הָעוֹלָמוֹת הָרוּחָנִים הוּא בִּשְׁבִיל עוֹלָם הַזֶּה הַתַּחְתּוֹן, לְפִי שֶׁכַּךְ עָלָה בִרְצוֹנוֹ יִתְבָּרֵךְ לִהְיוֹת נַחַת רוּחַ לְפָנָיו יִתְבָּרֵךְ כַּד אִתְכַּפְיָא סִטְרָא אָחֳרָא וְאִתְהַפְּכָא חֲשׁוֹכָא לִנְהוֹרָא, וּבִשְׁבִיל זֶה נִיתַּן לְאָדָם הַכֹּחַ וְהָעוֹז שֶׁיּוּכַל לְהִתְגַּבֵּר גַּם עַל הַנֶּפֶשׁ הַבַּהֲמִית,
וּכְלָלוּת עֲבוֹדָה זוֹ דְּבֵירוּר וְזִיכּוּךְ הַגּוּף וְנֶפֶשׁ הַבַּהֲמִית הוּא בְּדֶרֶךְ מִלְחָמָה, וְהָאָדָם צָרִיךְ לָדַעַת כִּי הַנֶּפֶשׁ הַבַּהֲמִית וְהַיֵּצֶר הָרָע הֵם שׂוֹנְאָיו הַיּוֹתֵר גְּדוֹלִים וַחֲפֵצִים בְּמִיתָתוֹ הָרוּחָנִית, וּבָאִים עָלָיו בְּעַרְמוּמִיּוּת לְפַתּוֹתוֹ בְּחֶלְקַת לְשׁוֹנָם וּבְחָכְמָה גְדוֹלָה מִתְנַכְּלִים אֵלָיו לְהַדִּיחוֹ מִדֶּרֶךְ הַחַיִּים הָאֲמִיתִּים בִּשְׁמִירַת הַתּוֹרָה וְהַמִּצְוֹת עַל יְדֵי כַּמָּה עִנְיְנֵי הֶיתֵּר לְמַלֹּאות תַּאֲוַת נַפְשׁוֹ, וּלְאַט לְאַט מַדִּיחִים אוֹתוֹ מִן הַהֶיתֵּר גָּמוּר אֶל הַהֶיתֵּר שֶׁל בְּדִיעֲבַד, עַד שֶׁמַּרְגִּילִים אוֹתוֹ בְּכַמָּה מִינֵי תַאֲווֹת, וּמְפַתְּחִים וּמַרְגִּילִים בּוֹ אֶת כֹּחַ הַמִּתְאַוֶּה מִבְּלִי שֶׁיּוּכַל לְהַעֲמִיד עַל עַצְמוֹ מִבְּלִי לְהַבִּיט אוֹ גַם לְהִסְתַּכֵּל בְּמַה שֶּׁאֵין צְרִיכִים לִרְאוֹת, וְהָעַיִן רוֹאָה וְהַלֵּב חוֹמֵד וְנֶחְלְשׁוּ כֹּחוֹתָיו הָרוּחָנִים לַעֲצוֹר בְּעַד הֲמוֹן רְצוֹנוֹתָיו הַגּוּפְנִים וְנוֹפֵל רַחֲמָנָא לִצְּלָן חָלָל בִּידֵי הַיֵּצֶר הָרָע וְהַנֶּפֶשׁ הַבַּהֲמִית.
[cup]וַאֲשֶׁר[/cup] עַל כֵּן הִנֵּה כְּלָלוּת הָעֲבוֹדָה בְּבֵירוּר וְזִיכּוּךְ הַגּוּף צְרִיכָה לִהְיוֹת בְּדֶרֶךְ מִלְחָמָה לְנַגֵּד לָהֶם, אֲשֶׁר בְּכָל דָּבָר וְעִנְיָן אֲשֶׁר הַיֵּצֶר הָרָע אוֹמֵר הֵן יֹאמַר הוּא לַאו לְעַצְמוֹ, וּבְכָל דָּבָר וְעִנְיָן אֲשֶׁר הַיֵּצֶר הָרָע אוֹמֵר לַאו יֹאמַר הוּא לְעַצְמוֹ הֵן, אֲשֶׁר בְּהַהִתְנַגְּדוּת הַכְּלָלִית הַלֵּזּוּ הוּא מוֹדִיעַ לְהַיֵּצֶר הָרָע שֶׁהוּא שׂוֹנְאוֹ וַאֲשֶׁר יִלְחוֹם עִמָּדוֹ,
וְאַחַת מִטַּכְסִיסֵי הַמִּלְחָמָה הוּא הַשִּׂימַת לֵב עַל עִנְיָנִים קַלֵּי הָעֶרֶךְ, כִּי הַנִּצָחוֹן אֵינוֹ תָּלוּי בְּהַגְּבוּרָה דְּרִיבּוּי אֲנָשִׁים וְרִיבּוּי כְּלֵי זַיֵּין בִּלְבָד, אֶלָּא הָעִיקָּר הוּא הַהַמְצָאוֹת שִׂכְלִיּוֹת אֵיךְ לְהַטְעוֹת אֶת הַמְנַגֵּד וּלְהוֹלִיכוֹ שׁוֹלָל, וְלִפְעָמִים הִנֵּה בְּהַמְצָאָה שִׂכְלִית שֶׁל עַרְמוּמִיּוּת מְנַצְּחִים הַרְבֵּה יוֹתֵר מִבִּגְבוּרָה שֶׁל רִיבּוּי אֲנָשִׁים וְרִיבּוּי כְּלֵי זַיֵּין, וְכֵן הוּא בְּמִלְחָמָה הָרוּחָנִית, דְּהַתַּאֲווֹת וַעֲבֵירוֹת הֵם הַכְּלֵי זַיֵּין וְהַגְּבוּרָה שֶׁל הַיֵּצֶר הָרָע, וּשְׁטוּתֵי הַהִצְטַדְּקוּת מַה שֶּׁהָאָדָם מַצְדִּיק אֶת עַצְמוֹ הֵן הֵמָּה הַהַמְצָאָה שִׂכְלִית שֶׁל הַיֵּצֶר הָרָע וְהַנֶּפֶשׁ הַבַּהֲמִית בְּעַרְמוּמִיּוּת גְּדוֹלָה, לִלְכּוֹד חַס וְשָׁלוֹם בְּכֹחַ גַּם אֵלּוּ אֲשֶׁר בְּעִנְיְנֵי תַאֲוָה וַעֲבֵירָה אִי אֶפְשָׁר לְהַחֲטִיאָם,
וַאֲשֶׁר עַל כֵּן הִנֵּה גַּם מִי שֶׁהוּא צַדִּיק גָּמוּר יָכוֹל לִמְצוֹא בְעַצְמוֹ שְׁטוּתִים כָּאֵלּוּ שֶׁעִם הֱיוֹת שֶׁהֵם דַּקִּים וְלֹא הֵבִיאוּ אוֹתוֹ לַעֲבוֹר עֲבֵירָה בְּפוֹעַל מַמָּשׁ חַס וְשָׁלוֹם, עִם כָּל זֶה הִנֵּה חֶסְרוֹנוֹת הַדַּקִּים גַּם כֵּן עוֹמְדִים בְּרוּמוֹ שֶׁל עוֹלָם וְגוֹרְמִים דְּבָרִים רָעִים מְאֹד, עַל דֶּרֶךְ מָשָׁל מִי שֶׁלּוֹמֵד תּוֹרָה הַרְבֵּה, הִנֵּה גַּם אִם לוֹמֵד לִשְׁמָהּ וּמִתְפַּלֵּל בְּכַוָּונַת הַלֵּב בְּטוּב טַעַם וָדַעַת וּבִדְבֵיקוּת נִפְלָאָה עַד שֶׁיָּבֹא לִבְחִינַת צִמָּאוֹן גָּדוֹל כָּל כַּךְ עַד שֶׁתִּגְעַל הַנֶּפֶשׁ אֶת הַבָּשָׂר, הִנֵּה כְּשֶׁמַּחֲזִיק טוֹבָה לְעַצְמוֹ וְכָל שֶׁכֵּן כְּשֶׁבָּא לִכְלַל גֵּיאוּת הִנֵּה גֵּיאוּת אָדָם תַּשְׁפִּילֶנּוּ כָּל כַּךְ עַד שֶׁנֶּאֱבָד כָּל הַטּוֹב שֶׁעָשָׂה.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]הָאָדָם צָרִיךְ לָדַעַת כִּי הַנֶּפֶשׁ הַבַּהֲמִית וְהַיֵּצֶר הָרָע הֵם שׂוֹנְאָיו בְּנֶפֶשׁ, וְעוֹלָם הַזֶּה הַגַּשְׁמִי הוּא תַּכְלִית הַהִשְׁתַּלְשְׁלוּת, וַעֲבוֹדָתוֹ לַהֲפוֹךְ הַחֹשֶׁךְ לָאוֹר צְרִיכָה לִהְיוֹת בְּדֶרֶךְ מִלְחָמָה בְּשִׂימַת לֵב גַּם עַל דְּבָרִים קַלֵּי הָעֶרֶךְ, עֲבֵירָה תַּאֲוָה, וּשְׁטוּתֵי הַהִצְטַדְּקוּת קָשֶׁה מֵהֶם, כְּמוֹ הַגַּאֲוָה שֶׁמְּאַבֵּד גַּם הַמְעוּלֶּה בְּתוֹרָה וּמִצְוֹת.[kitzur_end]
