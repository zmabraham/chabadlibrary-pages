שהתו' נק' עוז ותושי' שמתשת כח הנה"ב ומגברת כח הנה"א כו', וזה אם בהתיקון שע"י התורה עצמה או מה שהתורה מסייעה בעבודה דתפלה כו' וכמשנת"ל פקע"ו, והוא שיש חילוקי מדרי' בנשמות יש נשמות דעיקר עבודתן הוא בתפלה, דבתפלה הוא התגלות הכחות דנה"א ומתישים את כח הנה"ב כו', רק שזהו ע"י סיוע התורה שקודם התפלה כו', ויש נשמות שעיקר עבודתן הוא עסק התורה שע"י התורה עצמה נעשה התגברות הנה"א והתשת כח הנה"ב כו' (ולמוד התורה לשמה זהו שכר הנשמה כו', וגם בזה יש בירור ותיקון בכללות העולם וכמא' ר"י כולהו תנויי בנזיקין כו', וגם שהכוונה האמיתית היא ליחדא קוב"ה ושכינתא בתחתונים כו', וכמ"ש בסש"ב ספ"י ולא בכדי לדבקה בו ית' לבד לרוות צמאון נפשם כו' אלא כדפי' בתי' איזהו חסיד כו'), וכן בקיום המצות הוא בכדי לקשר נה"ט לאלקות כו', וזה במדרי' הבינונים דבצדיקים ענין קיום המצות הוא לאתקנא רזא דשמי' דהיינו להוסיף אורות באצי' כו' וכמ"ש במ"א, אבל עבודת הבינונים בקיום המצות הוא לקשר את הנה"ב כו', וכנודע דמצות יש להם סגולה פרטי' לקשר
את הנה"ט כו' כמ"ש בתו"א ד"ה מי כה' אלקינו המגביהי כו', וגם יש עבודה פרטיות בכל מצוה כמו תפלין לשעבד הלב והמוח כו' וכמ"ש בסש"ב פמ"א, וכמו אתרוג אל תבאני רגל גאוה כו' וכה"ג בכל מצוה כו' (ושכר עבודת המצות הו"ע מצותי, וכידוע שיש מצות ה' ומצותי כו' או שז"ע עבדי הם וכמ"ש כי לי בנ"י עבדים עבדי הם דענין עבדי הוא בחי' יר"ע וכמו משה עבדי כו', דמ"ש כי לי בנ"י עבדים הוא בחי' יר"ת ועבדי זהו בחי' יר"ע כו' וזהו שכר עבודתה כו'), והן בעבודה בהתבוננות אלקי יהי' בבחי' שייכות אל הגוף ונה"ט כו', והיינו שלא יתפעל בנפשו ביותר בהרגש הענין האלקי שבההתבוננות שיבוא לידי התפעלות יתירה ובבחי' הפשטה מהכחות הטבעים, כ"א אדרבא להשקיט התפעלות הנפש ולעצור בעד כחות הנפש שלא יתפעלו כ"כ ויהי' בבחי' הרגש פנימי יותר, והיינו שיהי' בבחי' התיישבות יותר שאז ההתבוננות הוא בבחי' הנחה טובה ובבחי' הרגש פנימי ונרגש גם בכחות הטבעים כו', ובכלל זה ג"כ שתהי' ההתבוננות בענינים השייכים לנה"ט, והיינו בההארה אלקית שבאה בבחי' התלבשות בעולמות
כו', ושתהי' ההתבוננות בבחי' השגה ממש בלבושי השגה דוקא בכדי שיהי' השגת הענין גם בהשכל הטבעי כו' וכמשנת"ל פ' רצ"ח (וכאשר ההתבוננות הוא באופן כזה זה גופא מעציר ומשקיט התפעלות הנפש מצ"ע כו', כי התפעלות הנפש עצמו הוא יותר בענינים נעלים באלקות וכאשר אינם בלבושי השגה כו', אך באופן הנ"ל בפ' ש"א היינו כאשר הנשמה אין לה שייכות אל הגוף הנה בד"כ אין זה בהשגה גמורה (לבד בנשמות גבוהות שבירידתם למטה ה"ה כמו שהן למעלה ממש כו', וכמו הנשמות שנבקע עליהם הפרגוד ובאו למטה כמו שהן למעלה כו') ומתפעל מהרגש כללות הענין האלקי כו', ואמיתת עבודה הנ"ל היא בשכר הנשמות להתענג על הוי' כו' כנ"ל). וכן בענין התשובה הוא שירגיש את הלא טוב מריחוקו מאלקות ויהי' בהרגש פנימי ויהי' צר לו מאד מזה כו', ואם כי זה עושה כליון בנפשו בבחי' יציאה מגדר כלי, אין זה שזהו הנוחם לנפשו כ"א זהו בבחי' תוקף הרצוא והצמאון לאלקות כו', או בדרך בריחה שבורח מהרע ונמשך ביותר (מיט א שטארקען ציא) לאלקות כו', אבל משו"ז דוקא נעשה הסכם חזק בנפשו בקיום
