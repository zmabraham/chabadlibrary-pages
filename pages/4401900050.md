מַאֲמָר ח.
וְהִנֵּה כָּל מַה שֶּׁנִּתְבָּאֵר בְּעִנְיַן לְמַעַן סְפוֹת הָרָוָה אֶת הַצְּמֵאָה, שֶׁעַל יְדֵי זֶה שֶׁסְּפִירַת הַמַּלְכוּת רַגְלֶיהָ יוֹרְדוֹת בִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה הִנֵּה מְחַבֵּר רָוָה שֶׁהִיא הַסִּטְרָא אָחֳרָא עִם הַצְּמֵאָה בְּחִינַת מַלְכוּת לְקַבֵּל תּוֹסֶפֶת חַיּוּת מֵהַקְּדוּשָּׁה, זֶהוּ לְפֵירוּשׁ רַשִׁ"י שֶׁמְּפָרֵשׁ סְפוֹת לְשׁוֹן חִבּוּר.
[cup]אָמְנָם[/cup] לְפִי פֵּירוּשׁ הָרְדַ"ק שֶׁמְּפָרֵשׁ לְשׁוֹן רִיבּוּי מַהוּ עִנְיַן שֶׁהָרָוָה מַרְבֶּה עוֹד עַל בְּחִינַת צְמֵאָה.
[cup]הָעִנְיָן[/cup] הוּא דְּהִנֵּה יָדוּעַ שֶׁיְּנִיקַת הַחִיצוֹנִים הִיא עַל ב' אוֹפַנִּים, אוֹ שֶׁמְּקַבְּלִים מִבְּחִינַת אֲחוֹרַיִים וְחִיצוֹנִיּוּת, דְּהַיְינוּ מִבְּחִינַת רִיבּוּי הַצִּמְצוּמִים דִּבְחִינַת חִיצוֹנִיּוּת הַכֵּלִים דְּמַלְכוּת. וְאוֹפֶן הַב' הוּא שֶׁמְּקַבְּלִים מִבְּחִינַת הַמַּקִּיף עֶלְיוֹן (וְהוּא גַם כֵּן מִבְּחִינַת אֲחוֹרַיִים וְחִיצוֹנִיּוּת הַמַּקִּיף).
[cup]וְהָעִנְיָן[/cup] הוּא דְּהִנֵּה אָמְרוּ (אָבוֹת פֶּרֶק ה) עֲשָׂרָה דוֹרוֹת מֵאָדָם עַד נֹחַ וּמִנֹּחַ עַד אַבְרָהָם לְהוֹדִיעַ כַּמָּה אֶרֶךְ אַפַּיִם לְפָנָיו שֶׁכָּל הַדּוֹרוֹת הָיוּ מַכְעִיסִין וּבָאִין כו', הַיְינוּ שֶׁכָּל הַדּוֹרוֹת הָאֵלּוּ קִבְּלוּ מִבְּחִינַת אֶרֶךְ אַפַּיִם, וּכְמוֹ שֶׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (עֵירוּבִין כב. א) מַאי דִּכְתִיב אֶרֶךְ אַפַּיִם, אֶרֶךְ אַף מִבָּעֵי לֵיהּ אֶלָּא אֶרֶךְ אַפַּיִם לַצַּדִּיקִים אֶרֶךְ אַפַּיִם לָרְשָׁעִים, וּפֵרֵשׁ רַשִׁ"י אַפַּיִם מַשְׁמָע פָּנִים בֵּין שׂוֹחֲקוֹת בֵּין זְעוּמוֹת, אַפַּיִם שׂוֹחֲקוֹת מַאֲרִיךְ וּמְאַחֵר לַצַּדִּיקִים לֶעָתִיד לָבֹא וְאַפַּיִם זְעוּמוֹת מַאֲרִיךְ לָרְשָׁעִים וּמְאַחֵר פּוּרְעָנוּתָם לָעוֹלָם הַבָּא.
[cup]וְהָעִנְיָן[/cup] הוּא דְּהִנֵּה יָדוּעַ דְּאוֹר אֵין סוֹף הוּא קָדוֹשׁ וּמוּבְדָּל מֵעוֹלָמוֹת שֶׁאֵינוֹ בְּעֶרֶךְ הָעוֹלָמוֹת כְּלָל, וְלָכֵן אֲנִי ה' לֹא שָׁנִיתִי (מַלְאָכִי ג. ו) שֶׁלֹּא יֵשׁ שׁוּם שִׁינּוּי בֵּין קוֹדֶם הַבְּרִיאָה לְאַחַר הַבְּרִיאָה, מִפְּנֵי שֶׁאֵינוֹ תּוֹפֵס מָקוֹם כְּלָל. וְזֶהוּ גַם כֵּן מַה שֶּׁכָּתוּב (אִיּוֹב לה. ו-ז) אִם חָטָאתָ מַה תִּפְעָל בּוֹ וְרַבּוּ פְשָׁעֶיךָ מַה תַּעֲשֶׂה לּוֹ אִם צָדַקְתָּ מַה תִּתֶּן לוֹ אוֹ מַה מִּיָּדְךָ יִקָּח, דְּכָל זֶה הוּא מִפְּנֵי שֶׁמַּעֲשֵׂה הַתַּחְתּוֹנִים אֵינָם תּוֹפְסִים מָקוֹם כְּלָל,
וּכְמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה בְּמִדְרָשׁ רַבָּה (לֶךְ לְךָ פַּרְשָׁה מ"ד) הָאֵל תָּמִים דַּרְכּוֹ אִמְרַת ה' צְרוּפָה מָגֵן הוּא לְכֹל הַחוֹסִים בּוֹ, אִם דְּרָכָיו תְּמִימִים הוּא עַל אַחַת כַּמָּה וְכַמָּה, רַב אָמַר לֹא נִתְּנוּ הַמִּצְוֹת אֶלָּא לְצָרֵף בָּהֶן אֶת הַבְּרִיּוֹת, וְכִי מָה אִיכְפַּת לֵיהּ לְהַקָּדוֹשׁ בָּרוּךְ הוּא לְמִי שֶׁשּׁוֹחֵט מִן הַצַּוָּאר אוֹ מִי שֶׁשּׁוֹחֵט מִן הָעוֹרֶף הָוֵי לֹא נִתְּנוּ הַמִּצְוֹת אֶלָּא לְצָרֵף בָּהֶם אֶת הַבְּרִיּוֹת, אֲבָל לְגַבֵּי אוֹר אֵין סוֹף אֵין הָעִנְיָן נוֹגֵעַ כְּלָל,
וּמַה שֶּׁכָּתוּב (שְׁמוֹת כג. כה) וַעֲבַדְתֶּם אֵת ה' אֱלֹקֵיכֶם, וּמְבוֹאָר בְּמָקוֹם אַחֵר עַל פָּסוּק זֶה דַּעֲבוֹדָה הִיא צוֹרֶךְ גָּבוֹהַּ, זֶהוּ בְּסֵדֶר הִשְׁתַּלְשְׁלוּת וְהַיְינוּ בְּעֶשֶׂר סְפִירוֹת דַּאֲצִילוּת דְּרֵאשִׁיתָן הִיא חָכְמָה שָׁם נוֹגֵעַ עֲבוֹדַת הַתַּחְתּוֹנִים כְּמוֹ שֶׁכָּתוּב בְּמָקוֹם אַחֵר וּכְמוֹ שֶׁיִּתְבָּאֵר בְּסָמוּךְ. אֲבָל בְּאוֹר אֵין סוֹף שֶׁלְּמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת אֵינָם תּוֹפְסִים מָקוֹם מַעֲשֵׂה הַתַּחְתּוֹנִים. וְזֶהוּ שֶׁאָמְרוּ בַּמִּשְׁנָה לְהוֹדִיעַ כַּמָּה אֶרֶךְ אַפַּיִם לְפָנָיו שֶׁכָּל הַדּוֹרוֹת הָיוּ מַכְעִיסִין וּבָאִין כו', שֶׁאַף עַל פִּי שֶׁהָיוּ מַכְעִיסִין מִכָּל מָקוֹם נִמְשַׁךְ לָהֶם הַשְׁפָּעַת הַחַיּוּת מִבְּחִינַת אֶרֶךְ אַפַּיִם שֶׁהוּא בְּחִינַת אוֹר אֵין סוֹף שֶׁלְּמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת, לְפִי שֶׁשָּׁם אֵינוֹ נוֹגֵעַ מַעֲשֵׂה הַתַּחְתּוֹנִים וִיכוֹלִים הַכֹּל לְקַבֵּל.
[cup]וְזֶהוּ[/cup] שֶׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (פְּסָחִים קיח. א) אָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי הַנִי כ"ו הוֹדוּ כְּנֶגֶד מִי, כְּנֶגֶד כ"ו דּוֹרוֹת שֶׁבָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא בְּעוֹלָמוֹ וְלֹא נָתַן לָהֶם תּוֹרָה וְזָן אוֹתָם בְּחַסְדּוֹ, וְהַיְינוּ שֶׁאַף עַל פִּי שֶׁהָיוּ עוֹבְדֵי עֲבוֹדָה זָרָה כְּמוֹ דּוֹרוֹ שֶׁל אֱנוֹשׁ הָיוּ נִיזּוֹנִין בְּחַסְדּוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא הַיְינוּ בְּחִינַת רַב חֶסֶד דַּאֲרִיךְ אַנְפִּין, שֶׁמִּשָּׁם נִמְשָׁכָה הַהַשְׁפָּעָה גַּם לְעוֹבְרֵי רְצוֹנוֹ. וְזֶהוּ וְלֹא נָתַן לָהֶם תּוֹרָה וְזָן אוֹתָם בְּחַסְדּוֹ, דְּעַל פִּי הַתּוֹרָה אֵינוֹ מַגִּיעַ לָהֶם הַהַשְׁפָּעָה, דְּאוֹרַיְיתָא מֵחָכְמָה נָפְקַת וּבְחָכְמָה יֵשׁ הַגְבָּלָה, לִהְיוֹת נִמְשָׁךְ הַחֶסֶד דַּוְקָא בְּאוֹפֶן כָּזֶה וְלֹא בְּאוֹפֶן אַחֵר, וְהַיְינוּ כַּאֲשֶׁר עוֹשֶׂה רְצוֹנוֹ שֶׁל מָקוֹם תּוּמְשַׁךְ לוֹ הַשְׁפָּעַת הַחֶסֶד וּבְאוֹפֶן אַחֵר לֹא תוּמְשַׁךְ, אֲבָל אוֹתָן כ"ו דּוֹרוֹת אַף שֶׁהָיוּ עוֹבְרֵי רְצוֹנוֹ נִמְשְׁכָה לָהֶם הַהַשְׁפָּעָה שֶׁלֹּא עַל פִּי הַתּוֹרָה כִּי אִם מֵחַסְדּוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בְּחִינַת רַב חֶסֶד.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]יְנִיקַת הַסִּטְרָא אָחֳרָא א) מֵרִיבּוּי הַצִּמְצוּמִים, ב) מֵהַמַּקִּיף הָעֶלְיוֹן. הָעֲשָׂרָה דוֹרוֹת מֵאָדָם עַד נֹחַ, מִנֹּחַ וְעַד אַבְרָהָם נִיזּוֹנִין בְּחַסְדּוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, אֶרֶךְ אַפַּיִם, אוֹר אֵין סוֹף שֶׁלְּמַעְלָה מֵהִשְׁתַּלְשְׁלוּת דְּאֵין מַעֲשֵׂה הַתַּחְתּוֹנִים נוֹגֵעַ, וְהַשְׁפָּעָתָם אֵינָהּ עַל פִּי מִשְׁפַּט הַתּוֹרָה אֶלָּא מֵרַב חֶסֶד.[kitzur_end]
