בניו כו' ושמרו דרך הוי"ה לעשות צדקה ומשפט דהיינו מה שעל ידו נעשה קשר וחבור להמשיך יח"ע הרוחנית למצות הגשמיות וזהו שיצוה לבניו לעשות ולשמור דרך הוי"ה שלמעל' לבא למטה בגשמיות וזהו למען הביא ה' על אברהם כו' שזהו גלוי אלקות בא"י ע"י קיום התו"מ דבנ"י כו' ונמצא דג' מדרגות הללו הן בירידת האור למטה יותר שזהו מורה על עוצם מעלת האור בשרשו יותר כפי כלל הנ"ל והיינו שקיים היח"ע שהיה בהעלם דח"ס עד שהיה בן ע"ה שנה ואח"כ יצא לגלוי באצי' בכ"ד שנה הנ"ל ומצ"ט שנה ע"י המילה שירד האור יותר לקשר ולחבר אצי' בבי"ע כנ"ל וד"ל: 
(נג) [cup]ומעתה[/cup] יובן סדר דברי הפרשה הזאת בתוס' באור במ"ש ויאמר ה' אל אברם לך לך מארצך כו' אל הארץ אשר אראך שזה מבואר למעלה ויש בו ב' פירושים כנ"ל הא' כפי' הזהר מלמעלה למטה דהיינו מארצך בבחי' בינה סתימאה שמקבלת מח"ס וע"ז אמר לך לך מארצך שבחי' נקודה דח"ס דטמיר וגניז בבינה לבא מן ההעלם לגלוי למטה בבחי' המל' דאצי' סופא דכל דרגין דאצי' שהוא א"י כו' ופי' הב' שבס' לק"ת הוא להיפך שיעלה מלמטה למעלה בבחי' או"ח להביא ולהעלות מעולם הפירוד שנפל למטה דהיינו מארצך אור כשדים ויעלה משם לעולם היחוד בפנימית דבינה עלאה שז"ש אל הארץ אשר אראך שהוא א"י העליונה ה"א עלאה דשם הוי"ה כידוע ושני הפירושים אמת כנ"ל בפי' וילך למסעיו שהיה קשר וחבור מלמעלה למטה ומלמטה למעלה וכמ"ש אל המקום אשר היה אהלה כו' וכן בפי' אראך דקאי על א"י או על אברהם שני הפירושים אמת וכנ"ל וד"ל: 
(נד) [cup]ומעתה[/cup] יש להבין מ"ש ואעשך לגוי גדול ואברכך ואגדלה שמך כו'. דהנה מה שלא אמר לו עד כה שיברכנו ויגדל שמו אע"פ שכבר היה ראוי לכך מצד מעשיו הטובים בכל יח"ע שעשה בעבודתו ותפלתו אבל היה הכל עדיין בבחי' ההעלם דח"ס הנ"ל אך כאשר יצא מההעלם לגלוי שז"ש לך לך מארצך כו' שאז יתגלו יח"ע שלו מעצמות המאציל באצי' בתוס' אור בכלים כנ"ל ע"כ אמר לו עכשיו דוקא ואברכך ואגדלה שמך למעלה בכל האורות העליונים מפני שיומשך מבחי' מל' דא"ס תוס' ברכה ע"י היח"ע הללו שנק' שמו הגדול ויתגדל שמך כשמו הגדול ממש כמו ענין יהא שמיה רבא כו' כידוע והוא לפי שיומשך הכל בא"י שהוא הארץ אשר אראך ששם יהיה לגוי גדול ע"י בירור הניצוצות דתהו שנפלו שם מז' מלכין קדמאין כו' וז"ש ואעשך לגוי גדול ואברכך כו' עד והיה ברכה שיתברכו בו כל האורות ומדרגות העליונות ביחודים שתעשה למעלה מצד בחי' הבירורים דניצוצות דתהו שע"י עולים למעלה ומוסיפים תוס' אור גדול כמו הגרים אשר עשה וכל אשר העלה ממצרים כמ"ש ויעל אברם ממצרים כו' ובפרט ע"י מצות מילה שנק' אב המון גוים שז"ש והיה ברכה לעתיד וד"ל:
