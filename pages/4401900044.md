# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ו. > פֶּרֶק ב

אָמְנָם עַכְשָׁיו הִנֵּה סְפִירַת הַמַּלְכוּת הִיא צְמֵאָה, מִצַּד יְרִידָתָהּ שֶׁיָּרְדָה וְנִתְלַבְּשָׁה בִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה הִיא בִּבְחִינַת צִמָּאוֹן לַעֲלוֹת מִירִידָתָהּ וְלִיכָּלֵל בְּשָׁרְשָׁהּ, וְכֵן הוּא גַּם בַּנְּשָׁמוֹת, שֶׁעַל יְדֵי הִתְלַבְּשׁוּתָם בַּגּוּף וְנֶפֶשׁ הַבַּהֲמִית, שֶׁיָּרְדָה הַנְּשָׁמָה מִמַּדְרֵיגָתָהּ מִכְּמוֹ שֶׁהָיְתָה קוֹדֶם הִתְלַבְּשׁוּתָהּ בַּגּוּף, שֶׁהָיְתָה בִּבְחִינַת תַּכְלִית הָרוּחָנִיּוּת וְלֹא הָיְתָה שַׁיֶּיכֶת לְיֵשׁוּת וְגַשְׁמִיּוּת כְּלָל וּכְלָל, וּמִכָּל שֶׁכֵּן שֶׁלֹּא הָיְתָה עֲלוּלָה חַס וְשָׁלוֹם לֵירֵד וּלְהִתְגָּאֵל לִהְיוֹת הֵיפֶךְ מֵאֱלֹקוּת חַס וְשָׁלוֹם, וּבִירִידָתָהּ בַּגּוּף הֲרֵי גַם כְּשֶׁעוֹבֶדֶת עֲבוֹדָתָהּ כַּנַ"ל וּפוֹעֶלֶת הַכְנָעַת הַחוֹמֶר וּבִיטּוּל הַיֵּשׁ מִכָּל מָקוֹם הֲרֵי הִיא שַׁיָּיכָה לְיֵשׁוּת וְגַשְׁמִיּוּת, שֶׁהֲרֵי הִיא בִּבְחִינַת הִתְלַבְּשׁוּת מַמָּשׁ בְּנֶפֶשׁ הַבַּהֲמִית,
וּכְמוֹ שֶׁפּוֹעֶלֶת פְּעוּלּוֹת גַּשְׁמִיּוֹת לְהַשְׂכִּיל בְּעִנְיָנִים גַּשְׁמִים, וְהַשְׂכָּלָה וְהַשָּׂגָה גַּשְׁמִיּוֹת, שֶׁכָּל זֶה לֹא הָיָה בָהּ קוֹדֶם הִתְלַבְּשׁוּתָהּ בַּגּוּף שֶׁהָיָה הַשְׂכָּלָתָהּ וְהַשָּׂגָתָהּ בְּתַכְלִית הָרוּחָנִיּוּת וְהַדַּקּוּת, וּבְעִנְיָנִים רוּחָנִיִּים וְנַעֲלִים מְאֹד, וְעִנְיָנִים גַּשְׁמִים, רְצוֹנִי לוֹמַר, גַּם עִנְיָנִים שִׂכְלִים אֶלָּא שֶׁהֵם גַּשְׁמִים לֹא יָדְעָה מִזֶּה כְּלָל, וּבִירִידָתָהּ בַּגּוּף הֲרֵי הַשָּׂגָתָהּ הִיא בְהַגְשָׁמָה, וּמַשֶּׂגֶת עִנְיָנִים וְשִׂכְלִים גַּשְׁמִים, וְכָל זֶה הוּא מִצַּד הַהִתְלַבְּשׁוּת בְּנֶפֶשׁ הַטִּבְעִי הַבַּהֲמִי, כְּמוֹ כֵן מִצַּד זֶה בִּכְלָלוּתָהּ נַעֲשָׂה לָהּ שַׁיָּיכוּת לְיֵשׁוּת וְגַשְׁמִיּוּת, אֲשֶׁר מִקּוֹדֶם לֹא הָיְתָה לָהּ שַׁיָּיכוּת לָזֶה כְּלָל, עַד שֶׁיָּכוֹל לִהְיוֹת חַס וְשָׁלוֹם יְרִידָה מַמָּשׁ לְהִתְגָּאֵל חַס וְשָׁלוֹם,
וּמִשּׁוּם זֶה נַעֲשָׂה תּוֹקֶף הַצִּמָּאוֹן בְּהַנְּשָׁמָה לַעֲלוֹת מִירִידָתָהּ וּלְהִכָּלֵל לְמַעְלָה, שֶׁהַצִּמָּאוֹן הַזֶּה גַּם כֵּן לֹא הָיָה בְּהַנְּשָׁמָה קוֹדֶם, אֶלָּא דַּוְקָא עַל יְדֵי יְרִידָתָהּ בַּגּוּף וְנֶפֶשׁ הַבַּהֲמִית מִצַּד הַיְרִידָה כַּנַ"ל נַעֲשֶׂה בָּהּ תּוֹקֶף הָרָצוֹא וְהַצִּמָּאוֹן,
וּכְמוֹ כֵן הוּא בִּבְחִינַת מַלְכוּת, שֶׁעַל יְדֵי יְרִידָתָהּ לִבְרִיאָה-יְצִירָה-עֲשִׂיָּה עַל יְדֵי זֶה הִיא בִּבְחִינַת צִמָּאוֹן לַעֲלוֹת לְמַעְלָה כו'.
[cup]וְרָוָה[/cup] נִקְרָא מַה שֶּׁרוֹצֶה לְמַלֹּאות תַּאֲוָתוֹ הַגַּשְׁמִית וְאֵין לוֹ רָצוֹן וְצִמָּאוֹן לֵאלֹקוּת, כִּי אִם עִיקַּר חֶפְצוֹ וְחֶשְׁקוֹ הוּא בְּתַעֲנוּגִים גַּשְׁמִים וּבָהֶם מְרַוֶּה צִמְאוֹנוֹ, וְגַם יָכוֹל לְרַוּוֹת צִמְאוֹנוֹ בָּזֶה לְפִי שֶׁהֵן מְזוּמָּנִים לְפָנָיו, כִּי הַנֶּפֶשׁ הָאֱלֹקִית צִמְאוֹנָהּ וְחֶשְׁקָהּ לֵאלֹקוּת וְאֵינָהּ מְרַוָּוה צִמְאוֹנָהּ חַס וְשָׁלוֹם בִּדְבָרִים הַגַּשְׁמִים כִּי אִם חֶפְצָהּ וְחֶשְׁקָהּ לֵאלֹקוּת, וַהֲרֵי הִיא בִּבְחִינַת צְמֵאָה בִּירִידָתָהּ לְמַטָּה וְתָמִיד הִיא בְּצִמָּאוֹן זֶה וְלֹא תוּכַל לְרַוּוֹת צִמְאוֹנָהּ, כִּי אֱלֹקוּת אֵינוֹ מֵאִיר וּמִתְגַּלֶּה עַתָּה בְּגִילּוּי מַמָּשׁ כִּי אִם בְּדֶרֶךְ יְדִיעַת הַמְּצִיאוּת לֹא הַשָּׂגַת הַמַּהוּת מַמָּשׁ, עַד לֶעָתִיד לָבֹא דִּכְתִיב (יְשַׁעְיָה ל. כ) וְלֹא יִכָּנֵף עוֹד מוֹרֶיךָ, שֶׁאָז יִתְגַּלֶּה בְּחִינַת הַמַּהוּת מַמָּשׁ, אֲבָל עַתָּה אֵינוֹ מִתְגַּלֶּה הַמַּהוּת, וְלָכֵן הִיא בִּבְחִינַת צְמֵאָה תָּמִיד,
אֲבָל הַנֶּפֶשׁ הַבַּהֲמִית חֶשְׁקָהּ וְצִמְאוֹנָהּ לִדְבָרִים גַּשְׁמִיִּים וּבָזֶה הִיא מְרַוָּוה צִמְאוֹנָהּ וּבְאֶפְשָׁרָהּ לְרַוּוֹת צִמְאוֹנָהּ בָּזֶה שֶׁמְּזוּמָּנִים לְפָנֶיהָ, וְעַל דֶּרֶךְ שֶׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (שַׁבָּת קנה. ב) לֵית עַתִּיר מֵחֲזִירָא, לְפִי שֶׁמַּאֲכָלוֹ פְּסוֹלֶת, וְלָכֵן מַאֲכָלוֹ מוּכָן וּמְזוּמָּן לְפָנָיו תָּמִיד וּבְנָקָל לְהַשִּׂיג מְבוּקָּשׁוֹ וְתַאֲוָותוֹ, וּכְמוֹ כֵן הוּא בַּתַּעֲנוּגִים הַגַּשְׁמִים דְּיָדוּעַ שֶׁהֵן פְּסוֹלֶת הַתַּעֲנוּגִים הָעֶלְיוֹנִים וּבְנָקָל לְהַשִּׂיגָם וּמְרַוָּוה צִמְאוֹנָהּ וְעַל כֵּן נִקְרָא רָוָה.

[kitzur_begin]סְפִירַת הַמַּלְכוּת בִּירִידָתָהּ מִתְגַּבֵּר צִמְאוֹן הַהִתְכַּלְלוּת בִּמְקוֹרָהּ, הַנְּשָׁמוֹת לְמַעְלָה הֵן בְּתַכְלִית הָרוּחָנִיּוּת, וּמוּשְׁלָלָה גַּם מִידִיעַת מוּשְׂכָּלוֹת הָעֲמוּקִים הַסְּבוּכִים בְּגֶשֶׁם, וּבִירִידָתָהּ בַּגּוּף הִיא שַׁיֶּיכֶת לְיֵשׁוּת וִיכוֹלָה לְהִתְגָּאֵל אָז מִתְגַּבֵּר הַצִּמָּאוֹן, אֲבָל הַנֶּפֶשׁ הַבַּהֲמִית שֶׁעִיקָּרוֹ חוּמְרִיּוּת נִקְרָא רָוָה.[kitzur_end]
