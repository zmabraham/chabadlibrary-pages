# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר כה. > פֶּרֶק ד

וְהִנֵּה יִשְׂרָאֵל הֵם מַאֲמִינִים בְּנֵי מַאֲמִינִים, דֶּאֱמוּנָה הִיא בְּטֶבַע תּוֹלַדְתָּם, שֶׁבָּזֶה חֲלוּקִים עַם יִשְׂרָאֵל מִכָּל הָעַמִּים אֲשֶׁר עַל פְּנֵי הָאֲדָמָה, דְּכָל הָעַמִּים הִנֵּה עִם הֱיוֹת שֶׁגַּם בֵּינֵיהֶם נִמְצָאִים בַּעֲלֵי אֱמוּנָה בֶּאֱמוּנָתָם וְאַף גַּם כָּאֵלּוּ שֶׁהֵם אֲדוּקִים בֶּאֱמוּנָתָם, הִנֵּה אֵין זֶה בְּטֶבַע תּוֹלְדוֹתָם וְהוּא רַק כְּדָבָר הַנִּקְנֶה עַל יְדֵי דְבַר מָה, וּכְמוֹ שֶׁאָנוּ רוֹאִים בְּמוּחָשׁ בְּעִנְיַן הַשֵּׂכֶל שֶׁיֵּשׁ שֵׂכֶל עַצְמִי וְיֵשׁ שֵׂכֶל הַנִּקְנֶה עַל יְדֵי דָבָר זוּלָתוֹ, שֶׁבִּכְלָלוּת הוּא הַהֶפְרֵשׁ בֵּין שִׂכְלֵי הַבַּעֲלֵי חַיִּים לְשֵׂכֶל הָאָדָם, שֶׁהָאָדָם הוּא שִׂכְלִי בְּעֶצֶם מַהוּתוֹ וְלָכֵן הִנֵּה כְּלָלוּת מַהוּת שֵׂכֶל הָאָדָם הוּא שֵׂכֶל עַצְמִי וְשֵׂכֶל הַבַּעֲלֵי חַיִּים הוּא שֵׂכֶל הַנִּקְנֶה, וּכְמוֹ יָדַע שׁוֹר קוֹנֵהוּ וַחֲמוֹר אֵבוּס בְּעָלָיו וְכֵן יֵשׁ כַּמָּה עִנְיָנִים מַה שֶּׁבָּהֶם וְעַל יָדָם מַכְנִיסִים שֵׂכֶל בְּבַעֲלֵי חַיִּים וּכְמוֹ הַמַּרְדַּעַת שֶׁמּוֹרֶה דֵעָה בַּבְּהֵמָה שֶׁכָּל זֶה הוּא רַק שֵׂכֶל הַנִּקְנֶה אֲבָל שֵׂכֶל הָאָדָם הוּא עַצְמִי,
וְכֵן הוּא בְּעִנְיַן הָאֱמוּנָה דְּכָל הָעַמִּים הִנֵּה אֱמוּנָתָם הִיא כְּדָבָר הַנִּקְנֶה עַל יְדֵי דָבָר שֶׁחוּץ מִמֶּנּוּ, אֲבָל לְהַבְדִּיל, יִשְׂרָאֵל הִנֵּה אֱמוּנָתָם בַּה' וּבְתוֹרָתוֹ יִתְבָּרֵךְ הִיא עַצְמִית, וּכְמוֹ שֶׁאָנוּ רוֹאִים בְּמוּחָשׁ אֲשֶׁר בְּכָל עִנְיָן וְעִנְיָן, בֵּין שֶׁהוּא טוֹב וּבֵין שֶׁהוּא הַפְכִּי, הִנֵּה שֵׁם שָׁמַיִם שָׁגוּר בְּפִי כֹל בְּנוּסְחָאוֹת שׁוֹנוֹת, יְרַחֵם הַשֵּׁם יִתְבָּרֵךְ, יָחוּס הַשֵּׁם יִתְבָּרֵךְ תּוֹדָה לְהַשֵּׁם יִתְבָּרֵךְ, וְכַיּוֹצֵא בָהֶם בְּסִגְנוֹנוֹת תְּפִלָּה וְתַחֲנוּנִים וְתוֹדָה, דְּכָל זֶה מִפְּנֵי שֶׁבְּנִשְׁמוֹת יִשְׂרָאֵל הִנֵּה אֱמוּנָתָם בְּהַשֵּׁם יִתְבָּרֵךְ וּבְתוֹרָתוֹ יִתְבָּרֵךְ הִיא עַצְמִית, הֵם חַיֵּינוּ וְאוֹרֶךְ יָמֵינוּ וַעֲלֵיהֶם הוֹרַגְנוּ כָּל הַיָּמִים בִּימֵי הַגָּלוּת הַמַּר,
וְכֵן הוּא בְּעִנְיַן הַפַּרְנָסָה דְּבֶאֱמֶת הִנֵּה הַכֹּל יוֹדְעִים שֶׁהָעִיקָּר הוּא בִּרְכַּת ה' וְכָל מִי שֶׁהוּא יְהוּדִי אֵינוֹ חוֹשֵׁב בְּשׁוּם אוֹפֶן שֶׁהָעֵסֶק הוּא מְקוֹר פַּרְנָסָתוֹ חַס וְשָׁלוֹם וְיוֹדֵעַ הֵיטֵב שֶׁהַהַשְׁפָּעָה מֵאֵת ה' הִיא וְהוּא הַנּוֹתֵן לוֹ אֶת פַּרְנָסָתוֹ עַל יְדֵי הָעֵסֶק, אֶלָּא שֶׁהָרוּחַ שְׁטוּת שֶׁהַיֵּצֶר הָרָע מַשְׁטֶה אוֹתוֹ שֶׁאֵינוֹ מֵשִׂים אֶל לִבּוֹ לָדַעַת אֲמִיתִּית הַדְּבָרִים הַנַ"ל דְּמֵאַחַר שֶׁהָעִיקָּר הוּא הַהַשְׁפָּעָה הָעֶלְיוֹנָה הֲרֵי אֵינוֹ צָרִיךְ לִהְיוֹת שׁוּם מְנִיעָה וְעִיכּוּב וְשׁוּם קוֹצֶר בְּהַדְּבָרִים אֲשֶׁר חִיֵּיב אוֹתוֹ הַבּוֹרֵא לַעֲשׂוֹת, וְאַף שֶׁבִּשְׁבִיל זֶה לֹא יוּכַל לַעֲסוֹק הַרְבֵּה בְּעֵסֶק מַשָּׂא וּמַתָּן יַסְפִּיק לוֹ גַּם מְעַט עֵסֶק וְהַשֵּׁם יִתְבָּרֵךְ יַמְצִיא לוֹ לְהַרְוִיחַ בִּמְעַט זְמַן שֶׁבִּמְעַט הָעֵסֶק שֶׁיַּעֲשֶׂה יוּמְשַׁךְ וְיִתְלַבֵּשׁ הַחֶסֶד הַנִּקְצַב וְהַנִּמְשַׁךְ אֵלָיו,
וְכַאֲשֶׁר נִפְתֶּה לַהֲסָתַת הַיֵּצֶר וּמַשְׁטֶה אֶת עַצְמוֹ בְּרוּחַ שְׁטוּת שֶׁצָּרִיךְ לְהִתְעַסֵּק הַרְבֵּה דַוְקָא וְאִי אֶפְשָׁר לוֹ לִקְבּוֹעַ עִתִּים לַתּוֹרָה וּלְהִתְפַּלֵּל כִּדְבָעֵי מִצַּד הַזְּמַן וְהַטִּרְדָּא וְנִדְמֶה לוֹ שֶׁנָּכוֹן כֵּן כְּפִי אֲשֶׁר מְדַמֶּה בְּשִׂכְלוֹ הַסָּכָל וְהַמְּעוּקָּל עַל יְדֵי רוּחַ שְׁטוּת הַנַ"ל, מַרְאֶה לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁאֵינוֹ מוֹעִיל לוֹ רִיבּוּי הָעֵסֶק וַהֲרֵי הוּא הוֹלֵךְ סוֹבֵב כָּל הַיּוֹם בַּשְּׁוָקִים וּבָרְחוֹבוֹת מִבְּלִי יִמְצָא שׁוּב רֶיוַח כְּלָל, וּמִדָּה כְּנֶגֶד מִדָּה מְשַׁלֵּם לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא כְּמוֹ שֶׁכָּתוּב (דְּבָרִים כח. לד) וְהָיִיתָ מְשׁוּגָּע כו' אֲשֶׁר תִּרְאֶה, וּמֵרוֹב הַדְּאָגָה מִשְׁתּוֹמֵם מוֹחוֹ עַד שֶׁכִּמְשׁוּגָּע יֵחָשֵׁב שֶׁזֶּהוּ עַצְמוֹ עִנְיַן שְׁטוּת שֶׁהֲרֵי הַמְשׁוּגָּע וְהַשּׁוֹטֶה עִנְיָן אֶחָד לָהֶם, כִּי כְּמוֹ שֶׁאֵצֶל הַמְּשׁוּגָּע נִדְמֶה שֶׁצָּרִיךְ לִהְיוֹת דָּבָר שֶׁהוּא מוּפְרָךְ אֵצֶל הַבָּרִיא וְחָלִים, כֵּן הַשּׁוֹטֶה בְּרוֹב שְׁטוּתָיו גַּם כֵּן מִתְנַהֵג עַל דֶּרֶךְ זֶה וּכְמוֹ שֶׁכָּתוּב (שְׁמוּאֵל-א כא. טז) חֲסַר מְשׁוּגָּעִים אָנִי שֶׁכִּינָּה לְדָוִד שֶׁהִנִּיחַ לִימָּשֵׁךְ רִיר אֶל זְקָנוֹ בְּשֵׁם מְשׁוּגָּע עַל שֵׁם שֶׁהוּא הֵיפֶךְ הַדַּעַת כו',
וְכָל זֶה גּוֹרֵם לוֹ מַה שֶּׁנִּפְתָּה לְהָרוּחַ שְׁטוּת שֶׁל הַיֵּצֶר הָרָע, אֲשֶׁר אִם הָיָה מֵשִׂים לִבּוֹ לְהַדְּבָרִים הָאֲמִיתִּים הַנַ"ל הָיָה בְּאוֹפֶן אַחֵר לְגַמְרֵי, שֶׁלֹּא הָיָה צָרִיךְ לְהִתְעַסֵּק בְּרִיבּוּי עֵסֶק כְּלָל וְהָיָה נִמְשָׁךְ לוֹ הַחֶסֶד עֶלְיוֹן גַּם בִּמְעַט עֵסֶק בְּרִיבּוּי שֶׁפַע פַּרְנָסָה בְּרֶיוַח.

[kitzur_begin]יִשְׂרָאֵל אֱמוּנָתָם בַּה' וּבְתוֹרָתוֹ הִיא עַצְמִית, אֱמוּנַת כָּל הָעַמִּים גַּם הָאֲדוּקִים הִיא כְּדָבָר הַנִּקְנֶה עַל יְדֵי דָבָר שֶׁחוּץ מִמֶּנּוּ, שֵׂכֶל עַצְמִי וְשֵׂכֶל הַנִּקְנֶה, הַכֹּל יוֹדְעִים שֶׁרַק בִּרְכַּת ה' הִיא תַעֲשִׁיר וְהַנִּפְתֶּה לַהֲסָתַת הַיֵּצֶר הָרָע מְאַבֵּד פַּרְנָסָתוֹ, שׁוֹטֶה וּמְשׁוּגָּע עִנְיָנָם אֶחָד.[kitzur_end]
