שאין כח זה בשמי' כלל והיינו ג"כ ל' רצון שרצתה לעשות רצון קונה שהוא עצמות הרצון שעלה ברצון ומחשבה הקדומה כנ"ל והוא בחי' רצוא כמו אם רץ לבך כו' כגודל התשוק' ברשפי אש שלהבת שהניצוץ נמשך אחר השלהבת כמו אחריך נרוצ'[ftnref_1261_19] ולכך נק' רצון מלשון מרוצ' וזהו למעל' מבחי' ראשי' דחכ' כנ"ל ולכך נא' והארץ אזכור כו' וכה"ג וד"ל (וכמ"כ בנש"י כמו שעלו במח'[ftnref_1262_20] הקדומה הנה נק' ארץ חפץ[ftnref_1263_21] לע"ל שרצתה לעשות רצון קונה בביטול כל עצמות שלמעל' מבחי' ביטול דחכמ' שבנפש האלקית כו' שזהו מצד העפר דגוף החומרי דוקא שמעפר יסודו[ftnref_1264_22] כשעושה תשובה ומעשים טובים בעולם הזה דוקא שאז נק' כנ"י כלה לשון כלתה נפשי[ftnref_1265_23] בבחי' או"ח דרצוא עד בחי' מהותו ועצמותו דוקא שלא תגיע הנשמ' לזה מצד עצמה בשרש' בח"ע מטעם הנ"ל בארץ שקדמה לשמים וד"ל): 
(לב) [cup]ואחר[/cup] כל זה יובן ענין המאמר דלעיל שקשה לזווגם כקי"ס דוקא דהנה בקי"ס נאמ' הפך ים ליבשה[ftnref_1266_24] כו' דהיינו ע"ד הנ"ל כמו שהמים היו מקיפים על הארץ ושרשם למעלה היינו ענין שמים שקדמו לארץ בסדר ההשתלשלות וא"כ כאשר הפך ים ליבשה זהו כמו שצוה יקוו המים כו' וירדו למטה שהגבוה הושפל וירד למטה להיות התגלות היבשה שהוא הארץ למעלה שזהו הגבהת השפל שעלה למעלה וכך הי' בקי"ס שהפך ים ליבשה הרי הים שהוא הגבוה הושפל למטה להיות התגלות היבשה בתוך הים שזהו העלאת והגבהת השפל ונמצא זהו כמשפיל גבוהים ומגביה שפלים בבת אחת ולמעלה הרי הוא ענין הגבהת השפל במדרגה שהוא הארץ עילאה וירידת הגבוה והשפלתו שהן השמים שקדמו לארץ מפני שהארץ קדמה במח' תחלה כנ"ל וע"כ אמר שקשה לזווגם כקי"ס לפי שנס דקי"ס היה קשה מכל הנסים מפני שקשה הדבר מאד להשפיל הגבוה ולרומם ולהגביה הנמוך ברגע אחת וזהו כמו עד"מ שקש' הדבר להגבי' את המקבל שהוא התלמיד ולהשפיל את[ftnref_1267_25] המשפיע שהוא הרב שיקבל מן התלמיד וכה"ג (כמו שמואל דיתיב קמיה ר"י תלמידו[ftnref_1268_26] ) או כמו שקשה הדבר להגבי' ולרומם את האשה ולהוריד להשפיל האיש המשפיע שיהי' כמו בחינת מקבל מאשתו שזהו היפך הסדר דהשתלשלות העולמות בבחי' עו"ע שהוא בחינת משפיע ומקבל שהמשפיע ודאי למעלה והמקבל למטה כמ"ש והוא ימשול בך[ftnref_1269_27] ולהיות כל איש שורר בביתו[ftnref_1270_28] ולהפוך זה הדבר שיהי' המקבל מושל במשפיע זהו קשה מאד אע"פ שראוי להיות כן מצד שיש יתרון מעלה במקבל שהמשפיע צריך לקבל הימנו כענין ארץ קדמ' דאו"ח דארץ נעוץ בתחלתן שלמעלה מן המשפיע וכמו שיהי' לע"ל דא"ח יהי' עט"ב[ftnref_1271_29] נקבה תסובב גבר[ftnref_1272_30] (וכמו שאמר לאברהם כל אשר תאמר[ftnref_1273_31] אליך שרה שמע בקול' אע"פ ששר' בחי' מקבל בחי' ב"ן הגביה' למעל' להיות אברהם בחי' מ"ה[ftnref_1274_32] שומע בקולה והוא לפי ששלשה האבות הטעימן מעין העוה"ב[ftnref_1275_33] שאז יהי' א"ח עט"ב כו' כמ"ש במ"א וד"ל) אך לפי סדר ההשתלשלות דעכשיו שהמשפיע עיקר והמקבל טפל וכמו שמים שנק' משפיע לארץ לפי שקדמו כמ"ש בראשית ברא כו' כנ"ל ע"כ הי' קשה מאד נס דקי"ס להשפיל
