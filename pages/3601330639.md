שצריך להעלים בזה האה' והחסד שלא יתגלה ביותר בכדי שתהי' ההשפעה לפ"ע הבן מה שיוכל לקבל כו', וכן לפעמים צריך להעלים האה' אליו כו', שבזה הגבו' היא רק לבוש אל האה' כו', דבאמת הגבו' אינו גבו' אמית' שה"ה רק להעלים בכדי שיהי' התגלות האה' באופן שתוכל להתקבל כו', וההעלם בזה אינו העלם גמור גם אם מעלים אהבתו ממנו כו', והאה' היא רק חיצוני' האה' כו'. משא"כ ברוגז הנ"ל ה"ז גבו' ממש דהרי מתרגז ומתכעס בשעת מעשה שהוא בכעס עליו, אך סבת הכעס הוא האה' עצמי', הרי שעצם מהותה של הגבו' היא אה' לא שהיא רק לבוש אל האה' כ"א עצם מהותה היא האה' ודוקא האה' עצמי' ממש, רק שבאה בציור הגבו' אבל עצם מהותה היא האה' עצמי' כו', ובהסיר החיצוני' מתגלה הפנימי', והוא מפני כי אדרבה הגבו' הזאת מביאה את האה' עצמי' בהתגלות דבלא זה ה"ה בהעלם בלבו (ומה שמתגלה הוא רק הארה חיצוני' כו'), ובזה בא האה' עצמי' בהתגלות רק שבא בציור הגבו' וע"כ בהסיר הציור החיצוני מתגלה הפנימי' כו', והיינו כאשר הבן מקבל זאת באה' ומתתקן עי"ז דאז מתגלה פנימי' האה' באה' יתירה דבחי' אה' עצמי' ממש כו'.
[kitzur][cup]קיצור.[/cup] דהנה כיסוי דבחי' חכ' על בחי' עתיק נת' שזהו שהתענוג נעשה חכ', ולא כמו כח תחתון שמלביש ומעלים כח בבחי' לבוש לבד, כ"א שנעשה חכ' וזהו ההעלם שהוא חכ', וא"כ ה"ז העלם עצמי, ומזה נמשך כח ההסתר, וכמו ואוהבו שחרו מוסר שהגבו' היא תוקף האהבה עצמיות שבא בציור הגבו' (ולבוש הגבו' הוא בהשפעה העלם בשביל הגילוי), וע"כ כשהבן מקבל מוסר ומתתקן מתגלה האהבה עצמיות.[/kitzur]
[cup]רכה)[/cup] והדוגמא מזה יובן בענין הגלות בריבוי העלמות והסתרים ביותר שזהו משום דהבן יקיר לי אפרים באה' גדולה ועצומה ביותר בבחי' התקשרות עצמי' שז"ע אהבתי אתכם כו', וכתיב ואוהב את יעקב, וידוע שזהו בבחי' עצמו' א"ס שלמע' מבחי' תהו ותיקון (שהן שם בבחי' אחים כו', וכמ"ש הלא אח עשו ליעקב דבהתגלות הרי תהו ותיקון מובדלים זמ"ז לגמרי וא"כ מה שהם אחים היינו בעצמות אוא"ס כו' וכמ"ש במ"א, ובבחי' ומדרי' זו) הנה בבחי' הפנימי' והעצמי' ואוהב את יעקב כו', וע"כ נוגע מאד הלכלוך שלהם שזהו מצד פנימי' האה' כו', ומשו"ז הן כל ההעלמות וההסתרים מפני שנוגע בפנימי' כו', דבעצם מהותם הוא האה' עצמי' דוקא כו', וכמ"ש הפרד"ס ש"ה פ"ג בשם הבהיר המים הרו וילדו אפלה מורה כי עלת וסבת הגבו' שהיא אפלה היא החסד שהן המים, והיינו דסבת הגבו' שהוא ההסתר הגדול הוא האה' עצמי' כו'. והכח ההעלם הזה שרשו מח"ס כמ"ש הרמ"ז ר"פ פנחס, וי"ל דהנה בחי' פנימי' ח"ס זהו בחי' כשמן הטוב שהוא בחי' הטוב העצמי שלמע' באין ערוך מכל הגילוים כו', והוא
בבחי' סתימות בעצם שאינו גם בבחי' מקור לגילוים כו', וע"כ משם הוא דוקא כח ההעלם שהוא טוב בעצם ובהתגלות הוא בבחי' הסתר לגמרי, אבל תכליתו ופנימיותו והיינו עצם מהותו הוא עצם הטוב כו'. וע"י העבודה בבחי' ההעלם הזה והיינו שגם בריבוי העלמות וההסתרים יהי' חזק בקיום התורה והמצות ולעמוד נגד כל מונע ששום דבר מהריבוי מניעות ועיכובים דזה"ג לא ימנע אותו מעסק התו' וקיום המצות כו',
