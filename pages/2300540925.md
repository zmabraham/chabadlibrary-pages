שנתאחדו עמו והן כאיש א' לומר בלשון יחיד אשירה כו'. והנה ידוע ומבואר במ"א (בס' ש"א[ftnref_8507_600] ) דמשה ובנ"י הן ב' מדרגות דאו"י ואו"ח שהן ב' הפכי' דחו"ג, דאו"י הן התו"מ ששרשם בבחי' החסדים דוקא שיורד מגבוה לנמוך מן המשפיע אל המקבל וכמו במ"ת שירד ה' על ה"ס וא' אנכי[ftnref_8508_601] כו', וזהו בחי' ומדרגה דמשה שהוא שרש התו"מ כמ"ש תורה צוה לנו משה[ftnref_8509_602] כו' כי משה ידבר והאלקי'[ftnref_8510_603] כו', ונק' מחוקק[ftnref_8511_604] כו' כידוע. ושרש בנ"י הוא בבחי' או"ח מלמטה למעלה מן המקבל אל המשפיע שהוא בבחי' ההעלא' דגבורו' כו'. והיינו ב' מדרגות הנ"ל בצ"ג ובע"ת, דצ"ג הן בבחי' או"י דתו"מ מלמעלה למטה ובע"ת הן בבחי' ההעלאה דאו"ח מלמטה למעלה והן כמו ב' הפכי' דאש ומים כו'. ועכ"ז בשירת הים א' אז ישיר משה ובנ"י גם [ש]המה ב' הפכי' דחו"ג נכללו יחדיו יודבקו והיו לאחדים ממש עד שא' בזה לשון יחיד אז ישיר כו', והוא מפני שבחי' ההעלא' דאו"ח דבנ"י נכללו באו"י דתו"מ דמשה ע"כ משה הוא העיקר בשירה זו כמ"ש אז ישיר משה כו', וכן בסיפא א' אשירה לה' כו' לשון יחיד דגם שכנ"י היו או' השירה עם משה יחד כמ"ש ויאמרו לאמר כו' נכללו עמו יחד כאיש א' ממש, והיינו מטעם ענין הנ"ל בבחי' תשובה ומע"ט שנכללי' כא' ממש (דאין תשובה בלא מע"ט ואין מע"ט בלא תשובה), וכל יריד' הנשמה בגוף הוא להיות מצ"ג בע"ת כדי שיעלו למעלה יותר שזהו הירידה לצורך עלי' כנ"ל. אך א"כ הי' צ"ל בהיפוך שיהיו בנ"י עיקר בשירה זאת כנ"ל שהרי יש יתרון מעלה בבע"ת מצ"ג עד שאין צ"ג יכולים לעמוד במקום בע"ת כנ"ל. אבל הענין הוא לפי שמבואר למעלה דקי"ס הי' הכנה למ"ת דוקא ואם לא קי"ס לא היו יכולים לעמוד בקבלת התורה פב"פ בה"ס, וז"ש הפך ים ליבשה כו' שם נשמחה בו שהוא בחי' ההעלא' דנש"י למעלה מעלה במהו"ע ית' ממש שנק' סדכ"ס כו' בחי' אדון הנפלאות שז"ש כי גאה גאה כו' כנ"ל, וזהו נשמחה בו ממש, וע"י בחי' העלאה זו הוא שגרם בחי' הירידה וההמשכה מלמעלה למטה במ"ת ש[שם] א' ביום חתונתו וביום שמח"ת לבו בשמחה דמשפיע במקבל כו' כנ"ל, וא"כ הרי בקי"ס הי' להיפוך שבחי' העלי' הי' לצורך הירידה דהיינו כמו בחי' ההעלא' דאו"ח דבע"ת הי' לצורך בחי' הירידה דאו"י דמ"ת שהוא בחי' צ"ג (וכך יכול להיות ענין תשובה ואח"כ מע"ט שזהו עלי' בשביל ירידה משא"כ תשו' אחרי מע"ט שזהו לעשות מצדיק בע"ת כנ"ל שנק' ירידה בשביל עלי' יותר כנ"ל) וכמו[ftnref_8512_605] ענין הספי' קודם מ"ת[ftnref_8513_606] שהוא בחי' הבירור והעלא' דז' מדות רעות דק"נ שיובררו ע"י הספי' ויוכללו בז' מדות דקדושה ואז יוכל להיות בחי' הירידה וההמשכה דמ"ת שהיא בפנימי' אבא כידוע).
