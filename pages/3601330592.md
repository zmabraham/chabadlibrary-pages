שהוא תגבורת העצמות גילוי בחי' פנימי' עתיק כו', וכמ"ש חשף ה' את זרוע קדשו כו' קירוב המאור כו', ולכן אז זמן התשו' לתקן כל הפגמים והחסרונות שחיסר באיזה מצוה או קיצר בעבודת הוי', דבכדי למלאות הפגמים והחסרונות צ"ל ההמשכה מבחי' עצמות א"ס דהיינו בחי' פנימי' עתיק בחי' עצמות התענוג כו', וכמשנת"ל פקצ"ג. וזהו ג"כ ענין תקיעת שופר שהו"ע הצעקה פנימי' מעומקא דלבא בחי' צעק לבם כו', ושופר הוא קול המטיל חרדה דהיינו בחי' גבו' דעתיק כו', והזמן המסוגל לזה הוא בעשי"ת דאז מאיר בחי' פנימי' עתיק כו', דבבחי' ומדרי' זו לא יש חילוקי מדרי' כלל וה"ה מצוי לכל דורשיו גם ליחיד כו', ובירושלמי אי' שהוא מצוי בבתי כנסיות ובבתי מדרשות היינו בתו', והוא כמשנת"ל דבח"ס מאיר בחי' פנימי' עתיק כו', וזהו שהתו' היא ג"כ תשו' דכמו שהתשו' היא רפואה וכמא' גדולה תשו' שמביאה רפואה לעולם שנא' ארפא משובתם, כמו"כ בתו' כתי' ולכל בשרו מרפא כו', והיינו מפני שבחכ' דתו' יש בזה גילוי בחי' פנימי' עתיק כו'. וזהו דרשו ה' בהמצאו אלו עשרה ימים שבין ר"ה
ליוהכ"פ דאז הוא מצוי היינו שנמשך בחי' עצמי' א"ס בחי' פנימי' עתיק כו', וה"ה מצוי כבי' לדורשיו גם ליחיד כו'.
[kitzur][cup]קיצור.[/cup] אמנם חכ' עם היותו רק כח, נק' הנפש ע"ש החכ', דעצם הנפש הוא חכ' שזהו מעלת האדם (והוא בחי' תיקון), וזהו סבת מציאת כח החכ', ולכן מתאחד עם העצם ונק' ע"ש, וזהו שהתענוג הוא סבת החכ', ובאמת התענוג הוא עצמו נעשה חכ', ולכן החכ' היא כפי התענוג דוקא, וכמו חיוב השכל בהבנין הוא כפי תענוג העצמי בו, וזהו דמוחין דגדלות הן בהבדלה, ובשרשו ומקורו הוא התענוג הנעלם, וזהו דח"ס הוא פנימיות עתיק ובבחי' פנימיות, וי"ל דז"ע גבו' דעתיק שמלובש במו"ס. וזהו דרשו ה' בהמצאו, דבכל קראנו איליו בחי' רצון, ובהמצאו בעשי"ת שהעצם מצוי בחי' עתיק, ובירוש' בבת"כ ובמ"ד זהו תורה בחי' ח"ס.[/kitzur]
