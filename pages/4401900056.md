# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר י. > פֶּרֶק א

מַאֲמָר י.
וְזֶהוּ דִּכְתִיב (דְּבָרִים כט. יח) וְהִתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר שָׁלוֹם יִהְיֶה לִּי כִּי בִּשְׁרִירוּת לִבִּי אֵלֵךְ שֶׁהָעוֹבֵר עֲבֵירָה מִתְבָּרֵךְ בִּלְבָבוֹ שֶׁיִּהְיֶה שָׁלוֹם לוֹ בְּהַשְׁפָּעַת הַחַיּוּת וְכָל הַטּוֹב, וְאַדְרַבָּא בְּרִיבּוּי הַהַשְׁפָּעָה, כִּי בִּשְׁרִירוּת לִבִּי אֵלֵךְ דִּלְפִי שֶׁהוֹלֵךְ בִּשְׁרִירוּת לִבּוֹ שֶׁלֹּא עַל פִּי הַתּוֹרָה דַּוְקָא עַל יְדֵי זֶה יְדַמֶּה בְנַפְשׁוֹ לְקַבֵּל רִיבּוּי הַשְׁפָּעָה מִבְּחִינַת מַקִּיף עֶלְיוֹן שֶׁלֹּא עַל פִּי חֶשְׁבּוֹן כְּמוֹ הַקְּלִיפּוֹת וְסִטְרָא אָחֳרָא, וְזֶהוּ כִּי בִּשְׁרִירוּת לִבִּי אֵלֵךְ, שֶׁהוּא כְּמוֹ נְתִינַת טַעַם עַל וְהִתְבָּרֵךְ כו' שָׁלוֹם יִהְיֶה כו' שֶׁעַל יְדֵי זֶה יְקַבֵּל מִבְּחִינַת הַמַּקִּיף שֶׁלֹּא עַל יְדֵי סֵדֶר הִשְׁתַּלְשְׁלוּת,
לְמַעַן סְפוֹת הָרָוָה אֶת הַצְּמֵאָה, דִּסְפוֹת הוּא לְשׁוֹן רִיבּוּי, וְהַיְינוּ שֶׁבְּהָרָוָה שֶׁהִיא הַסִּטְרָא אָחֳרָא נִמְשָׁךְ רִיבּוּי הַשְׁפָּעָה יוֹתֵר (כַּנַ"ל מַאֲמָר ט פֶּרֶק ג) מִכְּמוֹ בְּהַצְּמֵאָה שֶׁהוּא סִטְרָא דִקְדוּשָּׁה שֶׁנִּמְשֶׁכֶת הַהַשְׁפָּעָה עַל יְדֵי סֵדֶר הִשְׁתַּלְשְׁלוּת שֶׁהוּא בִּבְחִינַת גְּבוּל, אֲבָל הָרָוָה מְקַבֶּלֶת מִלְמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת בְּלִי דִין וְחֶשְׁבּוֹן,
וְעַל זֶה אוֹמֵר (שָׁם כט. יט) לֹא יֹאבֶה ה' סְלוֹחַ לוֹ, דְּדַוְקָא עַכּוּ"ם יְכוֹלִים לְקַבֵּל מִבְּחִינַת הַמַּקִּיף שֶׁלֹּא עַל יְדֵי סֵדֶר הִשְׁתַּלְשְׁלוּת אֲבָל לֹא יִשְׂרָאֵל, כִּי יִשְׂרָאֵל הֵם בְּעֶצֶם מִבְּחִינַת הַפְּנִימִית וּכְמוֹ שֶׁכָּתוּב (דְּבָרִים לב. ט) כִּי חֵלֶק ה' עַמּוֹ, וְיָדוּעַ דְּשֵׁם הוי' הוּא בְּסֵדֶר הִשְׁתַּלְשְׁלוּת י' בְּחָכְמָה, ה' בְּבִינָה ו' בְּזָ"א [בִּזְעֵיר אַנְפִּין] ה' אַחֲרוֹנָה בְּמַלְכוּת, וּבְכָל מָקוֹם שֵׁם הוי' הוּא בִּבְחִינַת הַפְּנִימִית דַּוְקָא, וּכְתִיב (דְּבָרִים יד. ב) וּבְךָ בָּחַר ה', כִּי יַעֲקֹב בָּחַר לוֹ יָהּ (תְּהִלִּים קלה. ד) וְעִנְיַן הַבְּחִירָה הוּא הַבֵּירוּר שֶׁמְּבָרֵר וּבוֹחֵר בָּזֶה דַוְקָא כִּדְאִיתָא בְּמִדְרָשׁ רַבָּה (בַּמִּדְבָּר פַּרְשָׁה כ) מֶלֶךְ שֶׁבֵּירֵר לְעַצְמוֹ מָנָה וְעָמַד אֶחָד וְאָמַר גְּנַאי עָלֶיהָ כְּלוּם יֵשׁ לוֹ חַיִּים, וְאֵלּוּ (יִשְׂרָאֵל) חֶבְלָתוֹ וְנַחֲלָתוֹ וּסְגוּלָּתוֹ שֶׁנֶּאֱמַר (דְּבָרִים לב. ט) כִּי חֵלֶק ה' עַמּוֹ יַעֲקֹב חֶבֶל נַחֲלָתוֹ (שְׁמוֹת יט. ה) וִהְיִיתֶם לִי סְגוּלָּה, וְזֶהוּ הַבְּחִירָה מַה שֶּׁבֵּירֵר יִשְׂרָאֵל לְעַצְמוֹ דְּזֶה שַׁיָּיךְ בִּבְחִינַת פְּנִימִית דַּוְקָא,
וּכְמוֹ כֵן נִשְׁמוֹת יִשְׂרָאֵל בָּחֲרוּ בַה', וּכְמוֹ שֶׁכָּתוּב (אֵיכָה ג. כד) חֶלְקִי ה' אָמְרָה נַפְשִׁי, וְאִיתָא בְּמִדְרָשׁ רַבָּה (אֵיכָה עַל פָּסוּק זֶה) רַבִּי אַבוּהָ בְּשֵׁם רַבִּי יוֹחָנָן אָמַר לַמֶּלֶךְ שֶׁנִּכְנַס לַמְּדִינָה וְהָיוּ עִמּוֹ דּוּכְסִין וְאִיפַרְכִין וְאִיסְטַרְטִילוּטִין וְהָיוּ גְּדוֹלֵי מְדִינָה יוֹשְׁבִים בְּאֶמְצַע הַמְּדִינָה, חַד אָמַר אֲנָא נָסִיב דּוּכְסִין לְגַבִּי, חַד אָמַר אֲנָא נָסִיב אִיפַרְכִין לְגַבִּי, וְחַד אָמַר אֲנָא נָסִיב אִיסְטַרְטִילוּטִין לְגַבִּי, הָיָה פִּקֵּחַ אֶחָד לְשָׁם אָמַר אֲנָא נָסִיב לְמַלְכָּא דְּכוּלְּהוּ מִתְחַלְּפִין וּמַלְכָּא אֵינוֹ מִתְחַלֵּף, כַּךְ עַכּוּ"ם מֵהֶן עוֹבְדִין לְחַמָּה מֵהֶן עוֹבְדִין לִלְבָנָה וּמֵהֶן עוֹבְדִין לְעֵץ וָאֶבֶן אֲבָל יִשְׂרָאֵל אֵינָן עוֹבְדִין אֶלָּא לְהַקָּדוֹשׁ בָּרוּךְ הוּא הֲדָא הוּא דִכְתִיב חֶלְקִי ה' אָמְרָה נַפְשִׁי.
[cup]וְזֶהוּ[/cup] שֶׁהַתּוֹרָה נִיתְּנָה לָנוּ, דְּאוֹרַיְיתָא מֵחָכְמָה נָפְקַת, דְּעִנְיַן הַתּוֹרָה וּמִצְוֹת הוּא מִצַּד הַפְּנִימִית שֶׁשָּׁם הוּא הַחֶשְׁבּוֹן וְהַמְּדִידָה שֶׁצָּרִיךְ לִהְיוֹת בְּאוֹפֶן כַּךְ וְכַךְ דַּוְקָא, וְנִיתְּנָה הַתּוֹרָה לְיִשְׂרָאֵל דַּוְקָא לְפִי שֶׁהֵם מִבְּחִינַת פְּנִימִית.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]הַחוֹטֵא הַחוֹשֵׁב דִּכְשֶׁיֵּלֵךְ בִּשְׁרִירוּת לִבּוֹ שָׁלוֹם יִהְיֶה לוֹ בְּדוּגְמַת הַסִּטְרָא אָחֳרָא שֶׁמְּקַבֶּלֶת מֵהַמַּקִּיף שֶׁפַע מְרוּבֶּה טוֹעֶה הוּא, לִהְיוֹת דְּיִשְׂרָאֵל הֵם מֵהַפְּנִימִית, וְהַיְינוּ דְּהֵן בְּחִירַת הָעַצְמוּת כִּבְיָכוֹל בְּיִשְׂרָאֵל וְהֵן בְּחִירַת יִשְׂרָאֵל בְּעַצְמוּת בָּרוּךְ הוּא הוּא מִצַּד הַפְּנִימִיּוּת.[kitzur_end]
