שמגיע עד אור הכתר דוקא שנק' עתיקא שכולו רחמים פשוטים בלי נטיה כלל לדין (וכמ"ש דלית שמאלא בהאי עתיקא כו' כמשי"ת) וכמ"ש בזהר ע"פ (שמות יז, ז) היש הוי"ה בקרבנו אם אין שהוא אין דכתר כמ"ש (איוב כח, יב) והחכמה מאין תמצא כידוע והענין הוא לפי שבחי' ת"ת הוא בבחי' קו האמצעי שנק' בריח התיכון שמבריח מן קצה התחתון לקצה העליון שהוא בחי' פנימית הכתר שלמעלה מהתחלקות לב' קוין דימין ושמאל שהן חו"ג בכלל ע"כ הוא שם בבחי' רחמים וחסדים פשוטים שאין שם נטיה לדין כלל (לפי שבחי' ת"ת כלול מדבר והפכו חו"ג שזה בא בהכרח מלמעלה מהתחלקות דחו"ג לגמרי ע"כ נכללין שניהם כאחד וכענין עושה שלום במרומיו כו' (איוב כה, ב) מצד האור העליון משניהם כו' והוא שם מ"ה מלגיו כו' ששרשו בפנימית הכתר דוקא שבו אא"ס עצמו שמושלל מכל הי"ס ולאו מכל אינון מדות כלל וזהו מי כמוך באלם הוי"ה דוקא (כמ"ש בפ' נח באריכות) וא"כ לפי זה גם שם הוי"ה שבבחי' ת"ת שכלול מחו"ג כמו שהוא בשרשו בכתר עצמו הרי הוא ג"כ כולו רחמים וחסדים פשוטים כמו אור הכתר עצמו ממש ולז"א להתיצב על הוי"ה בבחי' מנגד מכל צד לפי שאין בשם הוי"ה זה שום נטיה לדין כלל וזהו ג"כ מ"ש הכצעקתה הבאה אלי למנגד גמור מכל צד מן הקצה לקצה מטעם זה עצמו משום דאלי זה היינו אלי ממש שהוא בחי' שם הוי"ה שבפנימית הכתר ששם גלוי אא"ס עצמו כמ"ש בר"מ אא"ס מלגיו כ"ע מלבר כו' וכמו שאמר (שמות יט, ד) ואשא אתכם על כנפי נשרים ואביא אתכם אלי אלי ממש לעצמות אא"ס שבכתר כנ"ל והוא לפי שבחי' הרחמים שבת"ת שנק' כנפי נשרים כידוע עולה למעלה ברחמים הגדולים הפשוטים שבעצמות אא"ס ממש שכולו רחמים פשוטים וד"ל): 
(ו) [cup]אך[/cup] הנה עדיין יש להבין בזה ב' דברים הא' איך יש ביכולת כח הדין שבשם אלקים שעולה מלמטה למע' שהוא מדרגה האחרונה ביותר בבחי' הצמצום והסתר אא"ס שבאצי' להיות עולה למעלה כ"כ להתיצב על הוי"ה למנגד והוי"ה זה שרשו בעתיקא עצמו שכולו רחמים פשוטים דוקא וגם מטעם זה יפלא במ"ש הכצעקתה הבאה אלי למנגד כו' והב' מ"ש בתחלה ארדה נא ואראה כו' ירידה זאת היתה בשם הוי"ה שהזכיר תחלה באמרו ויאמר הוי"ה זעקת כו' והוא שאמר ארדה נא כו' כנ"ל וזהו פלא גדול איך שייך ירידה כזאת משם הוי"ה שכולו רחמים פשוטים שאין בו נטיה לדין כלל שלמעלה מבחי' הגבו' דשם אדני דמשמע שירד למטה לראות הכצעקתה כו' היינו לראות ולפקח בעין השכל אם עשו רק כצעקת מדה"ד עליהם אם לא עשו כו' ומאחר שכולו רחמים וחסד ואין בו נטיה לקו השמאל כח הדין כלל וכלל כנ"ל א"כ איך יוכל להטות עצמו לראות ולפקח בכח הדין באופן שיחייב אותם אם עשו רע ואיך יפול ספק אצלו פן עשו רע אשר מחמת זה ירצה לירד ולראות הדבר על אמתתו להוציא לאור את הספק שלו מאחר שכולו רחמים פשוטים ועצמיים ודאי לא ידין לכף חוב ודין כלל (כענין לא הביט און ביעקב (במדבר כג, כא) וכה"ג) ולא יפול ספק כלל פן עשו
