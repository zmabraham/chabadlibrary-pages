שהוא נמשך מאבא וכנודע וכמש"ל ב' הרמ"ז פ' בהעלותך דקמ"ט, עד"ז ממש ה"ע המשכ' שמן הטוב מבחי' יסוד דעתיק כו' ועד"ז נמשך ג"כ לעתיק גם מחכמה דעקודים וכמ"ש במ"א בביאור ע"פ ונקדשתי בתוך בנ"י, ובפרט לפמש"ל ב' המגיד נ"ע שבחי' עקודי' זה"ע שישנה בדרך קצרה כו' והוא ממש ענין המשכ' השמן הטוב ע"י היסוד כו', ובעה"מ שי"א פ"ח פי' שהשמן הטוב נמשך מיסוד דא"ק. ומכ"ז מובן מה דבהדרוש מבואר שבחי' שמן זהו מעשה המצות, והאור זהו התורה, דלכאורה שמן כיון שהוא מח"ע איך יקרא על מעהמ"צ שהם רצה"ע, אלא כי הרי השמן הטוב ראשית התגלותו הוא כמ"ש היורד על הראש דהיינו גולגלתא דא"א שהוא רצה"ע מקור המצות, וכ"פ בפ' בלק דקפ"ז ע"פ ושמן על ראשך אל יחסר, דהא נהורא דבראשו אצטריך למשחא ואינון עובדין טבאן כו' והיינו כמו שלמעלה נמשך בחי' כשמן הטוב על הראש כנ"ל כמ"כ ג"כ בבחי' ישראל לי ראש המבואר בד"ה אני דפ' ציצית נמשך בחי' כשמן הטוב ע"י מעשה המצות, ועמ"ש בד"ה ענין חנוכה אע"פ שהשמן אין בו אור מצ"ע כו' ככה ממש היא המצוה כו' רצה"ע כו' וסד"ה הנ"ל בבחי' מקיף כו', ויובן עוד עפמ"ש במ"א בענין ג' בחי' אמ"ר בד"ה הללוי' הללו אל בקדשו בסידור, ובד"ה קא קמיפלגי במתיבתא דרקיע, והנה ג"ע נק' מתיבתא דרקיע ע"ד ועל ראשי החי' רקיע כעין הקרח כו' ונמשך ע"י ונהר יוצא מעדן כו' והיינו בחי' מים, ועדן עצמו בחי' אור, והשמן נק' שמן למאור שהמאור הוא למעלה מבחי' אור והוא ענין עדן עילאה סתימאה הנז' באדר"ז דר"צ ע"א, ומזה יובן ענין המבואר בתוך הדרוש בענין שש"מ שהיא יותר מכל בחי' הג"ע דהיינו מרוב כל, והנה ענין הנז' במדרש והשמן אינו מתערב כו' יובן עד"מ מהמבואר בהגהות לד"ה ראה ריח בני בענין י"א סממני הקטורת בענין לבונה שהוא המקיף לי"ס דנוגה כי אין קדש מתערב בחול כמ"ש בע"ח ש"י פי"א, ועד"ז יובן אפי' בע"ס דקדושה עם היות שבהם איהו וגרמוהי חד היינו רק הארה מתלבש בהם אבל אא"ס ב"ה נק' חד ולא בחושבן והיינו בחי' עתיק שאינו בחשבון הע"ס, דכמו שאין קדש מתערב בחול, כך עד"ז ארז"ל קדושתי למעלה מקדושתכם, למעלה דוקא שאינו יכול להתלבש ולהתערב כלל בע"ס דאצי' הנק' קדושתכם כ"א בחי' מקיף מלמעלה, ועמ"ש סד"ה לסוסתי ברכבי פרעה בשה"ש שהמשל הנ"ל מהלבונה לע"כ כן ערך המקיף העליון הנמשך ע"י המצות לנש"י כו' נז' שם בפי' שכן הוא, וזה"ע אשר קדשנו במצותיו דעם היות קדש ישראל לה' מצ"ע, אך ע"י המצות קדשנו בבחי' קדושתו ית' שלמעלה מקדושת נש"י, וזה"ע השמן שהוא בחי' שאינו מתערב כלל בשאר משקין כו' וכמ"ש כשמן הטוב על הראש, על דייקא, וזהו ענין אתה ה' לבדך, ה' בדד ינחנו, וזהו ענין עד יערה עלינו רוח ממרום, דהנה פי' רם על כל גוים רם דוקא היינו לפי שמקבלים מבחי' המקיף דוקא בחי' לבונה כנז' בד"ה ויקח מן הבא בידו, כעד"ז בחי' המקיף העליון שאפי' לגבי ע"ס דאצי' הוא בבחי' רם חד ולא בחושבן קדושתי למעלה מקדושתכם הנה מבחי' זו יהי' הערה והגילוי וזהו בדד ינחנו וכנ"ל, וזהו שנחלקו רבנן וריב"ן רפ"ק דשבת אם שמן שצף ע"ג יין אם הוי אפי' חיבור זל"ז, אף שהיין ג"כ מדרגה גבוה כי נכנס יין יצא סוד גילוי עלמא דאתכסי' כו' מ"מ ס"ל לרבנן דאינו חיבור, ומ"ד דהוי חיבור י"ל כי בהשתל' השמן בח"ע הרי או"א הם תרין
