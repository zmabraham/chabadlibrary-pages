# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר כו. > פֶּרֶק ב

וּבֵיאוּר הָעִנְיָן הוּא דְּהִנֵּה הָאֱלֹקִים עָשָׂה אֶת הָאָדָם יָשָׁר וּמִצַּד הַכֹּחַ וְהַהֶרְגֵּשׁ הָאֱלֹקִי הַנִּיתַּן לוֹ הָיָה צָרִיךְ לֵידַע וּלְהַרְגִּישׁ אֶת הָאֱמֶת אֵיךְ שֶׁכָּל טוּבוֹ בָּעוֹלָם הוּא בִּבְחִינַת הַטּוֹב הָרוּחָנִי וְלַעֲלוֹת בְּמַעֲלַת הַדְּבֵקוּת בֵּאלֹקוּת וּבְתוֹרָתוֹ וּבְמִצְוֹתָיו וְאֵיךְ שֶׁהַהֵיפֶךְ רַע וּמַר בְּתַכְלִית וּכְכָל הַחֶשְׁבּוֹנוֹת שֶׁנִּתְבָּאֲרוּ לְעֵיל אֵיךְ שֶׁהֲסָתַת הַיֵּצֶר הוּא שֶׁקֶר גָּמוּר וְרָע, דְּבֶאֱמֶת הָיָה צָרִיךְ הָאָדָם לְהַרְגִּישׁ כָּל זֶה מִצַּד הַהֶרְגֵּשׁ אֱלֹקִי שֶׁבּוֹ וּמֵאַיִן הִיא הַסִּבָּה שֶׁנִּפְתֶּה לַהֲסָתַת הַיֵּצֶר, אָמְנָם זֶהוּ שֶׁגָּרַם חֵטְא דְּעֵץ הַדַּעַת טוֹב וָרָע,
דְּהִנֵּה גַּם קוֹדֶם הַחֵטְא הָיָה בוֹ יֵצֶר הָרָע וּכְמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (בְּרָכוֹת סא. א) עַל פָּסוּק וַיִּיצֶר בִּשְׁנֵי יוּדִי"ן שְׁנֵי יְצָרִים כו' אֶלָּא שֶׁאָז הָיָה בוֹ תּוֹקֶף הַהֶרְגֵּשׁ הָאֱלֹקִי עַד שֶׁלֹּא הִרְגִּישׁ עִנְיָנִים הַטִּבְעִיִּים וְכַנַ"ל בְּשֵׁם הַמּוֹרֵה נְבוּכִים שֶׁהָיָה בוֹ יְדִיעַת הַמּוּשְׂכָּלוֹת וְלֹא הִרְגִּישׁ כְּלָל יְדִיעַת הַמְפוּרְסָמוֹת מִפְּנֵי תּוֹקֶף הַהֶרְגֵּשׁ בְּעִנְיְנֵי אֱלֹקוּת, וּלְאַחַר חֵטְא עֵץ הַדַּעַת נַעֲשָׂה לְהֵיפֶךְ שֶׁנַעֲשָׂה בּוֹ תּוֹקֶף הַהֶרְגֵּשׁ הַטִּבְעִי, הַיְינוּ שֶׁמַּרְגִּישׁ מְאֹד הַטּוֹב שֶׁבְּעִנְיָנִים הַגַּשְׁמִים וְכָל עִנְיָן טִבְעִי וְגַשְׁמִי שֶׁבָּעוֹלָם הוּא מַרְגִּישׁ מְאֹד, וְהַהֶרְגֵּשׁ הַזֶּה הוּא גוֹרֵם לוֹ הֶעְדֵּר הַהֶרְגֵּשׁ הָאֱלֹקִי לֵידַע וּלְהַרְגִּישׁ אֲמִיתַּת טוֹבָתוֹ שֶׁנִּתְבָּאֵר לְעֵיל.
[cup]וְזֶהוּ[/cup] עִנְיַן הַשְּׁטוּת שֶׁמִּצַּד הַיֵּצֶר הָרָע, שֶׁמִּפְּנֵי תּוֹקֶף הַהֶרְגֵּשׁ בְּעִנְיָנִים הַגַּשְׁמִים הוּא מַשְׁטֶה אֶת עַצְמוֹ בְּיוֹתֵר וְיוֹתֵר הוֹלֵךְ וְיוֹרֵד רַחֲמָנָא לִצְּלָן וְאֵינוֹ מִתְנַהֵג בְּדַעַת וְחֶשְׁבּוֹן אֲמִיתִּי וְזֶהוּ הַשְּׁטוּת דִּלְעוּמַּת זֶה שֶׁהוּא לְמַטָּה מִן הַדַּעַת, וְהַהֵיפֶךְ מִזֶּה הוּא הַשְּׁטוּת דִּקְדוּשָּׁה שֶׁהוּא גַם כֵּן מַה שֶּׁמִּתְנַהֵג שֶׁלֹּא עַל פִּי דַעַת וְחֶשְׁבּוֹן אֶלָּא שֶׁהוּא לְמַעְלָה מִטַּעַם וָדַעַת, וְהַיְינוּ שֶׁאֵינוֹ עַל פִּי הַדַּעַת וְהַהֶרְגֵּשׁ הַטִּבְעִי דְּעֵץ הַדַּעַת טוֹב וָרָע וְגַם אֵינוֹ עַל פִּי הַדַּעַת דִּקְדוּשָּׁה גַּם כֵּן, וְהַיְינוּ שֶׁאֵינוֹ נִכְנָס כְּלָל בְּחֶשְׁבּוֹנוֹת עַל פִּי הַדַּעַת וְהַשָּׂגָה שֶׁצָּרִיךְ לִהְיוֹת כַּךְ וְכַךְ אֶלָּא עוֹשֶׂה כָּל הַדְּבָרִים מִצַּד תּוֹקֶף הַנֶּפֶשׁ שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת,
וּכְיָדוּעַ שֶׁבִּנְשָׁמוֹת יֵשׁ ב' מַדְרֵיגוֹת הָא' כֹּחוֹת הַגְּלוּיִם שֶׁל הַנְּשָׁמָה שֶׁהֵן בְּחִינַת מוֹחִין וּמִדּוֹת וְהַב' הוּא עֶצֶם הַנְּשָׁמָה שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת,
וְהִנֵּה בְּחִינָה זוֹ שֶׁהָיְתָה בְּהִתְגַּלּוּת וּבְהִתְגַּבְּרוּת בְּאָדָם הָרִאשׁוֹן קוֹדֶם הַחֵטְא שֶׁהָיָה כָּל עִיּוּנוֹ בֵּאלֹקוּת שֶׁזֶּהוּ מִצַּד הַמּוֹחִין דְּנֶפֶשׁ הָאֱלֹקִית שֶׁהָיוּ מְאִירִים בּוֹ בְּהִתְגַּלּוּת,
וּכְמוֹ כֵן יֵשׁ נְשָׁמוֹת שֶׁגַּם לְאַחַר הַחֵטְא לֹא יָרְדוּ מִמַּדְרֵיגָתָן כָּל כַּךְ וּמֵאִיר בָּהֶם אוֹר נַפְשָׁם הָאֱלֹקִית וְיֵשׁ לָהֶם דַּעַת וְהַשָּׂגָה בֵּאלֹקוּת וּבָזֶה כָּל עִסְקָם תָּמִיד וְאֵינָם שַׁיָּיכִים לְעִנְיָנִים אֲחֵרִים לְהַרְגִּישׁ כָּל כַּךְ אֶת הַטּוֹב הַטִּבְעִי הַבַּהֲמִי מֵעִנְיְנֵי עוֹלָם הַזֶּה, וַהֲגַם שֶׁאֵינוֹ בְּאוֹפֶן כְּמוֹ אָדָם הָרִאשׁוֹן קוֹדֶם הַחֵטְא שֶׁלֹּא הָיָה לוֹ הֶרְגֵּשׁ הַטִּבְעִי כְּלָל מִכָּל מָקוֹם אֵין בָּהֶם הֶרְגֵּשׁ הַטִּבְעִי כָּל כַּךְ, וְאֵינָם נִפְתִּים בִּשְׁטוּתֵי הַיֵּצֶר הָרָע וְיוֹדְעִים וּמַשִּׂיגִים אֶת הָאֱמֶת כְּפִי הַחֶשְׁבּוֹנוֹת הַנַ"ל.

[kitzur_begin]הָאֱלֹקִים נָתַן לְהָאָדָם הֶרְגֵּשׁ אֱלֹקִי אֲשֶׁר מִצַּד זֶה צָרִיךְ לֵידַע שֶׁטּוֹבָתוֹ הָאֲמִיתִּית הִיא בַּתּוֹרָה וּמִצְוֹת וַהֲסָתַת הַיֵּצֶר הָרָע שֶׁקֶר וְרָע, אָמְנָם הַהֶרְגֵּשׁ הַטִּבְעִי בְּסִבַּת חֵטְא עֵץ הַדַּעַת שׁוֹמֵעַ לִשְׁטוּתֵי הַיֵּצֶר הָרָע שֶׁלְּמַטָּה מִטַּעַם וָדַעַת, וּשְׁטוּת דִּקְדוּשָּׁה לְמַעְלָה מִטַּעַם וָדַעַת. אֵלּוּ שֶׁמֵּאִיר בָּהֶם אוֹר נַפְשָׁם הָאֱלֹקִית.[kitzur_end]
