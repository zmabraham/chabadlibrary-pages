מַאֲמָר ה.
וְהִנֵּה מַה שֶּׁהַיֵּצֶר הָרָע מַבְטִיחוֹ שֶׁגַּם אִם יַעֲשֶׂה מַעֲשֶׂה הָרָע מִכָּל מָקוֹם טוֹב יִהְיֶה לוֹ, וּכְמוֹ שֶׁכָּתוּב (דְּבָרִים כט. יח) וְהִתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר גו', גַּם זֶה מִפִּתּוּי הַיֵּצֶר אֲשֶׁר נוֹפֶת תִּטּוֹפְנָה שִׂפְתֵי זָרָה גו' וְאַחֲרִיתָהּ מָרָה כַלַּעֲנָה (מִשְׁלֵי ה. ג-ד).
[cup]וְהִנֵּה[/cup] צָרִיךְ לְהַקְדִּים תְּחִלָּה פֵּירוּשׁ הַפָּסוּק וְהִתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר שָׁלוֹם יִהְיֶה לִּי כִּי בִּשְׁרִירוּת לִבִּי אֵלֵךְ לְמַעַן סְפוֹת הָרָוָה אֶת הַצְּמֵאָה וּמַשְׁמָעוּת הַפָּסוּק דְּכִי בִּשְׁרִירוּת לִבִּי הוּא נְתִינַת טַעַם עַל מַה שֶּׁכָּתוּב קוֹדֶם וְהִתְבָּרֵךְ בִּלְבָבוֹ, דְּמִשּׁוּם זֶה שָׁלוֹם יִהְיֶה לִּי לְפִי שֶׁבִּשְׁרִירוּת לִבִּי אֵלֵךְ, וְלִכְאוֹרָה אֵינוֹ מוּבָן כְּלָל, דְּאֵיךְ יְדַמֶּה בְנַפְשׁוֹ דְּדַוְקָא כַּאֲשֶׁר יֵלֵךְ בִּשְׁרִירוּת לִבּוֹ עַל יְדֵי זֶה יִהְיֶה שָׁלוֹם לוֹ, וְגַם מַהוּ פֵּירוּשׁ לְמַעַן סְפוֹת הָרָוָה כו'.
[cup]וְרַשִׁ"י[/cup] (סַנְהֶדְרִין עו. ב) פֵּירֵשׁ סְפוֹת לְשׁוֹן חִיבּוּר, וּכְמוֹ סְפוּ שָׁנָה עַל שָׁנָה (יְשַׁעְיָה כט. א)
וְהָרְדַ"ק [רַבִּי דָּוִד קִמְחִי] פֵּירֵשׁ לְשׁוֹן תּוֹסֶפֶת וּכְמוֹ לִסְפּוֹת עוֹד (בַּמִּדְבָּר לב. יד) עֹלוֹתֵיכֶם סְפוּ עַל זִבְחֵיכֶם (יִרְמְיָה ז. כא)
וּפֵירוּשׁ רָוָה וּצְמֵאָה פֵּרֵשׁ רַשִׁ"י (שָׁם) רָוָה עַכּוּ"ם [עוֹבְדֵי כּוֹכָבִים וּמַזָּלוֹת] שֶׁשְּׂבֵעִים וְאֵינָן צְמֵאִים לְיוֹצְרָם. צְמֵאָה זוֹ כְּנֶסֶת יִשְׂרָאֵל שֶׁצְּמֵאָה וּתְאֵבָה לְיִרְאַת יוֹצְרָהּ וּלְקַיֵּים מִצְוֹתָיו. וְהִנֵּה יָדוּעַ דִּכְנֶסֶת יִשְׂרָאֵל לְמַעְלָה שֶׁהוּא בְּחִינַת מַלְכוּת הִיא נִשְׁמוֹת יִשְׂרָאֵל שֶׁלְּמַטָּה וְנִקְרֵאת גַּם כֵּן הַצְּמֵאָה שֶׁצְּמֵאָה לְיוֹצְרָהּ.
[cup]וְהָעִנְיָן[/cup] הוּא דְּהִנֵּה בְּחִינַת מַלְכוּת הִיא מָקוֹר הַמְהַוֶּוה וּמְחַיֶּה עוֹלָמוֹת וּכְמוֹ שֶׁכָּתוּב (תְּהִלִּים קמה. יג) מַלְכוּתְךָ מַלְכוּת כָּל עוֹלָמִים, דְּכָל עוֹלָמִים שֹׁרֶשׁ הִתְהַוּוּתָם מִבְּחִינַת מַלְכוּת,
וּכְתִיב (נְחֶמְיָה ט. ו) וְאַתָּה מְחַיֶּה אֶת כֻּלָּם, וְאַתָּה הֵן הָאוֹתִיּוֹת מֵא' עַד ת', וְהֵן אוֹתִיּוֹת הַדִּבּוּר דִּבְחִינַת מַלְכוּת שֶׁמֵּהֶם נִמְשָׁךְ חַיּוּת הָעוֹלָמוֹת, וְאַל תִּקְרֵי מְחַיֶּה אֶלָּא מְהַוֶּוה כו' כְּמוֹ שֶׁכָּתוּב בְּמָקוֹם אַחֵר, וְיָדוּעַ דִּבְחִינָה זוֹ שֶׁהוּא מָקוֹר הַמְהַוֶּוה וּמְחַיֶּה עוֹלָמוֹת הוּא רַק הֶאָרָה לְבָד שֶׁבְּאֵין עֲרוֹךְ לְגַבֵּי עַצְמוּת, דַּהֲרֵי מֵעַצְמוּת אוֹר אֵין סוֹף לֹא הָיָה אֶפְשָׁר לִהְיוֹת הִתְהַוּוּת הָעוֹלָמוֹת וְנִבְרָאִים בַּעֲלֵי גְבוּל, לְפִי שֶׁגְּבוּל וּבִלְתִּי בַּעַל גְּבוּל הֵם בְּאֵין עֲרוֹךְ כְּלָל כו', וּכְמוֹ שֶׁכָּתוּב בְּכַמָּה דוּכְתֵּי בַּאֲרִיכוּת, כִּי אִם רַק מֵהֶאָרָה לְבָד נַעֲשׂוּ וְנִתְהַוּוּ הָעוֹלָמוֹת,
וְזֶהוּ מַה שֶּׁבְּחִינַת מַלְכוּת הוּא בְּחִינַת שֵׁם, מֶלֶךְ שְׁמוֹ נִקְרָא עֲלֵיהֶם, דְּהַשֵּׁם הוּא רַק הֶאָרָה בִּלְבָד, וּכְמוֹ שֵׁם הָאָדָם שֶׁאֵינוֹ עַצְמוּת הָאָדָם, וּכְמוֹ כֵן בְּחִינַת הַחַיּוּת שֶׁל הָעוֹלָמוֹת הוּא רַק בְּחִינַת שֵׁם וְהֶאָרָה לְבָד.

[kitzur_begin]מֵהֶמְשֵׁךְ הַכָּתוּב כִּי בִּשְׁרִירוּת לִבִּי אֵלֵךְ מַשְׁמָע שֶׁהוּא נְתִינַת טַעַם עַל וְהִתְבָּרֵךְ שָׁלוֹם יִהְיֶה לִּי, וּמַהוּ סְפוֹת הָרָוָה אֶת הַצְּמֵאָה, רָוָה עַכּוּ"ם, צְמֵאָה נִשְׁמוֹת יִשְׂרָאֵל דְּשָׁרְשָׁם מִסְּפִירַת הַמַּלְכוּת מְקוֹר הָעוֹלָמוֹת וְהַנִּבְרָאִים, שֶׁהִיא הֶאָרָה לְבָד שֶׁבְּאֵין עֲרוֹךְ לְהָעַצְמוּת וְלָכֵן נִקְרֵאת שֵׁם.[kitzur_end]
