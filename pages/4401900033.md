# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר א. > פֶּרֶק ד

וְהִנֵּה עוֹד יִתְרוֹן בִּידִיעַת וְהַשָּׂגַת פְּנִימִיּוּת הַתּוֹרָה שֶׁעַל יְדֵי זֶה הוּא בָא לִידֵי אַהֲבָה וְיִרְאָה, שֶׁזֶּהוּ מַה שֶּׁכָּתוּב (דְּבָרִים ד. לט) וְיָדַעְתָּ הַיּוֹם וַהֲשֵׁבוֹתָ אֶל לְבָבֶךָ כו' שֶׁמְּבִיאָה לְלֵב שָׁלֵם כו', וּכְמוֹ שֶׁכָּתוּב מִזֶּה בְּאִגֶּרֶת הַקֹּדֶשׁ בְּסוֹפוֹ בְּדִבּוּר הַמַּתְחִיל לְהָבִין מַה שֶּׁכָּתוּב בִּפְרִי עֵץ חַיִּים וּבְלִקּוּטֵי תוֹרָה בְּהַבֵּיאוּר דִּוְלֹא תַשְׁבִּית, וְאֵין כַּאן מָקוֹם לְהַאֲרִיךְ בָּזֶה, וְאַחַר הַטּוֹב הַזֶּה צָרִיךְ הָאָדָם לִהְיוֹת נִמְשָׁךְ, וּבָזֶה יִהְיֶה כָּל חֶפְצוֹ וּמְגַמָּתוֹ, וּבָזֶה יִהְיֶה עוֹנֶג נַפְשׁוֹ, לְפִי שֶׁהַטּוֹב הַזֶּה הוּא לְפִי מַעֲלַת מַהוּתוֹ, וּבָזֶה נִשְׁלָם הַכַּוָּונָה שֶׁבָּא אֶל הַתַּכְלִית שֶׁבִּשְׁבִילוֹ נִבְרָא,
מַה שֶּׁאֵין כֵּן כְּשֶׁנִּמְשָׁךְ אַחֲרֵי הָעִנְיָנִים הַגַּשְׁמִים וְהַגּוּפְנִים, הֲרֵי לֹא זוֹ בִּלְבָד שֶׁלֹּא נִשְׁלַם הַכַּוָּונָה וְהַתַּכְלִית שֶׁנִּבְרָא בִּשְׁבִיל זֶה, הֲרֵי אַדְרַבָּה הוּא מוֹרִיד נַפְשׁוֹ מִמַּעֲלָתוֹ וּמַהוּתוֹ הַנַּעֲלָה וְיוֹרֵד לְמַדְרֵיגַת הַבַּעַל חַי הַגָּרוּעַ וּפָחוּת מִמֶּנּוּ בְּאֵין עֲרוֹךְ, וְהוּא בִּשְׁנֵי דְבָרִים, הָא' כְּיָדוּעַ שֶׁכָּל דָּבָר שֶׁהָאָדָם מִתְקַשֵּׁר בּוֹ הֲרֵי הוּא נַעֲשֶׂה כְּמַהוּת אוֹתוֹ הַדָּבָר וּכְשֶׁמְּקַשֵּׁר נַפְשׁוֹ וּרְצוֹנוֹ בְּדוֹמֵם אוֹ בְצוֹמֵחַ אוֹ בְחַי לְהִתְעַנֵּג מֵהֶם, הֲרֵי הוּא נַעֲשֶׂה כְּמוֹהֶם (וְעוֹד לְמַטָּה מֵהֶם שֶׁהֲרֵי הוּא מְקַבֵּל מֵהֶם, וְהוּא הַתַּעֲנוּג שֶׁמִּתְעַנֵּג מֵהֶם) וַהֲרֵי הוּא יוֹרֵד מִמַּעֲלַת הַמְדַבֵּר לְמַדְרֵיגַת הַדּוֹמֵם צוֹמֵחַ חַי, שֶׁזֶּה יְרִידָה גְדוֹלָה מְאֹד,
וְהַב' שֶׁמִּצַּד מַעֲלָתוֹ (מַעֲלַת הַמְדַבֵּר) הֲרֵי שִׁפְלוּתוֹ הוּא בְיוֹתֵר, דְּבִירִידַת וְהַמְשָׁכַת נַפְשׁוֹ שֶׁנִּמְשָׁךְ אַחֲרֵי עִנְיָנִים פְּחוּתִים וּשְׁפָלִים, וּבָהֶם הוּא מוֹצֵא תַּעֲנוּג נַפְשׁוֹ, הֲרֵי הוּא גָרוּעַ מֵהַבְּהֵמָה שֶׁהַבְּהֵמָה אֵין לָהּ דַּעַת וְאֵינָהּ יוֹדַעַת מֵעִנְיָנִים נַעֲלִים כְּלָל, אֲבָל הָאָדָם שֶׁנִּיתַּן לוֹ שֵׂכֶל וָדַעַת כְּשֶׁעוֹשֶׂה מַעֲשֵׂה בְהֵמָה הֲרֵי הוּא גָרוּעַ מִמֶּנָּה (וּבִפְרָט לְפִי מַה שֶּׁנִּתְבָּאֵר לְעֵיל (פֶּרֶק א) דְּקַדֵּשׁ עַצְמְךָ בְּמוּתָּר לְךָ הוּא מִצְוַת עֲשֵׂה דְאוֹרַיְיתָא וּכְשֶׁעוֹשֶׂה עִנְיָנִים הַגַּשְׁמִים בִּשְׁבִיל הַנָּאָתוֹ וְתַעֲנוּג נַפְשׁוֹ הֲרֵי הוּא יוֹרֵד בְּרַע גָּמוּר לְפִי שָׁעָה עַד אֲשֶׁר יָשׁוּב וְיַחֲזוֹר לַעֲבוֹדַת ה' וּלְתוֹרָתוֹ כְּמוֹ שֶׁכָּתוּב בְּסֵפֶר שֶׁל בֵּינוֹנִים (פֶּרֶק ז הַנַ"ל פֶּרֶק א) הֲרֵי עַל יְדֵי תַּאֲווֹת נַפְשׁוֹ הוּא גָרוּעַ עוֹד גַּם מִבַּעֲלֵי חַיִּים הַטְּמֵאִים כו' וּכְמוֹ שֶׁיִּתְבָּאֵר), וְאִם כֵּן הֲרֵי הֲסָתַת הַיֵּצֶר הָרָע שֶׁמְּסִיתוֹ בִּדְבָרִים הַנַ"ל, הֲרֵי זֶה שְׁטוּת נוֹרָא שֶׁמַּטֵּהוּ מִדֶּרֶךְ הָאֱמֶת וְהַצֶּדֶק, וְרַק מְסַמֵּא וּמְעַוֵּור עֵינָיו,
וְכַאֲשֶׁר מְסִיתוֹ לוֹמַר שֶׁזֶּה טוֹב לְפָנָיו, זֹאת יָשִׂים אֶל לִבּוֹ, הֲכִי זֶהוּ הַטּוֹב שֶׁלִּי לְפִי מַעֲלַת וּמַדְרֵיגַת נַפְשִׁי הַנִּיתַּן לִי, וְכִי זֶהוּ כַּוָּונַת בְּרִיאָתִי בָּעוֹלָם, וְאַדְרַבָּא בְּמַה שֶּׁאוֹמֵר לוֹ שֶׁתַּאֲוָה הוּא לָעֵינַיִם, הֲרֵי מִפְּנֵי זֶה עַצְמוֹ צָרִיךְ לְהִתְרַחֵק מִזֶּה בְּתַכְלִית שֶׁלֹּא לְהוֹרִיד נַפְשׁוֹ חַס וְשָׁלוֹם כַּנַ"ל, כִּי אִם הַטּוֹב שֶׁלּוֹ יִהְיֶה בְּעִנְיָנִים רוּחָנִים וְנַעֲלִים, שֶׁהֵם טוֹבִים בֶּאֱמֶת, וּבִפְרָט בְּעֵסֶק הַתּוֹרָה שֶׁזֶּהוּ טוֹב וְתַעֲנוּג אֲמִיתִּי, וּבָזֶה נִשְׁלָם כַּוָּונַת בְּרִיאָתוֹ כו'.

[kitzur_begin]הָאָדָם לְפִי מַעֲלַת מַהוּתוֹ תַּכְלִיתוֹ עֲבוֹדַת ה' בְּאַהֲבָה וְיִרְאָה. וּכְשֶׁהָאָדָם נִמְשָׁךְ אַחַר תַּעֲנוּגִים גּוּפְנִים מוֹרִיד נַפְשׁוֹ לְהִתְקַשֵּׁר בְּמַה שֶּׁפָּחוּת מִמֶּנּוּ לְקַבֵּל מֵאִתּוֹ וְגָרוּעַ מִבַּעֲלֵי חַיִּים שֶׁאֵין לָהֶם דַּעַת, וְהַתַּאֲוָה עוֹשָׂה גַּם אֶת הַמּוּתָּר לְרַע לְשָׁעָה, וְכָל זֶה בָּא בְּסִבַּת הַיֵּצֶר הָרָע הָעוֹשֶׂה אוֹתוֹ לְשׁוֹטֶה.[kitzur_end]
