שם ע"ס ומבו' בכ"ד שבאצי' הן בבחי' התחלקות כו', דזהו ההפרש בין למעלה מאצי' לאצי', דלמעלה מאצי' הכל הוא בבחי' אחדות ממש שלא יש חילוקי מדרי' כלל כו', ובאצי' הוא בבחי' התחלקות כו' והן בבחי' אורות בכלים כו', הנה אין זה שהן בבחי' התחלקות ממש במדרי' נבדלות ונפרדות זמ"ז כו'. והענין הוא כמשנת"ל (ד"ה ביום השמע"צ הג') מה שנז' ל' בחי' למעלה שזהו עד"מ בחי' הכסף שאין הסיגים מובדלים מהטוב כ"א הוא מהות א' שיש בו בחי' כסף הטוב במספר כך וכך כו', כמו"כ הוא עד"מ בע"ס דאצי' שהכל הוא עצם ומהות אחד ממש שיש בו עשרה כחות כח החכ' כח הבינה כו', ועם היות שהן בחי' אורות בכלים הכל הוא אלקות ממש (והכלים הן כהדין קמצא דלבושא מני' ובי' כו') וע"כ הכל עצם א' כו', רק שלמעלה מאצי' לא יש בחי' חכ' ובינה כלל כו' (וכמשנת"ל בהע"ס שבכתר עצמו שזהו שרוצה להתחכם ורוצה להשיג שהכל הוא רצון ולא חו"ב כו'), ובאצי' הן בחי' חו"ב אבל לא שהן מדרי' נבדלות כו'. וי"ל בדוגמא בבחי' עקודים שהן עשר אורות עקודים בכלי א' שהעשר אורות אינן בבחי'
התחלקות ומ"מ הן עשר אורות כו', ומבו' במ"א שזהו ע"ד שאמרו במשנה פ"ט דמנחות שנתות היו בהין ע"כ לפר ע"כ לאיל כו', היינו שאין כאן התחלקות ממש דהכל הוא עצם א' כו', ולמעלה מעקודים אינו שייך כלל לומר עשר אורות כו'. כמו"כ י"ל דאצי' עם היות שהן בחי' אורות בכלים מ"מ הכל הוא עצם ומהות א' כו', וזהו שנק' אצי' עולם האחדות וכמא' איהו וחיוהי חד איהו וגרמוהי חד שהכל הוא בבחי' אחדות ממש כו', והגם דמחולקין הם במעלתן ובמדריגתן וכמשנת"ל מ"מ הם בבחי' אחדות כו', וי"ל עד"מ באדם למטה הרי כח הראי' וכח השמיעה מובדלים ומופרדים זמ"ז דבראי' לא יש שמיעה ובשמיעה לא יש ראי' (ובפנימיותם הם קשורים ומיוחדים זע"ז כו' וכמ"ש במ"א), ובשרשן היינו בכח החכ' ובכח הבינה הגם שזה חכ' וזה בינה מ"מ הבינה כלולה בחכ' וחכ' בבינה שז"ע הבן בחכ' וחכם בבינה כו'. ואם בהאדם למטה הוא כן ריחוק הערך בין השרש להמסתעף ממנו, דבשרשן הם כלולים ומיוחדים וההסתעפות הן שתי כחות נפרדים כו', הרי יובן מזה דבאצי' הגם שהן חכ' ובינה אין החכ' במציאות ולא הבינה
במציאות בינה כו' והן בהתאחדות גמורה זע"ז כו'. ולכן במ"ת שהאיר בחי' האצי' היו רואים את הנשמע ושומעים את הנראה בלי שום התחלקות כלל כו' (ועמשנת"ל בענין רואין את הנשמע כו'), וכמו"כ הי' בזמן שבהמק"ד קיים שהי' גילוי אור האצי' בלי שום אמצעית והי' הגילוי כמו שהוא כו'. משא"כ בגלות הנה גילוי השכינה הי' ליחזקאל ע"י ריבוי התחלקות המדרי' בהתלבשות פני ארי' פני שור כו', דבזמן שבהמק"ד קיים היו הנשמות במעלה העליונה והי' כח השגתם בגילוי אלקות בכללות החיות היינו עצם האצי' והי' הגילוי שלא ע"י ממוצעים כו', אבל כשגלו לבבל הנה הגילוי הי' רק מבחי' מל' שהו"ע השכינה שגלה עמהם כו', והי' ע"י ירידה בהתלבשות בחיות המרכבה ובריבוי התחלקות כו', וע"י נמשך שפע האלקית לנשמה כו', דמיכאל באהבה וגבריאל ביראה, דכמו"כ האהוי"ר בנשמה היא מקבלת ע"י המלאכים וכמ"ש כי מלאכיו יצוה לך לשמרך כו' על כפים ישאונך כו', פי' כי מלאכיו יצוה לך להיות משפיע לך כו', והם ישאונך שע"י מתנשאת ומתעלת הנשמה ע"י פני ארי' במדרי' האהבה וע"י פני שור במדרי' היראה כו'.
