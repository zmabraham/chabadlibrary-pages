שברוממות העצמות בעלוי אחר עלוי גם שמים לא זכו בעיניו כמו בר"ה כנ"ל וע"כ יש ביכולתם לנגד שם הוי"ה הרי שניהם בכח העצמות דא"ס ממש כנ"ל אך מצד עצמות הטוב הפשוט שלו מגביר בעצמו כח החסד על כח הדין שבעצמו שזהו שעיניך תחזינה מישרים ע"כ גם למטה גובר שם הוי"ה דרחמים על כח הדין דבני האלקים המתיצבים לנגדו עד שנהפכו גם הם לחסדים שקטיגור נעשה סניגור שיומתקו בחסדים ממש עד שזדונות נעשו כזכיות כנ"ל בענין ולא אבה ה' כו' ויהפוך כו' הקללה לברכה כו' וד"ל: 
(נח) [cup]ואחר[/cup] כ"ז יתורצו כל הדקדוקים שבפ' זו במ"ש ויאמר ה' זעקת סדום ועמרה כי רבה כו' דהנה שם זה הוא שם הוי"ה שכולו רחמים פשוטים שבחי' ת"ת עולה בפנימיות הכתר ששם עצמות אא"ס דלית שמאלא כו' כנ"ל וז"ש ארדה נא ואראה הכצעקתה של מדה"ד הבאה אלי ואין הפי' הבאה אלי לבחי' מנגד לעשות בו שנוי ח"ו שהרי ברחמים וחסדים הפשוטים אין שם בחי' מנגד כלל וכלל כנ"ל רק מפני שגם בבחי' העצמות כלול גם מבחי' הגבורה והדין וצמצום דבחי' אלקים שהוא בחי' ב"ן ומ"ה חו"ג שבעצמות ע"כ א' הבאה אלי מפני שבחי' שם אדני ואלקים דשם הוא בא בבחי' משפט ודין ונק' שופט כל הארץ בכלל ולמטה הוא בחי' המל' דינא דמלכותא כו' ששומע ומקבל כל קטרוג ודין ממש ולז"א הכצעקתה של מדה"ד שבעצמות הבאה אלי ממש למנגד לעצם כח החסד ולכך נק' בשם מנגד להיותו מנגד לעצם כח החסד שהוא רק טוב וחסד לבד ואע"פ שאין בכחו להיות בחינת מנגד ממש כנ"ל מ"מ אע"פ שאין כח הדין הזה עושה בו שנוי כלל וכלל אבל עכ"ז מעורר אותו לבא במשפט בעצמו לראות הדבר בעצמו בב' הקצוות דחו"ג שבעצמו דהיינו בשקול הדעת שבעצמו שכולל לחו"ג הנ"ל שבעצמותו כנ"ל בפי' מלפניך משפטי כו' שמצד עצם טובו יכריע לימין דחסד בעצמו וז"ש ארדה נא ואראה בעצמי דוקא הכצעקתה כו' וד"ל: 
(נט) [cup]ויובן[/cup] זה עד"מ הנ"ל מלך שכל משפטי אנשי המדינה שולח למקום המשפט שממונים על כל דבר דין ומשפט שידונו הם רק לפרקים בדבר משפט של איש אח' או על איזה ענין גדול יצוה להביא לפניו הדבר שרוצה לשפוט ד"ז בעצמו ואע"פ שהמלך מצד עצמו הוא רק טוב וחסד לבד ולא היה מביא הדבר לעין משפט כלל. אך עכ"ז כדי לצאת י"ח משפט ודין הוא יורד ומשפיל א"ע לצוות להביא לפניו המשפט שיכוין ויפקח הוא בעצמו על הדבר בב' קצוות דימין ושמאל חסד ודין שבעצמות כפי ראות עיניו בשקול הדעת ומצמצם א"ע כאילו אינו איש טוב וחסד בעצם כלל שזהו ודאי בחי' ירידה וצמצום גדול אצלו. ואמנם ע"פ השגחה אלקית עצמו בד"ז יכריע ממילא לחסד וטוב שלא יהיה כזה במשפט אחר זולתו מטעם הנ"ל וכ"ה למעלה במ"ש ארדה נא ואראה בחי' ירידה זו היינו מפני שמצמצם א"ע לעיין ולפקח בראית עצמו להביא בשקול דב' קצוות דחו"ג שלו וז"ש ואראה דקדק בזה לומר אראה בעצמי דוקא שבא למעט בזה כל ראיות אחרות זולתו שכאשר יראה בעצמו יהיה המשפט באופן אחר ממילא
