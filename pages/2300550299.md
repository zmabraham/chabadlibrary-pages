שהעיקר המה המצות מעשיות שבקיומו המצות מעשיות עושה לעצמו לבושים לקבל הזיו השכינה ב"ה כנ"ל רק שהמצות נתלבשו בעניני' גשמיי' וצריך לזכך אותו שהזיכוך שלהם הוא ע"י מוח הדעת המתגלה בזמן התפלה כנ"ל וד"ל. 
[cup]והנה[/cup] נת' בתו' שע"י ג"פ קדוש שאו' המלאכים ממשיכים [אליהם] בחי' מוח הדעת ואין מקדשין אותו מלמעלה עד שיקדישו אותו מלמטה כו' וע"ז נא' כולם כא' עונים כו' וכמ"ש בזהר דכולהו מקדשין לי' כחדא[ftnref_1523_77] לעילא ולתתא כו' שהמלאכים [דלעילא] מחכים וממתינים על ישראל שלמטה לאמר קדוש כאחד ביחד כו'. והענין הוא דהנה המלאכים שלמעלה [ש]במרכבה העליונה נק' ג"כ בחי' בהמות[ftnref_1524_78] לגבי בחי' עולם האצי' כמ"ש פני ארי' אל הימין ופ"ש מהשמאל[ftnref_1525_79] כו', שאף שיש להם דעת גדול אעפ"כ לגבי הדעת [עליון] דאצי' נמשל כבהמות נדמו[ftnref_1526_80] כאלו אין להם דעת כלל, וע"ז צריכים להמשיך אליהם הדעת [עליון] מאחר שהן נקראים בע"ג ותכלית ונפרדים ממש אשר אין להם ערך ויחוס אל הבורא ית' כו'. ואף שנאמר בזהר בכמה דוכתי דבאתעדל"ת אתעדל"ע[ftnref_1527_81] דרוח אייתי רוח[ftnref_1528_82] ואמשיך רוח שלפ"ע ההעלאת מ"ן שהוא בחי' אתדל"ת כך ערך אופן המשכת מ"ד שהוא בחי' אתעדל"ע ממש, אעפ"כ אין המלאכים העליונים יכולים בעצמן להמשיך המ"ד כו' בחי' ד"ע הנ"ל אליהם מפני שאין בהם הכח ביכולתם[ftnref_1529_83] להתעורר בעצמן בהעלאת מ"ן[ftnref_1530_84] כמ"ש בע"ח שאף לצורך ההעלאת מ"ן מוכרח לישפע מלמעלה לסייעה לעוררה בהעלאת מ"ן כו', והיינו שנשפע מלמעלה מבחי' אתעדל"ע שמתעורר מצד עצמו בלתי אתעדל"ת כלל לעורר בהעלאת מ"ן[ftnref_1531_85] שיהי' בה הכח והיכולת להעלות מ"ן ואח"כ כשמעלה מ"ן ממשכת מ"ד כאופן המ"ן כך הוא המשכת המ"ד שהוא בחי' [oldpage_832] ּ אתעדל"ע הנמשך ע"י אתעדל"ת כו'. וההפרש בין המ"ד הנמשך ע"י העלאת מ"ן שלמטה ובין המ"ד הנשפע מעצמ[ו] וממילא לעוררה בהעלאת מ"ן כו', הוא שהמ"ד הנמשך ע"י העלאת מ"ן הוא בדרך וסדר הדרגת השתלשלו' עו"ע כו', והמ"ד היורד בעצמו לעורר בהעלאת מ"ן הוא נמשך מבחי' שלמעלה מהשתלשלות מבחי' העיגולים והמקיפי' שלמעלה דהיינו מבחי' ומדרגת משרע"ה שנמשך מבחי' עבר הנהר העליון הידוע לעוררה בהעלאת מ"ן כו'. והמשל לזה הוא כמו באדם שלמטה שמתעורר בתשובה פתאום יפלו לו הרהורי תשובה להתעורר על ריחוקו מאור פני מלך חיים כו' שזה לא באה מצד עבודתו שלמטה שהרי לא עשה עדיין כלום ומאין באה לו התעוררות בתשובה, והיינו בחי' אתעדל"ע שלמעלה מהשתלשלו' שמשפיעים לו הרהורי תשובה לעורר[ו] בתשובה ואח"כ כשישים אל לבו להתפלל ויתפלל
