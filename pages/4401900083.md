# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר כב. > פֶּרֶק ב

וְהִנֵּה כָּל זֶה הוּא לְפִי הַשִּׁיטָה שֶׁבְּהַחֶסֶד דַּאֲצִילוּת אֵין עֲדַיִין שׁוּם תּוֹאַר מֵחֶסֶד דַּעֲשִׂיָּה בִּכְלָל, וּמִכָּל שֶׁכֵּן מֵהַהַשְׁפָּעוֹת הַגַּשְׁמִיּוֹת הַנִּמְשָׁכוֹת לְמַטָּה, דְּהִנֵּה גַּם בַּעֲשִׂיָּה הָרוּחָנִית אֵינוֹ נוֹפֵל בּוֹ עִנְיַן הַהַגְשָׁמָה וּמִכָּל שֶׁכֵּן בָּעוֹלָמוֹת יְצִירָה וּבְרִיאָה וּבִפְרָט בְּעוֹלָם הָאֲצִילוּת, אֲשֶׁר מַלְכוּת דַּאֲצִילוּת כְּמוֹ שֶׁהִיא בַּאֲצִילוּת הִיא מִסְּפִירוֹת הָעַצְמִיּוֹת דַּאֲצִילוּת דְּאֵין סוֹף לְהִתְפַּשְּׁטוּתָן הִנֵּה גַּם הַחֶסֶד דַּעֲשִׂיָּה הַכָּלוּל בְּחֶסֶד דַּאֲצִילוּת אֵין בּוֹ שׁוּם צִיּוּר עֲדַיִין, וּבִכְדֵי שֶׁיִּהְיֶה אֵיזֶה צִיּוּר הִנֵּה עַל זֶה הוּא הַדִּין וּמִשְׁפָּט,
אָמְנָם לְפִי הַשִּׁיטָה דְּשָׁרְשֵׁי הַחֶסֶד דַּעֲשִׂיָּה הַכָּלוּל בְּמַלְכוּת דַּאֲצִילוּת הִנֵּה בְּחֶסֶד זֶה כְּבָר יֵשׁ בּוֹ הִצְטַיְּירוּת הַהַשְׁפָּעוֹת—שֶׁנִּמְשְׁכוּ וְנִתְהַוּוּ מִשָּׁרְשֵׁי הַחֶסֶד דַּעֲשִׂיָּה הַכָּלוּל בְּמַלְכוּת דַּאֲצִילוּת—אִם כֵּן הֲרֵי הַמְשָׁכָתָם בְּעוֹלָם הָעֲשִׂיָּה הָרוּחָנִית וְצִיּוּרָם אֵינוֹ הִתְחַדְּשׁוּת, וְלָמָּה יִהְיֶה עַל זֶה דִּין גָּמוּר מֵאַחַר שֶׁכְּבָר נִמְשַׁךְ וְנֶחְתַּם בְּהַדִּין דְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים.
[cup]אַךְ[/cup] הָעִנְיָן הוּא דְּהִנֵּה כָּל נִבְרָא וְנִבְרָא הֲרֵי הוּא כָּלוּל מִגּוּף וְנֶפֶשׁ, דְּהַגּוּף הוּא גַשְׁמִי וְהַנֶּפֶשׁ רוּחָנִי וְהֵם מְחוּלָּקִים בְּעִנְיַן וְאוֹפֶן הִתְהַוּוּתָם, דְּהִתְהַוּוּת הַנֶּפֶשׁ הִיא בָאָה בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת וּכְמוֹ נַפְשׁוֹת הַבְּהֵמוֹת שֶׁלְּמַטָּה בָּאוֹת בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת מִבְּחִינַת פְּנֵי שׁוֹר שֶׁבְּמֶרְכָּבָה וְנַפְשׁוֹת הַחַיּוֹת מִפְּנֵי אַרְיֵה וְנַפְשׁוֹת הָעוֹפוֹת מִפְּנֵי נֶשֶׁר וְנֶפֶשׁ הָאָדָם מִפְּנֵי אָדָם שֶׁבְּמֶרְכָּבָה,
וְיָדוּעַ דְּהַד' חַיּוֹת יֵשׁ בְּכָל עוֹלָם וְעוֹלָם וּבָא בִּבְחִינַת הִשְׁתַּלְשְׁלוּת מִמַּדְרֵיגָה לְמַדְרֵיגָה דְּהַיְינוּ מֵרוּחָנִיּוּת לְרוּחָנִיּוּת דְּגַם הַנְּפָשׁוֹת שֶׁלְּמַטָּה הֵן רוּחָנִיּוֹת, אֲבָל לִהְיוֹת הִתְהַוּוּת מְצִיאוּת הַגַּשְׁמִי כְּמוֹ גּוּף הַבַּעֲלֵי חַיִּים וְכַדּוֹמֶה בְּכָל הַדּוֹמֵם-צוֹמֵחַ-חַי-מְדַבֵּר, זֶה אִי אֶפְשָׁר לִהְיוֹת בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת עִלָּה וְעָלוּל, שֶׁאֲפִילוּ יִהְיֶה רִיבּוּי רְבָבוֹת מַדְרֵיגוֹת הִשְׁתַּלְשְׁלוּת לֹא יִתְהַוֶּוה מֵהָרוּחָנִי מְצִיאוּת גַּשְׁמִי כִּי אִם יִהְיֶה הַכֹּל בִּבְחִינַת רוּחָנִי, וְהִתְהַוּוּת הַגַּשְׁמִי מֵהָרוּחָנִי הוּא בְּדֶרֶךְ בְּרִיאָה יֵשׁ מֵאַיִן דַּוְקָא.
[cup]וּכְשֵׁם[/cup] שֶׁלְּמַטָּה יֵשׁ גּוּף וְנֶפֶשׁ וְהַנֶּפֶשׁ הוּא בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת וְהַגּוּף הוּא בְּדֶרֶךְ בְּרִיאָה יֵשׁ מֵאַיִן, כְּמוֹ כֵן הוּא בְּדוּגְמָא כָזֹאת לְמַעְלָה גַם כֵּן, וּכְמוֹ מַלְאָכִים הֲרֵי יָדוּעַ שֶׁכְּלוּלִים גַּם כֵּן מִגּוּף וְנֶפֶשׁ וּכְמוֹ שֶׁכָּתַב הָרַמְבַּ''ן דְּמַה שֶּׁכָּתוּב (תְּהִלִּים קד. ד) עֹשֶׂה מַלְאָכָיו רוּחוֹת מְשָׁרְתָיו אֵשׁ לֹהֵט, הַיְינוּ גּוּפוֹת הַמַּלְאָכִים שֶׁהֵן מִיסוֹד הָאֵשׁ וְהָרוּחַ, נִמְצָא יֵשׁ בָּהֶן גּוּף וְנֶפֶשׁ,
וְכֵן בְּהָעֶשֶׂר סְפִירוֹת שֶׁבָּעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה שֶׁכְּלוּלוֹת מֵאוֹרוֹת וְכֵלִים מְבוֹאָר בְּאִגֶּרֶת הַקֹּדֶשׁ (סִימָּן כ' דְּרוּשׁ הַמַּתְחִיל אִיהוּ וְחַיּוֹהִי חַד) שֶׁהַכֵּלִים דְּעֶשֶׂר סְפִירוֹת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה הֵם בְּחִינַת נִבְרָא וְנִבְרְאוּ בְּדֶרֶךְ יֵשׁ מֵאַיִן מִבְּחִינַת אוֹר הַנְּשָׁמָה שֶׁלָּהֶם, וְאִם כֵּן גַּם שָׁם בְּחִינַת הַנֶּפֶשׁ שֶׁהִיא הַנְּשָׁמָה הוּא בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת עִלָּה וְעָלוּל וּבְחִינַת הַכֵּלִים וְכָל שֶׁכֵּן הַלְּבוּשִׁים וְהַהֵיכָלוֹת וְהַמַּלְאָכִים שֶׁבְּכָל עוֹלָם הִתְהַוּוּתָם בְּדֶרֶךְ בְּרִיאָה יֵשׁ מֵאַיִן,
וּלְפִי זֶה מַה שֶּׁנִּתְבָּאֵר לְעֵיל דְּהַחֶסֶד דַּאֲצִילוּת כּוֹלֵל שָׁרְשֵׁי בְּחִינַת חֶסֶד דַּעֲשִׂיָּה וְהַהַשְׁפָּעוֹת הַבָּאוֹת מִזֶּה, וְהַהַמְשָׁכָה הִיא בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת לְבָד, זֶהוּ בִּבְחִינַת הָאוֹר וְהַחַיּוּת הַיְינוּ בִּבְחִינַת הַנְּשָׁמָה שֶׁבְּכָל הָעוֹלָמוֹת, אֲבָל בְּחִינַת הַכֵּלִים דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה וְכָל שֶׁכֵּן הַלְּבוּשִׁים דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה הֵם בְּדֶרֶךְ בְּרִיאָה יֵשׁ מֵאַיִן.

[kitzur_begin]הַחֶסֶד שֶׁנִּקְצָב בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים הוּא שָׁרְשֵׁי הַחֶסֶד דַּעֲשִׂיָּה שֶׁכָּלוּל בְּמַלְכוּת דַּאֲצִילוּת הֲרֵי כְּבָר יֵשׁ בּוֹ צִיּוּר מִשָּׁרְשֵׁי הַשְׁפָּעוֹת גַּשְׁמִיּוֹת אִם כֵּן מַהוּ עִנְיַן הַדִּין בְּכָל יוֹם, בְּכָל נִבְרָא יֵשׁ גּוּף וְנֶפֶשׁ, הַנֶּפֶשׁ הִתְהַוּוּתוֹ בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת וְהַגּוּף הִתְהַוּוּתוֹ בְּדֶרֶךְ בְּרִיאָה יֵשׁ מֵאַיִן, כֵּן הוּא בְּהָעֶשֶׂר סְפִירוֹת בְּאוֹרוֹת וְכֵלִים.[kitzur_end]
