# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ח. > פֶּרֶק ב

וְהִנֵּה הַהַשְׁפָּעָה מִבְּחִינַת רַב חֶסֶד הִיא הַשְׁפָּעָה בְּרִיבּוּי, וְזֶהוּ מַה שֶּׁיֵּשׁ לָהֶם עוֹשֶׁר וְכָבוֹד בְּרִיבּוּי מְאֹד שֶׁלֹּא עַל פִּי חֶשְׁבּוֹן כְּלָל, כְּמוֹ שֶׁכָּתוּב (קֹהֶלֶת ט. יו"ד) כִּי אֵין מַעֲשֶׂה וְחֶשְׁבּוֹן וְדַעַת וְחָכְמָה בִּשְׁאוֹל וגו', וְהַיְינוּ שֶׁמְּקַבְּלִים שֶׁלֹּא עַל פִּי הַחֶשְׁבּוֹן.
[cup]דְּהִנֵּה[/cup] אִיתָא בִּמְאוֹרֵי אוֹר (אוֹת ח' סְעִיף עה) חֶשְׁבּוֹן נִקְרֵאת הַחָכְמָה עַל שֵׁם הַמַּחֲשָׁבָה. וּבִגְמָרָא (בָּבָא בַתְרָא עח. ב) דָּרְשׁוּ עַל פָּסוּק עַל כֵּן יֹאמְרוּ הַמּוֹשְׁלִים בֹּאוּ חֶשְׁבּוֹן (בַּמִּדְבָּר כא. כז) בֹּאוּ וְנַחְשׁוֹב חֶשְׁבּוֹנוֹ שֶׁל עוֹלָם הֶפְסֵד מִצְוָה נֶגֶד שְׂכָרָהּ וּשְׂכַר עֲבֵירָה נֶגֶד הֶפְסֵדָהּ,
דְּחֶשְׁבּוֹן זֶה הוּא מִבְּחִינַת הַחָכְמָה דְּאוֹרַיְיתָא מֵחָכְמָה נָפְקַת וְהוּא בְּחִינַת קַו הַמִּדָּה, שֶׁמִּשָּׁם נִמְשֶׁכֶת הַמִּדָּה וְהַקִּצְבָה עַל כָּל הַשְׁפָּעוֹת רוּחָנִיּוֹת וְגַשְׁמִיּוֹת. וְעִנְיַן הַמִּשְׁקָל וְהַמִּדָּה הוּא בְּאוֹפֶן הַהִתְכַּלְלוּת דְּחֶסֶד וּגְבוּרָה, שֶׁזֶּהוּ עַל פִּי הַחָכְמָה שֶׁמּוֹדֶדֶת אֵיךְ וּבְאֵיזֶה אוֹפֶן תִּהְיֶה הַהִתְכַּלְלוּת דְּחֶסֶד וּגְבוּרָה בְּהִתְגַּבְּרוּת דְּמִדָּה אַחַת עַל זוּלָתָהּ, וּמִזֶּה נִמְשֶׁכֶת הַמְּדִידָה בְּהַשְׁפָּעַת הַמִּדּוֹת הַנַ"ל, כְּמוֹ בְּהַשְׁפָּעַת בָּנֵי חַיֵּי וּמְזוֹנֵי כַּמָּה תִהְיֶה הַהַשְׁפָּעָה וּבְאֵיזֶה אוֹפֶן, וּכְמוֹ מְזוֹנוֹתָיו שֶׁל אָדָם קְצוּבִים לוֹ מֵרֹאשׁ הַשָּׁנָה (בֵּיצָה טז. א) וְכֵן אָדָם נִידּוֹן בְּכָל יוֹם [שָׁעָה] (רֹאשׁ הַשָּׁנָה טז. א) שֶׁהַדִּין וְהַמִּשְׁפָּט הוּא עַל פִּי מַעֲשָׂיו וַעֲבוֹדָתוֹ שֶׁבְּאוֹפֶן כָּזֶה נִמְשָׁכָה לוֹ הַהַשְׁפָּעָה עַל פִּי הַמִּשְׁקָל דְּחֶסֶד וּגְבוּרָה, אִם בְּהִתְגַּבְּרוּת הַחֲסָדִים אוֹ כו' חַס וְשָׁלוֹם,
וּכְמוֹ שֶׁכָּתוּב (אִיּוֹב לז. יב) וְהוּא מְסִבּוֹת מִתְהַפֵּךְ בְּתַחְבּוּלוֹתָיו לְפָעֳלָם וּפֵירֵשׁ רַשִׁ"י שֶׁזֶּהוּ כְּמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (רֹאשׁ הַשָּׁנָה יז. ב) הֲרֵי שֶׁהָיוּ יִשְׂרָאֵל צַדִּיקִים בְּרֹאשׁ הַשָּׁנָה וּפָסְקוּ עֲלֵיהֶם גְּשָׁמִים מְרוּבִּים לְסוֹף חָזְרוּ כו' מוֹרִידָן שֶׁלֹּא בִזְמַנָּן עַל הָאָרֶץ שֶׁאֵינָהּ צְרִיכָה לָהֶם, וְכַאֲשֶׁר לֹא הָיוּ מוּכְשָׁרִים וּפָסְקוּ עֲלֵיהֶם גְּשָׁמִים מוּעָטִים לְסוֹף חָזְרוּ כו' מוֹרִידָן בִּזְמַנָּן עַל הָאָרֶץ הַצְּרִיכָה לָהֶם.
[cup]וּבְזֹהַר[/cup] (חֵלֶק א קט. ב) עַל פָּסוּק זֶה הַקָּדוֹשׁ בָּרוּךְ הוּא מְסַבֵּב סִבּוּבִין וְאַיְיתִי עוֹבָדִין בְּעָלְמָא לְאִתְקַיָּימָא וּלְבָתַר דַּחֲשִׁיבוּ בְּנֵי נָשָׁא דְּיִתְקַיְּימוּ אִינּוּן עוֹבָדִין, קוּדְשָׁא בְּרִיךְ הוּא מְהַפֵּךְ לוֹן לְאִינּוּן עוֹבָדִין מִכְּמָה דַּהֲוָה בְּקַדְמֵיתָא (הַיְינוּ שֶׁקּוֹרֵא הַדּוֹרוֹת וּבוֹרְאָן לְהִתְקַיֵּים כְּגוֹן דּוֹר הַמַּבּוּל וְהֵם חָשְׁבוּ שֶׁיִּהְיוּ קַיָּימִים לְעוֹלָם וְקִלְקְלוּ מַעֲשֵׂיהֶם וּבִטְּלָם וַהֲפָכָם מֵהַקִּיּוּם אֶל הַהֶעְדֵּר. אוֹר הַחַמָּה) בְּתַחְבּוּלֹתָו חָסֵר י' כו' דְּאִיהוּ טִקְלָא דְּאִסְתַּחֲרַת קַמֵּיהּ כו'. אִי מְטִיבִין בְּנֵי נָשָׁא עוֹבָדֵיהוֹן הַהִיא טִקְלָא דְּסָחֲרָא אַסְחָרַת לוֹן לְיַמִּינָא וּכְדֵין אִתְעֲבִידוּ עוֹבָדִין בְּעָלְמָא לְאוֹטָבָא לוֹן כו'. אָתוּ בְּנֵי נָשָׁא לְאַבְאָשָׁא כו' קוּדְשָׁא בְּרִיךְ הוּא אַסְחַר לֵיהּ בְּסִטְרָא דִשְׂמָאלָא וּמְהַפֵּךְ מְסִבּוֹת כו' וְאִתְעֲבִידוּ עוֹבָדִין בְּעָלְמָא לְאַבְאָשָׁא לוֹן לִבְנֵי נָשָׁא כו' עַד דִּבְנֵי נָשָׁא תָּיְיבִין לְאוֹטָבָא עוֹבָדֵיהֶן כו'.
[cup]וְהָעִנְיָן[/cup] הוּא דְּהִנֵּה טִקְלָא הוּא בְּחִינַת מַלְכוּת דַּאֲצִילוּת שֶׁהִיא מְקוֹר הַהַשְׁפָּעָה לִבְרִיאָה-יְצִירָה-עֲשִׂיָּה, וְהִיא מִסְתַּחָרַת כְּפוּם עוֹבָדֵיהוֹן דִּבְנֵי נָשָׁא, דְּאִם מְטִיבִין מַעֲשֵׂיהֶם אֲזַי הַהוּא טִקְלָא דְּסָחֲרָא אַסְחָרַת לוֹן לְיַמִּינָא, הַיְינוּ שֶׁאָז מְקַבֶּלֶת הַמַּלְכוּת מִבְּחִינַת קַו הַיָּמִין דַּאֲצִילוּת שֶׁהוּא חָכְמָה-חֶסֶד-נֶצַח, וְאָז הוּא בְּחִינַת הִתְגַּבְּרוּת הַחֲסָדִים וְנִמְשָׁךְ טוֹב וָחֶסֶד בָּעוֹלָם לְמַטָּה, וְאִם חַס וְשָׁלוֹם לֵיכָּא אִתְעֲרוּתָא דִלְתַתָּא וְאֵין עוֹשֵׂה טוֹב כו', אֲזַי גּוֹרְמִים בְּמַעֲשֵׂיהֶם שֶׁתְּקַבֵּל הַמַּלְכוּת מִבְּחִינַת קַו הַשְּׂמֹאל כו', דְּהַכֹּל תָּלוּי בְּמַעֲשֵׂה הַתַּחְתּוֹנִים.

[kitzur_begin]הַהַשְׁפָּעָה דְּרַב חֶסֶד הוּא בְּרִיבּוּי שֶׁלֹּא עַל פִּי הַחֶשְׁבּוֹן, חֶשְׁבּוֹן חָכְמָה בֹּאוּ וְנַחְשׁוֹב חֶשְׁבּוֹנוֹ שֶׁל עוֹלָם, כָּל הַהַשְׁפָּעוֹת בָּעוֹלָם עַל פִּי קַו הַמִּדָּה דְּתוֹרָה. אָדָם נִידּוֹן בְּכָל יוֹם (שָׁעָה) לְפִי מַעֲשָׂיו, שֶׁבָּזֶה תָּלוּי הַקַּבָּלָה וְהַהַשְׁפָּעָה דִסְפִירַת הַמַּלְכוּת אִם לְיָמִין וְטוֹב אוֹ לִשְׂמֹאל וְרָע.[kitzur_end]
