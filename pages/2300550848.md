ונכלל בבחי' פני אדם עכשיו בכל עולם. ואין זה פלא כ"כ מאחר שידוע שבסדר ההשתל' מעולם לעולם שיש בכל עולם [בחי'] פני שור ופני אדם ולפ"ע עליות המדרגות מתחתון לעליון יחשב פני אדם שבעולם התחתון בעלותו למעלה לעליון הימנו רק בחי' פני שור לבד (מאחר שמחיצוני' שבעליון נמשך לתחתון כנ"ל), וכמו ש[ה]אדם בעוה"ז אין לו ערך אף לצומח וחי שבגעה"ת כנ"ל. ואין זה חידוש כלל דגם בעוה"ז נמצא במין[ftnref_8520_1491] האדם מעין זה, כמו שכבר הי' לעולמים בהגלות נגלות אור האלקי בנשמות צדיקי' ונביאים עד שהיו מופשטים מחומר גופם בתכלית ועלו למעלה ממדריגת המלאכים העליוני' כאברהם ומשה ואליהו וכה"ג, עד שהי' ערך נפשות האדם לגביהם כבהמות ממש, וכמו שאברהם רדף אחר ד' מלכים[ftnref_8521_1492] במתי מספר והיינו לפי שסר צלם[ftnref_8522_1493] צל אדם מהם וברחו מפניו כמו איכה ירדוף אחד אלף[ftnref_8523_1494] כו' (וכערך נה"ב הגם שנק' אדם, לגבי נה"א כידוע). 
[cup]ודוגמא[/cup] לזה אנו מוצאים גם עכשיו בנש"י בגופי' שיש בהם ב' מדרגות בחי' אדם ובחי' בהמה (והוא בחי' בהמ"ה שבאדם) כמ"ש וזרעתי את ב"י[ftnref_8524_1495] זרע אדם וזרע בהמה דגם שנק' נשמות שבאים בזריעה דאור העליון מ"מ נק' זרע בהמה לגבי הנשמות העליונות מהם שנק' זרע אדם, וכמו שאנו רואים דנשמות ת"ח שיש בהן דעת והשגה אלקית בבחי' גדלות המוחי' והמדות דאוי"ר שכליי' כו' נק' זר"ע אדם העליון כו', והנשמות דע"ה אף שהן יריאים בקבלת עומ"ש כשור לעול למשא ועבודה אלקי' בבחי' עשי' בפ"מ לבד ואין בהן דעת והשגה אלקי' כלל וגם לא התפעלות אוי"ר רק בבחי' אוי"ר טבעי' כבהמה ממש רק שהוא בבחי' התפעלות אלקות כמרכבה דפני ארי' ושור כו' למעלה זהו שנק' זרע בהמה. ועכ"ז יש בהן חילוקי מדרגות זו למעלה מזו עד שיש בזרע בהמה שעובדי' ה' בגופם למעשה לבד ג"כ בחי' אדם שהוא קצת השגה אלקי' שמכיר לבוראו ומתפעל באוי"ר מצד השגתו המצומצם בדוגמא דאופנים וחיות דיצי'[ftnref_8525_1496] שהן בבחי' מדות דאוי"ר בקטנות המוחי' כו', נק' בחי' בהמה ממש לגבי הנשמות שבבחי' גדלות המוחי' דחב"ד שמתפשט ברחבות המדות (כשרפי' ונשמות דבריאה בגעה"ע כו'). וכן עד"ז גם נשמות דבריאה שבגעה"ע בגדלות המוחי' ומדות דאדם נק' רק בחי' קטנות המוחי' ומדות דבהמה לגבי נשמות דאצי' ממש (כנשמות דפני אדם בבנ"י במ"ת שנק' אנפי זוטרי לגבי הארת פני הוי' דאדם הגדול שנק' אנפי רברבי[ftnref_8526_1497] ) וכשרפים דבריאה לגבי בחי' המל' שנק' אדם תתאה. וכך גם עליות הנשמות דאצי' למעלה מן האצי' דמאציל עצמו וכידוע דכמו שאין ערך לבריאה עם האצי' כך אין ערך לאצי' לגבי
