# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר יג. > פֶּרֶק ב

וּבֵיאוּר הָעִנְיָן הוּא דְּהִנֵּה בֶּאֱמֶת אֵינוֹ מוּבָן כְּלָלוּת הָעִנְיָן מַה שֶּׁיֵּשׁ בְּכֹחַ הַנֶּפֶשׁ הָאֱלֹקִית לְהִתְגַּבֵּר עַל הַנֶּפֶשׁ הַבַּהֲמִית,
דְּהִנֵּה כְּתִיב (מִשְׁלֵי יח. יז) צַדִּיק הָרִאשׁוֹן בְּרִיבוֹ וּפֵירֵשׁ בִּמְצוּדַת דָּוִד כִּי יוּכַל לְהַטְעִים דְּבָרָיו וּלְהַכְנִיסָם בַּלֵּב הוֹאִיל וְלֹא אָמַר מִי בְּחִילּוּף, דְּכַאֲשֶׁר שְׁנַיִם הַחֲלוּקִים בְּטַעֲנוֹתֵיהֶם בָּאִים כְּאֶחָד הִנֵּה טַעֲנָתוֹ שֶׁל כָּל אֶחָד נֶחֱלֶשֶׁת מִטַּעֲנַת זוּלָתוֹ, אֲבָל כַּאֲשֶׁר הוּא רִאשׁוֹן יָכוֹל הוּא לְהַטְעִים דְּבָרָיו לְהַכְנִיסָם בְּלֵב הַשּׁוֹמֵעַ שֶׁאֵין מִי שֶׁיְּנַגֵּד לוֹ וְגַם יַחֲלִישׁ טַעֲנָתוֹ,
וְכֵן הוּא בְּנֶפֶשׁ הָאֱלֹקִית וְנֶפֶשׁ הַבַּהֲמִית שֶׁהֵם מְנַגְּדִים בְּעִנְיְנֵיהֶם, וְהַנֶּפֶשׁ הַבַּהֲמִית הִיא בְהָאָדָם מִתְּחִלַּת בְּרִיאָתוֹ וּכְמוֹ שֶׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (סַנְהֶדְרִין צא. ב) מֵאֵימָתַי יֵצֶר הָרָע שׁוֹלֵט בָּאָדָם כו' מִשַּׁעַת יְצִיאָה, וּבְמִדְרָשׁ רַבָּה (נֹחַ פַּרְשָׁה ל"ד. יב) דָּרַשׁ מִנְּעוּרָיו מִשָּׁעָה שֶׁנִּנְעָר לָצֵאת מִמְּעֵי אִמּוֹ. וְהַנֶּפֶשׁ הָאֱלֹקִית נִכְנָס בּוֹ כְּשֶׁהוּא בֶּן י"ג שָׁנָה וְיוֹם אֶחָד, שֶׁאָז נַעֲשֶׂה מְחוּיָּיב בְּמִצְוֹת. וְכִדְאִיתָא בְּמִדְרָשׁ רַבָּה (קֹהֶלֶת) עַל פָּסוּק (קֹהֶלֶת ד. יג) טוֹב יֶלֶד מִסְכֵּן וְחָכָם מִמֶּלֶךְ זָקֵן וּכְסִיל, טוֹב יֶלֶד מִסְכֵּן וְחָכָם זֶה יֵצֶר טוֹב, וְלָמָּה נִקְרָא שְׁמוֹ יֶלֶד כִּי אֵין מִזְדַּוֵּוג לְאָדָם אֶלָּא מִבֶּן י"ג שָׁנָה וּלְמַעְלָה כו' מִמֶּלֶךְ זָקֵן וּכְסִיל זֶה יֵצֶר הָרָע כו' וְלָמָּה קוֹרֵא אוֹתוֹ זָקֵן שֶׁהוּא מִזְדַּוֵּוג לוֹ מִיַּלְדוּתוֹ וְעַד זִקְנָתוֹ, וְכֵן הוּא בְּזֹהַר וַיֵּשֶׁב (קעט. א),
וַהֲגַם דְּבִגְמָרָא וּבְמִדְרָשׁ (פַּרְשַׁת נֹחַ הַנַ"ל) אִיתָא נְשָׁמָה מֵאֵימָתַי נִיתְּנָה בְּאָדָם כו' מִשַּׁעַת פְּקִידָה (וְעַיֵּן שָׁם בְּמַתְּנוֹת כְּהֻנָּה) זֶהוּ רַק שֶׁנִּיתְּנָה בּוֹ אֲבָל אֵינָהּ פּוֹעֶלֶת עַד הֱיוֹתוֹ בֶּן י"ג שָׁנָה, אִם כֵּן הֲרֵי הַנֶּפֶשׁ הַבַּהֲמִית אַקְדְּמֵיהּ טַעֲנָתֵיהּ.
[cup]וְכֵן[/cup] מִצַּד הַגּוּף הַגַּשְׁמִי, הֲרֵי כָּל עִנְיְנֵי הַתַּאֲוָה בַּאֲכִילָה וּשְׁתִיָּה וּשְׁאָר הַתַּעֲנוּגִים קְרוֹבִים לְהַיֶּלֶד יוֹתֵר וּמִתְרַגֵּל בָּהֶם וְהָרְגִילוּת מִשְׁתָּרֶשֶׁת בּוֹ בִּמְאוֹד, וּבִפְרָט בִּהְיוֹתוֹ בִּסְבִיבָה לֹא טוֹבָה הַנִּמְשָׁכִים אַחֲרֵי הַגַּשְׁמִיּוּת וְהַחוּמְרִיּוּת, הִנֵּה כָּל הָעִנְיָנִים הַלֹּא טוֹבִים אוֹ הָרָעִים שֶׁהִתְרַגֵּל בָּהֶם נִטְבָּעִים בּוֹ, כְּמַאֲמָר הֶרְגֵּל נַעֲשֶׂה טֶבַע אוֹ טֶבַע שֵׁנִי, שֶׁקָּשֶׁה בִּמְאֹד לְהִפָּטֵר מֵהֶם,
וְהַנֶּפֶשׁ הָאֱלֹקִית עֲדַיִין טֶרֶם הִתְאַחֲזָה בּוֹ בְּהִתְיַישְּׁבוּת, אִם כֵּן הֲרֵי הַנֶּפֶשׁ הַבַּהֲמִית הוּא חָזָק הַרְבֵּה יוֹתֵר מֵהַנֶּפֶשׁ הָאֱלֹקִית, דְּלָכֵן הִנֵּה הַנֶּפֶשׁ הַבַּהֲמִית נִקְרָא מֶלֶךְ זָקֵן, דִּלְבַד זֹאת דְּהִקְדִּים טַעֲנָתֵיהּ וּמוֹשֵׁל בְּכָל עִנְיְנֵי הַגּוּף, הִנֵּה הוּא זָקֵן וְרָגִיל אֵיךְ לְפַתּוֹת בְּנֵי אָדָם וְלַהֲסִיתָם לִהְיוֹת נִמְשָׁכִים אַחֲרֵי הַגַּשְׁמִיּוּת וְהַחוּמְרִיּוּת, וְהַנֶּפֶשׁ הָאֱלֹקִית הִנֵּה לְבַד זֹאת שֶׁהִיא יֶלֶד לְפִי עֶרֶךְ קְדִימָתוֹ שֶׁל הַנֶּפֶשׁ הַבַּהֲמִית, הִנֵּה עוֹד זֹאת הוּא מִסְכֵּן בְּעֵינֵי בְּנֵי אָדָם, שֶׁהֲרֵי כָּל עִנְיָנוֹ הוּא עִנְיַן הָרוּחָנִיּוּת וְהָאָדָם שֶׁנִּמְשָׁךְ אַחֲרֵי הַגַּשְׁמִיּוּת וְהוּרְגַּל בְּעִנְיְנֵי עוֹלָם הַחוּמְרִיּוּת נִרְאֶה לוֹ כָּל עִנְיָן רוּחָנִי לְמִסְכֵּנוּת, כִּי הָעִיקָּר אֶצְלוֹ הֵם עִנְיָנִים הַגַּשְׁמִים וְזֶה הוּא הָעֲשִׁירוּת וְהַחֲשִׁיבוּת שֶׁלּוֹ, אִם כֵּן הֲרֵי הַנֶּפֶשׁ הַבַּהֲמִית הוּא חָזָק יוֹתֵר מֵהַנֶּפֶשׁ הָאֱלֹקִית וְאִם כֵּן נִפְלָא הַדָּבָר בִּמְאֹד אֵיךְ יֵשׁ כֹּחַ בְּנֶפֶשׁ הָאֱלֹקִית לְהִתְגַּבֵּר עַל הַנֶּפֶשׁ הַבַּהֲמִית,
וּכְמוֹ כֵן מִצַּד שָׁרְשָׁן, הֲרֵי יָדוּעַ דְּשֹׁרֶשׁ הַנֶּפֶשׁ הַבַּהֲמִית הוּא מֵעוֹלָם הַתֹּהוּ וְשֹׁרֶשׁ הַנֶּפֶשׁ הָאֱלֹקִית מֵעוֹלָם הַתִּקּוּן וְתֹהוּ קָדַם אֶל הַתִּקּוּן, וּכְמוֹ שֶׁכָּתוּב (בְּרֵאשִׁית לו. לא) וְאֵלֶּה הַמְּלָכִים אֲשֶׁר מָלְכוּ בְּאֶרֶץ אֱדוֹם לִפְנֵי מְלָךְ מֶלֶךְ לִבְנֵי יִשְׂרָאֵל, וְהֵן ז' מַלְכִין קַדְמָאִין דְּתֹהוּ דְּקָדְמוּ אֶל הַתִּקּוּן, וּמִכָּל זֶה לִכְאוֹרָה אֵינוֹ מוּבָן אֵיךְ יֵשׁ כֹּחַ בְּנֶפֶשׁ הָאֱלֹקִית לְהִתְגַּבֵּר עָלָיו. אַךְ הָעִנְיָן הוּא דְּהִנֵּה בֶּאֱמֶת הַנֶּפֶשׁ הָאֱלֹקִית בְּשָׁרְשׁוֹ קָדוּם אֶל הַנֶּפֶשׁ הַבַּהֲמִית. דְּהִנֵּה כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל (מַאֲמָר י פֶּרֶק ב) דְּסֵדֶר הִשְׁתַּלְשְׁלוּת שָׁרְשׁוֹ בִּלְמַעְלָה מִסֵּדֶר הִשְׁתַּלְשְׁלוּת בִּבְחִינַת פְּנִימִית הַמַּקִּיף דַּוְקָא, כְּמוֹ כֵן הוּא בִּבְחִינַת הַתֹּהוּ וְתִקּוּן, דְּתִקּוּן שָׁרְשׁוֹ בְּעַצְמוּת אוֹר אֵין סוֹף לְמַעְלָה מִשֹּׁרֶשׁ הַתֹּהוּ, וְזֶהוּ מַה שֶּׁבֶּאֱמֶת הָיָה יַעֲקֹב הַבְּכוֹר, וְכַיָּדוּעַ דְּיַעֲקֹב וְעֵשָׂו הֵן תֹּהוּ וְתִקּוּן וְעֵשָׂו נוֹלַד קוֹדֶם שֶׁזֶּהוּ כְּמוֹ שֶׁעוֹלָם הַתֹּהוּ קָדַם לָבֹא לְעוֹלָם הַתִּקּוּן. אַךְ זֶהוּ בִּבְחִינַת לֵידָה וְהִתְגַּלּוּת לְבָד, אֲבָל בֶּאֱמֶת הָיָה יַעֲקֹב הַבְּכוֹר וּכְמוֹ שֶׁפֵּירֵשׁ רַשִׁ"י בְּפַרְשַׁת תּוֹלְדוֹת עַל פָּסוּק וְיָדוֹ אֹחֶזֶת בַּעֲקֵב עֵשָׂו (בְּרֵאשִׁית כה. כו) מָשָׁל לִשְׁפוֹפֶרֶת נִכְנָס רִאשׁוֹן יוֹצֵא אַחֲרוֹן, דְּיַעֲקֹב נוֹצַר מִטִּפָּה הָרִאשׁוֹנָה, וְהַיְינוּ דִּבְשָׁרְשׁוֹ יַעֲקֹב שֶׁהוּא בְּחִינַת הַתִּקּוּן שָׁרְשׁוֹ לְמַעְלָה מֵעֵשָׂו בְּחִינַת הַתֹּהוּ,
וְכֵן הוּא בְּנֶפֶשׁ הָאֱלֹקִית וְנֶפֶשׁ הַבַּהֲמִית, דְּהַנֶּפֶשׁ הָאֱלֹקִית שָׁרְשׁוֹ לְמַעְלָה מִנֶּפֶשׁ הַבַּהֲמִית, וְעַל יְדֵי זֶה יֵשׁ לוֹ כֹּחַ לְהִתְגַּבֵּר עַל הַנֶּפֶשׁ הַבַּהֲמִית מִצַּד שֶׁנִּמְשָׁךְ לוֹ כֹּחַ עַל זֶה מִשָּׁרְשׁוֹ הַקָּדוּם, דְּלִהְיוֹת שָׁרְשׁוֹ לְמַעְלָה מִשֹּׁרֶשׁ הַנֶּפֶשׁ הַבַּהֲמִית הֲרֵי בְּמֵילָא יֵשׁ בְּכֹחוֹ לְהִתְגַּבֵּר עָלָיו, אַף גַּם שֶׁהַנֶּפֶשׁ הַבַּהֲמִית קָדוּם לָבֹא וְלִפְעוֹל בְּגוּף הָאָדָם, מִכָּל מָקוֹם מֵאַחַר שֶׁשָּׁרְשׁוֹ קָדוּם אֵלָיו, הֲרֵי יֵשׁ בְּכֹחוֹ לְהִתְגַּבֵּר עָלָיו.

[kitzur_begin]הַנֶּפֶשׁ הַבַּהֲמִית שֶׁשָּׁרְשׁוֹ מֵהַתֹּהוּ בָּא אֶל הָאָדָם קוֹדֶם לַנֶּפֶשׁ הָאֱלֹקִית שֶׁשָּׁרְשׁוֹ מֵהַתִּקּוּן, וּמִשְׁתָּרֵשׁ בְּהֶרְגֵּלִים גּוּפְנִים הַנִּמְשָׁכִים בְּטִבְעָם אֶל הַחוּמְרִיּוּת, תֹּהוּ קָדַם אֶל הַתִּקּוּן בְּהִתְגַּלּוּת, וּבְשָׁרְשָׁם הַתִּקּוּן קָדַם, לָכֵן בְּכֹחַ הַנֶּפֶשׁ הָאֱלֹקִית לְהִתְגַּבֵּר עַל הַנֶּפֶשׁ הַבַּהֲמִית.[kitzur_end]
