מַאֲמָר כא.
אָמְנָם דַּעַת רַבָּנָן בְּעִנְיַן הַדִּין וּמִשְׁפָּט דִּמְזוֹנוֹתָיו שֶׁל אָדָם קְצוּבִין לוֹ מֵרֹאשׁ הַשָּׁנָה וְכָל צְרָכָיו שֶׁל הָאָדָם וּבְנֵי בֵיתוֹ בְּבָנֵי חַיֵּי וּמְזוֹנֵי הֵם בִּכְלַל הַמְּזוֹנוֹת, וּסְבִירָא לְהוּ דִּגְזַר דִּינוֹ שֶׁל אָדָם נֶחְתָּם בְּיוֹם הַכִּפּוּרִים (שָׁם בִּגְמָרָא) דִּבְרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים נִגְמָר פְּרָטֵי עִנְיְנֵי הָאָדָם עַל כָּל הַשָּׁנָה לְחַיִּים אוֹ כו' וְכֵן לְכָל הַמְּאוֹרָעוֹת שֶׁלּוֹ, אִם כֵּן לְפִי זֶה אִי אֶפְשָׁר לוֹמַר שֶׁבְּהַחֶסֶד עֶלְיוֹן הַנִּקְצָב לוֹ בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים לֹא נִצְטַיֵּיר מֵעִנְיְנֵי עוֹלָם הַזֶּה כְּלָל, דְּאִם כֵּן מַהוּ עִנְיָן שֶׁנִּכְתָּב וְנֶחְתָּם וּמַהוּ עִנְיָן שֶׁנִּגְמָר כָּל מַה שֶּׁיֶּאֱרַע לוֹ, אֶלָּא בְּהֶכְרַח לוֹמַר דִּבְהַחֶסֶד שֶׁנִּקְצָב לוֹ נִיתַּן תּוֹאַר וּתְמוּנָה מִמַּה שֶּׁיּוּמְשַׁךְ מִזֶּה בָּעוֹלָם הַזֶּה, וְלִכְאוֹרָה אֵין זֶה מוּבָן וַהֲלֹא אֵין עֲרוֹךְ עֲשִׂיָּה לְגַבֵּי אֲצִילוּת וְאִם כֵּן אֵיךְ בְּהַחֶסֶד שֶׁנִּמְשָׁךְ לְמַלְכוּת דַּאֲצִילוּת יִהְיֶה בוֹ תּוֹאַר וְצִיּוּר מֵעִנְיְנֵי עוֹלָם הַזֶּה.
[cup]אַךְ[/cup] הָעִנְיָן יוּבַן עַל פִּי מַה דְּאִיתָא בְּתוֹרָה אוֹר (פַּרְשַׁת וַיֵּרָא דְּרוּשׁ הַמַּתְחִיל פָּתַח אֵלִיָּהוּ) דְּיוֹתֵר מִמַּה שֶּׁאֵין עֲרוֹךְ עֲשִׂיָּה לְגַבֵּי אֲצִילוּת אֵין עֲרוֹךְ אֲצִילוּת לְגַבֵּי אוֹר אֵין סוֹף עַצְמוּת וּמַהוּת, דְּהִנֵּה הֲגַם שֶׁנִּתְבָּאֵר לְעֵיל (מַאֲמָר יט פֶּרֶק א) דַּאֲצִילוּת מוּבְדָּל מֵעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה, מִכָּל מָקוֹם הֲרֵי אֲצִילוּת הוּא גַם כֵּן עוֹלָם וְיֵשׁ בּוֹ עֶשֶׂר סְפִירוֹת בִּבְחִינַת אוֹרוֹת וְכֵלִים. וַהֲגַם שֶׁהוּא אֱלֹקוּת מַמָּשׁ מִכָּל מָקוֹם נִקְרָא אֱלֹקוּת וְלֹא אֱלֹקַהּ שֶׁמּוֹרֶה עַל הָעַצְמוּת, אֲבָל אֱלֹקות פֵּירוּשׁ גֶעטְלִיךְ, דְּהַיְינוּ שֶׁהוּא הִתְפַּשְּׁטוּת הֶאָרָה לְבָד, וּמִשּׁוּם זֶה שַׁיָּיךְ שָׁם עֶשֶׂר סְפִירוֹת וּבִפְרָט בְּהַכֵּלִים שֶׁהֵם בִּבְחִינַת מְצִיאוּת סְפִירוֹת חָכְמָה וָחֶסֶד, וְלָכֵן הִנֵּה מֵעֶשֶׂר סְפִירוֹת דַּאֲצִילוּת נִמְשָׁכִים עֶשֶׂר סְפִירוֹת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת עִלָּה וְעָלוּל שֶׁיֵּשׁ לָהֶם עֶרֶךְ וְשַׁיָּיכוּת זֶה לָזֶה, 
דְּהִשְׁתַּלְשְׁלוּת וְהִתְהַוּוּת חָכְמָה דַעֲשִׂיָּה מֵחָכְמָה דַאֲצִילוּת הוּא עַל דֶּרֶךְ מָשָׁל מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (בְּרֵאשִׁית רַבָּה פַּרְשָׁה יו"ד) אֵין לְךָ עֵשֶׂב מִלְּמַטָּה שֶׁאֵין לוֹ מַזָּל מִלְמַעְלָה הַמַּכֶּה בוֹ וְאוֹמֵר גְּדַל כו' וְגָבוֹהַּ מֵעַל גָּבוֹהַּ וְגָבוֹהַּ עֲלֵיהֶם דְּכָל הָעִנְיָנִים שֶׁיֶּשְׁנָם בְּעֵשֶׂב וּפְרִי הַגַּשְׁמִים הֲרֵי יֶשְׁנָם בְּמַזָּל הָרוּחָנִי וּבְכָל הַדַּרְגּוֹת דְּגָבוֹהַּ מֵעַל גָּבוֹהַּ. וְהִנֵּה הַתַּפּוּחַ הַגַּשְׁמִי יֵשׁ בּוֹ מְתִיקוּת גַּשְׁמִי הַאִם נֹאמַר שֶׁיֵּשׁ בְּכֹחַ הַצּוֹמֵחַ שֶׁבָּאָרֶץ גַּם כֵּן מְתִיקוּת, וּפְשִׁיטָא שֶׁאִי אֶפְשָׁר לוֹמַר כֵּן עַל הַמַּזָּל הָרוּחָנִי וְעַל הַדַּרְגּוֹת דְּגָבוֹהַּ מֵעַל גָּבוֹהַּ שֶׁאֵינָם בְּעֶרֶךְ וּמַהוּת זֶה, וּבֶאֱמֶת הִנֵּה הַטְּעָמִים דִּמְתִיקוּת וּמְרִירוּת גַּשְׁמִים נִמְשָׁכִים מִשָּׁרְשָׁם בְּחֶסֶד וּגְבוּרָה, שֶׁהַמְּתִיקוּת נִמְשָׁךְ מֵהַחֶסֶד וּמְרִירוּת מִגְּבוּרָה, וַהֲרֵי חֶסֶד וּגְבוּרָה הֵן עִנְיָנִים רוּחָנִים, רַק שֶׁבְּהִשְׁתַּלְשְׁלוּת וְסֵדֶר יְרִידַת הַמַּדְרֵיגוֹת מַדְרֵיגָה אַחַר מַדְרֵיגָה נִתְגַּשְּׁמוּ עַד שֶׁנִּמְשָׁךְ מִבְּחִינַת הַחֶסֶד מְתִיקוּת,
עִם הֱיוֹת שֶׁאֵין עֲרוֹךְ אֵלָיו מִכָּל מָקוֹם הוּא מִמִּזְגּוֹ וּתְכוּנָתוֹ, וּכְמוֹ שֶׁרוֹאִים כַּמָּה מִינֵי עֲרֵיבוּת וּמְתִיקוּת בְּרוּחָנִיּוּת, וּכְמוֹ תַּעֲנוּג שֶׁל דְּבַר חָכְמָה הוּא גַם כֵּן מְתִיקוּת, וְכֵן תַּעֲנוּג שֶׁל נִיגּוּן שֶׁהוּא לְמַטָּה בְּמַדְרֵיגָה מִמֶּנּוּ, וְלָכֵן בְּכַמָּה יְרִידוֹת וְהִשְׁתַּלְשְׁלֻיּוֹת נַעֲשָׂה מְתִיקוּת גַּשְׁמִי מַמָּשׁ,
וְכֵן הוּא הָעִנְיָן בְּחָכְמָה שֶׁיֵּשׁ בָּהּ גַּם כֵּן גָּבוֹהַּ מֵעַל גָּבוֹהַּ, וְהַחָכְמָה שֶׁבְּעֶשֶׂר סְפִירוֹת דַּעֲשִׂיָּה יֵשׁ לָהּ גַּם כֵּן עֶרֶךְ וְיִחוּס לְחָכְמָה דַאֲצִילוּת, דְּעַל יְדֵי רִיבּוּי רְבָבוֹת הִשְׁתַּלְשְׁלֻיּוֹת מַדְרֵיגוֹת נִמְשָׁךְ מֵחָכְמָה דַאֲצִילוּת חָכְמָה דַעֲשִׂיָּה,
וַהֲגַם דְּאֲצִילוּת מוּבְדָּל מִבְּרִיאָה-יְצִירָה-עֲשִׂיָּה וְאֵיךְ הוּא בְּדֶרֶךְ הִשְׁתַּלְשְׁלוּת עִלָּה וְעָלוּל, הָעִנְיָן הוּא דְּהַהִשְׁתַּלְשְׁלוּת זֹאת אֵינָהּ כְּמוֹ הַהִשְׁתַּלְשְׁלוּת בְּעוֹלָם אֲצִילוּת גּוּפָא בְּהָעֶשֶׂר סְפִירוֹת שֶׁבּוֹ, אוֹ בְּעוֹלָם הַבְּרִיאָה וּשְׁאָרֵי הָעוֹלָמוֹת בְּסֵדֶר הַהִשְׁתַּלְשְׁלוּת שֶׁבְּאוֹתוֹ הָעוֹלָם, אֶלָּא שֶׁהַהִשְׁתַּלְשְׁלוּת מֵאֲצִילוּת לִבְרִיאָה הוּא עַל יְדֵי הַפַּרְסָא, דְּעִנְיַן הַפַּרְסָא הוּא כְּמוֹ הַמָּשָׁל אֶל הַשֵּׂכֶל, שֶׁהַמָּשָׁל הוּא דָּבָר זָר מֵהַשֵּׂכֶל, וְלָכֵן הוּא מַעְלִים עָלָיו לְגַמְרֵי עַד שֶׁיֵּשׁ כָּאֵלּוּ שֶׁאֵינָם מְבִינִים כְּלָל שֶׁיֵּשׁ בָּזֶה אֵיזֶה נִמְשָׁל, אָמְנָם אַחַר כַּךְ כְּשֶׁמִּתְגַּלֶּה הַשֵּׂכֶל מִן הַמָּשָׁל וַדַּאי שֶׁאֵינוֹ בְּמַהוּת אוֹר הַשֵּׂכֶל כְּמוֹ שֶׁהָיָה קוֹדֶם שֶׁנִּתְלַבֵּשׁ בְּמָשָׁל, וּמִכָּל מָקוֹם הֲרֵי אֵינוֹ שֵׂכֶל אַחֵר, וַהֲרֵי הוּא אוֹתוֹ הַשֵּׂכֶל שֶׁנִּתְלַבֵּשׁ בְּהַמָּשָׁל אֶלָּא שֶׁהוּא בְּאוֹפֶן וּמַהוּת אַחֵר,
וּכְמוֹ כֵן הוּא עִנְיַן הַפַּרְסָא שֶׁמַּעְלִים וּמַסְתִּיר עַל בְּחִינַת חָכְמָה דַאֲצִילוּת וּכְשֶׁנַּעֲשָׂה בְּחִינַת חָכְמָה דִבְרִיאָה עַל יְדֵי הַפַּרְסָא הוּא כְּמוֹ אוֹר הַשֵּׂכֶל שֶׁמִּתְגַּלֶּה עַל יְדֵי הַמָּשָׁל, הֲרֵי הוּא אוֹר שֶׁל תּוֹלָדָה, וְהַיְינוּ אוֹר שֶׁנִּתְחַדֵּשׁ בְּעוֹלָם הַבְּרִיאָה, לְפִי שֶׁאֵינוֹ כְּלָל כְּמוֹ אוֹפֶן הַחָכְמָה שֶׁהִיא בַּאֲצִילוּת, אֲבָל מִכָּל מָקוֹם גַּם כֵּן אֵינָהּ מַהוּת וְדָבָר אַחֵר לְגַמְרֵי שֶׁהֲרֵי הִיא הַמְשָׁכָה וְהִתְהַוּוּת מִבְּחִינַת חָכְמָה דַאֲצִילוּת אֶלָּא שֶׁהוּא אוֹפֶן אַחֵר,
וְאֵינוֹ דוֹמֶה כְּמוֹ חָכְמָה דַאֲצִילוּת לְגַבֵּי אוֹר אֵין סוֹף, דִּבְאוֹר אֵין סוֹף עַצְמוּת וּמַהוּת הֲרֵי לֹא יֵשׁ שָׁם בְּחִינַת חָכְמָה כְּלָל וְאֵינוֹ מִגֶּדֶר הַחָכְמָה כְּלָל, וְאִם כֵּן בְּחִינַת חָכְמָה דַאֲצִילוּת הוּא בְּאֵין עֲרוֹךְ מַמָּשׁ לְגַבֵּי אוֹר אֵין סוֹף אֲבָל בַּאֲצִילוּת הֲרֵי כְּבָר יֵשׁ בְּחִינַת חָכְמָה, וְאִם כֵּן בְּחִינַת חָכְמָה דִבְרִיאָה וּכְמוֹ כֵן חָכְמָה דַעֲשִׂיָּה אֵינוֹ בִּבְחִינַת אֵין עֲרוֹךְ לְגַמְרֵי, אֶלָּא שֶׁחָכְמָה דַאֲצִילוּת מוּבְדֶּלֶת מֵחָכְמָה דִבְרִיאָה וְלָכֵן הִנֵּה הִתְהַוּוּתָם הוּא עַל יְדֵי הַפַּרְסָא דְּהַיְינוּ שֶׁבָּא בְּאוֹפֶן אַחֵר מִבְּחִינַת חָכְמָה דַאֲצִילוּת, אֲבָל מִכָּל מָקוֹם יֵשׁ לָהֶם עֶרֶךְ וְיַחַס אֶל חָכְמָה דַאֲצִילוּת.

[kitzur_begin]בְּעוֹלָם אֲצִילוּת עֶשֶׂר סְפִירוֹת אוֹרוֹת וְכֵלִים, חָכְמָה דַעֲשִׂיָּה לְגַבֵּי חָכְמָה דַאֲצִילוּת כְּעֶרֶךְ מְתִיקוּת גַּשְׁמִי לְגַבֵּי חֶסֶד הָרוּחָנִי וְעִם זֶה מִשְׁתַּלְשֵׁל מִמֶּנּוּ, הִשְׁתַּלְשְׁלוּת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה מֵאֲצִילוּת הוּא עַל יְדֵי הַפַּרְסָא כְּדוּגְמַת הַשֵּׂכֶל הַמִּתְגַּלֶּה מֵהַמָּשָׁל, אֲצִילוּת מוּבְדָּל מִבְּרִיאָה-יְצִירָה-עֲשִׂיָּה וְיֵשׁ לָהֶן יַחַס אֵלָיו.[kitzur_end]
