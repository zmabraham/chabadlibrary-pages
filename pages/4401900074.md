מַאֲמָר יז.
וְהִנֵּה הַבַּעֲלֵי עֲסָקִים שֶׁעוֹסְקִים כָּל יְמֵיהֶם בְּעֵסֶק מַשָּׂא וּמַתָּן וְאֵינָם שַׁיָּיכִים לְעִנְיָנִים אֲחֵרִים, כְּמוֹ לְהִתְאַוּוֹת תַּאֲוָה וּלְהִתְעַנֵּג בְּתַעֲנוּגִים גַּשְׁמִים, אֲשֶׁר הָעֵסֶק בְּמַשָּׂא וּמַתָּן מִצַּד עַצְמוֹ הוּא הֶיתֵּר, וְיֵשׁ בּוֹ צוֹרֶךְ, וּכְמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה בְּמִשְׁנָה (אָבוֹת פֶּרֶק ב) יָפֶה תַּלְמוּד תּוֹרָה עִם דֶּרֶךְ אֶרֶץ כו', אָמְנָם יֵשׁ בָּזֶה כַּמָּה שְׁטוּתִים מִצַּד הַיֵּצֶר הַמַּעֲבִירִים אֶת הָאָדָם עַל דַּעְתּוֹ וְעַל דַּעַת קוֹנוֹ, וְהַיְינוּ עִנְיַן רִיבּוּי הַטִּרְדָא בְּעֵסֶק, שֶׁמּוּטְרָד בָּזֶה יוֹמָם וָלַיְלָה עַד שֶׁמּוֹנֵעַ אוֹתוֹ מִלִּימּוּד הַתּוֹרָה וּמִלְּהִתְפַּלֵּל בְּצִבּוּר וּלְכַוֵּין פֵּירוּשׁ הַמִּלּוֹת עַל פִּי פָּשׁוּט עַל כָּל פָּנִים, מִפְּנֵי שֶׁמְּדַמֶּה בְנַפְשׁוֹ שֶׁעַל יְדֵי זֶה שֶׁיִּתְעַכֵּב בִּתְפִלָּה לִשְׁמוֹעַ בְּאָזְנָיו מַה שֶּׁהוּא מוֹצִיא בְּפִיו בְּכַוָּונָה פְשׁוּטָה כַּנַ"ל וּלְהִתְפַּלֵּל בְּצִבּוּר וּלְהִתְעַכֵּב זְמַן בְּלִימּוּד הַתּוֹרָה עַל יְדֵי זֶה יַפְסִיד שֶׁלֹּא יִקְנֶה וְיִמְכּוֹר בָּעֵת הַהִיא, וּבֶאֱמֶת זוֹהִי שְׁטוּת נִפְלָא,
דְּכַאֲשֶׁר הָאָדָם יָשִׂים לִבּוֹ וְדַעְתּוֹ לִרְאוֹת אֶת הַנַּעֲשֶׂה אִתּוֹ וּבִכְלָלוּת הָעוֹלָם, הֲלֹא יִרְאֶה לָעֵינַיִם אֲשֶׁר כַּמָּה פְעָמִים יֵלֵךְ בְּדַרְכֵי הַמִּסְחָר וְיִשְׁמוֹר הַדְּבָרִים בְּעִתָּם וּבִזְמַנָּם וְיִתְעַסֵּק הַרְבֵּה וְלֹא יַרְוִיחַ מְאוּמָה אוֹ גַם יַפְסִיד, וּבִמְעַט זְמַן וּבְמִיעוּט עֵסֶק יַרְוִיחַ, וְאִם כֵּן מִי יֹאמַר אֲשֶׁר בְּהָעַכָּבָה יַפְסִיד אוּלַי יוֹתֵר טוֹב לְפָנָיו שֶׁלֹּא יַרְבֶּה בְּהִתְעַסְּקוּת וּבְעֶזְרַת הַשֵּׁם יִתְבָּרֵךְ יַרְוִיחַ בִּמְעַט זְמַן, וְכַאֲשֶׁר עוֹד יִתְבָּאֵר לְקַמָּן.
[cup]וּבִפְרָט[/cup] אִם לֹא הָיָה שׁוֹכֵחַ עַל ה' אֲשֶׁר הוּא הַנּוֹתֵן לוֹ כֹּחַ לַעֲשׂוֹת חַיִל וּבִרְכַּת ה' דַּוְקָא הִיא תַעֲשִׁיר, וְעֶצֶם הָעֵסֶק הוּא רַק בְּדֶרֶךְ עֲשִׂיַּית כְּלִי בְּמַה שֶּׁתִּשְׁרֶה הַבְּרָכָה, אֲבָל הָעִיקָּר הוּא בִּרְכַּת ה', אִם כֵּן הֲלֹא צָרִיךְ לִהְיוֹת מָסוּר לַה' (דַארְף עֶר דָאךְ זַיְין אִיבֶּערְגֶעגֶעבֶּען דֶער עִיקָּר צוּ אֱלֹקוּת) שֶׁהוּא הַזָּן וּמְפַרְנֵס אוֹתוֹ, וְרַק צְרִיכִים כְּלִי לָזֶה וְהוּא הָעֵסֶק, אֲבָל דַּי בְּמִיעוּט הִתְעַסְּקוּת גַּם כֵּן.
[cup]וְהָעִנְיָן[/cup] הוּא דְּהִנֵּה כְּתִיב (דְּבָרִים טו. ח) וּבֵרַכְךָ ה' אֱלֹקֶיךָ בְּכֹל אֲשֶׁר תַּעֲשֶׂה, וְצָרִיךְ לְהָבִין לָמָּה צְרִיכִים הָעֲשִׂיָּה וְהָעֵסֶק כְּלָל מֵאַחַר שֶׁהַבְּרָכָה מֵאֵת ה' תָּבֹא אִם כֵּן גַּם בְּלִי עֵסֶק וְהִשְׁתַּדְּלוּת כְּלָל תָּבֹא בִּרְכַּת ה', וּמִמַּה שֶּׁאוֹמֵר בְּכֹל אֲשֶׁר תַּעֲשֶׂה מַשְׁמָע שֶׁצָּרִיךְ לִהְיוֹת עֲשִׂיָּה וְעֵסֶק, וּבְיוֹתֵר אֵינוֹ מוּבָן לְפִי מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (בֵּיצָה טז. א) כָּל מְזוֹנוֹתָיו שֶׁל אָדָם קְצוּבִים לוֹ מֵרֹאשׁ הַשָּׁנָה עַד יוֹם הַכִּפּוּרִים, וְאִם כֵּן מַהוּ הַצּוֹרֶךְ לַעֲסוֹק בְּאֵיזֶה מַשָּׂא וּמַתָּן אוֹ אוּמָּנוּת שֶׁעַל יְדֵי זֶה יִתְפַּרְנֵס, הֲלֹא וַדַּאי מַה שֶּׁנִּקְצְבָה לוֹ פַּרְנָסָה בְּחֶסֶד עֶלְיוֹן יוּמְשַׁךְ לוֹ וְלָמָּה צָרִיךְ עֵסֶק וְהִשְׁתַּדְּלוּת בָּזֶה, גַּם צָרִיךְ לְהָבִין מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (רֹאשׁ הַשָּׁנָה טז. א) אָדָם נִידּוֹן בְּכָל יוֹם וְאֵיךְ יִתַּוֵּוךְ זֶה עִם מַאֲמָר הַנַ"ל דִּמְזוֹנוֹתָיו קְצוּבִים לוֹ מֵרֹאשׁ הַשָּׁנָה כו', דְּוַדַּאי גַּם רַבִּי יוֹסֵי שֶׁאוֹמֵר דְּאָדָם נִידּוֹן בְּכָל יוֹם מוֹדֶה דְּהַדִּין וּמִשְׁפָּט הוּא בְּרֹאשׁ הַשָּׁנָה וְאָז נִקְצָב לוֹ כָּל הִצְטָרְכוּתוֹ בְּבָנֵי חַיֵּי וּמְזוֹנֵי, וְאִם כֵּן מַהוּ עִנְיַן הַדִּין וּמִשְׁפָּט שֶׁבְּכָל יוֹם.
[cup]וְגַם[/cup] צָרִיךְ לְהָבִין מַה שֶּׁאָנוּ מְבַקְשִׁים בְּכָל יוֹם בִּשְׁמוֹנֶה עֶשְׂרֵה עַל פַּרְנָסָה וּרְפוּאָה וכו' הֲלֹא כָּל זֶה נִקְצַב מֵרֹאשׁ הַשָּׁנָה כו'. וּבְזֹהַר (חֵלֶק ב סב. ב) אִיתָא דְּרַב יֵיבָא סָבָא הָיָה מִתְפַּלֵּל בְּכָל יוֹם קוֹדֶם הָאֲכִילָה עַל מְזוֹנוֹ שֶׁיִּנָּתֵן לוֹ, גַּם אַף שֶׁהָיָה מְזוֹנוֹ כְּבָר מוּכָן לְפָנָיו. דְּאַף עַל גַּב דַּהֲוָה לֵיהּ לְמֵיכְלָא לָא הֲוִי מְתַקֵּן לֵיהּ עַד דְּשָׁאִיל לֵיהּ מִבֵּי מַלְכָּא בִּצְלוֹתָא, וְצָרִיךְ לְהָבִין מַהוּ עִנְיַן תְּפִלָּה זוֹ.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]שְׁטוּתֵי בַּעֲלֵי הָעֲסָקִים שֶׁאֵינָם קוֹבְעִים עִתִּים לַתּוֹרָה וּמֵהֶם שֶׁבִּלְתִּי מִתְפַּלְלִים בְּצִבּוּר, שׁוֹכְחִים עַל הָעִיקָּר דְּבִרְכַּת ה' הִיא תַעֲשִׁיר, סְתִירַת הַמַּאֲמָרִים דִּמְזוֹנוֹת קְצוּבִים מֵרֹאשׁ הַשָּׁנָה וְאָדָם נִידּוֹן בְּכָל יוֹם.[kitzur_end]
