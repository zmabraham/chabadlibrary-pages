שמקבל מזה כו'. וכמו"כ הוא באה' אלקי' שע"י ההתבוננות והרגש הטוב והעילוי דאלקות ה"ז ג"כ הרגש עצמו כו', ובפרט מה שקרבת אלקים לו טוב שז"ע כי הוא חייך כו', שזה בבחי' הרגש עצמו כו', והגם שהוא הרגש אלקי כו' מ"מ ה"ז בבחי' הרגשה שמרגיש א"ע דנפשו יקרה בעיניו וחפץ במה שהוא טוב לו ורוצה בהגילוי כו', ולכן מהרגשת האה' יכול להיות אחר התפלה הרגשה חומרי', וכמו לא יחפוץ כסיל בתבונה כ"א בהתגלות הלב דמאהוי"ר בהתגלות הלב יכול להיות אחיזה כו', וזה"ע עה"ד טו"ר דאין הכוונה שהוא בחי' רע ח"ו שהרי עה"ד הוא במדרי' דקדושה, רק שיוכל להשתל' מזה רע להיות שדעת הוא בחי' הרגשה ע"כ יכול להשתל' מזה רע כו' וכמ"ש בלק"ת בהבי' דמי מנה (ובאמת זהו עבודה אמיתית ומי יתן והי' לבו תמיד בבחי' אה' זו, וע"ה שזהו הרגש עצמו עז"נ ה' מתיר אסורים כו' וכמשנת"ל, וצ"ל ההשתדלות שלא יהי' ההרגשה כ"כ, והיינו ע"י שההשגה האלקי' ממה שנתפעל באה', הגם שבעת האה' היא מתעלמת מ"מ צריכה להיות בבחי' מקיף עכ"פ על התפעלות האה', ואז אינו במורגש כ"כ כו', ובפרט
כאשר יש יסוד מהביטול דחכ', והיינו במי שיש בו ביטול בעצם (א אוועק גילייגטקייט) גם ההתפעלות שע"י ההתבוננות לא יהי' במורגש ולא יהי' מזה הרגשה חומרי' כו', רק שבפרטי מדרי' בעבודה הוא בחי' יש מי שאוהב כו'), ובכלל הוא בבחי' התפעלות מורגשת כו'. אבל בחי' אה' דבכל מאדך הוא האה' דבחי' יחידה שבנפש והיא האהבה וההתקשרות באוא"ס מצד האוא"ס עצמו כו' לא מצדו היינו מה שלו טוב כ"א מצד הא"ס עצמו כו', וכמו האה' עצמי' דבן אל האב שאינו מצד הטוב והעילוי ולא מצד ההשפעה כו', והיא אה' שאינה תלוי' בדבר שאינו חפץ בזה לעצמו דבר ואין בזה שום מבוקש כלל כ"א אה' עצמי' לפי שהוא שרשו ומקורו כו', וכמו"כ הוא האה' עצמי' דנש"י לאלקות מצד שרש ומקור נפשם שמושרשים בעצמות ומקושרים בעצמות כו', והוא התגלות בחי' פנימי' נקודת הלב בבחי' רצון פשוט לה' כו', והיא אה' הבלתי מורגשת כלל כו'. ועז"א ומל ה"א את לבבך היינו ההעברה דבחי' חיצוני' הלב שהיא האה' שבבחי' הרגשה, דלגבי האה' העליונה הזאת האה' שהיא בבחי' הרגשה הגם שהיא אה' אלקית ה"ז ג"כ
מסתיר כו', וע"כ ימול ויעביר האה' דהרגשה הנ"ל בכדי שיתגלה בחי' פנימי' הלב כו' (וי"ל שזהו ג"כ ע"י מרירות נפשו שמתמרמר על ענין ההרגשה כו', וכמו הכובד ראש קודם התפלה שעי"ז הוא שבירת החומרי' המסתיר ומתגלה אור האה' בתפלה כו' כמשנת"ל, כמו"כ הוא ג"כ ע"י המרירות ע"ז שהוא מורגש בעצמו ועבודתו היא בבחי' הרגש (והיינו במי ששייך להעבודה דרעו"ד ה"ה מתמרמר על ההרגשה הנ"ל), ועי"ז מתגלה בחי' פנימי' הלב כו', וזהו העבודה בר"ה ועשי"ת שגם עובדי ה' בכל השנה משפילים א"ע וה"ה כחוטאים בעיני עצמם, והרי אינם שייכים לחטא ועון ח"ו רק שהן בעבודתם כחוטאים בעיני עצמם בזה שהעבודה היא במורגש כו', ועי"ז הוא התגלות בחי' ההתקשרות עצמי דעצם הנשמה כו', וכמו"כ התגלות פנימי' הלב הוא ע"י מרירות הנ"ל, דבאמת בחי' פנימי' הלב דהיינו האה' דרעו"ד היא באה מלמע' רק הכלי לזה הוא ההעברה והביטול דהאה' המורגשת כו', והיינו ע"י המרירות כו'). ולכן לא נא' כאן ערלת הלב כי אינו שייך לקרוא זה בשם ערלה כלל כי זהו אה' לאלקות, רק שלהיותה עדיין בבחי' יש מי שאוהב היינו בבחי' הרגשה ע"כ צריכים
