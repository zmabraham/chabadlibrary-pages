שבמוחי' דחב"ד כו', וכתי' חכמת אדם תאיר פניו כר"א שמצא תוספ' חדשה וצהבו פניו שהוא בא מחידוש כח המשכיל כל שכל שנק' פנימי' ח"ס כו' והמוחי' דחו"ב יאירו בפני' ממש והן בחי' תרין תפוחי', וכמו במ"ת שנדמה להם כזקן כו' בהדרת אור פנים העליוני' דבחי' א"א כמ"ש באור פניך נתת לנו כו' וכן א' יאר ה' פני"ו אליך פני"ו העליוני' שנק' אנפי רברב"י כו' והוא שרש הי"ג מדה"ר כידוע (ויש פני' עליוני' מזה שמשם מקור ושרש להארת פני' די"ג מדה"ר דא"א והוא מ"ש ישא ה' פני"ו כו' והיינו בבחי' א"ק כו'), ובחי' אנפי זוטרי הוא במוחי' דחו"ב שבמדות עליונו' דז"א כו' וז"ש כתפארת אדם כו' תפארת בחורי' כחם כו', וכאשר א' ויעבור ה' על פני"ו היינו שעלה ועבר ודילג למעלה מכל בחי' פני' העליוני' דאנפי זוטרי ואנפי רברבי דא"א וא"ק כו' עד בחי' עצמותו ממש שלמעלה מבחי' אד"ם כמ"ש כי לא אד"ם הוא כו' (וכידוע דגם א"ק נק' אד"ם דבריא' בכללות), והיינו ענין ב' עליו' דגרשי"י' הנ"ל ואח"כ ויקרא ה' ה' כו' פי' אחרי ב' העליו' זו למעלה מזו הנה ממקום שעלה לשם משם ירד וכמו שעלה בב' עליו' זו למעלה מזו כך ירד ב' ירידו' זו למטה מזו. והענין הוא כי ב' הויו' הללו א' הוי' דחכמה שבכת"ר בדרך כלל שנק' אנפי רברבי כנ"ל והב' הוי' שבמדו' דאו"א שמלובש בז"א ולשניהם עבר ודילג בעליית"ו למעלה בעצמו', ואחרי העלי' צריך לירד באותן ב' מדרגו' דה' ה' ביריד' אחר יריד', ירידה א' מעצמו' אא"ס לבחי' הוי' דפנימי' ח"ס דא"א וע"י (או לחכמ' דא"ק כו') וירידה הב' מפנימי' ח"ס דא"א לחכמה ובינה דז"א שנק' אנפי זוטרי כו' דוקא כדי שמשם יוכל לבא האור גם לכל עולם לפי ערכו עד למטה ממש כנ"ל וד"ל. וזהו ה' ה' פסיק טעמא בגוייהו לפי שאין ערוך לה' שבז"א לה' דח"ס ועכ"ז דילג בירידתו מן העצמות כמו שדילג מלמטה למעלה בעלייתו עד"מ הדולג ועולה בסולם בשליבות ומדרגות ירידתו ג"כ באותן en.מדרגו' וד"ל.
