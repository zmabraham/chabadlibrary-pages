שהשם הוא המחבר כו') הרי עצם הנפש מוגדר בזה כו', וכידוע דבהתלבשות הנפש בגוף א"א שיתלבש בגוף אחר כו', וכמו שמצינו גבי אלי' ששאל אותו רשב"י מפני מה לא הי' בהאדרא והשיב מפני שהי' צריך להציל את רהמ"ס וחביריו, ואי' בכהאריז"ל דלכבודו של רהמ"ס הי' צ"ל בגוף דוקא וכן לכבודו של הרשב"י הי' צ"ל ג"כ בגוף וא"א לנשמה להיות בב' גופים כו', ומה שאלי' נמצא בברית מילה ברגע אחד בכמה מקומות זהו רק ניצוץ מנשמתו, אבל כאשר בא בגוף שהנשמה מתלבשת בזה א"א להיות בב' גופים כו', הרי שהנשמה מוגדרת בגוף כו', והשם הוא ג"כ להנפש כמו שמתלבש בגוף דוקא לא כמו שהוא למעלה כו', ובזה העצם מוגדר כו', ואנו רואין שהשם קורא וממשיך את חיות הנפש משרשו ומקורו וכמו בהתעלפות ר"ל שיכולים להשיב נפשו ע"י שקוראין אותו בשמו, וידוע דהשבת הנפש זהו בדרך התחדשות הבא מהנפש כמו שהוא למעלה מבחי' מקור להתפשטות וגילוי חיות הנפש כו', וכמ"ש במ"א בענין החיות שע"י אכילה ושע"י ריח חזק כו', ועמשנת"ל מפ' רט"ז ואילך בענין קרבנות וקטרת כו'. ובעבודה הוא העבודה
שע"פ טו"ד ובחי' רעו"ד כו', דעבודה שע"פ טו"ד היא בכחות הגלוים דשכל ומדות בהשגה והתבוננות והתעוררות אהוי"ר כו', וכמו"כ מגיע למעלה ג"כ בבחי' הגילוים דסדר השתל' ובשרש ומקור ההשתל' כו', ורעו"ד מגיע בבחי' עצמות אוא"ס כו', וכמ"ש בזהר דלית מח' תפיסא בי' אבל נתפס איהו ברעו"ד כו', דעצמות אוא"ס דלמחתב"כ נתפס איהו ברעו"ד כו', ובפרט בתשובה שמגיע בבחי' עצמות אוא"ס ומשם סליחת עונות כו', שהרי ע"י חטא ועון נעשה סילוק האור כו', ובכדי להמשיך את האור צ"ל שרש ההמשכה ממדרי' גבוה יותר והיינו מבחי' עצמות אוא"ס דוקא וכמ"ש כי עמך הסליחה עמך דוקא כו', והיינו ע"י התעוררות הנפש בבחי' פנימיות ועצמיות הנפש דזהו"ע ממעמקים קראתיך מעומק לעומק כו', וכמו עד"מ נהר שנפסק שצריכים לחפור בעומק המעין כו', והו"ע התשובה שהיא דוקא מעומק ופנימיות הנפש וזה מגיע בבחי' עצמות כו', וכמו"כ השם מגיע בעצמות כו', וז"ע שם המלך שקוראין בר"ה להיות דבר"ה הוא עליית המל' שעולה בשרשה ומקורה העצמי בבחי' עצמות א"ס, ואנו צועקים המלך להמשיך בחי'
המלוכה מן העצמות כו', ועל העדר עבודה כתי' אין קורא בשמך וע"י הקריאה ממשיכים בחי' המל' כו', וזהו אמרו לפני מלכיות כו' שע"י הקריאה ממשיכים בחי' המלוכה כו', וכמו"כ הוא בקריאת השמות פרטיים של הכחות ממשיכים את הכחות משרשם ומקורם העצמי, וכמו שע"י קריאת שם חכם ממשיכים כח חכמתו מהעלמו העצמי כו', דהיינו שגם כשלא יש גילוי החכ' בכח שכלו מתחדש אצלו חכ' ושכל מכח ההיולי העצמי ע"י קריאת שם חכם כו' ועמ"ש בהוספות לתו"א בהבי' דואני נתתי, ובזה יש יתרון מעלה בשם על אור שהאור אינו ממשיך את העצם והשם ממשיך את העצם כו' וזהו מצד עצם השם היינו אותיות השם כו' וכמשנת"ל פ' שו דגם עצם השם הוא בבחי' חיות כו', דעם היות שהשם הוא נבדל מן הנפש כנ"ל ה"ה מושרש בעצם הנפש ומשו"ז הוא ממשיך את העצם כו'.
[kitzur][cup]קיצור.[/cup] וכמו באור כ"ה בחיות הנפש שהוא גילוי, ויודעים שהנפש הוא חי בעצם לא בהתפשטות ולא בהמשכה, וכן הכחות הגם שהן חוץ להעצם, ה"ה גילוי, ויודעים מהם הנפש בשרשו, ויש להם קירוב אל העצם, שמתענג בהכחות[/kitzur]
