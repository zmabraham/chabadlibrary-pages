שנמשך ע"י התשו' דוקא דבע"ת משכין לי' בחילא יתיר בכח או"ח דוקא כו', וכמארז"ל דבמקום שבע"ת עומדים צ"ג אינם יכולים לעמוד דצ"ג בעבודה שלהם גם בבחי' אה' בתענוגים א"י לעמוד במקום בע"ת כו', דזהו כלל גדול דצער ומרירות הנפש מגיע למע' מעונג הנפש, דגם בחי' עונג עצמי אינו מגיע בעצמו' הנפש כמו הצער והמרירות מההיפך כו', וכמ"ש מן המיצר קראתי מן המיצר והדוחק דוקא מה שצר לו מאד מהריחוק מהאלקות ובפרט בחטאים ועוונות, וזה דוקא מגיע בעצם נקודת נפשו ומגיע למע' בפנימי' ועצמו' א"ס כו'. וכמו עד"מ לדלג חפירה גדולה שצ"ל בכח עצמו' הנפש שזהו דוקא כאשר בורח מן המות ר"ל כמו במלחמה וכה"ג שאז מתגלה כח עצמו' הנפש לדלג כו', וכמו"כ המיצר הגדול (דער ניט גוט) מהחטא ועון זהו דוקא אשר מביא ומגלה עצם הנשמה בבחינת הזזת כל עצמותה כו', וזהו דוקא שמגיע בפנימי' ועצמי' א"ס כו', דכאשר מן המיצר קראתי אז ענני במרחב י"ה בבחי' מרחב העצמי דא"ס כו', וכמ"ש ממעמקים קראתיך מבחי' עומק לעומק, דעומק ופנימי' הנפש מגיע בבחי' עומק ופנימי' א"ס כו'.
[cup]ועוד[/cup] זאת שבכאן הרע נהפך לטוב והיינו רע גמור לא רק בירורי נוגה כ"א שרע גמור נהפך לטוב ממש כו', דבתשובה זו הרי זדונות נעשות כזכיות כו', ונת"ל פרט"ז דכאשר דבר חריף וקיוהא נהפך למאכל טוב וערב הרי העריבות בזה יותר באין ערוך מהדבר הערב בעצם כו', והיינו דרע גמור כשנהפך לטוב ממש ה"ז ריח חזק שמגיע בבחי' חוטמא דעתיק בחי' פנימי' עתיק ופנימי' ועצמי' מו"ס כו'. והו"ע שמן הטוב שלמע' משמן המנורה שהוא בחי' שמן משחת קדש שלמע' מהחכ' דהיינו בחי' ח"ס, אבל הוא בחי' חיצוני' דח"ס שבבחי' מקור לחכ' בחי' מקיף דאו"י כו', אבל שמן הטוב הוא בחי' מוחא דעתיקא בחי' עצמי' ח"ס בחי' מקיף דאו"ח שסתום ונעלם בעצם כו', ומגיעים לזה דוקא ע"י תשו' בבחי' או"ח הבא מן הריחוק דאו"ח חוזר לקדמותו כו'. וזהו דגבי משיח כתי' והריחו ביראת ה' דמורח ודאין כו', וידוע דמשיח הוא בחי' פנימי' עתיק כו', אמנם מה שדאין היינו בחי' החכ' דהרי כל דין ומשפט הוא מבחי' החכ' כו', וכתיב בו לא למראה עיניו ישפוט ולא למשמע אזניו כו', דהיינו לא בבחי' חו"ב כ"א
מבחי' ח"ס כו', ומה שמריח ודאין היינו בחי' עצמי' ח"ס שבזה בחי' פנימי' עתיק ומגיעים לזה ע"י התשו' דוקא, דא"א להגיע לבחי' עצמו' ממש כ"א בכח או"ח דוקא כו'. וזהו דמשיחא אתא לאתבא צדיקייא בתיובתא דבכדי שצדיקים יזכו להגילוי דמשיח זהו ע"י התשו' דוקא כו', והיינו לפי שזהו בחי' עצמו' א"ס בחי' פנימי' עתיק כו', וא"א להגיע לזה כ"א בכח או"ח דוקא כו', וזהו דבענן הקטרת דיו"כ דוקא שז"ע התשו' כתי' כי בענן אראה בעצמותי כו'. וזהו ב' המדרי' דח"ס, בחי' חיצוני' ח"ס בחי' מקיף דאו"י שבבחי' מקור לחכ', והו"ע שמן המנורה שנמשך ע"י ריח הקטרת בחי' רעו"ד שעי"ז נעשה ג"כ ביטול מהות הנה"ב כו', אבל בחי' פנימי' ח"ס זהו בחי' פנימי' עתיק והו"ע שמן הטוב דעתיקא כו', ולכן להגיע לזה זהו דוקא ריח הקטרת דיו"כ שהו"ע התשו' שבאה מן המיצר דוקא בכח או"ח כו', שבזה ממשיכים בחי' פנימי' ועצמי' א"ס בחי' עצמי' מו"ס שבזה בחי' פנימי' עתיק כו'.
