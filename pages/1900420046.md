ויצא לחוץ כו' ופורץ גדר הקדום בהנהגה מרוממו' לשפלו' וענוה וכה"ג לפי שהשמח' מביא כל העלם לגילוי כו', וכך יובן למעלה שע"י בחי' השמח' עליונה שבמוה"ע ית' ממש יוצא מבחי' ההעלם אחר העלם שבפנימי' עצמותו שנק' סתימא דכל סתימין לידי גילוי דוקא שלא כפי גדר הקדום שהוא סתום ונעלם בהיכלו כולו אומר כבוד, וכמו שאו שערי' ראשיכם[ftnref_172_172] ויבא מלך הכבוד כו' שזהו בחי' מל' דא"ס בבחי' הפנימי' ועצמו' שלו ממש שמרומם לבדו (כמו ברוממו' העצמו' ביחידה שבנפש באדם), וכמו שא' בזוהר דבסוכו' בחמשה עשר יום כו' בי' מלכא בתרעא[ftnref_173_173] יתיב כו' (שהוא זמן שמחה למעלה. ואע"פ ששמחה זו רק בבחי' מוחי' דאימא ובחי' מקיפי' דחג הסוכו' ג"כ במקיפי' דאימא לבד כידוע, עכ"ז שמחה זו באה מלמעלה משם הוי' דמועדי' והוא בבחי' אתדל"ע בעצמו' א"ס ב"ה ממש רק שנמשך משם הארה משמחה העצמי' גם בבינ"ה ונקראי' מועדי' להוי' בכלל שאז נתגלה מהעלמו העצמי' מצד בחי' שמחה עצמי' דוקא, ולכך בג' רגלי' א' יראה כל זכורך[ftnref_174_174] את פני הוי' כו' שכל ישראל יכולי' לראות את פני הוי' בעזרה לפי שמתגלה לכל כקטן כגדול כמבוא' למעלה במלך ששמח ביותר שמתגלה בשער המלך לעין כל ומדבר עם כל אדם כו'). 
[cup]וענין[/cup] פרצת הגדר הזה שע"י השמחה שבמוה"ע ית' היינו שפורץ גדר של מא' קו המדה שהוא הקו"ח שנמדד במדה ושיעור אחר הצמצום הראשון הנק' מק"פ דישת חוש"ך סתר"ו[ftnref_175_175] וכל מה שנגלה מאור א"ס הכל הוא רק אפס קצהו בגדר ומדה מצומצמת מאד בקו"ח זה, וגם הקו"ח הזה נק' קו המדה שמודד שיעור ומדה לכל אור ושפע באבי"ע במקיף ופנימי ואו"כ (כמ"ש במ"א בענין וי"ו דוהנורא[ftnref_176_176]) כפי סדר ההשתלשלו' התמידי' (אף בשבתו' ויו"ט שבהן גילוי תוס' אורות מקיפי' בחו"ב שנק' קד"ש ומקראי קדש כנ"ל וכן בר"ה ויו"כ שמתגלה גם מפנימי' המל' דא"ס להיו' מקור למלכו' כ"ע כנ"ל וביו"כ נתגלה מבחי' רעוא דכ"ר בי"ג מדה"ר כמ"ש כי בענן אראה[ftnref_177_177] כו', הכל הוא בא במדה ושיעור דוקא, והיינו לפי ערך אתדל"ת בהמתקו' הגבו' בחסדי' כמו בכל יו"ט בקרבנו' ועולי רגל, וכן בשבת וחדש אע"פ שא' סגור יהי' כו' ביום השבת וחדש יפת"ח[ftnref_178_178] מעצמו שזהו גילוי כבוד אלקי ישראל בבחי' מל' דא"ס עצמו ממש, וכן בנשמו' בג"ע עולי' מדי חדש ושבת כו' שזהו יציאתו מהיכלו לחוץ
