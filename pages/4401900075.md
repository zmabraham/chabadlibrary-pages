# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר יח. > פֶּרֶק א

מַאֲמָר יח.
וְהִנֵּה לְהָבִין סְתִירַת הַמַּאֲמָרִים בְּעִנְיַן מְזוֹנוֹתָיו וּצְרָכָיו שֶׁל הָאָדָם, דְּפַעַם אָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה כָּל מְזוֹנוֹתָיו שֶׁל אָדָם קְצוּבִים לוֹ מֵרֹאשׁ הַשָּׁנָה עַד יוֹם הַכִּפּוּרִים, וּפַעַם אָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה אָדָם נִידּוֹן בְּכָל יוֹם, וְרַב יֵיבָא הָיָה מִתְפַּלֵּל בְּכָל יוֹם שֶׁיִּנָּתֵן לוֹ מְזוֹנוֹ וְכִדְאִיתָא בְּזֹהַר דְּאַף עַל גַּב דַּהֲוִי לֵיהּ לְמֵיכְלָא לָא הֲוָה מְתַקֵּן לֵיהּ עַד דְּשָׁאִיל לֵיהּ מִבֵּי מַלְכָּא בִּצְלוֹתָא,
הַיְינוּ שֶׁיֵּשׁ ג' עִנְיָנִים הָא' מַה שֶּׁמְּזוֹנוֹתָיו וּצְרָכָיו שֶׁל אָדָם קְצוּבִים לוֹ מֵרֹאשׁ הַשָּׁנָה הַב' דְּאָדָם נִידּוֹן בְּכָל יוֹם, וְהַג' דְּאַף גַּם שֶׁכְּבָר הַכֹּל מוּכָן לְפָנָיו הִנֵּה אֵינוֹ מִשְׁתַּמֵּשׁ בּוֹ עַד דְּשָׁאִיל לֵיהּ מִבֵּי מַלְכָּא בִּצְלוֹתָא,
דְּהִנֵּה מְבוֹאָר בִּפְרִי עֵץ חַיִּים אֲשֶׁר מֵרֹאשׁ הַשָּׁנָה עַד יוֹם הַכִּפּוּרִים הוּא תִּקּוּן וּבִנְיַן הַמַּלְכוּת דַּאֲצִילוּת עַד שֶׁבִּתְפִלַּת נְעִילָה הוּא תַּכְלִית הָעֲלִיָּה שֶׁל הַמַּלְכוּת שֶׁמִּתְעַלֵּית עַד דִּיקְנָא דַאֲרִיךְ אַנְפִּין, שֶׁגַּם בַּשַּׁבָּת אֵין לָהּ עֲלִיָּה כָּזוֹ, דְּבַשַּׁבָּת מִתְעַלֵּית בִּמְקוֹם אַבָּא, וְרַק זָ"א [זְעֵיר אַנְפִּין] עוֹלֶה עַד הַדִּיקְנָא, וּבִנְעִילָה דְיוֹם הַכִּפּוּרִים הַמַּלְכוּת עוֹלָה עַד הַדִּיקְנָא וּמְקַבֶּלֶת שָׁם ה' גְבוּרוֹת דְּעַתִּיק יוֹמִין, וְזֶהוּ עִנְיַן הַחוֹתָם דִּנְעִילָה דְיוֹם הַכִּפּוּרִים שֶׁהוּא עִנְיַן הַה' גְבוּרוֹת דְּעַתִּיק יוֹמִין.
[cup]וְצָרִיךְ[/cup] לְהָבִין מַהוּ עִנְיַן הַה' גְבוּרוֹת דְּעַתִּיק יוֹמִין שֶׁמְּקַבֶּלֶת הַמַּלְכוּת בִּתְפִלַּת נְעִילָה, וַהֲלֹא בִּנְעִילָה אָז עִיקַּר הַמְשָׁכַת בְּחִינַת רַחֲמִים פְּשׁוּטִים מַמָּשׁ, וּכְמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (פְּסִיקְתָּא דְרַב כַּהֲנָא פַּרְשָׁה וּלְקַחְתֶּם לָכֶם) הַסְּלִיחָה מוּפְקֶדֶת אֶצְלְךָ מֵרֹאשׁ הַשָּׁנָה עַד יוֹם הַכִּפּוּרִים, דְּיוֹם הַכִּפּוּרִים בִּכְלָל הוּא בִּבְחִינַת רַחֲמִים פְּשׁוּטִים, וּבִפְרָט בְּעֵת הַנְּעִילָה שֶׁאָז הוּא בְּחִינַת תַּכְלִית הָרַחֲמִים וְעֵת רָצוֹן כְּיָדוּעַ, וְאִם כֵּן מַהוּ עִנְיַן הַה' גְבוּרוֹת שֶׁמְּקַבֶּלֶת אָז, וְגַם מַהוּ עִנְיַן הַחוֹתָם.
[cup]אַךְ[/cup] הָעִנְיָן הוּא דְּהִנֵּה יָדוּעַ דִּבְחִינַת מַלְכוּת דַּאֲצִילוּת הִיא מְקוֹר הָעוֹלָמוֹת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה, וּכְמוֹ שֶׁכָּתוּב (תְּהִלִּים קמה. יג) מַלְכוּתְךָ מַלְכוּת כָּל עוֹלָמִים, דְּכָל עוֹלָמִים שֹׁרֶשׁ הִתְהַוּוּתָם הוּא מִבְּחִינַת הַמַּלְכוּת,
וְנִקְרֵאת אֵם כָּל חַי לְפִי שֶׁהִיא מְקוֹר הַחַיִּים וּמְקוֹר כָּל הַהַשְׁפָּעוֹת בְּכָל הָעוֹלָמוֹת, וְכִדְאִיתָא בְּפַרְדֵּס בְּשַׁעַר עֶרְכֵי הַכִּנּוּיִים עֶרֶךְ חַוָּה שֶׁהִיא סְפִירַת הַמַּלְכוּת שֶׁנִּקְרֵאת אֵם כָּל חַי.
[cup]וְהִנֵּה[/cup] בְּכָל יוֹם הֲרֵי מִדָּה זוֹ מְחַדֶּשֶׁת שֶׁפַע הָאוֹר וְהַחֶסֶד עֶלְיוֹן בְּכָל הָעוֹלָמוֹת, וְזֶהוּ שֶׁאָנוּ אוֹמְרִים בְּבִרְכַּת הַמָּזוֹן הַזָּן אֶת הָעוֹלָם כּוּלּוֹ בְּטוּבוֹ בְּחֵן בְּחֶסֶד כו' וּכְמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (עֲבוֹדָה זָרָה ג. ב) שְׁלִישִׁיּוֹת הַקָּדוֹשׁ בָּרוּךְ הוּא יוֹשֵׁב וְזָן אֶת כָּל הָעוֹלָם כּוּלּוֹ מִקַּרְנֵי רְאֵמִים עַד בֵּיצֵי כִנִּים, דְּהַיְינוּ מֵעוֹלָם הַבְּרִיאָה בְּגַן עֵדֶן הָעֶלְיוֹן עַד תַּחְתִּית עוֹלָם הָעֲשִׂיָּה בָּעוֹלָם הַשָּׁפָל, שֶׁכָּל הַנִּבְרָאִים נִכְלָלִים בִּכְלַל זֶה,
דִּסְפִירַת הַמַּלְכוּת חֲלוּקָה מִכָּל הַסְּפִירוֹת בָּזֶה דִּסְפִירַת הַמַּלְכוּת נֶאֶצְלָה בִּנְקוּדָּה וְהַט' סְפִירוֹת שֶׁבָּהּ בָּאוֹת בְּסוֹד תּוֹסָפוֹת, דְּמִסְּפִירַת הַמַּלְכוּת שֶׁבְּחָכְמָה נִמְשָׁךְ לִהְיוֹת סְפִירַת הַחָכְמָה שֶׁבְּמַלְכוּת וּמִסְּפִירַת הַמַּלְכוּת שֶׁבְּבִינָה נִמְשָׁךְ לִהְיוֹת סְפִירַת הַבִּינָה שֶׁבְּמַלְכוּת וְכֵן כּוּלָּם, וְהַיְינוּ דְּהָאוֹר וְהַחַיּוּת מְקַבֶּלֶת סְפִירַת הַמַּלְכוּת מֵהַסְּפִירוֹת שֶׁלְּמַעְלָה מִמֶּנָּה, וְשֹׁרֶשׁ הַהַמְשָׁכָה הוּא מִכֶּתֶר עֶלְיוֹן, וְזֶהוּ עִנְיַן בִּנְיַן הַמַּלְכוּת בְּרֹאשׁ הַשָּׁנָה, לִהְיוֹת שֶׁבְּרֹאשׁ הַשָּׁנָה נִמְשָׁךְ בְּחִינַת כֶּתֶר מַלְכוּת דְּהַיְינוּ הָרָצוֹן לִמְלוּכָה, כְּיָדוּעַ וּמְבוֹאָר בְּמָקוֹם אַחֵר בַּאֲרִיכוּת, וְלָכֵן אָז הוּא עִיקַּר הַמְשָׁכַת הָאוֹר בִּבְחִינַת מַלְכוּת (שֶׁזֶּהוּ עִנְיַן הָרָצוֹן לִמְלוּכָה)
וּבְכָל רֹאשׁ הַשָּׁנָה נִמְשָׁךְ אוֹר חָדָשׁ מֵאֵין סוֹף בָּרוּךְ הוּא לְמִדַּת מַלְכוּתוֹ יִתְבָּרֵךְ עַל כָּל הַשָּׁנָה כּוּלָּהּ לְהַחֲיוֹת עוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה לְזוּנָם וּלְפַרְנְסָם, שֶׁאוֹר וְחַיּוּת זֶה הוּא כּוֹלֵל חַיּוּת עוֹלָם שָׁנָה נֶפֶשׁ שֶׁל כָּל הַשָּׁנָה כּוּלָּהּ, אֶלָּא דִּבְרֹאשׁ הַשָּׁנָה הוּא בַּכֶּסֶה עֲדַיִין שֶׁמְּכוּסֶּה וְנֶעְלָם, וּבְיוֹם הַכִּפּוּרִים נִמְשָׁךְ בִּבְחִינַת הַמְשָׁכָה וְגִילּוּי לְמַעְלָה, וּבַסֻּכּוֹת הוּא לְיוֹם חַגֵּינוּ שֶׁנִּמְשָׁךְ בְּגִילּוּי יוֹתֵר אַךְ בִּבְחִינַת מַקִּיף, וּבַשְּׁמִינִי עֲצֶרֶת נִמְשָׁךְ בִּבְחִינַת פְּנִימִיּוּת שֶׁזֶּהוּ עִנְיַן הַקְּלִיטָה דִּשְׁמִינִי עֲצֶרֶת, וְאַחַר כַּךְ נִמְשָׁךְ עַל יְדֵי בְּחִינַת מַלְכוּת בְּכָל יוֹם וָיוֹם בְּגִילּוּי לְמַטָּה בָּעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה.
[cup]וְזֶהוּ[/cup] בְּחִינַת הַמְשָׁכַת הַחֶסֶד עֶלְיוֹן שֶׁנִּמְשָׁךְ מִבְּחִינַת הַכֶּתֶר בְּחִינַת חֲסָדִים דְּעַתִּיק בִּבְחִינַת מַלְכוּת בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים וּמִבְּחִינַת מַלְכוּת נִמְשָׁךְ בְּכָל הָעוֹלָמוֹת.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]אֲצִילוּת סְפִירַת הַמַּלְכוּת בְּסוֹד שֹׁרֶשׁ וּנְקוּדָּה אַחַת וּבְרֹאשׁ הַשָּׁנָה בִּנְיַן הַמַּלְכוּת שֶׁמְּקַבֶּלֶת מֵהַט' סְפִירוֹת שֶׁלְּמַעְלָה וְשֹׁרֶשׁ הַהַמְשָׁכָה מִכֶּתֶר מַלְכוּת, שֶׁבְּרֹאשׁ הַשָּׁנָה עֲדַיִין מְכוּסֶּה וּבְיוֹם הַכִּפּוּרִים מִתְגַּלֶּה וּבַסֻּכּוֹת בְּמַקִּיף וּבַשְּׁמִינִי עֲצֶרֶת בִּקְלִיטָה וְנִמְשָׁךְ עַל יְדֵי סְפִירַת הַמַּלְכוּת.[kitzur_end]
