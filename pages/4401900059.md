מַאֲמָר יא.
וְזֶהוּ שֶׁכָּתוּב (נְחֶמְיָה ט. ד) וַיִּזְעֲקוּ בְּקוֹל גָּדוֹל אֶל ה' אֱלֹקֵיהֶם, דְּלִכְאוֹרָה אֵינוֹ מוּבָן מַהוּ עִנְיַן קוֹל גָּדוֹל וַהֲלֹא אָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (בְּרָכוֹת כד. ב) הַמַּשְׁמִיעַ קוֹלוֹ בִּתְפִלָּתוֹ הֲרֵי זֶה מִקְּטַנֵּי אֲמָנָה, הַמַּגְבִּיהַּ קוֹלוֹ בִּתְפִלָּתוֹ הֲרֵי זֶה מִנְּבִיאֵי הַשֶּׁקֶר וּפֵרֵשׁ רַשִׁ"י דִּכְתִיב (מְלָכִים א יח. כח) וַיִּקְרְאוּ בְּקוֹל גָּדוֹל, וּמִכָּל מָקוֹם בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים נָהֲגוּ לְהַשְׁמִיעַ קוֹלָם וּכְמוֹ שֶׁכָּתַב הַבֵּית-יוֹסֵף בְּאוֹרַח חַיִּים סִימָּן קא. וּבְסֵפֶר מָגֵן דָּוִד לְהָרַדְבַּ"ז [רַבִּי דָּוִד בֶּן זִמְרָא] זִכְרוֹנוֹ לִבְרָכָה אוֹת ט כָּתַב וְזֶה לְשׁוֹנוֹ וְהִנֵּה אוֹדִיעֲךָ עִנְיָן נִפְלָא מְאֹד הֵבַנְתִּי מִתּוֹךְ דִּבְרֵי רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה, כִּי הַדָּבָר הַיּוֹצֵא מִפִּי הָאָדָם הוּא נִרְשָׁם בָּאֲוִיר וּבוֹקֵעַ כַּמָּה אֲוִירִים וּרְקִיעִים וְעוֹלֶה לְמַעְלָה אִם טוֹב וְיָשָׁר הוּא וְיוֹצֵא בִּקְדוּשָּׁה אֲזַי מַקְרִיבִין אוֹתוֹ לִפְנֵי מִי שֶׁאָמַר וְהָיָה הָעוֹלָם כו' וְעַל כֵּן נָהֲגוּ כָּל יִשְׂרָאֵל בְּצַעֲקָתָם עַל הַצָּרָה לִקְרֹא בְּקוֹל גָּדוֹל לַעֲשׂוֹת רוֹשֶׁם גָּדוֹל וְיִבְקְעוּ הָאֲוִירִים עַד כַּאן לְשׁוֹנוֹ,
הֲרֵי חִילֵּק דְּלִפְעָמִים צְרִיכִים לִקְרֹא בְּקוֹל גָּדוֹל דַּוְקָא וְעַל יְדֵי זֶה עוֹשִׂים רוֹשֶׁם גָּדוֹל.
[cup]וְהָעִנְיָן[/cup] הוּא דְּהִנֵּה כְּלָלוּת עִנְיַן הָאוֹתִיּוֹת הוּא עִנְיַן הַגִּילּוּי, וּכְמוֹ (יְשַׁעְיָה כא. יב) אָתָא בֹקֶר, דְּקוֹדֶם לָזֶה הָיָה חֹשֶׁךְ וְהֶעְלֵם שֶׁהוּא עִנְיַן הַלַּיְלָה וּכְשֶׁבָּא הָאוֹר שֶׁהוּא בֹקֶר אוֹמֵר אָתָא בֹקֶר שֶׁבָּא הַגִּילּוּי,
וְלִהְיוֹת שֶׁיֵּשׁ כַּמָּה אוֹפַנֵּי גִילּוּיִם כְּמוֹ גִילּוּי אוֹר הָעוֹנֶג, גִּילּוּי אוֹר הָרָצוֹן, גִּילּוּי אוֹר הַשֵּׂכֶל, גִּילּוּי אוֹר הַמִּדּוֹת שֶׁהֵם גִּילּוּיִם עֲצוּמִים לְפִי עֶרֶךְ מַהוּתָם, וְיֵשׁ גִּילּוּיֵי אוֹרוֹת לְבוּשֵׁי הַנֶּפֶשׁ בְּמַחֲשָׁבָה וְדִבּוּר, דְּכָל הַגִּילּוּיִם הֵם אוֹתִיּוֹת כֹּחוֹת וּלְבוּשֵׁי הַנֶּפֶשׁ,
וּכְשֵׁם שֶׁבְּאוֹתִיּוֹת הַכְּתָב הֲרֵי בִּכְלָלוּת ג' מִינֵי אוֹתִיּוֹת הֵן גְּדוֹלוֹת בֵּינוֹנִיּוֹת וּקְטַנּוֹת, אֲשֶׁר בִּסְפִירוֹת הָעֶלְיוֹנוֹת הֵן בְּדַרְגּוֹת חֲלוּקוֹת כְּמַאֲמָר אַתְוָון רַבְרְבִין בִּסְפִירַת הַבִּינָה אַתְוָון בֵּינוֹנִין בִּסְפִירַת זָ"א וְאַתְוָון זְעֵירִין בִּסְפִירַת הַמַּלְכוּת, הֲרֵי מַה שֶּׁהָאוֹת נִכְתֶּבֶת גְּדוֹלָה יוֹתֵר מוֹרֶה שֶׁהוּא בִּבְחִינָה עֶלְיוֹנָה יוֹתֵר אַף שֶׁהַכֹּל תְּמוּנָה אַחַת,
כְּמוֹ כֵן הוּא בְּקוֹל, דְּקוֹל גָּדוֹל מוֹרֶה עַל הַמַּעֲלָה שֶׁהוּא לְמַעְלָה בְּמַדְרֵיגָה, וְהַיְינוּ דְּהֵן מִלְמַעְלָה לְמַטָּה וּכְמוֹ בְּמַתַּן תּוֹרָה דִּכְתִיב (שְׁמוֹת יט. טז) וְקוֹל שׁוֹפָר חָזָק מְאֹד וּכְתִיב (דְּבָרִים ה. יט) קוֹל גָּדוֹל וְלֹא יָסָף כו' וְהֵן מִלְּמַטָּה לְמַעְלָה וּכְמוֹ (נְחֶמְיָה ט. ד) וַיִּזְעֲקוּ בְּקוֹל גָּדוֹל וּכְתִיב (יוֹנָה ג. ח) וְיִקְרְאוּ אֶל אֱלֹקִים בְּחָזְקָה, דְּקוֹל גָּדוֹל מַגִּיעַ לְמַעְלָה יוֹתֵר, לְפִי שֶׁהוּא מִפְּנִימִיּוּת הַלֵּב, דִּלְפִי שֶׁנּוֹגֵעַ לוֹ הָעִנְיָן בִּפְנִימִיּוּת לְבָבוֹ מִשּׁוּם זֶה הוּא צוֹעֵק בְּקוֹל גָּדוֹל.
[cup]וּמַה[/cup] שֶּׁבִּתְפִלָּה הַמַּשְׁמִיעַ קוֹלוֹ בִּתְפִלָּתוֹ הֲרֵי זֶה מִקְּטַנֵּי אֲמָנָה, שֶׁהַטַּעַם הוּא כְּפֵרוּשׁ רַשִׁ"י כְּאִלּוּ אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ תְּפִלַּת לַחַשׁ וּמַגְבִּיהַּ הַרְבֵּה וְגַם שֶׁלֹּא לְבַלְבֵּל לַאֲחֵרִים הַמִּתְפַּלְלִים, כָּל זֶה הוּא בִּסְתָם תְּפִלָּה שֶׁצְּרִיכָה לִהְיוֹת בַּחֲשָׁאי, אֲבָל בְּעֵת צָרָה רַחֲמָנָא לִצְּלָן שֶׁאָז הִנֵּה הַנְּקוּדָּה הַפְּנִימִית, שֶׁהִיא נְקוּדַּת הַהִתְקַשְּׁרוּת בֵּאלֹקוּת הִיא בְּכָל אָדָם בְּהִתְגַּלּוּת בֶּאֱמוּנָה פְשׁוּטָה בַּה' עוֹשֵׂה פֶלֶא, וְהַטַּעַם דְּבִלְבּוּל לַאֲחֵרִים גַּם כֵּן אֵינוֹ שַׁיָּיךְ אָז וְאַדְרַבָּא הַהִתְעוֹרְרוּת מֵעוּמְקָא דְלִיבָּא מְעוֹרֶרֶת גַּם לַאֲחֵרִים לְהִתְעוֹרֵר בְּהִתְעוֹרְרוּת תְּשׁוּבָה, צָרִיךְ לִהְיוֹת צְעָקָה בְּקוֹל גָּדוֹל דַּוְקָא, וְכֵן בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים שֶׁצְּרִיכִים לְעוֹרֵר בִּבְחִינַת פְּנִימִית וְעַצְמוּת כְּמוֹ שֶׁכָּתוּב בְּמָקוֹם אַחֵר, צְרִיכִים לִהְיוֹת בְּהִתְעוֹרְרוּת פְּנִימִית הַלֵּב וּבְקוֹל גָּדוֹל.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]אוֹת הוּא גִילּוּי אָתָא בֹקֶר, הַגִּילּוּיִם דְּכֹחוֹת וּלְבוּשֵׁי הַנֶּפֶשׁ הֵם אוֹתִיּוֹת. בְּאוֹתִיּוֹת הַכְּתָב ג' דַּרְגּוֹת גְּדוֹלוֹת בֵּינוֹנִיּוֹת וּקְטַנּוֹת כֵּן הוּא גַם בְּקוֹל, וּתְפִלָּה צְרִיכָה לִהְיוֹת בְּקוֹל גָּדוֹל מֵעוּמְקָא דְלִיבָּא שֶׁאָז הִיא הִתְגַּלּוּת נְקוּדַּת הַהִתְקַשְּׁרוּת וּמְעוֹרֵר גַּם אֲחֵרִים לִתְשׁוּבָה.[kitzur_end]
