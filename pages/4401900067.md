מַאֲמָר טו.
וְהִנֵּה הָאָדָם בַּאֲשֶׁר הוּא אָדָם צָרִיךְ לָדַעַת וְכָל אָדָם שֶׁיֵּשׁ לוֹ מוֹחַ בְּקָדְקֳדוֹ יוֹדֵעַ, אֲשֶׁר הָאָדָם הוּא בְּחִיר הַנִּבְרָאִים אֲשֶׁר בָּרָא וְיָצַר הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא לְבָד בִּצְבָא הָאָרֶץ אֶלָּא גַּם בִּצְבָא הַשָּׁמַיִם, וְהוּא (הָאָדָם) בְּרִיאָה הַתִּיכוֹנִית מִכָּל הַנִּבְרָאִים כּוּלָּם, הַיְינוּ הֵן מֵהַנִּבְרָאִים הָעֶלְיוֹנִים וְהֵן מֵהַנִּבְרָאִים הַתַּחְתּוֹנִים, דְּהִנֵּה כָּל הַנִּבְרָאִים אֲשֶׁר בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא וְהָאָדָם בִּכְלָל הֵם בַּעֲלֵי גוּף וָנֶפֶשׁ, אֲשֶׁר בָּזֶה שָׁוִים הֵמָּה צְבָא הָאָרֶץ עִם צְבָא הַשָּׁמַיִם, דְּעִם הֱיוֹת אֲשֶׁר צְבָא הַשָּׁמַיִם חָלוּק מִצְּבָא הָאָרֶץ, דִּצְבָא הָאָרֶץ בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹתָם מֵהַרְכָּבַת הַד' יְסוֹדוֹת אֵשׁ רוּחַ מַיִם עָפָר, וּצְבָא הַשָּׁמַיִם בָּרָא אוֹתָם הַקָּדוֹשׁ בָּרוּךְ הוּא רַק מֵהַרְכָּבַת שְׁנֵי יְסוֹדוֹת אֵשׁ וָרוּחַ כְּמוֹ שֶׁכָּתוּב עוֹשֶׂה מַלְאָכָיו רוּחוֹת מְשָׁרְתָיו אֵשׁ לוֹהֵט, בְּכָל זֶה הִנֵּה בָּזֶה מִשְׁתַּוִּים צְבָא הַשָּׁמַיִם וּצְבָא הָאָרֶץ שֶׁהֵם בַּעֲלֵי גוּף וָנֶפֶשׁ, וְהַיְינוּ דְּעִם הֱיוֹת דִּצְבָא הַשָּׁמַיִם בְּרָאָם מֵהַיְסוֹדוֹת הַדַּקִּים דְּאֵשׁ וָרוּחַ, בְּכָל זֶה הִנֵּה גַּם הֵם בַּעֲלֵי גוּף, וּצְבָא הָאָרֶץ עִם הֱיוֹת שֶׁבְּרָאָם בְּהַרְכָּבַת הַיְסוֹדוֹת מַיִם וְעָפָר גַּם כֵּן בְּכָל זֶה הִנֵּה גַּם הֵם בַּעֲלֵי נֶפֶשׁ, וְרַק הַהֶבְדֵּל הוּא בְּטִבְעָם, דְּטִבְעֵי צְבָא הַשָּׁמַיִם הוּא בְּהַגְבָּרַת הַצּוּרָה עַל הַחוֹמֶר, וְטִבְעֵי צְבָא הָאָרֶץ בְּטֶבַע תּוֹלְדוֹתָם בְּהַגְבָּרַת הַחוֹמֶר עַל הַצּוּרָה.
[cup]וְהִנֵּה[/cup] הַגּוּף וְהַנֶּפֶשׁ מַקְבִּילִים וּמַתְאִימִים זֶה לָזֶה, וְהַיְינוּ דְּעִם הֱיוֹת אֲשֶׁר הַנֶּפֶשׁ וְהַגּוּף הִנֵּה מִצַּד מַהוּתָם הָעַצְמִי הֵם הֲפָכִים זֶה מִזֶּה, דְּהַנֶּפֶשׁ הוּא רוּחָנִי וְהַגּוּף גַּשְׁמִי, וְרוּחָנִי וְגַשְׁמִי בְּמַהוּתָם הָעַצְמִי הֲרֵי הֵם הֲפָכִים זֶה מִזֶּה, בְּכָל זֶה הִנֵּה הַנֶּפֶשׁ וְהַגּוּף מַקְבִּילִים וּמַתְאִימִים זֶה לָזֶה, דְּזוֹהִי אַחַת מֵהַפְּלִיאוֹת דִּפְלִיאוֹת תְּמִים דֵּעִים יִתְבָּרֵךְ וְיִתְעַלֶּה בְּעִנְיַן בְּרִיאָה יֵשׁ מֵאַיִן וְהִתְחַבְּרוּתָן שֶׁהוּא רַק בְּחֵיק הַבּוֹרֵא בָּרוּךְ הוּא שֶׁמַּפְלִיא לַעֲשׂוֹת מְקַשֵּׁר רוּחָנִיּוּת בְּגַשְׁמִיּוּת בְּאוֹפֶן הַמַּקְבִּיל וּמַתְאִים כְּאִלּוּ הָיוּ לַאֲחָדִים מִתְּחִלַּת יְצִירָתָם בְּמַהוּתָם הָעַצְמִי, וְלָכֵן הִנֵּה הַנֶּפֶשׁ וְהַגּוּף נִקְרָאִים אוֹר וּכְלִי, דְּהַנֶּפֶשׁ הוּא אוֹר וְהַגּוּף הוּא כְלִי.
[cup]וְהִנֵּה[/cup] גּוּף הָאָדָם שִׁינָּה הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת אוֹפֶן יְצִירָתוֹ מִיצִירַת גּוּפוֹת שְׁאָר הַנִּבְרָאִים אֲשֶׁר בָּרָא וְיָצַר הַקָּדוֹשׁ בָּרוּךְ הוּא, דְּכָל הַנְּבִרָאִים הִנֵּה כְּשֶׁאָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹתוֹ הַמַּאֲמָר הַשַּׁיָּיךְ לְהַנִּבְרָאִים הָהֵם, מִיָּד נִבְרָא גוּפָם וְנִכְנְסָה בָהֶם נַפְשָׁם וּשְׁנֵיהֶם (הַגּוּף וְהַנֶּפֶשׁ) כְּאֶחָד יָצְאוּ עַל פְּנֵי הָאֲדָמָה, כְּמוֹ בְּיוֹם ג' כְּשֶׁאָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא תַּדְשֵׁא הָאָרֶץ דֶּשֶׁא עֵשֶׂב מַזְרִיעַ זֶרַע עֵץ פְּרִי עוֹשֶׂה פְּרִי לְמִינוֹ אֲשֶׁר זַרְעוֹ בוֹ עַל הָאָרֶץ וַיְהִי כֵן, הוֹצִיאָה הָאָרֶץ אֶת הָעֲשָׂבִים וְאֶת הָאִילָנוֹת כָּל אֶחָד וְאֶחָד כְּמוֹ שֶׁהֵם עִם נֶפֶשׁ הַצּוֹמַחַת שֶׁבָּהֶם, וְכֵן גַּם בְּבַעֲלֵי חַיִּים הוֹצִיאָה אוֹתָם הָאָרֶץ כְּמוֹ שֶׁהֵם חַיִּים וְגַם הַמַּיִם שָׁרְצוּ כֵן שֶׁרֶץ נֶפֶשׁ חַיָּה, וְהַיְינוּ דְּכָל הַנִּבְרָאִים הִנֵּה אוֹפֶן יְצִירָתָם בְּרָאָם הַקָּדוֹשׁ בָּרוּךְ הוּא הַגּוּף עִם הַנֶּפֶשׁ הַחִיּוּנִית אֲשֶׁר בְּתוֹכָם יַחַד,
מַה שֶּׁאֵין כֵּן אוֹפֶן יְצִירַת הָאָדָם שִׁינָּה הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת בְּרִיאָתוֹ, שֶׁבָּרָא תְּחִלָּה אֶת גּוּפוֹ בִּפְנֵי עַצְמוֹ עָפָר מִן הָאֲדָמָה בְּלִי שׁוּם נֶפֶשׁ חִיּוּנִית בְּתוֹכוֹ. וּכְמוֹ שֶׁכָּתוּב וַיִּיצֶר ה' אֱלֹקִים אֶת הָאָדָם עָפָר מִן הָאֲדָמָה, דְּמִתְּחִלָּה הָיָה גוֹלֶם, וְאַחַר כַּךְ וַיִּפַּח בְּאַפָּיו נִשְׁמַת חַיִּים, וְהַיְינוּ שֶׁלֹּא יָצָא גוּפוֹ מִן הָאָרֶץ עִם הָרוּחַ חַיִּים שֶׁבּוֹ כְּמוֹ שֶׁהָיָה בְּכָל הַנִּבְרָאִים, אֶלָּא דִּתְחִלָּה בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת גּוּפוֹ וּבָרָא אוֹתוֹ מִן הַחוֹמֶר הַיּוֹתֵר עָב וְגַס שֶׁהוּא דוֹמֵם. וְזֶהוּ עָפָר מִן הָאֲדָמָה, וְהָיָה גוֹלֶם בְּלִי שׁוּם חַיּוּת, דְּהָוֵי שִׁפְלוּת גְּדוֹלָה לְהָאָדָם לְעוּמַּת שְׁאָר הַבְּרוּאִים, דְּעִם הֱיוֹת דְּהַכֹּל הָיָה מִן הֶעָפָר, אֲבָל עִם זֶה הִנֵּה אוֹפֶן יְצִירָתָם שֶׁיָּצְאוּ גוּפָם וְנַפְשָׁם כְּאֶחָד, וְהָאָדָם הִנֵּה תְּחִלָּה הָיָה גוֹלֶם וְרַק אַחַר כַּךְ וַיִּפַּח נִשְׁמַת חַיִּים,
וְצָרִיךְ לְהָבִין מִפְּנֵי מָה שִׁינָּה הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹפֶן יְצִירַת גּוּף הָאָדָם מִגּוּפוֹת שְׁאָרֵי הַנִּבְרָאִים, דְּלִכְאוֹרָה הִנֵּה הַשִּׁינּוּי הוּא לִגְרִיעוּתָא, שֶׁהֲרֵי כָּל הַבְּרוּאִים, עִם הֱיוֹת שֶׁיְּצִירָתָם הִיא מִן הֶעָפָר, אֲבָל עִם זֶה לֹא נִרְאוּ בְּנַוְולוּתָם בִּהְיוֹתָם גּוֹלָמִים כִּי אִם בִּהְיוֹת גּוּפָם וְנַפְשָׁם כְּאֶחָד, וְהָאָדָם הִנֵּה אוֹפֶן יְצִירָתוֹ אֲשֶׁר תְּחִלָּה נִרְאָה בְּנַוְולוּתוֹ בִּהְיוֹתוֹ גוֹלֶם, וְרַק אַחֲרֵי כֵן הִפִּיחַ הַקָּדוֹשׁ בָּרוּךְ הוּא נִשְׁמַת חַיִּים, וְאַחַר כָּל זֶה הִנֵּה הָאָדָם דַּוְקָא הוּא בְּחִיר הַנִּבְרָאִים וְהַבְּרִיאָה הַתִּיכוֹנִית בְּכָל הַבְּרִיאָה כוּלָּהּ.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]כָּל הַנִּבְרָאִים צְבָא הַשָּׁמַיִם וּצְבָא הָאָרֶץ בַּעֲלֵי גוּף וָנֶפֶשׁ הֵמָּה, צְבָא הַשָּׁמַיִם שֶׁיְּצִירָתָם מִיסוֹדוֹת הַדַּקִּים דְּאֵשׁ וָרוּחַ טִבְעָם בְּהַגְבָּרַת הַצּוּרָה עַל הַחוֹמֶר, וּצְבָא הָאָרֶץ שֶׁיְּצִירָתָם גַּם מִיסוֹדוֹת הַגַּסִּים טֶבַע תּוֹלְדוֹתָם בְּהַגְבָּרַת הַחוֹמֶר עַל הַצּוּרָה, גּוּף וָנֶפֶשׁ אוֹר וּכְלִי מַתְאִימִים, הַקָּדוֹשׁ בָּרוּךְ הוּא שִׁינָּה אוֹפֶן יְצִירַת גּוּף הָאָדָם.[kitzur_end]
