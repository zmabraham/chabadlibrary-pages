כמשנת"ל (ד"ה ראשי המטות) דאצי' מובדל בערך מן העולמות רק שהוא בבחי' ממוצע לבד כו', ומה שבא מאצי' לבי"ע הוא רק ההארה דההארה לבד שבא בבחי' אור של תולדה כו', וזה בא בבחי' התלבשות ממש בבי"ע כו', וההשפעה למטה הוא מהמדרי' דבי"ע כו', וכידוע דכל מדרי' שבאה בבחי' התלבשות, במדרי' התחתונה היינו בבחי' עכבה ה"ז נעשה כמהות המדרי' התחתונה ונק' ע"ש המדרי' התחתונה כו', וכמ"ש בע"ח של"ג פ"ב בענין מעבר והתלבשות, ומכ"ש בבחי' אור של תולדה שבא בבחי' התלבשות ותפיסא ממש בבי"ע, ע"כ אין ההשפעה נק' ע"ש אצי' רק שהוא השפעה מבי"ע כו'. ובזה הי' חלוק דור המדבר שהיו מושפעים מאצי', וזהו שא' משה מאין לי בשר לפי שמשה הי' בדוגמת בחי' המאציל לא הי' יכול להשפיל עצמו בהשפעת בשר, וא"ל הקב"ה אספה אלי כו' וירדתי ואצלתי כו' שיהיו במדרי' אצי' ויהי' ע"י השפעת הבשר כו', והיינו שדור המדבר קבלו השפע מאצי' כו', אבל כל השפעות הן מבי"ע כו'. ולכן כל הדברים שלמטה יש בהם פסולת, להיות דעולמות בי"ע הגם שנתבררו הרי לא נתבררו בבחי' בירור בתכלית
כמו האצי' שנעשה בבחי' אלקות ממש משא"כ בי"ע כו', לפי שהבירור הוא מבחי' בי"ע דתהו שאין הטוב בזה כמו באצי' כו' וכמשנת"ל (ד"ה פנחס), ובפרט הלבושים דבי"ע הרי יש בהם טו"ר כו', ולכן כל הדברים המשתלשלים מבי"ע יש בהם תערובות פסולת כו', אבל השפעת המן הי' מאצי' שאין שם תערובות פסולת ח"ו כלל, וע"כ הי' המן למטה ג"כ מזון רוחני בלי שום תערובות פסולת כו', כי גם בחי' אור הנשמה דבי"ע אינו בבחי' תפיסא בבי"ע וכ"ש מדרי' דאצי' שאינם בבחי' תפיסא בבי"ע כו', וע"כ הי' למטה מזון רוחני כו', ובאמת גם באצי' זהו שירד ונתלבש באצי', ושרשו הוא למע' מבחי' אצי' ולמע' גם ממקור האצי' כו'. והענין הוא דהנה ידוע דכללות ההשתל' נבחן בג' מדרי' כללי' והן התלת עלמין שבזהר שלח דקנ"ט סע"א, המדרי' הא' היא עולמות בי"ע שהן מוגבלים ממש במקום וזמן, וכמשנת"ל שגם המלאכים נתפסים במקום וזמן כו', והמדרי' הב' הוא עולם האצי' שאין שם ההגבלה דמקום וזמן כלל רק שם הוא הגבלה רוחני', והו"ע המציאות של הספי' בבחי' חסד וגבו' כו', שהחסד הוא במציאות חסד
כו', ולא שהוא במציאות ח"ו כמו חסד דבי"ע, וכמ"ש באגה"ק סי' ט"ו בענין שא' אאע"ה ואנכי עפר ואפר שא' זה על מדריגתו בחי' החסד ומדת האה' דשרשו מבחי' חסד דאצי', שלא נא' שיש לו איזה ערך ודמיון לגבי חסד דאצי', וכמו עו"ע שהן בערך זל"ז דגם העילה היותר עליונה יש לה ערך עם העלול היותר אחרון כו', עז"א ואנכי עפר ואפר כמו שהאפר אין לו ערך ודמיון כלל אל מהות העץ כמו"כ הוא באין ערוך לגבי בחי' חסד דאצי' כו', ומ"מ ה"ז בחי' חסד בבחי' אור וכלי וכמא' חסד דרועא ימינא כו', ויש הגבלה למדת החסד דעד פה תבא בכדי שיוכל להיות אצי' מדת הגבו', וכן יש הגבלה למדת הגבו' וכה"ג לכל המדות יש להם גדר וגבול בעצם מהותם ובהתפשטות שלהם כו', והיינו בהאיכות והכמות של הספי' איך ומה היא הספי' ובכמות התפשטותה כו', והן בבחי' מעומ"ט ובבחי' קדימה ואיחור כו', וכ"ז הוא ע"פ מא' קו המדה כו' וכמ"ש במ"א באורך ונת"ל ג"כ (ד"ה אלה הדברים), שכל זה הוא הגבלה רוחני' כו' (ועמשנת"ל ד"ה ביום השמע"צ דמ"מ ע"ס דאצי' אינם בבחי' התחלקות כו'). והמדרי' הג' הוא בחי' עצמות המאציל
