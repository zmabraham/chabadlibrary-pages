# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ח. > פֶּרֶק ג

וְהִנֵּה הַמֵּנִיעַ אֶת הַטִּקְלָא וּמְסַבֵּב סִיבּוּב הַגַּלְגַּל שֶׁיִּהְיֶה מִתְהַפֵּךְ לְפָעֳלָם דִּבְנֵי נָשָׁא, הוּא בְּחִינַת קַו הַמִּדָּה דְחָכְמָה שֶׁהוּא הַמְהַפֵּךְ אֶת הַגַּלְגַּל לְפִי הָאִתְעֲרוּתָא דִלְתַתָּא, לְפִי שֶׁבְּחָכְמָה הִיא הַמְּדִידָה שֶׁיִּהְיֶה בְּאוֹפֶן כַּךְ וְכַךְ דַּוְקָא, הַיְינוּ שֶׁיִּהְיֶה קִיּוּם הַתּוֹרָה וּמִצְוֹת לִהְיוֹת סוּר מֵרָע בִּשְׁסָ"ה לֹא תַעֲשֶׂה וְעַנְפֵיהֶן וְלִהְיוֹת וַעֲשֵׂה טוֹב בְּקִיּוּם רְמַ"ח מִצְוֹת עֲשֵׂה וְעַנְפֵיהֶן, וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל (מַאֲמָר ז פֶּרֶק ב) וְלָכֵן נִמְשָׁךְ מִשָּׁם תְּנוּעַת הַגַּלְגַּל, דְּאִם מְקַיְּימִים תּוֹרָה וּמִצְוֹת אָז תְּקַבֵּל הַמַּלְכוּת מִקַּו הַיָּמִין וְתִהְיֶה הַשְׁפָּעַת הַטּוֹב, וְהַיְינוּ שֶׁתִּהְיֶה הַהַנְהָגָה בִּבְחִינַת חֶסֶד, וְכַאֲשֶׁר חַס וְשָׁלוֹם לָא אִכְשִׁיר דָּרָא תִּהְיֶה הַהַנְהָגָה בִּבְחִינַת גְּבוּרָה כו'.
[cup]וְזֶהוּ[/cup] עִנְיַן הַחֶשְׁבּוֹן בְּב' דְּבָרִים, הָא' בְּעוֹבָדֵיהוֹן דִּבְנֵי נָשָׁא לְמַטָּה אִם מוּכְשָׁרִים מַעֲשֵׂיהֶם עַל פִּי הַמְּדִידָה דְחָכְמַת הַתּוֹרָה, וְהַב' הַחֶשְׁבּוֹן בְּאוֹפֶן הַהִתְכַּלְלוּת דְּחֶסֶד וּגְבוּרָה, אִם תִּהְיֶה הִתְגַּבְּרוּת קַו הַיָּמִין שֶׁתִּהְיֶה הַהַנְהָגָה בִּבְחִינַת חֶסֶד אוֹ חַס וְשָׁלוֹם כו'.
[cup]וְזֶהוּ[/cup] דְּאִיתָא בְּמִשְׁנָה (אָבוֹת סוֹף פֶּרֶק ד) וְדַע שֶׁהַכֹּל לְפִי הַחֶשְׁבּוֹן, וּפֵרֵשׁ רַשִׁ"י אַחַת לְאַחַת לִמְצוֹא חֶשְׁבּוֹן, פְּרוּטָה וּפְרוּטָה מִצְטָרֶפֶת לְחֶשְׁבּוֹן גָּדוֹל, וְכֵן יֵשׁ חֶשְׁבּוֹן לְהֵיפֶךְ, וּכְמוֹ שֶׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (חֲגִיגָה ה. ב הוּבָא בְּמִדְרַשׁ שְׁמוּאֵל עַל מִשְׁנָה זוֹ) עַל פָּסוּק וּמַגִּיד לְאָדָם מַה שֵּׂיחוֹ (עָמוֹס ד. יג) שֶׁאֲפִילוּ עַל שִׂיחָה שֶׁבֵּין אִישׁ לְאִשְׁתּוֹ, עֲבֵירוֹת שֶׁאָדָם דָּשׁ בַּעֲקֵבָיו, הַכֹּל בָּא לְפִי הַחֶשְׁבּוֹן כו'.
[cup]וְיֵשׁ[/cup] מְפָרְשִׁים הַמִּשְׁנָה וְדַע שֶׁהַכֹּל לְפִי הַחֶשְׁבּוֹן, שֶׁאִם ב' אֲנָשִׁים יֶחֶטְאוּ חֵטְא אֶחָד אֵין עוֹנְשָׁם וּפְגָמָם שָׁוֶה, שֶׁכְּפִי גּוֹדֶל יִקְרַת עֶרֶךְ הַנְּשָׁמָה שֶׁל זֶה יִהְיֶה פְגָמוֹ יוֹתֵר גָּדוֹל מִשֶּׁל חֲבֵירוֹ כו'.
[cup]וְכָל[/cup] זֶה הוּא עַל פִּי סֵדֶר הִשְׁתַּלְשְׁלוּת, וְהַיְינוּ הַחֶשְׁבּוֹן וְהַדִּין וּמִשְׁפָּט שֶׁנִּמְשָׁךְ מִן הַחָכְמָה שֶׁהִיא רֵאשִׁית הַהִשְׁתַּלְשְׁלוּת,
אֲבָל מִצַּד אוֹר אֵין סוֹף שֶׁלְּמַעְלָה מֵהִשְׁתַּלְשְׁלוּת שָׁם אֵין דִּין וְחֶשְׁבּוֹן, לְפִי שֶׁאֵינוֹ תּוֹפֵס מָקוֹם מַעֲשֵׂה הַתַּחְתּוֹנִים וְנִמְשֶׁכֶת הַהַשְׁפָּעָה גַּם לְעוֹבְרֵי רְצוֹנוֹ בְּרִיבּוּי הַשֶּׁפַע שֶׁלֹּא עַל פִּי הַחֶשְׁבּוֹן כְּלָל. וְזֶהוּ הַטַּעַם אֲשֶׁר דֶּרֶךְ רְשָׁעִים צָלֵחָה, לְפִי שֶׁהוּא מַאֲרִיךְ אַף, שֶׁמְּקַבְּלִים מִבְּחִינַת אֶרֶךְ אַפַּיִם שֶׁלְּמַעְלָה מִבְּחִינַת חֶשְׁבּוֹן, שֶׁנִּמְשָׁךְ לָהֶם הַהַשְׁפָּעָה גַּם שֶׁאֵינָם רְאוּיִם, וּמִמֵּילָא נִמְשָׁךְ רִיבּוּי הַשְׁפָּעָה לְפִי שֶׁהִיא בִּבְחִינַת בְּלִי גְבוּל וּכְמוֹ שֶׁיִּתְבָּאֵר בְּסָמוּךְ.

[kitzur_begin]תְּנוּעַת הַטִּקְלָא לְיָמִין בְּהַשְׁפָּעַת הַחֶסֶד וְלִשְׂמֹאל בְּהַשְׁפָּעַת הַגְּבוּרָה הוּא בְּחֶשְׁבּוֹן מְדוּיָּיק, א) לְפִי קִיּוּם הַתּוֹרָה וּמִצְוֹת. ב) בְּאוֹפֶן הַהִתְכַּלְלוּת דְּחֶסֶד וּגְבוּרָה, וְהַחֶשְׁבּוֹן בְּקִיּוּם הַתּוֹרָה וּמִצְוֹת וְהֶפְכּוֹ הוּא לְפִי חֲשִׁיבוּת הַפְּרָטִי שֶׁל כָּל נְשָׁמָה, דֶּרֶךְ רְשָׁעִים צָלֵחָה שֶׁמְּקַבְּלִים מִלְמַעְלָה מֵהִשְׁתַּלְשְׁלוּת.[kitzur_end]
