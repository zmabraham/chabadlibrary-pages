יו"ד: קצז, י (נפל למשאל"ס ולא שהו והי' אומדנות והוכחות ומצאו מכתביו בלבושיו. אהע"ז סשמ"ו).
אהע"ז ח"א: מח (נאבד זכרו כ"ב שנים, ע"א שנמצא הרוג, כמה שמועות שהחליף דתו שהוכחשו). מט-נב (ע. דחומץ. ראוהו נטבע ונמצא בכמה סי"ב ובסי' בגדיו. בא' מהם יש הכחשה בדיוק המקום). נג (ע. מקאליסק. שנה אחר שהלך מביתו בא מכתב מהמ"ץ ששחל"ח פב"פ וסי' הראוים, רק ששיקר וא' שהי' לו בן ומת). נד (ע. דהאראדאק. כמה שנים אחר שהלך מביתו נמצא רשום אצל השמש בשקלאוו שקודם מותו אמר שמו ושם אשתו וכו' רק בשם הבת הי' שינוי). נה (ע. דביעשינקאוויץ. הלך בשליחות לסענע ונטבע והכירוהו בסי' בגדיו ומעות). נו (נמצא הרוג ואכלוהו חיות אבל הכירוהו בטב"ע בתמונת גופו ומלבושים שעליו ושאצלו וכלים דלא מושלי בטב"ע ובסי'). נח (כנ"ל וגם מצאו אצלו האיגרת והמעות שהוליך). נט (ר"צ ור"ש נסעו בספינה שנתהפכה ונתפסו בחבל ור"צ נפל למים). ס (נטבע שא"י סיפר לע"א במסל"ת ששט אחריו עם ספינה ג' וויארסט ולא מצאו, ונתן לו הטו"ת וסידור שלו). סא (ב' הלכו ביחד ונמצאו הרוגים והוכרו בצבע השערות ובגודל האצבעות ובסי' וטב"ע בגדיהם שעליהם ושסמוכים להם). סב (ביום א' נמצא הרוג ראשו חתוך מהגוף ופ"פ מוכה אבל הכירו יפה ואת מלבושיו שראוהו בו בש"ק וביום ג' הביאוהו לעיר והכירוהו יפה ואשתו נתנה סי' בכל בגדיו). סג (ע. דקרעמענטשוג. הלך לשאוב מים ואחר כמה שבועות נמצא בלי חוטם וחצי פ"פ והכירוהו בטב"ע וסי"ב וסי"ב בבגדיו). סד (ע. חי' שרה. נטבע ושהו עשת"נ ואשתהי והכירוהו וסי' בגדיו). סה (יעקב בר"ש הכהן מכפר קאחאנייקי הלך לשוט בנהר ונטבע והכירוהו בטב"ע וקצת סי' בגופו ובסי' א' הי' הכחשה או טעות אם שיניים העליונים חסרים או התחתונים). סו (ע. מליוואווא. נטבע והוכר ע"י הרבה גמסל"ת, וע"י יהודי אחר דאשתהי). סח-ע (נטבע עם הסוס תחת הקרח וחיפשו כל היום במהלך ווייארסט ולא מצאו רק הסוס הוציאו מת). ע, ז (ע. דטשערנאביל, שנטבע ומצאוהו אחר ט"ו חודש והכירוהו בטב"ע. לקמן סשמ"ז-ח). עא (הלך למדה"י והעיד א' משם שהרבה אנשים אמרו לו שמת ומשרת א"י אמר שמת וראיתי קבורתו). עב (פייוויש הלך למרחק ואבד זכרו ג"ש ונמצא הרוג והוכרו המלבושים הסמוכים לו בטב"ע. גם א"י אמר שראה אותו אחר שנהרג והי' חם והכירו וס' אם הוי כבכי' דפונדקית). עג (ע. מנארוליייע. בעלה הלך לחטוף סוס שנלקח מאתו ונטבע. האשה אמרה סי"מ בבגדי בעלה. כעבור חצי שנה נמצא והכירוהו בסי' והבגדים בסי"מ). עד (נטבע שנחבט בכח בדף ועמד קצת על האבן ונפל לנהר ואבד זכרו). עה (ע. מטאבאלסק. גוים העידו שנטבע, וכעבור שנה נמצא רגלו במגפיים שהוכרו בטב"ע ונפסק הרגל בצוה"ג). עו (ע. מסענע שהלך בעלה למוהילוב ונאבד ונמצא הרוג בדרך שהוכר בסי' אצבע כפוף ובבגדיו). עז (ר' נחום מאיר בר"צ חלה בחולי-רע ומסרוהו לשפיטאל, וע"פ ציווי האדון העתיקו מניירותיהם שמת ונקבר). עח (ר' צבי הערש נטבע בנהר דניעסטור והלך א' על שפת הנהר יותר מב' שעות, גם הנכרים פרשו מצודות ולא מצאו, ואח"ז מצאו הנכרים נטבע ואמרו שהכירו הבגדים בסי"ב). פ (ר' אהרן בע"ג מאומן הי' בבאהסלוב והאכסניא נשרף. הלך להציל הסוסים והגוי מסל"ת שראה שנפל הגג וזקנו ופיאותיו התחילו לשרוף ונפל ולמחר בוקר נמצא גולגולת אדם ואוזן השמאלי). פא (יעקב ב"ר מן משקלאב התגורר לבוקארערשט ואמרו לאשתו שמת אם חוששין לב' יעקב ב"ר מן משקלאב המתגורר בבוקארעסט). פב (ר' זאב בר"י אבוהב מזערעץ נסע ממארפול עם ד' א"י ואח"כ הודה א' מהם שהרגוהו בדרך). פג (ע. טעמא ב"ר יצחק בדחן מהארקי. בעלה משה ב"ר ארי' ליב המיר וגר באופיא והעידו שמת א' שנק' כן ביהדותו). פד (ר' ישראל דוד ב"ר יקותיאל בכרך הלך מקלימאוויץ לבארדיטשוב עם שט"ח שנכתב בו ישראל ב"ר דוב בריסקין, ובא גב"ע מבארדיטשוב שישראל ב"ר דוב בריסקין מקלימאוויץ שחל"ח). פה (יצחק ב"ר מאיר זאב מטשערקאס הי' בצבא בסאוויסטאפל ונשא את חי' בת שלמה ומת, וכתב השו"ב דסאוויסטאפל שבעלה יצחק ב"ר אברהם מת, ואח"כ כתב שטעה בשם אביו. לאביו הנ"ל הי' בנים מאשה אחרת והם מכחישים שאינו אחיהם).
פו (ע. שהי' הכחשה בחקירות ובגוף המיתה לא הי' הכחשות, ונשאת. אח"כ בא אחד ואמר שהוא בעלה והיא מכחישתו). פח (נמצא הרוג בדרך שהלך בה בעלה ונקבר ואחר עשרה ימים נפתח קברו והוכר בטב"ע ובסי' בלחי השמאלי ובסי' ובטב"ע בגדיו). צ, י (בתשו' בדיני ע. דאשת ר"א בר"ד הארכתי).
אהע"ז ח"ב: שלז (א' מהארדישץ הלך עם ב' נכרים לרחוץ ליד מיר וראוהו נטבע במקום דאיכא עכולי ופשורי ואח"כ הוכר ע"י הבגדים וקצת ע"י פניו שנתפח). שלח (נטבע שהוכר בארבע כנפות מצמר עבה, קאפטען מריפש עם אונטערשלאק עבה, אבנט מקרייקא). שלט-מ (נמצא הרוג ואכלוהו חיות אבל הכירוהו בטב"ע בתמונת גופו ומלבושים שעליו ושאצלו וכלים דלא מושלי בטב"ע ובסי', ואשתו אמרה כמה סי' בבגדיו לפני שראתה). שמא (ב' יהודים מנעויל הלכו לגבות חוב בחצר בקלימאווע ונהרגו). שמב (נאבד בדרך מדובראוונא לארשא והלך אחיו עם ערלים לחפשו ומצאוהו מת ביער והכירו בפ"פ ובגדים וס' אשתהי ג"י). שמג (ידעו מטביעתו ונמצא והכירוהו בפ"פ אם יש לחשוש בדדמי). שמה (נטבע שהכירוהו בסי' הגוף בטב"ע בבגדיו וע"י אגרת). שמו (ר' שמואל משקלאוו נפל למשאל"ס ולא שהו ושמעו מערלים שמצאוהו עם מעות ואגרות השייכים לו ואבד זכרו והכירוהו בכמה סי' ואומדנות שהיו בהם גמגומים). שמז-ח (יונה מטשערנאביל נטבע ומצאוהו צף אחר ט"ו חודש והכירוהו בפ"פ ובסי' בגדיו). שמט (טב"ע אחר שהי' טמון בעפר ג"י בצירוף סי' בלחי ובחוטם ובגדים ובילעט דלא מושלי אינשי). שנ (א' מקמאיי הלך לדענעבורג ונטבע ומצאוהו אחר ג"י והכירוהו אחר כמה שעות שהוציאוהו מהמים וגם ראו שאצבע הקטן עקום ונכווץ וגם הכירו הבגדים שאמרה חמותו). שנב (ראוהו נטבע ונמצא בכמה סי"ב ובסי' בגדיו. בא' מהם יש הכחשה בדיוק המקום). שנג-ד (הרוצח יוזיפאט הודה שהרב את משה עפ הגדת הנפח הבא להנצל, וגם סיפר טעם ואופן ההריגה והרה המקום ביער ומצאו שם לחי העליון ותחתון. פונדקית מצאה ההרוג וסיפרה ע"י מסל"ת מפי מסל"ת). שנה (ר' נחום מאיר בר"צ חלה בחולי-רע ומסרוהו לשפיטאל, וע"פ ציווי האדון העתיקו מניירותיהם שמת ונקבר). שנז (שכ"מ אמר לפני מותו באוסוועט, לפני ב' עדים, ששמו אליעזר ברד"ס מדובראוונא, ושיש לו ב' בנות. א' שמע וא' אמר שאשתקיל מילולי' ולא הבין דבריו). שנח (ע. מטאבאלסק. גוים העידו שנטבע, וכעבור שנה נמצא רגלו במגפיים שהוכרו בטב"ע ונפסק הרגל בצוה"ג). שנט (הכירה בעלה ונתנה סי' בגופו והוא מכחיש ואח"כ). הוספות: נ, ב (לפני מותו אמר שמו יעקב וש"ע זיטאמיר ושם אשתו בבאהסלאוו ושם אחיו יוסף וסי' בפניו וכ' העדים דבריו, וכעבור ב' שנים העידו ויש איזה שינויים בעדות). נא, ב (ר' ליב נסע מבאבער לשקלאוו ונאבד ונמצא הרוג ביער ואתפח וס' אשתהי ג"י והכירו חותנו ר' קאפיל בטב"ע ובגומות בפניו ובגדיו). נב, א (ישראל יודל סאחאראוו משפאלע נאסר ונחלה ומת ונמצא רשום שמת באסטראג ובפאליצע). נב, ד (שומרת יבם שיבמה נלקח לעבודת הצבא והשיבו מהדואר שמת בלאזארעט וכ' לאנ"ח והשיבו ד' יהודים שמת בז' מ"ח).
חידושים: קלט, ב (ב' הלכו ביחד ונמצאו הרוגים והוכרו בצבע השערות ובגודל האצבעות ובסי' וטב"ע בגדיהם).
