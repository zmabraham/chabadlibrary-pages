# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר יא. > פֶּרֶק ג

וְזֶהוּ דִכְתִיב (יְחֶזְקֵאל כ. לב-לג) וְהָעוֹלָה עַל רוּחֲכֶם (שֶׁעוֹלֶה עַל דַּעְתְּכֶם. רַשִׁ"י) הָיוֹ לֹא תִהְיֶה, אֲשֶׁר אַתֶּם אוֹמְרִים נִהְיֶה כַגּוֹיִם (נִפְרוֹק עוּלּוֹ מֵעַל צַוָּארֵנוּ מֵאַחַר שֶׁשִּׁלַּח מֵעָלָיו. רַשִׁ"י) כְּמִשְׁפְּחוֹת הָאֲרָצוֹת לְשָׁרֵת עֵץ וָאָבֶן חַי אָנִי נְאֻם ה' אֱלֹקִים אִם לֹא בְּיָד חֲזָקָה וּבִזְרוֹעַ נְטוּיָה וּבְחֵמָה שְׁפוּכָה (לְכַלּוֹת הַפּוֹשְׁעִים וְהַמּוֹרְדִים. רְדַ"ק) אֶמְלוֹךְ עֲלֵיכֶם.
[cup]וּבְמִדְרַשׁ[/cup] תַּנְחוּמָא (נִצָּבִים פֶּרֶק ג) אִיתָא כְּשֶׁבִּקְשׁוּ (יִשְׂרָאֵל) לִפְרוֹק עוֹל שְׁבוּעָתוֹ (שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא) בִּימֵי יְחֶזְקֵאל מַה כְּתִיב שָׁם (יְחֶזְקֵאל כ. א) בָּאוּ אֲנָשִׁים מִזִּקְנֵי יִשְׂרָאֵל לִדְרֹשׁ אֶת ה' אָמְרוּ לוֹ בֶּן כֹּהֵן הַקּוֹנֶה אֶת הָעֶבֶד מַהוּ שֶׁיֹּאכַל בִּתְרוּמָה אָמַר לָהֶם אוֹכֵל, אָמְרוּ לוֹ אִם חָזַר כֹּהֵן וּמְכָרוֹ לְיִשְׂרָאֵל לֹא יָצָא מֵרְשׁוּתוֹ (בִּתְמִיָּה) אָמַר לָהֶם הֵן (שֶׁיּוֹצֵא מֵרְשׁוּת הַכֹּהֵן לִרְשׁוּת יִשְׂרָאֵל) אָמְרוּ לוֹ אַף אָנוּ כְּבָר יָצָאנוּ מֵרְשׁוּתוֹ נִהְיֶה כְּכָל הָאוּמּוֹת, אָמַר לָהֶם יְחֶזְקֵאל לְיִשְׂרָאֵל וְהָעוֹלָה עַל רוּחֲכֶם הָיוֹ לֹא תִהְיֶה וְאוֹמֵר חַי אָנִי נְאוּם ה' אֱלֹקִים אִם לֹא בְּיָד חֲזָקָה וגו' אָמַר לָהֶם כָּל זְמַן שֶׁלֹּא מְכָרוֹ בִּרְשׁוּתוֹ הוּא וְאַתֶּם לֹא נִמְכַּרְתֶּם בְּדָמִים שֶׁנֶּאֱמַר (יְשַׁעְיָה נב. ג) כִּי כֹה אָמַר ה' חִנָּם נִמְכַּרְתֶּם, (דְּבָרִים כט. יב) לְמַעַן הָקִים אוֹתְךָ הַיּוֹם לוֹ לְעָם וְהוּא יִהְיֶה לְּךָ לֵאלֹקִים כַּאֲשֶׁר דִּבֶּר לָךְ וְכַאֲשֶׁר נִשְׁבַּע לַאֲבוֹתֶיךָ לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב שֶׁלֹּא אֶחֱזֹר בַּדָּבָר שֶׁנִּשְׁבַּעְתִּי לַאֲבוֹתֵיכֶם.
[cup]וְזֶהוּ[/cup] וְהָעוֹלָה עַל רוּחֲכֶם שֶׁיִּהְיוּ כְּכָל הַגּוֹיִם וִיקַבְּלוּ הַשְׁפָּעָתָם חַס וְשָׁלוֹם מִסִּטְרָא אָחֳרָא עַל יְדֵי יְנִיקָתָם מֵהַמַּקִּיף עֶלְיוֹן, הִנֵּה הָיוֹ לֹא תִהְיֶה, וְאוֹמֵר חַי אָנִי נְאֻם ה' אֱלֹקִים אִם לֹא בְּיָד חֲזָקָה אֶמְלוֹךְ עֲלֵיכֶם, כִּי הֲלֹא כְּבָר אַתֶּם עִמָּדִי בִּשְׁבוּעָה, שֶׁהֵן חֵלֶק ה' שֶׁבָּחַר בָּהֶם וְהֵם בָּחֲרוּ בוֹ יִתְבָּרֵךְ וַהֲרֵי הֵם שַׁיָּיכִים לִבְחִינַת הַפְּנִימִית דַּוְקָא. וְלָזֹאת אִי אֶפְשָׁר לָהֶם לְקַבֵּל חַיּוּתָם וְהַשְׁפָּעָתָם מִבְּחִינַת חִיצוֹנִיּוּת הַמַּקִּיף כְּכָל הַגּוֹיִם, כִּי אִם מִבְּחִינַת הַפְּנִימִית עַל יְדֵי קִיּוּם הַתּוֹרָה וּמִצְוֹת,
וּמִצַּד הָאַהֲבָה פְנִימִית דִּבְחִינַת פְּנִימִית וְעַצְמוּת אוֹר אֵין סוֹף (בְּחִינַת פְּנִימִית הַמַּקִּיף כַּנַ"ל) הוּא מַכְרִיחֵם שֶׁיְּקַבְּלוּ עֲלֵיהֶם עוֹל מַלְכוּתוֹ יִתְבָּרֵךְ דְּזֶהוּ שֶׁכָּתוּב אִם לֹא בְּיָד חֲזָקָה וגו'
וְכִדְאִיתָא בְּמִדְרָשׁ רַבָּה (בַּמִּדְבָּר פַּרְשָׁה ב פִּיסְקָא טז) אַף עַל פִּי שֶׁאַתֶּם לֹא עַמִּי שֶׁתְּבַקְשׁוּ לְהִפָּרֵשׁ מִמֶּנִּי אֲנִי לֹא אֶהְיֶה לָכֶם אֵין דַּעְתִּי מִשְׁתַּוֵּוית עִמָּכֶם אֶלָּא בְּעַל כָּרְחֲכֶם תִּהְיוּ עַמִּי וְכֵן הוּא אוֹמֵר וְהָעוֹלָה עַל רוּחֲכֶם הָיוֹ לֹא תִהְיֶה כו' אִם לֹא בְּיָד חֲזָקָה כו' אֶמְלוֹךְ עֲלֵיכֶם הָא לָמַדְתָּ חִיבָּתָן שֶׁל יִשְׂרָאֵל לִפְנֵי הַמָּקוֹם,
שֶׁזֶּה מִצַּד הָאַהֲבָה פְנִימִית דַּוְקָא דִּבְחִינָה זוֹ שַׁיֶּיכֶת לְיִשְׂרָאֵל דַּוְקָא, וּמִשָּׁם נִמְשָׁךְ לְהַכְרִיחָם לְטוֹבָתָם גַּם בְּאוֹפַנִּים קָשִׁים חַס וְשָׁלוֹם, וּכְמוֹ שֶׁאָמְרוּ (סַנְהֶדְרִין קה. א) אָמַר רַב נַחְמָן כָּל כִּי הַאי רִיתְחָא (מִי יִתֵּן וְיָבֹא לָנוּ זֶה הַכַּעַס בְּחֵימָה שְׁפוּכָה אֶמְלוֹךְ עֲלֵיכֶם שֶׁיִּגְאָלֵנוּ בְּעַל כָּרְחֵנוּ וְיִמְלוֹךְ עָלֵינוּ. רַשִׁ"י) לִירְתַּח רַחֲמָנָא עֲלָן וְלִפְרוֹקִינָן,
וְעַל יְדֵי צְעָקָה פְנִימִית מְעוֹרְרִים בְּחִינַת פְּנִימִית וְעַצְמוּת אוֹר אֵין סוֹף בָּרוּךְ הוּא דִּלְהָתָם מָטָא עַקְתָא דְיִשְׂרָאֵל כו'.
[cup]הַיּוֹצֵא[/cup] מִכָּל הַנַ"ל דְּמַה שֶּׁיִּתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר שָׁלוֹם יִהְיֶה לִּי גַּם אִם יַעֲשֶׂה מַעֲשֵׂה הַחֵטְא, הוּא רַק מִפִּתּוּיֵי וַהֲסָתַת הַיֵּצֶר, רַק שֶׁמְּכַסֶּה וּמַטֵּהוּ מִדֶּרֶךְ הָאֱמֶת, וּמִי הָאִישׁ הֶחָפֵץ חַיִּים צָרִיךְ לִהְיוֹת לִבּוֹ בִּרְשׁוּתוֹ וְלֹא יִתְפַּתֶּה לְדִבְרֵי הֲבַאי שֶׁל הַיֵּצֶר הָרָע, וְיָבִין הֵיטֵב דַּרְכּוֹ וַאֲשֶׁר נוֹטֵל עָלָיו וְאָז יֵלֵךְ לָבֶטַח דַּרְכּוֹ בְּקִיּוּם הַתּוֹרָה וּמִצְוֹת וְלֹא יָסוּר יָמִין וּשְׂמֹאל עַל יְדֵי פִּתּוּיֵי הַיֵּצֶר וְלֹא יִשְׁמַע אֵלָיו, כִּי כָל דְּבָרָיו בְּשֶׁקֶר יְסוֹדָם וְהַכַּוָּונָה רַק לְנַסּוֹתוֹ, וּלְבָבוֹ יָבִין אֶת הָאֱמֶת וְיֵלֵךְ לָבֶטַח בְּדֶרֶךְ הַתּוֹרָה וּמִצְוֹת וְעַל יְדֵי זֶה יִחְיֶה בְּחַיִּים רוּחָנִים וְחַיִּים גַּשְׁמִים.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]הָאוֹמְרִים נִהְיֶה כְּכָל הַגּוֹיִם לְקַבֵּל מֵחִיצוֹנִיּוּת הַמַּקִּיף נֶעֱנָשִׁים בְּיָד חֲזָקָה בִּכְדֵי שֶׁיַּחְזְרוּ בִּתְשׁוּבָה, וּמִצַּד הָאַהֲבָה פְנִימִית מַכְרִיחִים אוֹתָם עַל זֶה גַּם בְּאוֹפַנִּים קָשִׁים, לִירְתַּח רַחֲמָנָא עֲלָן וְלִפְרְקִינָן, יָבִין הֲסָתַת הַיֵּצֶר וְיִזָּהֵר בְּקִיּוּם הַתּוֹרָה וּמִצְוֹת וְיִחְיֶה בְּחַיִּים רוּחָנִים וְגַשְׁמִים.[kitzur_end]
