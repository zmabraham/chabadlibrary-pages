# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ו. > פֶּרֶק א

מַאֲמָר ו.
וְהִנֵּה הַפִּיּוּס עַל זֶה שֶׁאָמַר לָהּ הַקָּדוֹשׁ בָּרוּךְ הוּא לְכִי וּמַעֲטִי אֶת עַצְמֵךְ הוּא צַדִּיקִים יִקָּרְאוּ עַל שְׁמֵךְ, שְׁמוּאֵל הַקָּטָן, דָּוִד הַקָּטָן. וְהָעִנְיָן הוּא דִּכְמוֹ כֵן בִּירִידַת הַנְּשָׁמָה לְהִתְלַבֵּשׁ בַּגּוּף וְנֶפֶשׁ הַבַּהֲמִית אֲשֶׁר הוּא עָלוּל לְהִתְגָּאֵל כו' וְכָל הַדְּרָכִים בְּחֶזְקַת סַכָּנָה, כִּי לַפֶּתַח חַטָּאת רוֹבֵץ, וְאַף עַל פִּי כֵן כְּדַאי יְרִידָה זוֹ בִּשְׁבִיל הָעֲלִיָּה הַנִּפְלָאָה שֶׁנַּעֲשָׂה עַל יְדֵי זֶה,
שֶׁעַל יְדֵי שֶׁהוּא מְלוּבָּשׁ בַּגּוּף וְנֶפֶשׁ הַבַּהֲמִית, אֲשֶׁר עָלוּל לְהִתְגָּאֵל חַס וְשָׁלוֹם וּמִכָּל מָקוֹם עוֹמֵד בְּנִסָּיוֹן בְּכָל פְּרָט וְכוֹבֵשׁ רוּחַ תַּאֲוָתוֹ, שֶׁאֵינוֹ נִמְשָׁךְ אַחַר הֲסָתַת הַיֵּצֶר וְהוּא נִזְהָר בְּסוּר מֵרָע וַעֲשֵׂה טוֹב בְּכָל פְּרָטֵיהֶן וְדִקְדּוּקֵיהֶן, וְעִנְיְנֵי הָעוֹלָם הַזֶּה אֵינָם מוֹנְעִים וּמְעַכְּבִים אוֹתוֹ מִתּוֹרָה וַעֲבוֹדָה, אַדְרַבָּא, כָּל עִנְיָנָיו הַגַּשְׁמִים שֶׁמּוּכְרָח בָּהֶם הֵם בְּכַוָּונָה לְשֵׁם שָׁמַיִם, כְּמוֹ בַּאֲכִילָה שֶׁאֵינָהּ לְמַלֹּאות תַּאֲוַת נַפְשׁוֹ כִּי אִם לְקִיּוּם גּוּפוֹ, וּבִכְדֵי שֶׁבְּכֹחַ הָאֲכִילָה הַהִיא יוּכַל לִלְמוֹד וּלְהִתְפַּלֵּל וְהוּא לוֹמֵד וּמִתְפַּלֵּל בֶּאֱמֶת, וְכֵן בְּעֵסֶק מַשָּׂא וּמַתָּן, שֶׁעוֹשֶׂה כָּל דָּבָר בְּתַכְלִית הַהֶכְשֵׁר, וְאֵינוֹ מְשׁוּקָּע וּמוּטְרָד בָּזֶה, בְּאוֹפֶן שֶׁאֵינוֹ מוֹנֵעַ אוֹתוֹ מִתּוֹרָה וּתְפִלָּה, וְכַוָּונַת הַמַּשָּׂא וּמַתָּן הוּא כְּדֵי לְהַרְוִיחַ וְלִיתֵּן צְדָקָה וְשֶׁיִּהְיֶה לוֹ פְּנַאי לַעֲסוֹק בְּתוֹרָה וַעֲבוֹדָה,
עַל יְדֵי זֶה נַעֲשָׂה עֲלִיָּה בְּהַנְּשָׁמָה הַרְבֵּה יוֹתֵר מִכְּמוֹ שֶׁהָיְתָה קוֹדֶם הִתְלַבְּשׁוּתָהּ בַּגּוּף וְנֶפֶשׁ הַבַּהֲמִית, שֶׁלֹּא הָיָה לָהּ דְּבָרִים הַמּוֹנְעִים כְּלָל וְלֹא יָדְעָה כְּלָל מֵהַהֵיפֶךְ, וּמִמֵּילָא לֹא הָיָה עֲבוֹדָה כְּלָל, וּבְהִתְלַבְּשׁוּתָהּ בַּגּוּף כְּשֶׁעוֹבֶדֶת עֲבוֹדָתָהּ לְהַכְנִיעַ אֶת הַגּוּף וְהַנֶּפֶשׁ הַבַּהֲמִית לִהְיוֹת שָׁלֵם בַּעֲבוֹדָתָהּ כַּנַ"ל, עַל יְדֵי זֶה נַעֲשָׂה עֲלִיָּה גְדוֹלָה לְהַנְּשָׁמָה וְנַעֲשָׂה מִזֶּה תַּעֲנוּג גָּדוֹל לְמַעְלָה, וּכְמוֹ שֶׁכָּתוּב (בְּרֵאשִׁית כא. ו) צְחֹק עָשָׂה לִי אֱלֹקִים, דְּדַוְקָא עַל יְדֵי שֵׁם אֱלֹקִים, שֶׁעַל יְדֵי זֶה נַעֲשָׂה מְצִיאוּת הַיֵּשׁ וְכָל עִנְיְנֵי הָעוֹלָם הַגַּשְׁמִי וְהַחוּמְרִי אֲשֶׁר הַכֹּל הוּא מִשֵּׁם אֱלֹקִים, כְּמוֹ שֶׁכָּתוּב (בְּרֵאשִׁית א. א) בְּרֵאשִׁית בָּרָא אֱלֹקִים, וְכֵן הַנֶּפֶשׁ הַבַּהֲמִית שָׁרְשׁוֹ מִשֵּׁם אֱלֹקִים כְּיָדוּעַ וּמְבוֹאָר בְּמָקוֹם אַחֵר, הִנֵּה עַל יְדֵי עֲבוֹדָה לְמַטָּה בְּהַכְנָעַת הַחוֹמֶר בְּכָל הָעִנְיָנִים הַגַּשְׁמִים, וְלִהְיוֹת סוּר מֵרָע וַעֲשֵׂה טוֹב בְּפוֹעַל מַמָּשׁ מִזֶּה דַוְקָא נַעֲשֶׂה הַצְּחוֹק וְהַתַּעֲנוּג הָעֶלְיוֹן,
וּכְמוֹ כֵן הוּא בִּבְחִינַת מַלְכוּת שֶׁעַל יְדֵי יְרִידָתָהּ לִבְרִיאָה-יְצִירָה-עֲשִׂיָּה, הֲרֵי הִיא מְבָרֶרֶת בֵּירוּרִים דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה, בְּדוּגְמָא כְּמוֹ שֶׁהִתְלַבְּשׁוּת הַנְּשָׁמָה בַּגּוּף שֶׁפּוֹעֶלֶת הַהַכְנָעָה וְהַבִּיטּוּל כַּנַ"ל כְּמוֹ כֵן הוּא בְּהִתְלַבְּשׁוּת דִּבְחִינַת מַלְכוּת בִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה, הִיא מְבָרֶרֶת בֵּירוּרִים דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה. וּכְמוֹ שֶׁכָּתוּב (מִשְׁלֵי לא. טו) וַתִּתֵּן טֶרֶף לְבֵיתָהּ, טֶרֶ"ף בְּגִימַטְרִיָּא רפ"ח עִם הַכּוֹלֵל, וְהַיְינוּ שֶׁמְּבָרֶרֶת הָרפ"ח נִיצוֹצוֹת בִּירִידָתָהּ לְמַטָּה,
וְעַל יְדֵי זֶה נִתּוֹסֵף בָּהּ רִיבּוּי אוֹר עַד שֶׁלֶּעָתִיד תִּהְיֶה אֵשֶׁת חַיִל עֲטֶרֶת בַּעְלָהּ, שֶׁבְּחִינַת מַלְכוּת תִּהְיֶה בִּבְחִינַת מַשְׁפִּיעַ בְּהַסְּפִירוֹת שֶׁלְּמַעְלָה. וּכְמוֹ שֶׁכָּתוּב (זְכַרְיָה יד. ח) וְהָיָה בַּיּוֹם הַהוּא יֵצְאוּ מַיִם חַיִּים מִירוּשָׁלַיִם (בְּחִינַת מַלְכוּת) חֶצְיָם אֶל הַיָּם הַקַּדְמוֹנִי (יַם הַחָכְמָה) כו' וְזֶה יִהְיֶה בִּגְמַר הַבֵּירוּרִים שֶׁיִּתְבָּרְרוּ עוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה.
