# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר כז. > פֶּרֶק ב

וְזֶהוּ וּמַעְיָן מִבֵּית ה' יֵצֵא, דְּהִנֵּה מַעְיָן הוּא בְּחִינַת חָכְמָה וּכְמוֹ שֶׁכָּתוּב בְּסֵפֶר קְהִלַּת יַעֲקֹב בְּשֵׁם הָרַעְיָא מְהֵימְנָא, וּכְיָדוּעַ בְּעִנְיַן נָהָר וּמַעְיָן דְּנָהָר הוּא סְפִירַת הַבִּינָה וְנִקְרֵאת רְחוֹבוֹת הַנָּהָר לִהְיוֹת כָּל עִנְיַן הַנָּהָר אוֹרֶךְ וְרוֹחַב הַנִּמְשָׁךְ לְמֵרָחוֹק וּמִתְפַּשֵּׁט בְּרוֹחַב גָּדוֹל וּמַעְיָן הוּא סְפִירַת הַחָכְמָה, שֶׁהַהֶפְרֵשׁ בֵּינֵיהֶם דְּנָהָר הֲגַם שֶׁמִּתְפַּשֵּׁט בַּאֲרִיכוּת וּבְרוֹחַב גָּדוֹל שֶׁמִּתְמַלֵּא עַל כָּל גְּדוֹתָיו, בְּכָל זֶה אֶפְשָׁר שֶׁנִּפְסָק וּכְמוֹ שֶׁכָּתוּב (יְשַׁעְיָה יט. ה) וְנָהָר יֶחֱרַב וְיָבֵשׁ, וְהַמַּעְיָן עִם הֱיוֹת שֶׁנּוֹטֵף טִיפִּין טִיפִּין הִנֵּה אֵין לוֹ הֶפְסֵק,
וְרַבֵּינוּ נִשְׁמָתוֹ עֵדֶן בֵּיאֵר (סֵפֶר שֶׁל בֵּינוֹנִים פֶּרֶק י"ח וּפֶרֶק י"ט) דְּחָכְמָה שֶׁבְּנֶפֶשׁ הוּא בְּחִינָה שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת וּמִשָּׁם כֹּחַ הַמְּסִירַת נֶפֶשׁ שֶׁיֵּשׁ בְּכָל אֶחָד וְאֶחָד לִמְסוֹר נַפְשׁוֹ עַל קִדּוּשׁ הַשֵּׁם וּמֵבִיא עַל זֶה פָּסוּק פֶּתִי יַאֲמִין לְכָל דָּבָר (מִשְׁלֵי יד. טו) שֶׁזֶּהוּ בְּחִינַת שְׁטוּת דִּקְדוּשָּׁה שֶׁנִּתְבָּאֵר לְעֵיל שֶׁזֶּהוּ מִבְּחִינַת כֹּחַ הַחָכְמָה שֶׁבְּנֶפֶשׁ כֹּחַ מָה, וּצְרִיכִים לוֹמַר דִּבְחִינַת כֹּחַ מָה שֶׁבְּנֶפֶשׁ הוּא בְּחִינָה עֶלְיוֹנָה שֶׁבְּחָכְמָה,
דְּהִנֵּה בְּחָכְמָה יֵשׁ ב' מַדְרֵיגוֹת הָא' מַה שֶּׁחָכְמָה הִיא רֵאשִׁית הַגִּילּוּי וְרֵאשִׁית הַהִשְׁתַּלְשְׁלוּת וְהַב' מַה שֶּׁלְּמַעְלָה מִזֶּה וְהוּא לְמַעְלָה מֵהִשְׁתַּלְשְׁלוּת וְאֵינֶנָּה מִתְגַּלֵּית בְּאוֹתוֹ הָאוֹפֶן וּבְאוֹתוֹ הַסֵּדֶר כְּמוֹ שֶׁמִּתְגַּלָּה מַדְרֵיגָה הָא' שֶׁבְּחָכְמָה שֶׁהִיא רֵאשִׁית הַהִשְׁתַּלְשְׁלוּת,
וְעִנְיָנָם בְּכֹחַ הַחָכְמָה שֶׁבְּנֶפֶשׁ הוּא כְּעִנְיַן בְּרִיקַת וּסְקִירַת הַשֵּׂכֶל, דְּעִם הֱיוֹת שֶׁהִיא עֲדַיִין לְמַעְלָה מֵהַשָּׂגָה מַמָּשׁ לְפִי שֶׁהִיא בִּבְחִינַת נְקוּדָּה כְּלָלִית עֲדַיִין וְאֵינוֹ נִתְפָּס בְּהַשָּׂגָה, מִכָּל מָקוֹם הוּא בִּבְחִינַת שֵׂכֶל גָּלוּי עַל כָּל פָּנִים שֶׁהֲרֵי כְּשֶׁנּוֹפֵל לוֹ הַבָּרָק הַמַּבְרִיק הֲרֵי הוּא יוֹדֵעַ אֶת הַשֵּׂכֶל בִּכְלָל עַל כָּל פָּנִים, וְזֶהוּ בְּחִינַת חִיצוֹנִיּוּת הַחָכְמָה, אֲבָל בְּחִינַת פְּנִימִית הַחָכְמָה הוּא מַה שֶּׁמּוֹחִין דְּאַבָּא הֵן לְמַעְלָה מֵהִשְׁתַּלְשְׁלוּת, וְהֵם ב' הַבְּחִינוֹת וּמַדְרֵיגוֹת כֹּחַ וּמָה דְּחָכְמָה, דִּבְחִינַת כֹּחַ דְּחָכְמָה זֶהוּ עִנְיַן הִתְפַּשְּׁטוּתָהּ אֶל הַבִּינָה וּבְחִינַת מָה דְחָכְמָה הוּא עִנְיַן הִתְדַּבְּקוּתָהּ אֶל הַכֶּתֶר, וּבְחִינָה זוֹ דְחָכְמָה הִנֵּה זֶהוּ מַה שֶּׁחָכְמָה הוּא בְּחִינַת הַבִּיטּוּל שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת.
[cup]וְזֶהוּ[/cup] שֶׁכָּתוּב וּמַעְיָן מִבֵּית ה' יֵצֵא הִנֵּה בַּיִת הוּא בְּחִינַת מַקִּיף, וְעַיֵּן מַה שֶּׁכָּתוּב מֵעִנְיַן בַּיִת בְּלִקּוּטֵי תוֹרָה בְּסוֹפוֹ בִּדְּרוּשׁ הַמַּתְחִיל מִזְמוֹר שִׁיר חֲנוּכַּת הַבַּיִת דְּרוּשׁ הַב' שֶׁזֶּהוּ בְּחִינַת פְּנִימִיּוּת הַכֶּתֶר, וְזֶהוּ עִנְיַן וּמַעְיָן מִבֵּית ה' יֵצֵא הַיְינוּ בְּחִינַת הַחָכְמָה כְּמוֹ שֶׁנִּמְשֶׁכֶת מִבְּחִינַת הַכֶּתֶר (וְעַיֵּן מַה שֶּׁכָּתַב הַפַּרְדֵּס בְּשַׁעַר עֶרְכֵי הַכִּנּוּיִם עֶרֶךְ מַעְיָן) וְהוּא בְּחִינַת מָה דְחָכְמָה הַנַ"ל, וּמַעְיָן זֶה יַשְׁקֶה אֶת נַחַל הַשִּׁטִּים בְּחִינַת הַשְּׁטוּת דִּלְעוּמַּת זֶה שֶׁיִּתְהַפֵּךְ לִשְׁטוּת דִּקְדוּשָּׁה וְהַיְינוּ בַּעֲבוֹדָה בְּפוֹעַל לִהְיוֹת בִּבְחִינַת שְׁטוּת דִּקְדוּשָּׁה שֶׁלֹּא עַל פִּי דַעַת וְחֶשְׁבּוֹן כַּנַ"ל,
דְּהִנֵּה מַה שֶּׁחָכְמָה הִיא לְמַעְלָה מִטַּעַם וָדַעַת זֶהוּ בַּעֲבוֹדָה בִּבְחִינַת רְעוּתָא דְלִבָּא שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת, וְזֶהוּ בְּחִינַת מַעְיָן וְהִיא הָעֲבוֹדָה שֶׁמִּצַּד עֶצֶם הַנְּשָׁמָה, וְזֶה יַשְׁקֶה אֶת נַחַל הַשִּׁטִּים שֶׁמִּצַּד הַנֶּפֶשׁ הַבַּהֲמִית לִהְיוֹת הַהַנְהָגָה בְּפוֹעַל בְּכָל דָּבָר בִּבְחִינַת שְׁטוּת דִּקְדוּשָּׁה, הֵן בְּעֵסֶק מַשָּׂא וּמַתָּן שֶׁלֹּא יִהְיֶה לוֹ שׁוּם מְנִיעוֹת וְעִכּוּבִים כְּלָל, וּכְמוֹ כֵן לַעֲמוֹד עַל נַפְשׁוֹ בְּכָל הַתַּאֲווֹת וְתַעֲנוּגִים שֶׁלֹּא לְהִמָּשֵׁךְ אַחֲרֵיהֶם כְּלָל וּכְלָל, וְעַל יְדֵי זֶה יִפְעוֹל הַהַכְנָעָה בְּהַנֶּפֶשׁ הַבַּהֲמִית בִּכְלָל שֶׁבֶּאֱמֶת לֹא יִהְיֶה אֶצְלוֹ כָּל כַּךְ הֶרְגֵּשׁ הַטּוֹב דְּהָעִנְיָנִים הַגַּשְׁמִים, וְגַם יִהְיֶה עַל יְדֵי זֶה הִתְגַּלּוּת הַכֹּחוֹת מוֹחִין וּמִדּוֹת דְּנֶפֶשׁ הָאֱלֹקִית שֶׁיִּהְיֶה אָז הַנְהָגָתוֹ בְּכָל עִנְיָנָיו עַל פִּי בְּחִינַת הַדַּעַת דִּקְדוּשָּׁה.

[kitzur_begin]חָכְמָה פֵּירֵשׁ רַבֵּינוּ שֶׁהִיא בְּחִינָה שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת, מַעְיָן, וּמִשָּׁם כֹּחַ הַמְּסִירַת נֶפֶשׁ עַל קִדּוּשׁ הַשֵּׁם וְנִסְיוֹנוֹת, חָכְמָ"ה, כֹּחַ הִתְפַּשְּׁטוּת אֶל הַבִּינָה, מָ"ה הִתְדַּבְּקוּתָהּ אֶל הַכֶּתֶר, מָ"ה דְחָכְמָה מַשְׁקֶה נַחַל הַשִּׁטִּים שֶׁמְּהַפֵּךְ הַשְּׁטוּת דִּלְעוּמַּת זֶה לִשְׁטוּת דִּקְדוּשָּׁה וְאָז הַנְהָגָתוֹ בְּכָל עִנְיָנָיו עַל פִּי דַעַת דִּקְדוּשָּׁה.[kitzur_end]
