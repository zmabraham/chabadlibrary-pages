מַאֲמָר כח.
תּוֹכֶן הַדְּבָרִים בְּעִנְיַן וּמַעְיָן מִבֵּית ה' יֵצֵא וְהִשְׁקָה כו' הוּא
כִּי מִי שֶׁכְּבָר נִפְתָּה בְּפִיתּוּי וּשְׁטוּתֵי הַיֵּצֶר הָרָע מֵאֶחָד מֵהַדְּבָרִים הַנַ"ל, אִם בְּתַאֲווֹת וְתַעֲנוּגֵי עוֹלָם הַזֶּה שֶׁנִּמְשָׁךְ אַחַר הַטּוֹב הַטִּבְעִי הַבַּהֲמִי בְּעִנְיָנִים הַגַּשְׁמִיִים מִפְּנֵי שֶׁמִּצַּד חֻמְרוֹ הַגַּס מַרְגִּישׁ מְאֹד הַטּוֹב בָּזֶה וְזֶה מַעֲבִירוֹ עַל דַּעְתּוֹ וְשִׂכְלוֹ לֵידַע וּלְהָבִין שֶׁאֵין זֶה טוֹב שֶׁלּוֹ כְּלָל, וּבִפְרָט בִּדְבָרִים הָאֲסוּרִים רַחֲמָנָא לִצְּלָן שֶׁגַּם בָּזֶה הָרוּחַ שְׁטוּת שֶׁל הַיֵּצֶר הָרָע שׁוֹלֵט עָלָיו לַהֲסִיתוֹ וּלְכַסּוֹת עַל הָאֱמֶת אֵיךְ שֶׁאֵינוֹ נַעֲשֶׂה נִפְרָד עַל יְדֵי זֶה וַעֲדַיִין הוּא בְּיַהֲדוּתוֹ וְיִתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר שָׁלוֹם יִהְיֶה לוֹ וּמִי יִרְאֶנּוּ וּמִי יוֹדֵעַ אֶת עִנְיָנוֹ וּמַעֲשָׂיו, דְּכָל אֵלֶּה הֵמָּה רַק דִּמְיוֹנוֹת כּוֹזְבִים וְשֶׁקֶר מוּחְלָט כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל בְּאוֹרֶךְ, וְהוּא מִצַּד חוּמְרוֹ הַגַּס נִתְפָּס בְּרוּחַ שְׁטוּת הַנַ"ל,
אוֹ בְּעֵסֶק מַשָּׂא וּמַתָּן בְּרִיבּוּי הִתְעַסְּקוּת וְרִיבּוּי טִרְדּוֹת וְתַחְבּוּלוֹת עַד שֶׁמּוֹנֵעַ אוֹתוֹ מִקְּבִיעוּת עִתִּים לַתּוֹרָה וּמִתְּפִלָּה כִּדְבָעֵי,
וּמִפְּנֵי רִיבּוּי הַשִּׁיקּוּעַ בְּאֶחָד מֵהַדְּבָרִים הַנַ"ל שֶׁהַיֵּצֶר הָרָע הִכְתִּירוֹ וְהִלְבִּישׁוֹ לְהַטּוֹתוֹ מִדֶּרֶךְ הָאֱמֶת וְאֵין בִּיכָלְתּוֹ לְהִתְגַּבֵּר עָלָיו עַל פִּי הַדַּעַת וְחֶשְׁבּוֹן צֶדֶק הַנַ"ל מִפְּנֵי תּוֹקֶף חוֹמֶר יִצְרוֹ (אֲשֶׁר בְּמֶשֶׁךְ הַזְּמַן מִתְגַּבֵּר וְהוֹלֵךְ וּכְמוֹ שֶׁכָּתוּב בְּסֵפֶר שֶׁל בֵּינוֹנִים פֶּרֶק י"ג וְזֶה לְשׁוֹנוֹ וְאַדְרַבָּא נִתְחַזֵּק יוֹתֵר בְּהֶמְשֵׁךְ הַזְּמַן שֶׁנִּשְׁתַּמֵּשׁ בּוֹ הַרְבֵּה בַּאֲכִילָה וּשְׁתִיָּה וּשְׁאָר עִנְיְנֵי עוֹלָם הַזֶּה עַד כַּאן לְשׁוֹנוֹ) שֶׁבְּבוֹאוֹ לְאֶחָד מֵהַדְּבָרִים שֶׁהוּא מְשׁוּקָּע בָּהֶם הוּא מִתְפָּעֵל מְאֹד בְּהֶרְגֵּשׁ הַטִּבְעִי הַבַּהֲמִי וּמְאַבֵּד דַּעְתּוֹ וְשִׂכְלוֹ וּמַשְׁטֶה אֶת עַצְמוֹ מִמֵּילָא,
וְעוֹד זֹאת שֶׁהֲרֵי הַכֹּחוֹת דְּנַפְשׁוֹ הָאֱלֹקִית חֲלוּשִׁים אֶצְלוֹ דְּתוּקְפָּא דְגוּפָא חוּלְשָׁא דְנִשְׁמְתָא, וּמִפְּנֵי כָּל זֶה אִי אֶפְשָׁר לוֹ לָלֶכֶת עַל פִּי דַעַת וְחֶשְׁבּוֹן צֶדֶק
כִּי אִם זֹאת הִיא הָעֵצָה הַיְּעוּצָה לְהִתְגַּבֵּר עַל הָרוּחַ שְׁטוּת דְּהַיֵּצֶר הָרָע דְּזֶה עִנְיַן נַחַל הַשִּׁטִּים עַל יְדֵי זֶה שֶׁיִּתְנַהֵג בִּבְחִינַת שְׁטוּת דִּקְדוּשָּׁה וְהוּא שֶׁיִּתְגַּבֵּר כָּאֲרִי לַעֲשׂוֹת רַק הַהֵיפֶךְ מִיִּצְרוֹ הָרָע בְּלִי שׁוּם טַעֲנָה וּמַעֲנֶה וּבְלִי שׁוּם חֶשְׁבּוֹן כִּי אִם בְּתוֹקֶף גָּדוֹל, שֶׁבְּהַדָּבָר הַהוּא שֶׁשָּׁקוּעַ בּוֹ מֵאֶחָד מֵהַדְּבָרִים הַנַ"ל יַעֲשֶׂה הֵיפֶךְ לְגַמְרֵי וְלֹא יַחְשׁוֹב כְּלָל מַה יִּהְיֶה מֵאִתּוֹ, וְזֶה יֵשׁ בְּכֹחַ נִשְׁמוֹת יִשְׂרָאֵל מִצַּד עֶצֶם נִשְׁמָתָם שֶׁיֵּשׁ בָּהֶם תּוֹקֶף גָּדוֹל לְמַעְלָה מִטַּעַם וָדַעַת שֶׁזֶּהוּ עִנְיַן בְּחִינַת קְשֵׁה עוֹרֶף שֶׁהוּא לְמַעַלְיוּתָא וּבְחִינָה זוֹ זֶהוּ עִנְיַן הַמַּעְיָן שֶׁיֵּצֵא מִבֵּית ה', דִּלְמַעְלָה הוּא בְּחִינַת פְּנִימִית הַחָכְמָה כְּמוֹ שֶׁהִיא דְבוּקָה בַּכֶּתֶר, וּבַנֶּפֶשׁ הוּא בְּחִינַת עֶצֶם נִשְׁמָתוֹ שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת.
[cup]וְזֶהוּ[/cup] וּמַעְיָן מִבֵּית ה' יֵצֵא וְהִשְׁקָה אֶת נַחַל הַשִּׁטִּים, דְּהַתּוֹקֶף שֶׁמִּצַּד עֶצֶם הַנְּשָׁמָה יֵשׁ בְּכֹחוֹ לְהִתְגַּבֵּר עַל הָרוּחַ שְׁטוּת דְּיֵצֶר הָרָע וּלְהַפְּכוֹ לִבְחִינַת שְׁטוּת דִּקְדוּשָּׁה, לַעֲמוֹד עַל נַפְשׁוֹ בְּתוֹקֶף גָּדוֹל לַעֲשׂוֹת בְּפוֹעַל הֵיפֶךְ מֵהַהֶרְגֵּשׁ הַטִּבְעִי שֶׁלּוֹ בְּהַפְּרָט שֶׁהוּא שָׁקוּעַ בּוֹ רַחֲמָנָא לִצְּלָן. וְכַאֲשֶׁר כָּכָה יִתְנַהֵג בְּפוֹעַל מֶשֶׁךְ זְמַן עַל יְדֵי זֶה יַחֲלִישׁ אֶת חוּמְרוֹ הַגַּס וְהַהֶרְגֵּשׁ הַטִּבְעִי הַבַּהֲמִי וְיִתְגַּלֶּה וְיִתְגַּבֵּר בּוֹ הַהֶרְגֵּשׁ מֵהַכֹּחוֹת הַגְּלוּיִם דְּנַפְשׁוֹ הָאֱלֹקִית וְיוּכַל אָז לִהְיוֹת בִּבְחִינַת דַּעַת דִּקְדוּשָּׁה וּכְמוֹ שֶׁכָּתוּב (מִשְׁלֵי ג. ו) בְּכָל דְּרָכֶיךָ דָעֵהוּ, דְּהַיְינוּ שֶׁגַּם בִּדְרָכָיו הַגַּשְׁמִים הַמּוּכְרָחִים לְקִיּוּם וְחַיּוּת גּוּפוֹ וּבְעִנְיְנֵי מַשָּׂא וּמַתָּן לֹא יַשְׁטֶה אֶת עַצְמוֹ חַס וְשָׁלוֹם בְּרוּחַ שְׁטוּת שֶׁל הַיֵּצֶר הָרָע כִּי אִם יִהְיֶה בִּבְחִינַת הַדַּעַת הָאֱלֹקִי הָאֲמִיתִּי וְיֵלֵךְ לָבֶטַח דַּרְכּוֹ.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]בְּכָל אֶחָד מִיִּשְׂרָאֵל יֵשׁ בּוֹ הַתּוֹקֶף דְּמָ"ה דְחָכְמָה שֶׁבַּנֶּפֶשׁ וְהוּא הַמַּעְיָן הַמַּשְׁקֶה גַּם אֶת נַחַל הַשִּׁטִּים, הַנִּלְכָּד בְּפִתּוּיֵי הַיֵּצֶר הָרָע צָרִיךְ לְעוֹרֵר תּוֹקֶף זֶה, וּבְכֹחוֹ לַעֲשׂוֹת בְּפוֹעַל הַהֵיפֶךְ מֵהַהֶרְגֵּשׁ הַטִּבְעִי וּבָזֶה יְבָרֵר חוּמְרוֹ הַגַּס וְיִתְנַהֵג עַל פִּי הַדַּעַת דִּקְדוּשָּׁה.[kitzur_end]
