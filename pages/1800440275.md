והוי' יגיה חשכי שזהו למעלה מבחי' כי אתה נרי ה', וז"ע הב' זתים א' על ימין הגולה בחי' כי אתה נרי וא' על שמאלה להמשיך השמן בנ"ח שמהשמאל שהמשכה זו נמשך מבחי' והוי' יגיה חשכי ועמ"ש כה"ג בד"ה וארשתיך לי בענין כלתה נפשי כו' אכן לבי ובשרי כו' ע"ש. 
ומש"ר בענין שבעה ושבעה מוצקות ב"פ ז', כ"פ בזהר ס"פ מקץ דר"ד סע"ב שזהו כענין שבעת ימים ושבעת ימים, ובמק"מ שם פי' ז"ק דזעיר וז"ק דמל', אמנם לפ"ז כאן שהוא שרש ההמשכה לזו"נ צ"ל שהם ז"מ דאבא וז"מ דאי' והיינו מוצק א' לשמן שהוא מבחי' אבא ומוצק ב' לזהב שהוא מבחי' בינה, ועד"ז י"ל ג"כ למעלה יותר, וע' בע"ח שמ"ו שער כסה"כ פ"ב מענין שבע ושבע מוצקות שהוא ע"ד לאתקשרא שביעאה בשביעאה הנז' בפ' בראשית מ"ה סע"א ובפ' פקודי ר"ס סע"ב, וע' ברע"מ פנחס רנ"ה סע"ב. ולכן ג"כ בענין שבע ביום הללתיך יש ב"פ שבע הא' ע"י מעשה היינו ד' ציצית וב' תפילין ומזוזה כמ"ש בגמרא בפ' התכלת דמ"ג, הב' ע"י דבור היינו ז' ברכות דק"ש בשחר מברך שתים לפניה וא' לאחריה ובערב ב' לפניה וב' לאחריה ופרש"י ספ"ק דברכות ע"ש שבע ביום כו', והיינו כי הם שבעה כפולות בחי' שבעה ושבעה לאתקשרא שביעאה בשביעאה כו' והיינו ג"כ חיבור דיבור ומעשה רוחני' וגשמי' כו' דכר ונוק' כו'. 
ועוי"ל דהנה פי' מוצקות היינו שהשמן ניצוק בהם מעצמו וכמ"ש רש"י שם ע"פ לא בחיל ולא בכח כ"א ברוחי כשם שהזתים והשמן הזה נגמר מאליו לכל דבריו כך לא בחיל ולא בכח שלכם תעשו את בנין ביתי כ"א ברוחי כו' וה"ע אתעדל"ע שלמעלה מאתעדל"ת כ"א באה מאלי' וממילא כענין קדושתי למעלה מקדושתכם, וז"ע ז' תיקוני' דרישא דא"א שבהם נמשך מז"ת דעת"י כו' וע' בעה"מ ש"ח פ"ח, ושי"א פ"ח, והיינו מ"ש כשמן הטוב כו' כטל חרמון ופי' בזח"א פ"ח א' איהו שמן ואיהו טל כו' ונודע דטל הוא בחי' שלמעלה מאתעדל"ת כמ"ש בד"ה האזינו השמים כו' וגם טל הוא מעשר מגימט' שמן כי טל במו"ס טל אורות ותורה אור טל תורה ושמן הוא על הראש בחי' גלגלתא כשב"כ בחי' מצות כו', אמנם ידוע שכדי שיהי' גילוי בחי' זו שלמעלה מאתעדל"ת בבחי' פנימי' ממש הוא דוקא ע"י שבתחלה נמשך האתעדל"ע הבאה ע"י אתעדל"ת כמ"ש מזה בביאור ע"פ אני ישנה כו' ועמ"ש מזה בד"ה כי כאשר השמים החדשים כו' כי שמים וארץ דעכשיו זהו ענין אתעדל"ע הבאה ע"י אתעדל"ת ואזי אח"כ יהי' גלוי בחי' השמים החדשים כו' אשר אני עושה כלומר שהוא אתעדל"ע מאליו ומעצמו כו', וע' מענין אתעדל"ע זו ג"כ סד"ה רני ושמחי כו' שעז"נ כי הנני בא כו' וזה"ע נ"ח כו' ע"ש, ואפ"ל שלזה הן ז' וז' מוצקות פי' הז' בתחלה הן האתעדל"ע הבאה ע"י אתערותא דלתתא ונק' גם כן מוצקות אף שבאו ע"י אתעדל"ת כי באמת האתעדל"ת הוא כגרגיר חרדל לגבי האתעדל"ע כמ"ש בד"ה כי כאשר השמים כו' אלא שכך עלה ברצונו שיהי' נמשך אתעדל"ע על ידי אתעדל"ת הגם שאין ערוך זל"ז, וכיון שכן שזהו לפי שכך עלה ברצונו לכן נק' ג"כ מוצקות, ועוי"ל שיש בזה ג"כ רמז להאתעדל"ת שהוא מל' ד' בצר פקדוך צקון לחש כו' בישעי' כ"ו י"ו ופרש"י צקון
