מַאֲמָר כז.
וְהִנֵּה אֵלּוּ אֲשֶׁר אוֹר נַפְשָׁם הָאֱלֹקִית מֵאִיר בָּהֶם בְּגִילּוּי וְיֵשׁ לָהֶם דַּעַת וְהַשָּׂגָה בֵּאלֹקוּת וְהָעִיקָּר שֶׁהֵם בַּעֲלֵי עֲבוֹדָה וְהַיְינוּ דְּאֵין זֶה מַה שֶּׁהֵם בַּעֲלֵי יְדִיעָה וְהַשָּׂגָה בֵּאלֹקוּת אֶלָּא שֶׁהֵם בַּעֲלֵי עֲבוֹדָה, אֲשֶׁר מִכָּל דְּבַר הַשָּׂגָה יֵשׁ לָהֶם תּוֹעֶלֶת בַּעֲבוֹדָה בְּפוֹעַל, וְהַהֶרְגֵּשׁ הָאֱלֹקִי מִתְגַּבֵּר עַל הַהֶרְגֵּשׁ הַטִּבְעִי דְּלֹא זוֹ בִּלְבָד שֶׁאֵינָם נִפְתִּים בִּשְׁטוּתֵי הַיֵּצֶר הָרָע אֶלָּא עוֹד זֹאת מִתְחַדֵּשׁ אֶצְלָם עִנְיְנֵי בֵּיאוּרִים וְהֶסְבֵּרִים שִׂכְלִים בְּחֶשְׁבּוֹנוֹת הַנַ"ל (מַאֲמָר א' פֶּרֶק ג-ד) בְּמַעֲלוֹת מִין אֱנוֹשִׁי וּבְתַכְלִית בְּרִיאָתוֹ בָּעוֹלָם, וְטַעַם הַדָּבָר הוּא לְפִי שֶׁהֵם בַּעֲלֵי עֲבוֹדָה וּמִזְדַּכְּכִים מִזְּמַן לִזְמַן בְּעִילּוּי אַחַר עִילּוּי בְּהַגְבָּרַת הַצּוּרָה עַל הַחוֹמֶר,
אֲבָל מִי שֶׁאֵין אוֹר נִשְׁמָתוֹ מֵאִיר בּוֹ בְּגִילּוּי כָּל כַּךְ וְנִפְתֶּה בְּפִיתּוּי הַיֵּצֶר, זֹאת הִיא עֲצָתוֹ שֶׁלֹּא לָבֹא כְּלָל לִידֵי חֶשְׁבּוֹנוֹת פְּרָטִים אֶלָּא לַעֲמוֹד נֶגֶד כָּל מוֹנֵעַ וּמְעַכֵּב בְּתוֹקֶף גָּדוֹל מִצַּד הַבְּחִינָה הַב' שֶׁבַּנֶּפֶשׁ וְהוּא בְּחִינַת עֶצֶם הַנְּשָׁמָה שֶׁלְּמַעְלָה מִטַּעַם וָדַעַת.
[cup]דְּהִנֵּה[/cup] שֵׂכֶל וּמִדּוֹת הֵן רַק כֹּחוֹת הַנֶּפֶשׁ וְלֹא הָעֶצֶם אֲבָל עֶצֶם הַנְּשָׁמָה לְמַעְלָה מִבְּחִינַת הַשֵּׂכֶל וְעַל זֶה נֶאֱמַר (שְׁמוֹת לד. ט) כִּי עַם קְשֵׁה עוֹרֶף הוּא וְסָלַחְתָּ דְּמַה שֶּׁהֵם קְשֵׁה עוֹרֶף הוּא לְמַעַלְיוּתָא (דְּמִשּׁוּם זֶה הוּא טַעַם שֶׁבִּשְׁבִיל זֶה תִּהְיֶה הַסְּלִיחָה) וְהַיְינוּ הַתּוֹקֶף שֶׁיֵּשׁ בִּנְשָׁמוֹת לִהְיוֹת סוּר מֵרָע וַעֲשֵׂה טוֹב בְּלִי שׁוּם טַעַם וּבְלִי שׁוּם טַעֲנָה וּמַעֲנֶה כְּלָל,
וּכְמוֹ הָעִיקֵּשׁ הוּא שֶׁאֵינוֹ מִתְנַהֵג עַל פִּי הַטַּעַם וְאָנוּ רוֹאִים בְּחוּשׁ שֶׁהָעִיקֵּשׁ רְצוֹנוֹ חָזָק בְּיוֹתֵר הַרְבֵּה מִזֶּה שֶׁמִּתְנַהֵג עַל פִּי הַשֵּׂכֶל, דְּזֶה שֶׁמִּתְנַהֵג עַל פִּי הַשֵּׂכֶל הֲרֵי רְצוֹנוֹתָיו הֵן עַל פִּי הַדַּעַת וְחִיּוּב שִׂכְלִי, וּכְיָדוּעַ שֶׁהַשֵּׂכֶל מַחֲלִישׁ אֶת הָרָצוֹן וּמַגְבִּיל אוֹתוֹ כַּמָּה יִתְפַּשֵּׁט אֲבָל הָרָצוֹן שֶׁאֵינוֹ עַל פִּי הַשֵּׂכֶל הוּא מִתְפַּשֵּׁט בְּיוֹתֵר בְּלִי הַגְבָּלָה. וְזֶהוּ שֶׁהָעִיקֵּשׁ רְצוֹנוֹ בְּתוֹקֶף בְּיוֹתֵר.
[cup]וּכְמוֹ[/cup] כֵן יֵשׁ בְּנִשְׁמוֹת יִשְׂרָאֵל עִנְיַן הָעִקְשׁוּת בִּקְדוּשָּׁה בְּתוֹקֶף גָּדוֹל וּכְמוֹ בִּמְסִירַת נֶפֶשׁ עַל קִדּוּשׁ הַשֵּׁם שֶׁיֵּהָרֵג וְלֹא יַעֲבוֹר חַס וְשָׁלוֹם, וְכֵן יָכוֹל לִהְיוֹת בְּכָל דָּבָר לִהְיוֹת בְּתוֹקֶף גָּדוֹל שֶׁלֹּא לָלֶכֶת בְּדַרְכֵי הַיֵּצֶר הָרָע וְהַנֶּפֶשׁ הַבַּהֲמִית וְלֹא לְהִמָּשֵׁךְ אַחֲרֵי תַּאֲווֹת לִבּוֹ, אַחֲרֵי שֶׁיּוֹדֵעַ כְּלַל הַדָּבָר שֶׁכָּל הַתַּאֲווֹת וְהַתַּעֲנוּגִים הַגַּשְׁמִים הֵמָּה תּוֹעֲבַת ה' אֲשֶׁר שָׂנֵא, וּבִפְרָט דְּבַר אִיסּוּר וְרָע מַמָּשׁ אֵינוֹ נִכְנָס כְּלָל בִּטְעָנוֹת וְחֶשְׁבּוֹנוֹת בָּזֶה,
וּכְמוֹ כֵן אַחֲרֵי שֶׁיּוֹדֵעַ שֶׁצָּרִיךְ לְהִתְפַּלֵּל וְלִשְׁמוֹעַ פֵּירוּשׁ הַמִּלּוֹת עַל כָּל פָּנִים וְלִקְבּוֹעַ עִתִּים לַתּוֹרָה, אֵינוֹ נִכְנָס כְּלָל בְּשׁוּם טַעֲנָה וּמַעֲנֶה וְאֵינוֹ מַבִּיט כְּלָל אִם יַפְסִיד אוֹ לֹא יַפְסִיד, לִהְיוֹת כִּי מוּחְלָט אֶצְלוֹ בְּהֶחְלֵט גָּמוּר אֲשֶׁר הוּא צָרִיךְ לַעֲשׂוֹת אֶת שֶׁלּוֹ הַמְצֻוֶּוה עָלָיו עַל פִּי הַתּוֹרָה וְהַמִּצְוָה וְלַעֲמוֹד חָזָק נֶגֶד תַּאֲווֹת הַנֶּפֶשׁ הַבַּהֲמִית שֶׁלֹּא לַעֲשׂוֹת מַה שֶּׁמִּתְאַוֶּה לִבּוֹ בְּשׁוּם אוֹפֶן בְּכֹחַ וְתוֹקֶף גָּדוֹל, וְהוּא עִנְיַן בְּחִינַת הַשְּׁטוּת דִּקְדוּשָּׁה שֶׁהוּא הֵיפֶךְ הַדַּעַת דְּעֵץ הַדַּעַת טוֹב וָרָע וְגַם לֹא עַל פִּי הַדַּעַת דִּקְדוּשָּׁה כִּי אִם בְּהַתּוֹקֶף שֶׁמִּצַּד עֶצֶם הַנְּשָׁמָה וְאָז יוּכְשַׁר אַחַר כַּךְ לִבְחִינַת הַדַּעַת דִּקְדוּשָּׁה, כִּי עַל יְדֵי שֶׁיִּתְנַהֵג בִּבְחִינַת הַתּוֹקֶף שֶׁמִּצַּד עֶצֶם הַנְּשָׁמָה יִפְעוֹל עַל יְדֵי זֶה חֲלִישׁוּת הֶרְגֵּשׁ הַטִּבְעִי דְּנֶפֶשׁ הַבַּהֲמִית וְגַם שֶׁיִּהְיֶה הִתְגַּלּוּת הַכֹּחוֹת דְּנֶפֶשׁ הָאֱלֹקִית, וּמִמֵּילָא יָכוֹל לִהְיוֹת אַחַר כַּךְ בְּחִינַת הַדַּעַת וְהַהֶרְגֵּשׁ אֱלֹקִי בְּנַפְשׁוֹ לַעֲשׂוֹת מַעֲשָׂיו בְּהַשְׂכֵּל עַל פִּי הַדַּעַת וְהַחֶשְׁבּוֹנוֹת שֶׁנִּתְבָּאֲרוּ לְעֵיל.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]מִי שֶׁאוֹר נַפְשׁוֹ הָאֱלֹקִית בִּלְתִּי מֵאִיר לוֹ בְּגִילּוּי, צָרִיךְ בְּקַבָּלַת עוֹל לְעוֹרֵר אֶת תּוֹקֶף הַהִתְקַשְּׁרוּת שֶׁבְּנַפְשׁוֹ לַה' וְלַתּוֹרָה וּמִצְוֹת בְּלִי חֶשְׁבּוֹנוֹת פְּרָטִים. שֵׂכֶל מַחֲלִישׁ אֶת הָרָצוֹן, מַעֲלַת הָעִקְשׁוּת בְּרָצוֹן בִּלְתִּי מוּגְבָּל, עִקְשׁוּת דִּקְדוּשָּׁה בִּקְבִיעוּת עִתִּים לַתּוֹרָה וּבְהִידּוּר מִצְוָה מַחֲלִישׁ אֶת הַהֶרְגֵּשׁ הַטִּבְעִי.[kitzur_end]
