מַאֲמָר כג.
וְהִנֵּה לִהְיוֹת שֶׁהַחֶסֶד הַנִּמְשָׁךְ לְמַטָּה בָּעוֹלָם הַזֶּה צָרִיךְ לִהְיוֹת עַל יְדֵי עֶשֶׂר סְפִירוֹת דַּעֲשִׂיָּה וְדַוְקָא עַל יְדֵי הַכֵּלִים וְהַלְּבוּשִׁים כו', בִּכְדֵי שֶׁתִּצְטַיֵּיר הַהַשְׁפָּעָה בְּצִיּוּר גַּשְׁמִי, יוּבַן עִנְיַן וּבֵרַכְךָ ה' אֱלֹקֶיךָ בְּכֹל אֲשֶׁר תַּעֲשֶׂה (דְּבָרִים טו. יח)
דְּהִנֵּה בִּפְרִי עֵץ חַיִּים שַׁעַר הַתְּפִלָּה בְּכַוָּונַת פַּרְשַׁת הַתָּמִיד עַל פָּסוּק צַו אֶת בְּנֵי יִשְׂרָאֵל בְּגִימַטְרִיָּא אֵל אד' וְהוּא בַּעֲשִׂיָּה וְלָכֵן אָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה אֵין צַו אֶלָּא עֲבוֹדָה זָרָה לְפִי שֶׁיּוֹנְקִים מִשֵּׁם אֵל אד', וְלִכְאוֹרָה לְפִי דְבָרָיו אֵינוֹ מוּבָן כְּלָל מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה אֵין צַו אֶלָּא כו' מֵאַחַר דְּשֵׁמוֹת אֵל אד' הֵן בְּגִימַטְרִיָּא צַו וְהֵם שֵׁמוֹת עֶלְיוֹנִים, וְאַף שֶׁמֵּהֶם יוֹנְקִים הַקְּלִיפּוֹת דַּעֲשִׂיָּה מִכָּל מָקוֹם הֵם עַצְמָם הֵם שְׁמוֹת הַקָּדוֹשׁ בָּרוּךְ הוּא וְאִם כֵּן מַהוּ עִנְיַן אֵין צַו אֶלָּא כו'.
[cup]אַךְ[/cup] הָעִנְיָן הוּא דְּהִנֵּה עִנְיַן עֲבוֹדָה זָרָה הוּא שֶׁמִּשְׁתַּחֲוֶה לַשֶּׁמֶשׁ וְלַיָּרֵחַ, וְהַיְינוּ מִפְּנֵי שֶׁמֵּהֶם תֵּצֵא הַשְׁפָּעַת הַמָּזוֹן וּכְמוֹ שֶׁכָּתוּב (דְּבָרִים לג. יד) וּמִמֶּגֶד תְּבוּאֹת שָׁמֶשׁ וּמִמֶּגֶד גֶּרֶשׁ יְרָחִים, וּמִפְּנֵי שֶׁמְּחַשֵּׁב אוֹתָם לִמְקוֹרֵי הַהַשְׁפָּעָה לָכֵן הֲרֵי הוּא מִשְׁתַּחֲוֶה וְעוֹבֵד לָהֶם, וּבֶאֱמֶת זוֹהִי טָעוּת גְּדוֹלָה, דְּעִם הֱיוֹת שֶׁנִּמְשָׁךְ הַהַשְׁפָּעָה עַל יָדָם הֲרֵי בֶּאֱמֶת אֵינָם אֶלָּא שְׁלוּחֵי הַשְׁפָּעָה לְבָד וְאֵין בְּיָדָם לְהָרַע אוֹ לְהֵטִיב מֵעַצְמָם, שֶׁהֵם רַק כַּגַּרְזֶן בְּיַד הַחוֹצֵב בּוֹ שֶׁהַגַּרְזֶן בְּעַצְמוֹ אֵין בִּיכָלְתּוֹ לַעֲשׂוֹת שׁוּם דָּבָר כִּי אִם הָאוּמָּן בּוֹנֶה עַל יְדֵי הַגַּרְזֶן, וּכְמוֹ כֵן כָּל הַכּוֹכָבִים וְהַמַּזָּלוֹת הֵם רַק כַּגַּרְזֶן שֶׁעַל יָדָם נִמְשֶׁכֶת הַהַשְׁפָּעָה, אָמְנָם הַשּׁוֹפֵעַ וְהַנּוֹתֵן אֶת הַהַשְׁפָּעָה הוּא הָאֱלֹקוּת אֶלָּא שֶׁמַּשְׁפִּיעַ עַל יְדֵי הַמַּזָּלוֹת, אֲבָל הֵם מֵעַצְמָם אֵין לָהֶם שׁוּם כֹּחַ לַעֲשׂוֹת דָּבָר כְּלָל, וּמִמֵּילָא אֵין לְחַשְּׁבָם כְּלָל וּכְלָל וּכְמוֹ שֶׁכָּתוּב (יְשַׁעְיָה י. טו) הֲיִתְפָּאֵר הַגַּרְזֶן עַל הַחוֹצֵב בּוֹ,
וְעַל דֶּרֶךְ מָשָׁל אָדָם הָרוֹאֶה בִּנְיָן גָּדוֹל וּמְפוֹאָר הַאִם יַעֲלֶה עַל הַדַּעַת שֶׁיְּשַׁבַּח אֶת הַגַּרְזֶן שֶׁבָּנָה אֶת הַבִּנְיָן שֶׁיֹּאמַר כַּמָּה מְשׁוּבָּח הַגַּרְזֶן שֶׁבָּנָה בִּנְיָן גָּדוֹל וּמְפוֹאָר כָּזֶה אוֹ יֹאמַר כַּמָּה מְפוֹאָר הַבִּנְיָן שֶׁבָּנָה הַגַּרְזֶן, אֶלָּא יְשַׁבַּח וִיפָאֵר אֶת הָאוּמָּן שֶׁבָּנָה אֶת הַבִּנְיָן וְהַבִּנְיָן נִקְרָא עַל שֵׁם הָאוּמָּן, הֲגַם שֶׁהָאוּמָּן בָּנָה אוֹתוֹ עַל יְדֵי הַגַּרְזֶן מִכָּל מָקוֹם אֵין הַגַּרְזֶן נֶחְשָׁב לִמְאוּמָה אֶלָּא הָאוּמָּן שֶׁבּוֹנֶה עַל יְדֵי הַגַּרְזֶן, וּכְמוֹ כֵן הוּא בְּכָל הַשְׁפָּעוֹת שֶׁפַע חַיּוּת וּמָזוֹן בָּעוֹלָם הֲרֵי מִי הוּא הַשּׁוֹפֵעַ הוּא הָאֱלֹקוּת אֶלָּא שֶׁנִּמְשָׁךְ עַל יְדֵי הַמַּזָּלוֹת (מֵהַטַּעַם שֶׁיִּתְבָּאֵר לְקַמָּן), אֲבָל אֱלֹקוּת הוּא הַשּׁוֹפֵעַ עַל יָדָם וְהוּא הָעִיקָּרִי וְהַמַּזָּלוֹת אֵין נֶחְשָׁבִים לִכְלוּם אֲפִילוּ לֹא כִּטְפֵלִים גַּם כֵּן, וּכְמוֹ שֶׁהַגַּרְזֶן אֵינוֹ נֶחְשָׁב אֲפִילוּ כְּשׁוּתָּף טָפֵל אֶל הָאוּמָּן כְּמוֹ כֵן הַמַּזָּלוֹת אֵינָן אֲפִילוּ כִּטְפֵלִים גַּם כֵּן שֶׁאֵינָן נֶחְשָׁבִים לִמְאוּמָה מַמָּשׁ, מֵאַחַר שֶׁהֵן רַק כִּשְׁלוּחִים לְבָד וְאֵין בָּהֶם כֹּחַ מֵעַצְמָן כְּלָל וּכְלָל וְאֵין בָּהֶם שׁוּם מַמָּשׁ, וְלִפְעָמִים מַרְאִים זֶה לְמַעְלָה אֵיךְ שֶׁאֵין שׁוּם כֹּחַ בַּמַּזָּלוֹת לַעֲשׂוֹת דָּבָר מֵעַצְמָם, וּכְמוֹ בְּמִצְרַיִם שֶׁנִּדּוֹנוּ בְּהָעֶשֶׂר מַכּוֹת וְלֹא הוֹעִילָה לָהֶם יִרְאָתָם שֶׁעָבְדוּ לְמַזָּל טָלֶה לְהַצִּילָם וְאַדְרַבָּא וּבְכָל אֱלֹקֵי מִצְרַיִם אֶעֱשֶׂה שְׁפָטִים כְּתִיב (שְׁמוֹת יב. יב) וּבִזְמַן הַחוּרְבָּן הֶחֱלִיף הַמַּזָּלוֹת וְהַשָּׂרִים הָעֶלְיוֹנִים, וְהַיְינוּ לְפִי שֶׁהֵן רַק שְׁלוּחֵי הַשֶּׁפַע לְבָד וְהָאֱלֹקוּת הוּא הַשּׁוֹפֵעַ עַל יָדָם.

[kitzur_begin]אֵל אד' בְּגִימַטְרִיָּא צַו שֵׁם דַּעֲשִׂיָּה, הַהִשְׁתַּחֲוָאָה לַשֶּׁמֶשׁ וְיָרֵחַ וְהַמַּזָּלוֹת שֶׁעַל יָדָם רַק בָּאָה הַשְׁפָּעָה גַשְׁמִית אֲבָל הַמַּזָּלוֹת עַצְמָם אֵינָם יְכוֹלִים לֹא לְהַשְׁפִּיעַ וְלֹא לִמְנוֹעַ הַהַשְׁפָּעָה, וְהֵם רַק כַּגַּרְזֶן בְּיַד הַחוֹצֵב בּוֹ וְהַשּׁוֹפֵעַ הוּא אֱלֹקוּת וּבִרְצוֹנוֹ מְשַׁדֵּד הַמַּעֲרָכוֹת שֶׁל הַמַּזָלוֹת.[kitzur_end]
