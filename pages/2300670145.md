שנבדל מכל חיצוניות זולתו שהוא למעלה הרבה מג' בריתות דתו"מ שהן רק בבחי' נר"נ דאדם כנ"ל. שיש בזה הפסק וביטול כמו אם בחוקותי כו'. ואם לאו תופר הברי' הזה להתהפך מאהבה לשנאה להמשיך להם קללה כו'. וכן בסיבת הריחוק זה מזה משך זמן רב בגלות שנשכח הברית שבגלות נאמ' (ישעי' נ, א) ובפשעיכם שולחה אמכם דאתתא אתתרכ' מבעלה וקודב"ה בגלותא אסתלק לעילא לעילא כו'. ע"כ ומצאהו רעו' רבות וצרו' כו' כי אין אלקי בקרבו כו' (דברים לא, יז) וכה"ג אבל ברית דמילה שהוא בבחי' העצמיות שלמעלה מבחי' אדם דתו"מ שנכרתו עליה י"ג בריתות שהוא בבחי' פנימיות דע"י שלמעלה גם מן הח"ס דא"א כו'. וע"ז א' ואף גם זאת בהיותם בארץ אויביהם לא מאסתים כו' לכלותם להפר בריתי אתם (ויקרא כו, מד) והוא מ"ש ז"ל גדולה מילה שנכרתו עליה י"ג בריתות שהן י"ג מדה"ר שנמשכים מלמעלה מן החכ' והוא מבחי' רדכ"ר שלמעלה ממשפט התורה למחול ולסלוח גם לעוברי רצונו והזדונות נעשו לו כזכיות ביו"כ כידוע לפי שזהו בבחי' כריתות ברית העצמיות הנ"ל שבלתי מורכב כלל בדבר מה גם לא בתו"מ הגם שהן פנימיות רצה"ע כו' וד"ל: 
(ח) [cup]וזהו[/cup] ומל ה' את לבבך ואת לבב זרעך לאהבה את ה' אלקיך בכל לבבך כו' למען חייך פי' למען חייך ממש שהוא חיים עצמיים שיחי' מחיי החיים דבחי' עצמות אא"ס ממש שהוא בחי' התקשרות הפנימית ועצמיות דבחי' יחידה הנ"ל שמקושרת בפנימית דעצמות אא"ס שבפנימית ג"ר דע"י שהן י"ג מדה"ר העצמיים שלמעלה מי"ג מדה"ר שנק' י"ג ת"ד כו' דכתיב (במדבר ו, כה) יאר ה' פניו אליך שמאירים גם בזמן הגלות כמו בזמן בהמ"ק ממש בשוה לפי שהן שלא ע"י התלבשות בח"ס שגבורות דע"י מלובש בח"ס וחסד דע"י בגלגלתא ואו"א מלבישים לזרועות דא"א כו'. אבל י"ג בריתות הללו הן הבאים ונמשכים מפנימית ג"ר דע"י בפנימית יסוד דע"י וזהו במילה עליונה מלמעלה למטה כמ"ש ומל ה' את לבבך כו' למען חייך העצמיים בבחי' אהבה העצמית ממש שלא יתכן בזה ענין המשכה ותשוקה כמים כלל אלא כמו אהבה עצמה ממש שהיא למעל' מברית דתו"מ שמורכב בדבר מה ברצון וחכ' ומדות כו' וד"ל. אך הנה א"א לבא בחי' מילה עליונה זו מלמעלה למטה שהוא למען חייך ממש כנ"ל כ"א בהקדים תחלה המילה מלמט' למעלה ע"י תו"מ ותשובה דוקא שזהו להבדיל הפסולת מן הפנימיות לגמרי שהוא ע"י בחי' החכמ' דוקא שהחכ' הוא המברר ומבדיל בין טו"ר אמת ושקר. כמשל התלמידי' שיושבים בפני רבם ולומדים דבר הלכה שיש בזה תערובות טו"ר אמת ושקר בלי נודע מהו האמת ומהו השקר וטעות אך ע"י העיון האמיתי שבחכ' מתברר האמ' ונבדל השקר והטעו' כמצרף לכסף שמבדיל הפסול' והסיגים לגמרי ונשאר הנקי בתכלית כך מצרף לחכמה ממש ללבן ולברר להוציא תכלית האמת ולהפריד ולהבדיל כל סיג ופסולת כו'. וכך הוא בחכ' שבתורה דבחכ' אתברירו רפ"ח ניצוצין שנפלו בק"נ כידוע
