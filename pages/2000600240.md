ענין הקרבנות הוא שהיו מקריבין בהמה גשמי' ע"ג המזבח והי' נכללת באש שלמעלה, וכך עכשיו שתפלות כנגד תמידין תקנו, הוא ג"כ כדי להפוך את הנה"ב היורדת למטה וליכלל באש של קדושה שיהי' ואהבת בכל לבבך בשני יצריך וזהו בחי' העלאה ממטה למעלה, ועי"ז נמשך המשכה מלמעלה למטה וכמ"ש ריח ניחוח להוי' שע"י הריח של הקרבנות ממטה למעלה עי"ז נמשך ההמשכה בבחי' שם הוי' מלמעלה משם הוי' וכמ"כ בתפלות יש ג"כ בחי' שוב וכמארז"ל כל הקורא ק"ש בלא תפילין כאלו הקריב עולה בלא מנחה וזבח בלא נסכים, שהמנחה ונסכים הוא בחי' שוב (וע' בלק"ת פ' שלח בדרוש בפ' נסכים כתיב) וכך בק"ש ותפילין הוא ג"כ בחי' רצוא ושוב כו', וזהו שע"י הקרבנות עי"ז נמשך תוס' אור בג"ע כו', וזהו מ"ש באתי לגני אחותי כלה אריתי מורי עם בשמי הוא קטורת, אכלתי יערי כו' שתיתי ייני עם חלבי שהוא הכל ענין הקרבנות כו', ונק' בהמ"ק באתי לגני, והיינו לפי שע"י ביהמ"ק וקרבנות עי"ז ממשיכים המשכת גילוי תוס' אורות בג"ע, וע"כ נק' באתי לגני.
ה) [cup]אך[/cup] הנה ענין פרה אדומה הוא שיהי' בחוץ דוקא, והענין הוא דהנה כתיב ונהר יוצא מעדן להשקות את הגן ומשם יפרד והי' לד' ראשים, היינו שאף שד' ראשים הוא בחי' ד' חיות המרכבה מ"מ מבחי' ד' ראשים אלו יכול להשתלשל שיהי' ד' גליות ד' מלכיות, וז"ע פרה אדומה שרומז לד' מלכיות, וכמ"ש במדרש ובילקוט פרה זו מצרים כמ"ש עגלה יפיפי' מצרים, אדומה זו בבל וכמ"ש אנת רישא דדהבא כו', והיינו שמהתענוג עליון נפל בשבירה למטה ונפל בד' מלכיות ונתהוה מזה בחי' תענוגים גשמי' וכמשארז"ל על ר"ע כשראה כרך גדול של רומי א' אם לעוברי רצונו כך לעושי רצונו עאכו"כ, והיינו לפי שכשהתענוג שנפל בשבירה שהוא בחי' פסולת ואחוריי' של התענוג העליון הוא כך, כ"ש לעושי רצונו שהוא בחי' תענוג העצמי שלא ע"י שבירה, וזהו שד' מלכיות נק' ד' מראות נגעים, לפי שנגע הוא היפך הענג דהיינו העונג שנפל בשבירה. וז"ע פרה שהיא בחוץ דוקא שהוא בחי' תשובה, ולהפוך את היצה"ר שנפל בשבירה בבחי' תענוגי' גשמי' וכמ"ש מי יתנך כאח לי יונק שדי אמי אמצאך בחוץ, מי יתנך כאח לי קאי על צדיקים שנקראי' אחים וריעי' להקב"ה, אמצאך בחוץ הוא בחי' בע"ת, ונק' בחוץ לפי שנפלו בשבירה, וכך פרה אדומה היא בחוץ דוקא, דהנה פרה איתא בזהר דקבילת משמאלא ומאן
