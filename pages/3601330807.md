צו גיטראגענקייט) ה"ז פועל ביטול גדול בהנה"ב כו', וכידוע המשל מאמת המים כשסותמין אותה ה"ה שוברת את הסתימה (לבד מה שלוקחת הסתימה אתה), וכמו"כ כאשר הנה"א הוא במיצר והסתר הנה"ב, הנה כאשר יוצא מהמיצר הזה ה"ז שובר את החומריות והעביות דנה"ב שמתבטל במהות חומריותו כו', ובאמת צ"ל שגם הנה"ב יתפעל בתשובה (שזהו עד"מ שהמים לוקחים אתם את הסתימה כו'), והיינו לא רק שהנה"א צר לו מעניני הנה"ב ומההעלמות כו', כ"א שהנה"ב יהי' לו צר מאד מעניניו כו', והיא אמיתית התשו' כשהיא בשניהם בנה"א ובנה"ב כו', ובנה"ב דוקא שייך ענין זה שצר לו כ"כ עד שאינו מוצא נוחם לנפשו כלל מפני שהוא החייב בזה, ועוד מפני שהוא מן הקלי' כו', דהנה בנה"א באמת אינו שייך ענין זה שלא ימצא נוחם לנפשו, הא' שהרי הוא אינו החייב בזה, והעיקר הוא מפני שהוא אלקות בעצם מהותו, ובאלקות אינו שייך שפלות כזו (אזא ארא(גיפאלנקייט בא זיך כו'), והאמת הוא שהמיצר והלא טוב מהענינים הלא טובים הוא בנה"א יותר הרבה מבנה"ב, אך זהו מפני שזה היפך רצה"ע לגמרי, וזהו מורגש
בהנה"א ביותר כו' דגם בהנה"ב צ"ל נרגש איך שהוא היפך רצה"ע כו', אבל בנה"א להיותו אלקות ה"ז נרגש ביותר כו', ועוד זאת מצד טומאת נפשו כו', שהרי הנה"ב הוא מהקלי' (ועם היות שמצ"ע הוא בק"נ וע"י הענינים הלא טובים אפי' כשהן תאוות היתר ה"ה יורד ברע גמור ומכ"ש באיזה חטא ועון ר"ל, מ"מ אין זה דומה כמו בהנה"א כו'), והנה"א הרי הוא אלקות והרי טומאת נפשו נוגע מאד וכמ"ש בסש"ב דאין לך עלבון גדול מזה כו', שזהו בירידת נפשו, וכמו"כ מה שפעל הירידה למעלה בבחי' גלות השכינה כו', וע"כ המיצר דנה"א הוא גדול מאד כו', אבל מ"מ אינו שייך בו שפלות זו (דיא ארא(גיפאלינקייט) שלא ימצא נוחם לנפשו כלל (שזהו כמו רקבון דעצבות כו'), רק בהנה"ב שייך שפלות זו מפני שהוא החייב בזה, ומפני שהוא מהקלי' ע"כ המיצר שלו הוא שלא ימצא מקום לעצמו כלל כו', ובזה נשבר בעצם מהותו עד שנשתנה ממהותו כו' (וע"ד מיני' ובי' לישדי' בי' נרגא כו'), וכנודע בבע"ת אמיתים שמתהפכים מן הקצה אל הקצה ומשתנים ממהות למהות אחר ממש כו', אבל כאשר התשובה היא מצד הנה"א
בלבד (והיא באמת אינו אמיתית גם מצד הנה"א והיא במדמה לבד כו' כנ"ל, וזה בלבד מספיק דמשו"ז אינו פועל מאומה בתיקון דרכיו ומעשיו שמצד הנה"ב כו', וגם מפני שאינו בבחי' חשבון בנפשו ולא בבחי' הרגש פנימי כו' כנ"ל) אין זה שייך להנה"ב כלל כו'. וכמו"כ הוא ג"כ בעבודה בתפלה שיכול להיות התעוררות גדולה בתפלה מצד הנה"א (וזה ג"כ רק בבחי' מקיף כו') ואינו בבחי' שייכות כלל להנה"ב כו', ולאחר התפלה אינו נשאר אפי' רושם ממנה, וע"כ כאשר בא לאיזה דבר תאוה אינו יכול לכבוש רוח תאותו וגם אינו חושב כלל לאכפי' עצמו כו', דכאשר הוא בעניני הגוף אין להנשמה חלק בזה שתפעול על הגוף והטבעית כו', דכשם שבההתעוררות לא יש חלק להנה"ב, כמו"כ כשהוא בעניני העולם אין בזה חלק להנה"א כו', והיינו שאין להנשמה והגוף שייכות זה לזה וכל א' מהם עומד על טבעו העצמי הנשמה היא בבחי' אין ביותר והגוף הוא בבחי' היש ביותר כו' (ושניהם אינם כפי הכוונה האמיתית כלל כו'). והב' הוא להיפך מן הקצה שגופו בחי' ב"ן בגימ' בהמה הוא בבחי' ביטול היש בקל מאד כמו מי
שלבו רך כקנה לשוב בתשובה תמיד בהכנעה אמיתית וברוח נשברה מאד, וגם בעבודה שבלב בתפלה מיד יתפעל לבו הטבעי גם הוא להבטל לפני ה' יותר
