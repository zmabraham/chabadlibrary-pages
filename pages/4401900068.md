# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר טו. > פֶּרֶק ב

וְהִנֵּה טַעַם וְסִבַּת הַדָּבָר מַה שֶּׁהַקָּדוֹשׁ בָּרוּךְ הוּא שִׁינָּה אוֹפֶן יְצִירַת גּוּף הָאָדָם מֵאוֹפֶן יְצִירַת גּוּפוֹת כָּל הַבַּעֲלֵי חַיִּים, דְּכָל הַבַּעֲלֵי חַיִּים בְּקוֹמָתָן נִבְרְאוּ וּבְצִבְיוֹנָן נִבְרְאוּ וְהָאָדָם נִבְרָא עָפָר מִן הָאֲדָמָה, דִּתְחִלָּה נִבְרָא גוּפוֹ וְאַחַר כֵּן נִיתְּנָה בוֹ נִשְׁמָתוֹ, אֵינוֹ לִגְרִיעוּתָא אֶלָּא אַדְרַבָּא הִיא לְמַעֲלִיּוּתָא, וְגַם זֶה מַה שֶּׁהָאָדָם נִבְרָא מֵעָפָר מִן הָאֲדָמָה מוֹרֶה עַל גּוֹדֶל הַפְלָאַת מַעֲלַת נִשְׁמַת אָדָם.
[cup]וְהָעִנְיָן[/cup] הוּא דְּהִנֵּה בְּהַד' חֲלוּקּוֹת שֶׁבָּהֶם נֶחְלְקוּ כָּל הַנִּבְרָאִים וְהֵם דּוֹמֵם צוֹמֵחַ חַי וּמְדַבֵּר, הִנֵּה הַחֵלֶק הַיּוֹתֵר גָּרוּעַ וְשָׁפָל הוּא חֵלֶק הַדּוֹמֵם וְהַחֵלֶק הַיּוֹתֵר נַעֲלֶה וּמְשׁוּבָּח הוּא חֵלֶק הַמְדַבֵּר, וְלִהְיוֹת כִּי הָאָדָם שֶׁהוּא הַמְעוּלֶּה וּבְחִיר הַנִּבְרָאִים הִנֵּה אוֹפֶן יְצִירַת גּוּפוֹ הוּא בְּשִׁינּוּי מֵאוֹפֶן יְצִירַת גּוּפוֹת שְׁאָרֵי הַבַּעֲלֵי חַיִּים, וְגַם גּוּף וְעֶצֶם הַחוֹמֶר אֲשֶׁר מִמֶּנּוּ בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת גּוּף הָאָדָם עָפָר מִן הָאֲדָמָה הַגַּס בְּיוֹתֵר מוֹרֶה עַל נִשְׁמַת הָאָדָם שֶׁהִיא בְּתַכְלִית הָרוֹמְמוּת, כְּהַכְּלָל הַיָּדוּעַ דְּכָל הַגָּבוֹהַּ גָּבוֹהַּ בְּיוֹתֵר יוֹרֵד לְמַטָּה מַּטָּה יוֹתֵר, וּכְמוֹ בְּגַשְׁמִיּוּת אָנוּ רוֹאִים בְּכוֹתֶל אֲבָנִים דִּכְשֶׁהָאֲבָנִים שֶׁבְּתַחְתִּית הַכּוֹתֶל נוֹפְלִים הִנֵּה הֵם בְּקֵירוּב אֶל הַכּוֹתֶל וּכְשֶׁנּוֹפְלִים הָאֲבָנִים שֶׁבְּרֹאשׁ הַכּוֹתֶל הֵם בְּרִיחוּק מִן הַכּוֹתֶל, וְכָל אֶבֶן שֶׁהִיא גָבוֹהַּ גָּבוֹהַּ יוֹתֵר הִנֵּה כְּשֶׁנּוֹפֶלֶת הִיא בְּרִיחוּק מִן הַכּוֹתֶל בְּיוֹתֵר,
וְכֵן בְּרוּחָנִיּוּת בְּעִנְיַן הַנְּשָׁמוֹת וְהַנְּפָשׁוֹת בְּנִשְׁמַת כָּל חַי, דְּלִהְיוֹת דְּנִשְׁמַת הָאָדָם הִיא גָבוֹהַּ גָּבוֹהַּ בְּיוֹתֵר לָכֵן הִנֵּה נָתַן לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא גּוּף מֵחוֹמֶר עָב וְגַס בְּיוֹתֵר, לְפִי שֶׁבְּכֹחוֹ לְבָרֵר וּלְזַכֵּךְ אֶת הַחוֹמֶר הַיּוֹתֵר עָב וְהַגַּס בְּיוֹתֵר,
וְלָכֵן אָנוּ רוֹאִים בְּמוּחָשׁ בְּטִבְעֵי בְּנֵי אָדָם, דְּיֵשׁ כָּאֵלּוּ שֶׁהֵם בַּעֲלֵי מִדּוֹת רָעוֹת רַחֲמָנָא לִצְּלָן יוֹתֵר הַרְבֵּה מֵחַיּוֹת רָעוֹת הַדּוֹרְסוֹת וְטוֹרְפוֹת וַחֲכָמִים הֵמָּה לְהָרַע לְהָעִיק לְזוּלָתָם, מַאכֶען עֶנְג אוּן שְׁלֶעכְט דֶעם צְוֵוייטֶען, וְגוֹזְלִים וְרוֹצְחִים בִּכְדֵי לְמַלֹּאות תַּאֲוַת נַפְשָׁם, דְּעַל הָרוֹב הוּא מִצַּד הַנְּטִיּוֹת הָרָעוֹת שֶׁל הַגּוּף, וּנְטִיּוֹת הָרָעוֹת שֶׁל הַגּוּף פּוֹעֲלוֹת לְרָעָה עַל נַפְשָׁם, הַיְינוּ הַגְבָּרַת הַגּוּף עַל הַנֶּפֶשׁ שֶׁיַּשְׂכִּיל וְיִתְחַכֵּם אֵיךְ לְהָפִיק מְזִמּוֹתָיו בִּגְזֵילָה וּרְצִיחָה, אֲבָל הַנֶּפֶשׁ מִצַּד עַצְמָהּ הִיא בְּתַכְלִית הַמַּעֲלָה, וּבִיכוֹלֶת הָאָדָם לַעֲלוֹת בְּעִילּוּי אַחַר עִילּוּי בְּמַעֲשֵׂה הַטּוֹב וּלְהַגִּיעַ לְדַרְגוֹת רוּחָנִיּוֹת הַנַּעֲלוֹת בְּיוֹתֵר, וּכְהַמַּעֲשֶׂה הַמּוּבָא בְּרֵאשִׁית חָכְמָה (שַׁעַר הָאַהֲבָה פֶּרֶק ד) בְּאֶחָד שֶׁבָּא מֵעוֹמֶק הַשִּׁפְלוּת הָרוּחָנִית לְמַעֲלַת הָרוּם הָרוּחָנִי.
[cup]וּבָזֶה[/cup] יוּבַן מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (חֲגִיגָה טז. א) שִׁשָּׁה דְבָרִים נֶאֶמְרוּ בִּבְנֵי אָדָם שְׁלֹשָׁה כְּמַלְאֲכֵי הַשָּׁרֵת שְׁלֹשָׁה כַּבְּהֵמָה, וְהַיְינוּ דְּהָאָדָם הִנֵּה מִצַּד טֶבַע גּוּפוֹ וּנְטִיּוֹתָיו הוּא כִּבְהֵמָה, וְהָאָדָם שֶׁעוֹשֶׂה מַעֲשֵׂה בְהֵמָה הֲרֵי הוּא גָרוּעַ מִן הַבְּהֵמָה, דִּבְהֵמָה אֵין לָהּ דַּעַת וְאָדָם יֵשׁ לוֹ דַעַת, וּמִצַּד טֶבַע נַפְשׁוֹ וּנְטִיּוֹתָיו הוּא כְּמַלְאֲכֵי הַשָּׁרֵת, שֶׁיָּכוֹל לְהַגִּיעַ לְהַדַּרְגּוֹת הָרוּחָנִיּוֹת הַנַּעֲלוֹת בְּיוֹתֵר, וְהָאָדָם כְּשֶׁעוֹלֶה בְּמַעֲלַת הַדַּעַת הוּא גָבוֹהַּ בְּמַעֲלָה מִמַּלְאֲכֵי הַשָּׁרֵת, דְּמַלְאֲכֵי הַשָּׁרֵת הִנֵּה עֲמִידָתָם בְּעוֹלָם רוּחָנִי, וּכְמַאֲמָר כְּלוּם יֵצֶר הָרָע יֵשׁ בָּכֶם, אֲבָל הָאָדָם שֶׁהוּא בָּעוֹלָם הַזֶּה הַגַּשְׁמִי וּמַשְׂכִּיל בְּהָעִנְיָנִים הַיּוֹתֵר נַעֲלִים בְּהַשָּׂגוֹת אֱלֹקוּת וְהַשָּׂגָתוֹ זֹאת פּוֹעֶלֶת בּוֹ רָצוֹא וּתְשׁוּקָה לְהִדָּבֵק בֵּאלֹקוּת הֲרֵי הוּא לְמַעְלָה יוֹתֵר מִמַּלְאֲכֵי הַשָּׁרֵת, כִּי בַּעֲבוֹדָתוֹ זֹאת הֲרֵי הוּא מַעֲלֶה אֶת הַגַּשְׁמִיּוּת דְּעוֹלָם הַזֶּה,
וּמֵהַאי טַעְמָא הִנֵּה נִשְׁתַּנָּה אוֹפֶן יְצִירַת גּוּף הָאָדָם מִיצִירַת שְׁאָר הַבַּעֲלֵי חַיִּים וְגוּפוֹ הוּא מֵהַחוֹמֶר הָעָב וְגַס בְּיוֹתֵר, לִהְיוֹת דִּבְכֹחַ נִשְׁמָתוֹ לְהָאִיר גַּם אֶת הַחֹשֶׁךְ הַגַּס וּלְאַהֲפָכָא חֲשׁוֹכָא שֶׁל הַגּוּף הַגַּס לִנְהוֹרָא, לִהְיוֹת כִּי זֶהוּ תַּכְלִית הַכַּוָּונָה בִּבְרִיאַת הָאָדָם וִירִידַת נִשְׁמָתוֹ לָעוֹלָם הַזֶּה.

[kitzur_begin]כָּל הַגָּבוֹהַּ בְּיוֹתֵר יוֹרֵד לְמַטָּה בְּיוֹתֵר, הָאֶבֶן שֶׁהִיא בְּתַחְתִּית הַכּוֹתֶל כְּשֶׁנּוֹפֶלֶת הִיא סְמוּכָה לַכּוֹתֶל וַאֲשֶׁר בְּרֹאשׁ הַכּוֹתֶל בִּנְפִילָתָהּ הִיא בְּרִיחוּק מִן הַכּוֹתֶל. הַנְּטִיּוֹת הַגּוּפְנִיּוֹת שֶׁבָּאָדָם גְּרוּעִים מֵהַבַּעֲלֵי חַיִּים, וְהַנְּטִיּוֹת הַנַּפְשִׁיּוֹת נַעֲלוֹת מִבְּמַלְאֲכֵי הַשָּׁרֵת, וּבִיכָלְתּוֹ לְאַהֲפָכָא חֲשׁוֹכָא לִנְהוֹרָא.[kitzur_end]
