שאינה בבחי' מקור לחכ' כו', מ"מ לפעמים נמשך מבחי' הפנימי' ושנעשה עי"ז שלימות יתרה בבחי' החכ' כו'. והענין הוא דהנה בחי' פנימי' ח"ס זהו בחי' אוירא שע"ג קרומא דחפי' על מוחא וכמ"ש בע"ח שער א"א פ"ג, שזהו בחי' מוח סתום מאד והוא בחי' מוחא דעתיקא שלמע' מבחי' מוחא דאריך (וזהו בחי' חיצוני' ופנימי' דמו"ס כו'), דדעת דעתיק אחזי נהורא בהאי אוירא כו', וידוע דדעת הוא בחי' התקשרות ובחי' דעת דעתיק שבאוירא הוא שמקשר ומחבר בחי' ח"ס (היינו חיצוני' ח"ס) לבא לידי גילוי כו'. וכמו עד"מ כשאדם משכיל ומחדש איזה השכלה חדשה זהו המשכה שנמשך ממקור השכל הנק' משכיל, ולפעמים אינו נופל לו שכל מלמע' מהשכל (והיינו שאין הגילוי מוכרח להיות כו', ובד"כ כל הגילוים שבנפש אינם שמוכרחים הגילוים כו' וכמ"ש במ"א, וידוע דבכדי שיהי' גילוי השכל הוא ע"י הרצון כו', אך לפעמים גם הרצון אינו מועיל דהגם שרוצה להתחכם מ"מ אינו מתחכם כו', או שלא יש התגלות הרצון שגם גילוי הרצון אינו מוכרח, וא"א להכריח את הרצון כו' וכמ"ש במ"א), הרי בהכרח לומר שיש
דבר שלמע' מבחי' מקור השכל המשכיל שהוא המקשרו ומחברו לבא לידי גילוי בשכל כו' (וי"ל שזהו בדרך הכרח שמכריח את כח המשכיל לבא לידי שכל כו', שז"ע הדעת שהוא ההתקשרות בדרך הכרח כו', דכאשר יש לו רצון הרי ההתקשרות הוא מצד הרצון כו' (וג"ז הוא ע"י הדעת), אך עיקר ענין ההתקשרות שמצד הדעת הוא שמכריח א"ע להתקשר כו' וכמ"ש במ"א, והיינו דמה שהדעת עושה התחברות כח המשכיל כו' א"ז בבחי' מקור שהוא בהדרגה, כ"א הוא כמו כח עליון שפועל שלא בהדרגה כו'). והנה כח המשכיל זהו הנק' ח"ס והבחי' שלמע' מזה זהו בחי' אוירא שע"ג קרומא כו', שהוא המקשר ומחבר בחי' ח"ס לבא לידי גילוי כו', והו"ע עסק התו' בדרך קב"ע שגם שאין לו רצון הוא מכריח א"ע, וכאשר מכריח עצמו ה"ה נותן עצמו אל הלימוד כו', דאין הכוונה שההכרח הוא רק חיצוני וכאלו כפאו כו', כ"א שמכריח עצמו ונותן א"ע בכל עצמותו אל הלימוד כו' (שז"ע הדעת שמקשר את העצמות ומשו"ז שייך בזה הכרח כו' וכמ"ש במ"א), ועי"ז ג"כ באים גילוים מכח המשכיל, אך שרש הגילוים הם מלמע' מכח המשכיל והיינו
דשרשם הוא מבחי' אוירא הנ"ל וממילא הגילוים המה נעלים ביותר כו', וה"ז ע"ד ב' הכחות הנ"ל שבחכ' כמו"כ הוא בח"ס דהתגלות מבחי' חיצוני' ח"ס בא בכח בחי' הפנימי' וממילא יש בזה מבחי' הפנימי' שזהו שלימות יתירה בגילוי החכ' כו' (ובחכ' הגלוי' נת"ל פקצ"ח דכשלא יש כח ההפלאה גם כח הירידה הוא חסר כו', וכאן בח"ס הוא שמוסיף שלימות כו'). אך העיקר מה שאפשר לבא גם להפנימי' דח"ס א"ז רק מה שהפנימי' מוסיף אור בהחיצוני' וההתגלות היא רק חיצוני' ח"ס כו' (ובאמת גם בהנ"ל הוא התגלות מבחי' הפנימי' כו'), כ"א הכוונה שיהי' התגלות בחי' פנימי' ח"ס ממש כו', וזהו בכח או"ח דוקא.
