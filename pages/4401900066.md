# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר יד. > פֶּרֶק ב

וְהַטַּעֲנָה הַשֵּׁנִית מַה שֶּׁפְּלוֹנִי הֱסִיתוֹ כו' אֵינָהּ טַעֲנָה אֲמִיתִּית שֶׁהֲרֵי בֶּאֱמֶת צָרִיךְ לְהִתְרַחֵק מֵאִישׁ רָע וְרָשָׁע וְלֹא יִתְחַבֵּר אֵלָיו, כִּי הַרְבֵּה חֲבֵרוּת עוֹשָׂה, וַחֲבֵרִים טוֹבִים מְבִיאִים טוֹבָה הַרְבֵּה אֶל הָאָדָם, כִּי מִלְּבַד שֶׁהַלִּימּוּד טוֹב עִם חָבֵר, כְּמוֹ שֶׁכָּתוּב (קֹהֶלֶת ד. ט) טוֹבִים הַשְּׁנַיִם מִן הָאֶחָד וּכְמוֹ שֶׁדָּרְשׁוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (בְּרָכוֹת סג. ב) עַל פָּסוּק חֶרֶב אֶל הַבַּדִּים כו' (יִרְמְיָה נ. לו) הִנֵּה לְבַד זֹאת מַה שֶּׁיְּדַבְּרוּ יִרְאֵי ה' אִישׁ אֶל רֵעֵהוּ זֶה פּוֹעֵל הַרְבֵּה מְאֹד, כִּי אֵין אָדָם רוֹאֶה נִגְעֵי עַצְמוֹ מִצַּד אַהֲבַת עַצְמוֹ שֶׁעַל כָּל פְּשָׁעִים תְּכַסֶּה אַהֲבָה מַה שֶּׁאוֹהֵב אֶת עַצְמוֹ, וְאֵינוֹ רוֹאֶה הַפְּשָׁעִים שֶׁלּוֹ, וַחֲבֵרוֹ רוֹאֶה וְיוֹדֵעַ יוֹתֵר מִמֶּנּוּ וּמוֹכִיחוֹ עַל זֶה וְגַם מְדַבֵּר עַל לִבּוֹ אֵיךְ שֶׁלֹּא טוֹב לַעֲשׂוֹת כַּאֲשֶׁר עָשָׂה וְזֶה כָּל הָאָדָם שֶׁיֵּלֵךְ בְּדֶרֶךְ הַטּוֹב וְהַיָּשָׁר, וְכָל זֶה פּוֹעֵל עַל הָאָדָם הַרְבֵּה טוֹבָה.
[cup]וּלְעוּמַּת[/cup] זֶה חָבֵר רָע פּוֹעֵל הַרְבֵּה לְרָעָה, שֶׁמְּסִיתוֹ וּמַטֵּהוּ מִדֶּרֶךְ הַטּוֹב וְהַיָּשָׁר וְגוֹרֵם שֶׁיַּעֲשֶׂה דְבָרִים אֲשֶׁר מֵעַצְמוֹ בֶּאֱמֶת לֹא הָיָה עוֹשֶׂה דְבָרִים כָּאֵלּוּ כַּאֲשֶׁר אָנוּ רוֹאִים זֶה בְּחוּשׁ בְּהַרְבֵּה בְּנֵי אָדָם שֶׁהָיוּ טוֹבִים מִצַּד עַצְמָם וְעַל יְדֵי הִתְחַבְּרוּתָם לַאֲנָשִׁים רָעִים נַעֲשִׂים רָעִים וְחַטָּאִים רַחֲמָנָא לִצְּלָן, וְעַל זֶה נִצְטַוִּינוּ בְּמִשְׁנָה דְאָבוֹת (פֶּרֶק א) קְנֵה לְךָ חָבֵר, הַיְינוּ חָבֵר טוֹב, שֶׁהַרְבֵּה טוֹבָה תְּקַבֵּל מֵאִתּוֹ וּלְהֵיפֶךְ הַרְחֵק מִשָּׁכֵן רָע, וְאַל תִּתְחַבֵּר לָרָשָׁע, וּכְמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (סוּכָּה נו. ב) אוֹי לָרָשָׁע וְאוֹי לִשְׁכֵנוֹ, וּכְמוֹ שֶׁפֵּרֵשׁ רַשִׁ"י עַל פָּסוּק וַיִּקַּח קֹרַח (בַּמִּדְבָּר לח. א), וְכָל שֶׁכֵּן כְּשֶׁיִּהְיֶה חָבֵר אֵלָיו שֶׁגּוֹרֵם לוֹ רָעָה הַרְבֵּה, וְאִם כֵּן הֲרֵי הוּא מִתְחַיֵּיב בְּנַפְשׁוֹ עַל זֶה שֶׁהִתְחַבֵּר לָרָשָׁע וּבַמֶּה הוּא מַצְדִּיק אֶת עַצְמוֹ,
רַק בְּהַחֵטְא הָרִאשׁוֹן שֶׁעָשָׂה מַה שֶּׁהִתְחַבֵּר אֵלָיו שֶׁאֵין זֶה הַצְדָּקָה כְּלָל, כִּי אִם מַה שֶּׁמַּטְעֶה אֶת עַצְמוֹ מִצַּד אַהֲבַת עַצְמוֹ הֲרֵי הוּא מְחַפֵּשׂ וּמוֹצֵא לְהַצְדִּיק אֶת עַצְמוֹ בְּדִבְרֵי שֶׁקֶר, וּבַדָּבָר שֶׁבָּזֶה עַצְמוֹ הוּא אָשֵׁם אֲשֶׁר חָטָא בָּזֶה. וּפְעוּלַּת הַהַצְדָּקָה הַלֵּזּוּ שֶׁמַּצְדִּיק אֶת עַצְמוֹ הוּא שֶׁאֵין תְּשׁוּבָתוֹ כִּדְבָעֵי לְמֶהֱוֵי, הֵן בִּתְשׁוּבַת הַמִּשְׁקָל הַיְינוּ לְצַעֵר וּלְסַגֵּף אֶת עַצְמוֹ כְּעֶרֶךְ הַתַּעֲנוּג שֶׁהָיָה לוֹ, מֵאַחַר שֶׁחוֹשֵׁב שֶׁלֹּא הָיָה אָשֵׁם בָּזֶה כָּל כַּךְ, אִם כֵּן אֵינוֹ צָרִיךְ לְצַעֵר אֶת עַצְמוֹ כָּל כַּךְ בִּתְשׁוּבָתוֹ, וּבִפְרָט בְּעֶצֶם הַתְּשׁוּבָה הֲרֵי זֶה נוֹגֵעַ הַרְבֵּה, שֶׁאֵין הַחֲרָטָה עַל הֶעָבָר וְהַקַּבָּלָה עַל לְהַבָּא אֲמִיתִּית כָּל כַּךְ, כִּי הַחֲרָטָה וְהַקַּבָּלָה תְּלוּיוֹת בִּמְרִירוּת נַפְשׁוֹ, דְּכַאֲשֶׁר מִתְמַרְמֵר בְּנַפְשׁוֹ מְאֹד עַל הַחֵטְא וְהֶעָוֹן, אָז הוּא מִתְחָרֵט בֶּאֱמֶת וְעוֹקֵר רְצוֹנוֹ לְגַמְרֵי מִזֶּה וּמִמֵּילָא הַקַּבָּלָה עַל לְהַבָּא אֲמִיתִּית, אֲבָל עַל יְדֵי שֶׁמַּצְדִּיק אֶת עַצְמוֹ הֲרֵי אֵין מְרִירוּת נַפְשׁוֹ גְּדוֹלָה כָּל כַּךְ, מֵאַחַר שֶׁמּוֹצֵא זְכוּת עַל עַצְמוֹ, וּמֵאַחַר שֶׁהַמְּרִירוּת אֵינָהּ גְּדוֹלָה כָּל כַּךְ מִמֵּילָא הַחֲרָטָה וְהַקַּבָּלָה אֵינָם אֲמִיתִּים,
וְאִם כֵּן הֲרֵי הַהַצְדָּקָה הַזֹּאת חוֹשֶׂכֶת מִמֶּנּוּ טוֹבָה הַרְבֵּה בְּעֶצֶם תְּשׁוּבָתוֹ וּבְשִׁינּוּי מַהוּתוֹ (שֶׁעַל יְדֵי הַתְּשׁוּבָה אֲמִיתִּית מִתְהַפֵּךְ מַהוּתוֹ מֵרָשָׁע לַצַּדִּיק, דְּבֶאֱמֶת מִשְׁתַּנֶּה מַהוּתוֹ וְנַעֲשֶׂה רָחוֹק מִכָּל הַדְּבָרִים הַהֲפָכִים מֵאֱלֹקוּת, וְכַאֲשֶׁר אֵין תְּשׁוּבָתוֹ אֲמִיתִּית מִמֵּילָא גַּם כֵּן אֵינוֹ מִשְׁתַּנֶּה וּמִתְרַחֵק מֵהָרָע כָּל כַּךְ),
וַהֲרֵי זֶה כְּמוֹ עִנְיַן הַשְּׁטוּת שֶׁקָּדַם לִדְבַר עֲבֵירָה, אֶלָּא שֶׁהַשְּׁטוּת הַהִיא הֵבִיאָה שֶׁיִּכָּשֵׁל בִּדְבַר הָעֲבֵירָה, וְהַשְּׁטוּת שֶׁמַּצְדִּיק אֶת עַצְמוֹ הוּא הַגּוֹרֵם שֶׁתְּשׁוּבָתוֹ לֹא תִהְיֶה נְכוֹנָה כָּל כַּךְ כִּדְבָעֵי לְמֶהֱוֵי, וּבִפְרָט אֶחָד הֲרֵי זֶה יוֹתֵר גָרוּעַ מִשְּׁטוּת הַקּוֹדֵם, כִּי הֲרֵי לְאַחַר שֶׁכְּבָר עָבַר עֲבֵירָה רַחֲמָנָא לִצְּלָן כְּשֶׁעוֹשֶׂה תְשׁוּבָה מֵעוּמְקָא דְלִיבָּא הֲרֵי אָמְרוּ (בְּרָכוֹת לד. ב) מָקוֹם שֶׁבַּעֲלֵי תְשׁוּבָה עוֹמְדִים צַדִּיקִים גְּמוּרִים אֵינָן עוֹמְדִין, וַהֲרֵי הַשְּׁטוּת הַזֶּה מוֹנֵעַ אוֹתוֹ מִזֶּה, אִם כֵּן הֲרֵי שְׁטוּת זֶה עוֹד גָּרוּעַ יוֹתֵר שֶׁחוֹשֵׂךְ וּמוֹנֵעַ אוֹתוֹ מִטּוֹבָה הַרְבֵּה, שֶׁעַל יְדֵי תְשׁוּבָתוֹ הָאֲמִיתִּית הָיָה בָא לְמַעֲלָה וּמַדְרֵיגָה גָּבוֹהַּ מְאֹד.

[kitzur_begin]חָבֵר טוֹב מְסַיֵּיעַ לִמְצוֹא מוּמֵי הָרוּחַ וּמוֹעִיל לַעֲקוֹר הַדֵּעוֹת וּמִדּוֹת רָעוֹת וּלְהַקְנוֹת הַטּוֹבוֹת, וְחָבֵר רָע גּוֹרֵם הַשְׁרָשַׁת הַיְדִיעוֹת וּמִדּוֹת רָעוֹת. הַהִצְטַדְּקוּת בַּהֲסָתַת זוּלָתוֹ סִיבָּה שֶׁאֵין תְּשׁוּבָה שְׁלֵימָה, וּשְׁטוּתוֹ גְרוּעָה מִשְּׁטוּת הַגּוֹרֵם עֲבֵירָה.[kitzur_end]
