מַאֲמָר יב.
עוֹד זֹאת הוּא מִפִּתּוּיֵי הַיֵּצֶר הָרָע שֶׁמְּפַתֵּהוּ שֶׁלֹּא יִרְאֶנּוּ אָדָם, כִּי הַבַּעַל עֲבֵירָה רַחֲמָנָא לִצְּלָן אֵינוֹ רוֹצֶה שֶׁיֵּדְעוּ בְּנֵי אָדָם רָעָתוֹ, וּכְמוֹ שֶׁאָמַר רַבָּן יוֹחָנָן בֶּן זַכַּאי לְתַלְמִידָיו (בְּרָכוֹת כח. ב) יְהִי רָצוֹן שֶׁתְּהֵא מוֹרָא שָׁמַיִם עֲלֵיכֶם כְּמוֹרָא בָּשָׂר וָדָם אָמְרוּ לוֹ תַּלְמִידָיו עַד כַּאן (בִּתְמִיָּה כְּלוֹמַר וְלֹא יוֹתֵר מִמּוֹרָא בָּשָׂר וָדָם, רַשִׁ"י) אָמַר וּלְוַאי (שֶׁיְּהֵא כְּמוֹרָא בָּשָׂר וָדָם שֶׁאִם כֵּן תֶּחְדְּלוּ מֵעֲבֵירוֹת הַרְבֵּה. רַשִׁ"י) תֵּדְעוּ כְּשֶׁאָדָם עוֹבֵר עֲבֵירָה (בְּסֵתֶר מִמּוֹרָא הַבְּרִיּוֹת וְיוֹדֵעַ שֶׁהַכֹּל הוּא גָלוּי לְהַקָּדוֹשׁ בָּרוּךְ הוּא וְאֵינוֹ מַנִּיחַ בְּכַךְ. רַשִׁ"י) אוֹמֵר שֶׁלֹּא יִרְאַנִי אָדָם,
וּבָזֶה מְפַתֵּהוּ יִצְרוֹ שֶׁלֹּא יִרְאֶנּוּ אָדָם וְלֹא יֵדְעוּ עִנְיָנָיו הָרָעִים וּבֶאֱמֶת אֵינוֹ כֵן שֶׁהַכֹּל רוֹאִים וְיוֹדְעִים וּמַכִּירִים, כִּי לֹא יִמָּלֵט שֶׁיַּעֲשֶׂה אֵיזֶה דְבָרִים שֶׁיִּהְיוּ חֲשׁוּדִים בְּעֵינֵי הַבְּרִיּוֹת, כִּי בֶּאֱמֶת מֵרוֹב הַתַּאֲווֹת שֶׁמִּצַּד הֲסָתַת הַיֵּצֶר עוֹשֶׂה כַּמָּה שְׁטוּתִים אֲשֶׁר הַשֵּׂכֶל הַבָּרִיא לֹא יַסְכִּים עַל זֶה כְּלָל, רַק שֶׁהַיֵּצֶר הָרָע מַעֲבִירוֹ עַל דַּעְתּוֹ וְעַל דַּעַת קוֹנוֹ וְעוֹשֶׂה כַּמָּה עִנְיְנֵי שְׁטוּתים, שֶׁבָּזֶה נַעֲשֶׂה חָשׁוּד בְּעֵינֵי בְּנֵי אָדָם וְנוֹתְנִים דַּעְתָּם עָלָיו לֵידַע עִנְיָנָיו, וּכְמוֹ שֶׁאָנוּ רוֹאִים בְּמוּחָשׁ בְּכַמָּה עִנְיָנִים שׁוֹנִים בְּדִבּוּר וּבִתְנוּעוֹת שׁוֹנוֹת שֶׁהָאָדָם בְּעַצְמוֹ מְגַלֶּה עִנְיָנָיו הַפְּנִימִיִּים כִּי בְרָע הוּא, וְאֵין הַכַּוָּונָה עַל עִנְיְנֵי אִיסּוּר בְּדִבּוּר כְּמוֹ שֶׁקֶר רְכִילוּת וְלָשׁוֹן הָרָע וְדִבְרֵי נִיבּוּל בִּלְבָד, אֶלָּא הַכַּוָּונָה דְּגַם כְּשֶׁמְּסַפֵּר אֵיזֶה סִפּוּר דְּלִכְאוֹרָה אֵינוֹ אֶלָּא כְּמוֹ שְׂפַת יֶתֶר וְדִבּוּר שֶׁל בַּטָּלָה, אֲבָל בֶּאֱמֶת הִנֵּה בְּדִבּוּר זֶה יֵשׁ בּוֹ גִּילּוּי פְּנִימִית הָרָע,
וְכֵן הוּא בְּעִנְיַן הָרְאִיָּה דְּלִפְעָמִים הִנֵּה בְּהַבָּטָה שֶׁל הִסְתַּכְּלוּת שֶׁמַּבִּיט בְּמָקוֹם שֶׁאֵין צְרִיכִים לְהַבִּיט הִנֵּה בָּזֶה הָאָדָם מְגַלֶּה אֶת כָּל עוֹמֶק הָרָע שֶׁבּוֹ,
וְגַם אִם מִבְּנֵי אָדָם מוּסְתָּר וְנֶעְלָם מַעֲשָׂיו הָרָעִים, הֲלֹא לְפָנָיו יִתְבָּרֵךְ גָּלוּי וְיָדוּעַ הַכֹּל, וּכְמוֹ שֶׁכָּתוּב (יִרְמְיָה כג. כד) אִם יִסָּתֵר אִישׁ בַּמִּסְתָּרִים וַאֲנִי לֹא אֶרְאֶנּוּ וּכְמַאֲמָר הַכֹּל גָּלוּי וְיָדוּעַ לְפָנֶיךָ, אֵין דָּבָר נֶעְלָם מִמֶּךָּ וְאֵין נִסְתָּר מִנֶּגֶד עֵינֶיךָ כו' וְהַקָּדוֹשׁ בָּרוּךְ הוּא מוֹדִיעַ וּמְגַלֶּה עִנְיָנָיו לַכֹּל, וּכְמַאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה בְּמִשְׁנָה (אָבוֹת פֶּרֶק ד) כָּל הַמְחַלֵּל שֵׁם שָׁמַיִם בְּסֵתֶר נִפְרָעִין מִמֶּנּוּ בְּגָלוּי, וּפֵרֵשׁ רַשִׁ"י הַמְחַלֵּל שֵׁם שָׁמַיִם בְּסֵתֶר שֶׁעוֹבֵר עֲבֵירָה בְּסֵתֶר (כִּי אִי אֶפְשָׁר לוֹמַר שֶׁהַכַּוָּונָה עַל חִילּוּל הַשֵּׁם מַמָּשׁ, דְּחִלּוּל הַשֵּׁם הוּא בְּפַרְהֶסְיָא דַוְקָא וְאֵיךְ שַׁיָּיךְ שֶׁמְּחַלֵּל שֵׁם שָׁמַיִם בְּסֵתֶר, אֶלָּא הַכַּוָּונָה שֶׁעוֹבֵר עֲבֵירָה בְּסֵתֶר) נִפְרָעִין מִמֶּנּוּ בְּגָלוּי, הַקָּדוֹשׁ בָּרוּךְ הוּא מְגַלֶּה לַכֹּל חֶרְפָּתוֹ שֶׁנֶּאֱמַר (מִשְׁלֵי כו. כו) תִּכַּסֶּה שִׂנְאָה בְּמַשָּׁאוֹן תִּגָּלֶה רָעָתוֹ בְקָהָל (הָעוֹשֶׂה מַעֲשָׂיו בְּמַחְשָׁךְ וּמְכַסֶּה בְּמַשָּׁאוֹן שֶׁהוּא שׁוֹאַת חֹשֶׁךְ אֶת הַדָּבָר הַשָּׂנאוּי לְהַקָּדוֹשׁ בָּרוּךְ הוּא סוֹף שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מְגַלֶּה רָעָתוֹ בְּקָהָל שֶׁיַּכִּירוּ בוֹ שֶׁהוּא רָשָׁע, פֵּרוּשׁ רַשִׁ"י) וְכָל כַּךְ לָמָּה שֶׁלֹּא יִתְחַלֵּל שֵׁם שָׁמַיִם עַל יָדוֹ, שֶׁלֹּא יֹאמְרוּ רָאִיתָ פְלוֹנִי שֶׁהוּא חָכָם וְעוֹשֶׂה מַעֲשִׂים טוֹבִים וּפָגְעוּ בוֹ צָרוֹת כָּל כַּךְ, עַד כַּאן לְשׁוֹנוֹ,
וּמַה שֶּׁמְּפַתֵּהוּ שֶׁלֹּא יִרְאֶנּוּ אָדָם הוּא רַק מִשְּׁטוּתֵי הַיֵּצֶר הָרָע, ה' יִשְׁמְרֵנוּ מֵהֶם, שֶׁאָנוּ רוֹאִים בְּמוּחָשׁ בְּטִבְעֵי בְּנֵי אָדָם דְּאֵלּוּ שֶׁהֵם בַּעֲלֵי מֶזֶג דַּק, אֵיידֶעלֶע מֶענְשֶׁען, וּבַעֲלֵי שֵׂכֶל אֵינָם עֲלוּלִים כָּל כַּךְ לְקַבֵּל עִנְיְנֵי שְׁטוּת כְּמוֹ בַּעֲלֵי מֶזֶג גַּס, גְרָאבֶּע מֶענְשֶׁען, וּבַעֲלֵי מִדּוֹת, שֶׁהֵם הַבַּעֲלֵי מִדּוֹת הִנֵּה בְּהִתְרַגְּשׁוּתָם וְגוֹדֶל הִתְפַּעֲלוּתָם עֲלוּלִים לְקַבֵּל עִנְיְנֵי הַשְּׁטוּת וְהַהַטְעָיָה שֶׁמַּטְעִים אֶת עַצְמָם, זֵיי נַארֶען זִיךְ אַלֵיין, וְהַיֵּצֶר הָרָע לִהְיוֹתוֹ בַּהֲמִי וּבַעַל מִדּוֹת הִנֵּה מְפַתֶּה אֶת הָאָדָם בְּכַמָּה עִנְיָנִים, וְרַחֲמָנָא לִצְּלָן הַמִּתְנַהֵג עַל פִּי הוֹרָאַת הַיֵּצֶר הָרָע מְקַבֵּל שְׁטוּתו וַהֲסָתַת הַיֵּצֶר הָרָע וְנִדְמֶה לוֹ שֶׁאֵין אִישׁ יוֹדֵעַ עִנְיָנָיו הַלֹּא טוֹבִים.

[kitzur_begin]הָאָדָם בְּטִבְעוֹ יֵשׁ לוֹ מוֹרָא שֶׁהַזּוּלַת לֹא יֵדַע עִנְיָנָיו וּמַה גַּם עִנְיָנָיו הָרָעִים, מוֹרָא שָׁמַיִם כְּמוֹרָא בָּשָׂר וָדָם, הָאָדָם בְּדִבּוּרוֹ וּבִתְנוּעוֹתָיו מְגַלֶּה עוֹמֶק הָרָע שֶׁבּוֹ, וְהַקָּדוֹשׁ בָּרוּךְ הוּא מְגַלֶּה חֶרְפָּתוֹ בָּרַבִּים, הַיֵּצֶר הָרָע בְּפַחֲזוּתוֹ מְקַבֵּל שְׁטוּתִים וּמְפַתֶּה אֶת הָאָדָם.[kitzur_end]
