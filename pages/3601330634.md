ועצמי' מו"ס בחי' מוחא דעתיקא ממש כו', עז"א לך לך שיגיע לשרשו ומקורו והיינו לשרש ומקור נשמתו בחי' מקיף דיחידה כו'. ועז"א מארצך פי' מארציות וחומרי' וכנ"ל דלהגיע לבחי' מקיף דיחידה בבחי' הזזה עצמי' זהו שבא מן המנגד בכח או"ח דוקא כו', וזהו לך לך מארצך שההילוך יהי' מהארציות והחומריות שעי"ז יגיע לבחי' שרש ומקור העצמי דנשמה כו'. וזהו ג"כ מה שא' במדרש משל לצלוחית של אפופלסימון כו' שיהי' ריחה נודף דהיינו ריח הקטרת שהו"ע ריח התשו' כו', ובאאע"ה הי' זה ענין הנסיונות שעי"ז בא לבחי' מקיף דיחידה כו', וזהו טלטל עצמך ממקום למקום להיות בחי' ההילוך אך שיהי' בכח או"ח דהיינו מארצך דוקא כו'. ועי"ז שמך מתגדל דענין הגדלת השם הוא המשכת בחי' שמו העצמי בחי' העלם העצמי דא"ס כו' דעד שלא נבה"ע הי' הוא ושמו לבד, ונת"ל פרט"ז שז"ע שם א' שלמע' מהז' שמות והוא שרש בחי' ש' מ"ה דתיקון כו', וי"ל ג"כ דהגדלת השם הוא שיהי' הגדלת השם מבחי' שמו העצמי כו'. וזהו ואגדלה שמך שזהו מעלה גדולה באברהם כו', וכ"ז הוא ע"י ההילוך בבחי' או"ח
כו'. וזהו ויאמר ה' אל אברם בחי' שכל הנעלם כו', לך לך לשרשך ומקורך העצמי והו"ע טלטל עצמך כו', ויהי' ההילוך מארצך בכח או"ח כו', ועי"ז שמך מתגדל דבחי' שמו העצמי יאיר בגילוי כו'.
[kitzur][cup]קיצור.[/cup] אמנם קטרת דיוה"כ בקה"ק הו"ע התשובה שממשיך מפנימיות הכתר, דהנה אי' באד"ר מארי דחוטמא מחד נוקבא חיין ומחד נוקבא חיין דחיין, חוטם הוא כלי הנשמה, וזהו שהוא שער המוחין, והיינו עצמות המוחין (ופה שער לחיצוניות המוחין) דשם קישור עצם הנשמה, ומשם נמשך ב' המשכות, חיין הוא כללות החיות, וזהו הנשמה כמו הדפיקא בהאו"פ, חיין דחיין עצמיות הנשמה. והדוגמא למעלה חיין מקיף דאו"י חיות כללי דעולמות, והגילוי או שמאיר המקיף או שמוסיף אור בפנימי, וחיין דחיין מקיף דאו"ח פנימיות עתיק ע"י עצמיות מו"ס בחי' אוירא, וזהו דבמקום שבע"ת עומדים צ"ג גם ברעו"ד א"י לעמוד, ועי"ז גם זדונות נעשים זכיות, וזהו שמן משחת קדש למע' מחיצוניות ח"ס, וז"ע משיח דמורח ודאין עצמיות ח"ס, וע"כ אתא לאתבא צדיקייא בתיובתא. וזהו ויאמר ה' אל אברם אב רם דבכדי להגיע לעצמות ח"ס לך לך מקיף דיחידה, ואגדלה שמך בחי' שם א' וי"ל מבחי' שמו העצמי.[/kitzur]
