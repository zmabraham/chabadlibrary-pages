מַאֲמָר יט.
וְהִנֵּה הַחוֹתָם דִּנְעִילָה דְיוֹם הַכִּפּוּרִים הִנֵּה מִלְּבַד זֹאת שֶׁעִנְיָנוֹ הוּא שֶׁהַשֶּׁפַע הַכְּלָלִי שֶׁנִּמְשָׁךְ בִּסְפִירַת הַמַּלְכוּת יַגִּיעַ לְמָקוֹם הָרָאוּי וּלְכָל אֶחָד בִּפְרָט, הִנֵּה עוֹד זֹאת עִנְיַן הַחוֹתָם דִּנְעִילָה הוּא הַשְּׁמִירָה כְלָלִית שֶׁלֹּא תִהְיֶה יְנִיקַת הַחִיצוֹנִים מֵהַחֶסֶד הַנִּמְשָׁךְ, וּכְמוֹ חוֹתָם שֶׁמַּנִּיחִים עַל אִגֶּרֶת הוּא בִּכְדֵי שֶׁלֹּא יִקְרָאֶנָּה זָר, וּכְמוֹ כֵן הוּא עִנְיַן הַחוֹתָם לְמַעְלָה שֶׁלֹּא יִנְקוּ הַחִיצוֹנִים מֵהַהַשְׁפָּעָה, וְהַיְינוּ עַל יְדֵי ה' גְבוּרוֹת דְּעַתִּיק שֶׁמְּקַבֶּלֶת אָז בְּחִינַת הַמַּלְכוּת, שֶׁבָּהֶם וְעַל יָדָם נִמְשָׁךְ הִתְחַלְּקוּת הַחֶסֶד עֶלְיוֹן דְּעַתִּיק לְכָל הַנִּבְרָאִים דְּעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה הַמְקַבְּלִים מִבְּחִינַת מַלְכוּת דַּאֲצִילוּת.
[cup]וְזֶהוּ[/cup] מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה דִּמְזוֹנוֹתָיו שֶׁל אָדָם קְצוּבִים לוֹ מֵרֹאשׁ הַשָּׁנָה עַד יוֹם הַכִּפּוּרִים, הַיְינוּ שֶׁעַל יְדֵי ה' גְבוּרוֹת נִמְשָׁךְ הַקִּצְבָה שֶׁהוּא כַּמָּה יוּמְשַׁךְ הַחֶסֶד עֶלְיוֹן לְכָל אָדָם בִּפְרָט, מַה שֶּׁאֵין כֵּן מִתְּחִלָּה הָיָה הַחֶסֶד אוֹר וְחַיּוּת כְּלָלִי לִכְלָלוּת הָעוֹלָמוֹת, וְעַל יְדֵי ה' גְבוּרוֹת הַחוֹתָם דִּנְעִילָה נַעֲשׂוּ שְׁנֵי עִנְיָנִים כְּאֶחָד, וְהוּא שֶׁלֹּא יִקְרַב זָר לְקַבֵּל גַּם שֶׁמֶץ מִשֶּׁפַע זוּ וַאֲשֶׁר יוּמְשַׁךְ לְמָקוֹם הָרָאוּי וְנוֹתֵן קִצְבָה שֶׁיִּתְחַלֵּק וְיוּמְשַׁךְ לְכָל נִבְרָא וּלְכָל אָדָם בְּדֶרֶךְ פְּרָט.
[cup]דְּהִנֵּה[/cup] כְּשֶׁמַּלְכוּת דַּאֲצִילוּת מְקַבֶּלֶת בִּנְעִילָה דְּיוֹם הַכִּפּוּרִים מֵה' גְבוּרוֹת דְּעַתִּיק אוֹר הַחֶסֶד עֶלְיוֹן שֶׁנִּקְצַב עַל יְדֵי הַה' גְבוּרוֹת לְכָל הָעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה וּלְכָל בְּרִיָּה בִּפְרָט, הִנֵּה גַּם אָז עֲדַיִין הַחֶסֶד הוּא כְּמוֹ שֶׁהוּא לְמַעְלָה בַּאֲצִילוּת שֶׁהוּא בִּבְחִינַת תַּכְלִית הָרוּחָנִיּוּת וְלֹא נִמְצָא וְלֹא נִצְטַיֵּיר בּוֹ עֲדַיִין מֵהַשְׁפָּעוֹת וְחַיּוּת גַּשְׁמִי דְּעִנְיְנֵי עוֹלָם הַזֶּה כְּלָל, כְּמוֹ שֶׁאֵין עֲרוֹךְ חַיּוּת עוֹלָם הַזֶּה לְעוֹלָם הָאֲצִילוּת,
וּכְיָדוּעַ דַּאֲצִילוּת מוּבְדָּל בְּעֶרֶךְ מֵעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה וְלָכֵן בְּהַד' יְסוֹדוֹת אֵשׁ-רוּחַ-מַיִם-עָפָר, שֶׁכְּלָלוּת הַד' עוֹלָמוֹת אֲצִילוּת-בְּרִיאָה-יְצִירָה-עֲשִׂיָּה נִכְלָלִים בְּד' יְסוֹדוֹת אֵלּוּ כְּמוֹ שֶׁכָּתוּב בְּעֵץ חַיִּים, הִנֵּה אֲצִילוּת הוּא בְּחִינַת יְסוֹד הָאֵשׁ וּכְמוֹ שֶׁכָּתוּב (דְּבָרִים ד. כד) כִּי ה' אֱלֹקֶיךָ אֵשׁ אֹכְלָה הוּא שֶׁנִּמְשָׁל לְאֵשׁ, וְהַיְינוּ כְּמוֹ שֶׁהָאֵשׁ אֵין לוֹ הִתְחַבְּרוּת עִם שְׁאָר הַיְסוֹדוֹת וְלָכֵן הוּא תָּמִיד בִּבְחִינַת עֲלִיָּה וְהִסְתַּלְּקוּת לְמַעְלָה, וּכְמוֹ כֵן עוֹלָם הָאֲצִילוּת אֵין לוֹ הִתְחַבְּרוּת עִם עוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה, דַּאֲצִילוּת הוּא אֱלֹקוּת מַמָּשׁ וְאֵין שָׁם בְּחִינַת יֵשׁ כְּלָל,
וְעוֹלָמוֹת בְּרִיאָה-יְצִירָה-עֲשִׂיָּה הֲרֵי מֵעוֹלָם הַבְּרִיאָה מַתְחִיל בְּחִינַת מְצִיאוּת הַיֵּשׁ, שֶׁעַל זֶה מוֹרֶה לְשׁוֹן בָּרָא כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ן [רַבִּי מֹשֶׁה בֶּן נַחְמָן] בְּפֵירוּשׁוֹ עַל הַתּוֹרָה בְּפָסוּק בְּרֵאשִׁית בָּרָא, וְלָכֵן אֲצִילוּת מוּבְדָּל מִבְּרִיאָה-יְצִירָה-עֲשִׂיָּה וְכָל שֶׁכֵּן מֵעוֹלָם עֲשִׂיָּה הַגַּשְׁמִי מַמָּשׁ, וְלָזֹאת בִּבְחִינַת הַשְׁפָּעַת הַחֶסֶד שֶׁבְּעוֹלָם הָאֲצִילוּת אֵין שָׁם שׁוּם צִיּוּר עֲדַיִין מִבְּחִינַת הַחֶסֶד הַגַּשְׁמִי דְּעוֹלָם הַזֶּה רַק שֶׁנִּמְשָׁךְ סְתָם הַשְׁפָּעַת הַחֶסֶד לִכְנֶסֶת יִשְׂרָאֵל, וְנֶחֱלָק עַל יְדֵי הה' גְבוּרוֹת לְכָל אֶחָד וְאֶחָד בִּפְנֵי עַצְמוֹ, אֲבָל הַכֹּל הוּא בִּבְחִינַת חֶסֶד רוּחָנִי עֲדַיִין.
[cup]וּבִכְדֵי[/cup] שֶׁיּוּמְשַׁךְ אַחַר כַּךְ מֵחֶסֶד זֶה חַיּוּת וְהַשְׁפָּעוֹת גַּשְׁמִיּוֹת בְּבָנֵי חַיֵּי וּמְזוֹנֵי, הוּא עַל יְדֵי הִתְלַבְּשׁוּת הַמַּלְכוּת דַּאֲצִילוּת בִּבְרִיאָה יְצִירָה וַעֲשִׂיָּה, וּכְמוֹ שֶׁכָּתוּב בְּאִגֶּרֶת הַקֹּדֶשׁ בְּסוֹפוֹ בִּדְרוּשׁ הַמַּתְחִיל לְהָבִין מַה שֶּׁכָּתוּב בִּפְרִי עֵץ חַיִּים שֶׁאֲפִילוּ בְּיוֹם טוֹב בְּחִינַת חֶסֶד דַּאֲצִילוּת מִתְלַבֵּשׁ לְגַמְרֵי בְּחֶסֶד דִּבְרִיאָה וּמְחַיֶּה עוֹלָם הַזֶּה הַגַּשְׁמִי עַל יְדֵי מַעֲבַר חֶסֶד דִּיצִירָה וַעֲשִׂיָּה הַנִּקְרָא גַם כֵּן הִתְלַבְּשׁוּת שֶׁאִם לֹא כֵן לֹא הָיָה פוֹעֵל בְּגַשְׁמִיּוּת עוֹלָם הַזֶּה עַד כַּאן דִּבּוּרוֹ.
[cup]וּבְיוֹמִין[/cup] דְּחוֹל הוּא עַל יְדֵי הִתְלַבְּשׁוּת גְּמוּרָה בְּחֶסֶד דִּיצִירָה וַעֲשִׂיָּה, וְהַיְינוּ גַם כֵּן עַל יְדֵי יִחוּד זוּ"נ [זְעֵיר אַנְפִּין וְנוּקְבָא] דִּיצִירָה וְאֵין מַסְפִּיק בְּחִינַת הַהִתְלַבְּשׁוּת בְּחֶסֶד דִּיצִירָה לְבָד, כִּי זָ"א דִּיצִירָה גַּם כֵּן עֲדַיִין אֵינוֹ בְגֶדֶר שֶׁתִּהְיֶה מִזֶּה הַשְׁפָּעַת חֶסֶד דַּעֲשִׂיָּה, וְלֹא הָיָה יָכוֹל לְהִתְהַוּוֹת מִזֶּה חַיּוּת הָעֲשִׂיָּה, כִּי אִם כְּשֶׁהוֹלֵךְ הַשֶּׁפַע עַל יְדֵי מַלְכוּת דִּיצִירָה וְנִתְצַמְצֵם עֶצֶם הָאוֹר דְּזָ"א דִּיצִירָה,
וּבַעֲשִׂיָּה צָרִיךְ לִהְיוֹת גַּם כֵּן יִחוּד זוּ"נ דַּעֲשִׂיָּה, כִּי עִיקַּר הִתְהַוּוּת הַיֵּשׁ וְדָבָר נִפְרָד לְגַמְרֵי הוּא מִבְּחִינַת מַלְכוּת דַּוְקָא לְפִי שֶׁאֵין מֶלֶךְ בְּלֹא עָם כו', וְאָז נִמְשָׁךְ מִזֶּה הַהַשְׁפָּעוֹת בַּעֲשִׂיָּה הַגַּשְׁמִית לְמַטָּה דְּבַר יוֹם בְּיוֹמוֹ עַל יְדֵי הַמִּדָּה שֶׁבַּיּוֹם הַהוּא מִו' קְצָווֹת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה, דְּכָל הַהִתְחַלְּקוּת בָּא מֵה' גְבוּרוֹת דְּעַתִּיק.
[kitzur_begin]קִיצּוּר.[kitzur_end]
[kitzur_begin]הַחוֹתָם הוּא שֶׁיַּגִּיעַ הַשֶּׁפַע לְמָקוֹם הָרָאוּי וּלְכָל אֶחָד בִּפְרָט. הַשֶּׁפַע הַכְּלָלִי הַמִּתְקַבֵּל בְּמַלְכוּת דַּאֲצִילוּת בִּנְעִילָה הוּא חֶסֶד רוּחָנִי דַּאֲצִילוּת. אֲצִילוּת מוּבְדָּל מִבְּרִיאָה-יְצִירָה-עֲשִׂיָּה כְּאֵשׁ הַנִּבְדָּל מִשְּׁאָר הַיְסוֹדוֹת, וְרַק בְּהִשְׁתַּלְשְׁלוּת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה הִנֵּה עַל יְדֵי ה' גְבוּרוֹת דְּעַתִּיק נַעֲשֶׂה חֶסֶד שֶׁבְּעֶרֶךְ הַנִּבְרָאִים.[kitzur_end]
