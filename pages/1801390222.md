רבו מכה אותו ומלמדו כיון שהגדיל בדבור הוא מייסרו, כך מתחילה והכית בצור, בפ' בשלח י"ז ו', וכאן ודברתם אל הסלע, וכיון שהכה משה את הסלע טפטפה הסלע דם שנאמר ויזובו וכתיב ואשה כי תזוב כו' עכ"ל (משא"כ בענין והכית בצור לא טפטפה דם כי היתה עדין נערה קטנה שאין לה דם נדות עדיין), וכ"ה בילקוט פ' חוקת סוף רמז תשס"ג (אפ"ל צור בחי' יעקב כמ"ש יבקע צורים יבקע אותיות יעקב, וסלע בחי' ישראל), הלימוד ע"ד מכל מלמדי פי' כי כל בשמים ובארץ הוא המלמד לבחי' דוד בחי' מ' שם ב"ן בגימט' בהמה, ע"ד מרדע שמורה דיעה בפרה במד"ר נשא דרנ"ח ונק' מלמד, ולימוד זה מבחי' בשכמל"ו ששם השגת המלאכים הנק' בהמה בד"ה בכ"ה בכסליו ובד"ה רני ושמחי השני ובד"ה ואלה המשפטים בענין ונתתי עשב בשדך לבהמתך, וזהו נק' כשהנער קטן שהשגתו רק ביחו"ת שלמטה היש ולמעלה אין, ולכן לפעמים רבו מכה אותו, וכמארז"ל במד"ר אמור פכ"ט ע"פ אני ה' מלמדך להועיל מסקיד לך ע"ד אעא דלא סליק ביה נהורא כו' מבטשין לי' זח"ג פ' שלח קס"ח הובא בסש"ב פכ"ט, ובבה"ז פ' קרח דקע"ח ב' על המאמר שם ע"פ ועבד הלוי הוא אלמלא דינא כו', נמצא הדין והביטוש עושה לבחי' אעא דהיינו כי האדם עץ השדה שזהו פי' אעא שיהי' סליק בי' נהורא דנשמתא, ובסש"ב שם פי' ענין ומהות הביטוש להיות לב נשבר, נבזה בעיניו נמאס, וע' בלקו"ת בדרושי עשי"ת בדרוש להבין פי' יג"מ בענין לב נשבר ונדכה אלקים לא תבזה שיש דכא ויש נדכה, וע' מזה בזח"ב משפטים דק"ח ע"א שזהו הנק' זבחי אלקים, ובחי' ביטוש זה נמשל להכאה, ומצינו שהדיבור נק' הכאה כמ"ש ויך את המצרי ופירש"י ע"פ הלהרגני אתה אומר שהרגו בשם המפורש, וע' מד"ר ר"פ תצוה שלכן נק' ישראל זית כמ"ש זית רענן יפה פרי תואר כמו שהזית חובטין אותו וטוחנין אותו וטוענין עליו אבנים אז נותן שמנו כו' כך ישראל ויאנחו בנ"י מן העבודה, וכתיב בצר לך ומצאוך כו' כי אל רחום ה' אלקיך בפ' ואתחנן סי' ד' למד ל"א, אכן משהגדיל ע"ד כי מי גוי גדול אשר לו אלקים קרובים אליו, וע' מד"ר פ' מקץ פ"צ ע"פ רק הכסא אגדל ממך שכמו"כ נא' והיית רק למעלה יכול כמוני ת"ל רק גדולתי למעלה מגדולתכם הרי לא מיעוט רק שגדולתי למעלה מגדולתכם, ואפ"ל ע"ד האדם הגדול בענקים או ע"ד המאור הגדול, והגדלה זו בנש"י ע"ד והי' אור הלבנה כאור החמה, או כמו שאומרים גדול הדור, לכן בבחי' זו אין שייך הכאה ואעפ"כ שייך בדברים הוא מייסרו, כי צריך לעשותו כלי ריקן כמ"ש בת"א פ' משפטים בד"ה לא תהי' משכלה ועקרה ושיהי' בבחי' אין ממש, וע' בלק"ת פ' אחרי בד"ה כי ביום הזה יכפר דרוש השני פ"ג גבי אלא היינו ביטול היש וגסות הרוח שבאדם כו', וע' בלק"ת פ' חקת בד"ה אז ישיר ישראל כו' עלי באר ושם פ"ג בפי' במחוקק שצ"ל ב' חקיקות לעשותה כלי שיתן מימיו בחי' מ"נ, והיינו ע"י התורה כמ"ש אם בחוקותי תלכו שתהיו עמלים בתורה, וזהו ענין בדברים אתה מייסרו כו'.
ג) קיצור מש"ש סעי' ג'. מ"ש בזח"א פ' בראשית כח"ב ובגינייהו חב בסלע דמחא בי' דלא אמר ליה אלא ודברתם אל הסלע, ובסה"מ סי' קי"ג דבור לשון זווג כמו ראוה מדברת כו', והענין כי כנס"י עם הקב"ה
