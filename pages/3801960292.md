שניאור זלמן, אדמו"ר הזקן[ftnref_5_5]
או"ח: יח , ד (ששמעתי ממנו בעצמו שיש כמה דברים שחוזר בו ממש"ש [בשו"ע] מפני שנסמך אז יותר מדאי על המ"א). כח, א (שאומרים בשם הגאון אאזמו"ר). לה, ז (הי' קורא בעצמו בתורה). סא (שמעתי ב' רבינו הגדול). סב (זכורני שכן עשה מעשה). פב (ששמעת בשם רבנו ז"ל … אלו הייתי שומע כן מכ"ק רבינו ז"ל ודאי הייתי מחמיר, אכן אני לא שמעתי כן בפי', ואין ראי' ג"כ מזה להיפך, ואעפ"כ מי שיוכל להחמיר לעצמו … הואיל וי"א כן משמו ז"ל). פד (שמעתי קבלה אמיתית מאדמו"ר). קיא (בפירוש שמיע לי מני דמרן כאאזמו"ר).
יו"ד: ח, א (והכשירו אאזמו"ר הגאון). כה, א-ב (אמר לנו בשם הרב אדמו"ר … שהי' שאלה … וציוה לשרוף). נו [נ, א] (פעם א' פסק אאזמו"ר). סג, ז (תשו' כת"י מאאזמו"ר הגאון נ"ע זי"ע). סג, ח (כשם שבין קדשי קדשים לחולי חולין כך גבהו דרכיו מדרכנו). עד (ואמר לו בזה"ל). קב (בסוף ימיו חזר בו … כשם שבין קדשי קדשים לחולי חולין גבהו דרכיהם מדרכינו). קד, ד (אין לנו להתחכם נגד דעתו הק'). קיד, ד (הגהה כתי"ק עה"ג מעיל צדקה). קלח, א (רבינו זקני מ"כ). קנד, ד (נשים מומחות דלאזני' אמרו שרבינו הגאון ז"ל החמיר). קסג, ו (זכורני כד הוינא טליא התיר אאזמו"ר הגאון נ"ע). קעב [קג, ג] (מאד הי' מקפיד ומזהיר שיעשו מקוה חמה). קעה (זכורני שבלאזני ראיתי ממה שהובא לו חצי עגול ככדור מקליפת רמון … רמונים מאה"ק). קעו ח"ג, ב (התפאר מאד במה שתיקן מקואות שיהיו יכולים לחממן … א' מן ג' דברים שהתקין הוא ז"ל בחייו). קפ, ה (הי' שו"ב עפ"י ציווי אאזמו"ר הגאון). רא [כג, ב] (שמעתי מדודי הרב הג' מהרי"ל זצ"ל ששמע מאחיו רבינו הג' נ"ע). ריח [לח, ד] (שמעתי מדו"ז הרב מוהרי"ל ז"ל ששמע מאחיו אאזמו"ר ז"ל). רכ (בל' השואל: שמעתי מהה"מ דיאנוויץ ששמע מאחיו אדמו"ר שיישב הט"ז [בהיתר חדש] … ולפ"ז התיר … כ"ז שמעתי מהרב הנ"ל). רכג (כמדומה לי שא' ששמע זאת מפי אדמו"ר ז"ל). רכט (זכרתי המעשה שהי' מפורסם בלאדי בשנים הראשונים של דירתינו … ובא עם השאלה לפני כאאזמו"ר הגאון נ"ע ונשא ונתן טובא בדין זה … וכמדומה שכ' ע"ז שו"ת באריכות). רצט (כשם שבין קדשי קדשים … גבהו דרכיהם מדרכינו). דש (העיד … ששמע מכ' אאזמו"ר נ"ע). שיד, יח (שצוה כאזמו"ר הגאון נ"ע להדאקטער … שיבאר לו). שכ (אח"כ העלה שיעור הריס והתורמוס פחות מעט ממ"ש בשולחנו). שלד, ה (כנודע מדרכו בקודש לדקדק בשיעורי' ומדות, וכמו בענין הרביעית). שלו, א (ידעתי מיעוט ערכי לעמוד על כוונת כ"ק נ"ע אשר מאד גבהו דרכיו מדרכינו). שמט (מ"ש בשם כאאמו"ז נ"ע). הוספות: כא, א (עיני ראו בלאדי שאדמו"ר נבג"מ אסר).
אהע"ז ח"א: ל [לז, ג] (מהרי"ל: שמעתי מפי קדשו). מט, ה (ודודי הרב הגדול … אמר לי בשם אאזמו"ר הגאון נ"ע). סג, ב (שמעתי מדודי הרב הגדול החסיד … שדעת רבינו ז"ל). פז (הגהות על תשובת אאזמו"ר הגאון ז"ל). קמג (השיב לענין שם כלה כחמותה כו').
אהע"ז ח"ב: קנב (תשו' כאאזמו"ר הגאון נ"ע כת"י). קפח, ג (כתי"ק הגהה לשו"ת מוהר"א ששון). קפט, ה (שהוגד לי משני אנשים לומדים מורי הוראות, שנשאל מלפני רבינו הגדול כאאזמו"ר הגאון נ"ע). ריד (הגהתו לשו"ת מהרא"ש). ריט (שמעתי מר' דוד מרודני' מ"ץ דאולע שהי' מלמד בלאדי בקיץ תק"ע ועל ידו נשאל ממנו נ"ע שאילת גט ההוא וצוה לכתוב … כמדומה ששמעתי מדודי הרב ר"ל זצ"ל שאח"כ חזר בו). רכב (נמצא כ' בשם אאזמו"ר נ"ע). דעת הרב שהיה רוצה להקל בזה). שו, א (הגהתו לשו"ת מהרא"ש). שז, ח (הגהת מהרי"ל בגליון תויק"ו בשמו). שכז (והטעם ע"פ הסוד שמעתי ממו"ר א"ז נ"ע). שצב (כמדומה לי ששמעתי כבר מאיזה למדן … שאלו ג"כ מאזמו"ר נבג"מ הגאון ז"ל איך יעשו הם, כמדומה לי שאמר לי האיש הנ"ל שהקיל בזה, אך איני זוכר זה על בירור). תי (הגהתו לשו"ת מהרא"ש). תיב (העדות שהעיד כת"ר בש' אדמו"ר נ"ע). תיג (ומ"ש כת"ר שנשמע לו בשם כאאזמו"ר הגאון נ"ע … ודאי אילו הי' בירור שאמר כן בלי ספק הייתי מבטל דעתי בזה, כי מי יבוא אחר המלך כו' וכגבוה שמים מעל הארץ גבהו דרכיו מדרכינו כו', אך ידוע לנו שיוכלו לומר בשמו מה שאינו כו', ולזאת אין לסמוך על השמועות). תטו, ג (הגהתו לב"ש). תטו, ה (הגהתו לשו"ת מהרא"ש). תכ (לפי מה ששמעתי שאאזמו"ר הגאון צוה … אך כמדומה לי ששמעתי מדו"ז הה"ג ר' ליב ז"ל שחזר בו רבינו ז"ל בסוף ימיו וצוה לכתוב). תכג (כמדומה לי ששמעתי שרבינו הגדול ז"ל נ"ע שצוה לההמ"ץ מו"ה אליעזר בהיות דירתו במחנינו [לאזניא] … ושאלתיו הטעם ואמר … ואחר כמה שנים בהיות רבינו הגדול ז"ל בק"ק ליאדי … ואמר … נהירנא שרבינו ז"ל הי' מסופק … וצוה … זוכרני בעת שהיתי משמש את אדמו"ר הגאון נ"ע ואירע).
חו"מ: ע, ט (אדמו"ר נ"ע דרשותיו הי' כותב אחיו הרי"ל ז"ל והוא הי' משלם לו).
חידושים: ז, ד (ההג"ה שם שהיא כת"י אחיו הרי"ל ז"ל … אך שאמר שכן פסק רבינו הגדול מר אחיו אאזמו"ר הגאון ז"ל). נז, ב (וכ"ד אאזמו"ר ז"ל הובא בתשו' דודי מהרי"ל ז"ל). סז, ד (וזו ההג"ה היא כת"י אחיו דו"ז הה"ג בעל שארית יהודא שאמר ששמע כן מפיו הלכה למעשה). קעב, א (הדברים שכתבתי כן דברי אאזמו"ר הגאון ז"ל בדרוש).
פס"ד: יט, ד (והרב כ' … וצריך להבין זה היטב דעתו דעת תורה). נ, א (עכ"ל כאאזמו"ר הג' זצ"ל שמצאתי אצלי). נ, ג (וכ' אאזמו"ר הגאון ז"ל). קכג, ד (ומשם אאזמו"ר הגאן ז"ל שמעתי מדודי הרב רי"ל זצ"ל ומהשו"ב הישיש דרודני'). קלב, א (ע"ז כתב … דודי השא"י בשם אאזמו"ר הגאון ז"ל). קנד, ג (ונשאל ע"ז כאאמו"ר זקני הגאון ז"ל מן א' מתלמידיו … והשיב לו). קצט, ד (וכתב אאזמו"ר ז"ל בחי' ליו"ד סי' ק"ט [נדפס בשו"ת סי' יח]). ריג, ב (דו"ז ההגרי"ל ז"ל א' לי משם רבינו הגאון ז"ל נ"ע זי"ע). ריג, ג (סיפר לי א' משום רבינו ז"ל ששאלוהו … והשיב). ריג, ד (שמעתיה מרבינו ז"ל). רטז, ג (ובשם אאזמו"ר הגיד לי הרר"פ). רצ, ד (לפי מה ששמעתי שאאזמו"ר הגאון צוה … אך כמדומה לי ששמעתי מדו"ז הה"ג ר' ליב ז"ל שחזר בו רבינו ז"ל בסוף ימיו וצוה לכתוב). ג, ד (ומ"ש כת"ר שנשמע לו בשם כאאזמו"ר הגאון נ"ע … ודאי אילו הי' בירור שאמר כן בלי ספק הייתי מבטל דעתי בזה, כי מי יבוא אחר המלך כו' וכגבוה שמים מעל הארץ גבהו דרכיו מדרכינו כו', אך ידוע לנו שיוכלו לומר בשמו מה שאינו כו', ולזאת אין לסמוך על השמועות).
