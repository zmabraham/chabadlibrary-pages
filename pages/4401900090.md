# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר כה. > פֶּרֶק ב

וְהִנֵּה לְבוּשֵׁי הַטֶּבַע עִם הֱיוֹת שֶׁזֶּהוּ כְּפִי הַכַּוָּונָה הָעֶלְיוֹנָה יִתְבָּרֵךְ וְיִתְעַלֶּה שֶׁעָלָה בִּרְצוֹנוֹ יִתְבָּרֵךְ לִבְרֹא עוֹלָם גַּשְׁמִי וַאֲשֶׁר כָּל עִנְיָן וְדָבָר אַף גַּם הָאוֹר הָאֱלֹקִי בָּעוֹלָם יִהְיֶה מִתְעַלֵּם וּמִסְתַּתֵּר בִּלְבוּשֵׁי הַטֶּבַע, הִנֵּה עִם כָּל זֶה הִנֵּה לְבוּשֵׁי הַטֶּבַע מִצַּד עֶצֶם מַהוּתָם הֲרֵי הֵם מַעְלִימִים וּמַסְתִּירִים אֶת הָאוֹר וְהַשֶּׁפַע הָאֱלֹקִי, עַד אֲשֶׁר הָאָדָם הָעוֹסֵק בְּמִסְחָר יוּכַל לִטְעוֹת וּלְדַמּוֹת בְּעַצְמוֹ כְּאִלּוּ חַס וְשָׁלוֹם כֹּחוֹ וְעוֹצֶם יָדוֹ עָשָׂה לוֹ אֶת הַחַיִל הַזֶּה, וְחָכְמָתוֹ עָמְדָה לוֹ שֶׁנִּתְחַכֵּם בְּעֵסֶק הַמִּקָּח וְהַמִּמְכָּר שֶׁהִתְנַהֵג עַל פִּי הַטֶּבַע לַחֲקוֹר וְלִדְרוֹשׁ בְּכָל עִנְיְנֵי הַקְּנִיּוֹת בָּחַר בִּדְרָכִים הַיּוֹתֵר נְאוֹתִים לְפִי דַרְכֵי הַטֶּבַע, וְכֵן בְּעִנְיַן אוֹפַנֵּי הַמְּכִירָה שֶׁעָשָׂה בָזֶה כַּמָּה תַּחְבּוּלוֹת וְהִתְחַכְּמוּת עַל פִּי דַרְכֵי הַטֶּבַע וְכָל אֵלֶּה עָמְדוּ לוֹ לְהַרְוִיחַ רֶיוַח גָּדוֹל,
וּבֶאֱמֶת הִנֵּה כָּל אֵלֶּה לְבוּשֵׁי הַטֶּבַע אֵינָם אֶלָּא לְבוּשׁ לְבָד אֲבָל הַבְּרָכָה נִמְשֶׁכֶת מִלְמַעְלָה עַל יְדֵי הַלְּבוּשׁ וְלֹא מִצַּד עַצְמוּת הַלְּבוּשׁ חַס וְשָׁלוֹם, וּכְמוֹ שֶׁגִּידּוּל הַתְּבוּאוֹת נִמְשָׁךְ עַל יְדֵי הַשֶּׁמֶשׁ וְהַיָּרֵחַ וּכְמוֹ שֶׁכָּתוּב (דְּבָרִים לג. יד) וּמִמֶּגֶד תְּבוּאוֹת שָׁמֶשׁ וּמִמֶּגֶד גֶּרֶשׁ יְרָחִים וְעִם זֶה אֵינָן אֶלָּא שְׁלוּחֵי הַהַשְׁפָּעָה וְאֵין בְּיָדָם לְהָרַע וּלְהֵטִיב כִּי אִם אֶת אֲשֶׁר יֹאמַר הַקָּדוֹשׁ בָּרוּךְ הוּא, וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל (מַאֲמָר כג פֶּרֶק א), כַּךְ הָעֵסֶק וְהַלְּבוּשׁ הוּא רַק כְּשָׁלִיחַ לְבָד וְהַהַשְׁפָּעָה הִיא בְּיַד הַקָּדוֹשׁ בָּרוּךְ הוּא אֶלָּא שֶׁצָּרִיךְ לְהִתְלַבֵּשׁ בִּלְבוּשׁ וְעֵסֶק שֶׁהֵן לְבוּשֵׁי הַטֶּבַע, 
וּלְפִי זֶה מוּבָן שֶׁאֵין צְרִיכִים לְהַאֲרִיךְ בְּהַלְּבוּשׁ יוֹתֵר מִדַּאי כִּי זֶה לְלֹא צוֹרֶךְ מֵאַחַר שֶׁלֹּא הַלְּבוּשׁ הוּא הָעִיקָּר כִּי אִם הַשְׁפָּעָתוֹ יִתְבָּרֵךְ שֶׁמַּשְׁפִּיעַ עַל יְדֵי הַלְּבוּשׁ, וְאִם כֵּן הָעִיקָּר הוּא שֶׁיִּתְפַּלֵּל לְמִי שֶׁהָעוֹשֶׁר וְהַנְּכָסִים שֶׁלּוֹ וִייַשֵּׁר דַּרְכּוֹ בְּדֶרֶךְ הַתּוֹרָה וְהַמִּצְוֹת שֶׁיִּהְיֶה רָאוּי לְבִרְכַּת ה' וְהַשְׁפָּעַת חֶסֶד, וְהַיְינוּ שֶׁיּוּמְשַׁךְ לוֹ קִצְבַת הַחֶסֶד בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים עַל יְדֵי קַבָּלַת עוֹל מַלְכוּתוֹ יִתְבָּרֵךְ וְעַל יְדֵי תְּשׁוּבָתוֹ הָאֲמִיתִּית,
וְהַיְינוּ שֶׁלֹּא בִּשְׁבִיל זֶה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יַשְׁפִּיעַ לוֹ קִצְבַת הַחֶסֶד שֶׁהוּקְצְבָה לוֹ יְקַבֵּל עָלָיו עוֹל מַלְכוּתוֹ וְיַעֲשֶׂה תְשׁוּבָה, אֶלָּא שֶׁהַקַּבָּלַת עוֹל מַלְכוּת שָׁמַיִם וְהַתְּשׁוּבָה תִּהְיֶה אֲמִיתִּית מִצַּד עַצְמָהּ מִפְּנֵי שֶׁכֵּן צָרִיךְ לִהְיוֹת וְיִהְיֶה חָזָק בְּקִיּוּם הַתּוֹרָה וּמִצְוֹת כָּל הַשָּׁנָה לְהִתְפַּלֵּל תְּפִלָּה בִּזְמַנָּהּ וּבְכַוָּונָה וְיִשְׁאַל צְרָכָיו גַּם כֵּן וְיִלְמוֹד תּוֹרָה בִּזְמַנִּים קְבוּעִים בְּלִי שׁוּם מְנִיעוֹת וְעִכּוּבִים וְאַחַר כַּךְ יַעֲסוֹק בְּמַשָּׂא וּמַתָּן וְיַסְפִּיק לוֹ גַּם מְעַט זְמַן עַל עֵסֶק, דְּמֵאַחַר שֶׁכְּבָר נִקְצַב לוֹ הַחֶסֶד עֶלְיוֹן וְנִמְשַׁךְ וְנִתְלַבֵּשׁ בִּלְבוּשִׁים שֶׁהֵם הַכֵּלִים דְּעֶשֶׂר סְפִירוֹת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה וְאֵינוֹ חָסֵר רַק לְבוּשׁ הַטֶּבַע שֶׁל הַמַּזָּלוֹת וְכֵן לְבוּשֵׁי הַטֶּבַע שֶׁל כְּלִי הַפַּרְנָסָה וַאֲשֶׁר עַל כֵּן הִנֵּה בִּמְעַט עֵסֶק שֶׁיַּעֲשֶׂה יוּמְשַׁךְ וְיִתְלַבֵּשׁ הַחֶסֶד בִּלְבוּשׁ הָעֵסֶק וְתוּמְשַׁךְ הַהַשְׁפָּעָה,
וְהַיְינוּ כַּאֲשֶׁר הָאָדָם הוּא רָאוּי עַל פִּי מַעֲשָׂיו בַּתּוֹרָה וּמִצְוֹת לְחֶסֶד עֶלְיוֹן הֲרֵי אָז נִמְשָׁךְ הַחֶסֶד מֵעַתִּיקָא קַדִּישָׁא עַל יְדֵי הִתְעוֹרְרוּת הַסְּפִירוֹת דַּאֲצִילוּת כַּנַ"ל (מַאֲמָר כד פֶּרֶק א) בְּמַלְכוּת דַּאֲצִילוּת וְיוֹרֵד וְנִמְשָׁךְ בִּלְבוּשִׁים וְכֵלִים דְּעֶשֶׂר סְפִירוֹת דִּבְרִיאָה-יְצִירָה-עֲשִׂיָּה וּמִתְגַּשֵּׁם בִּלְבוּשֵׁי הַמַּזָּלוֹת וַהֲרֵי הוּא נִמְשָׁךְ לְמַטָּה עַל יְדֵי לְבוּשׁ הָעֵסֶק וְאָז דַּי גַּם מְעַט עֵסֶק,
אֲבָל אִם חַס וְשָׁלוֹם אֵינוֹ רָאוּי לְהַשְׁפָּעַת הַחֶסֶד וְהַיְינוּ כַּאֲשֶׁר לֹא נִמְשַׁךְ קִצְבַת הַחֶסֶד לְמַעְלָה בַּאֲצִילוּת, וְהַיְינוּ דְּמִי יֹאמַר כְּשֶׁלֹּא נִמְשַׁךְ חַס וְשָׁלוֹם קִצְבַת הַחֶסֶד בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים כְּלָל, אֶלָּא אֲפִילוּ גַּם אִם נִמְשַׁךְ לְמַעְלָה בְּמַלְכוּת דַּאֲצִילוּת הִנֵּה אִם אֵינוֹ נִמְשָׁךְ וּמִצְטַיֵּיר בִּלְבוּשֵׁי בְּרִיאָה-יְצִירָה-עֲשִׂיָּה בְּאוֹפֶן שֶׁיּוּכַל לִהְיוֹת מִזֶּה הַשְׁפָּעָה גַשְׁמִית וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל (מַאֲמָר יט פֶּרֶק א), מַה מּוֹעִיל לוֹ רִיבּוּי הַהִתְעַסְּקוּת מֵאַחַר שֶׁאֵין חַס וְשָׁלוֹם הַמְשָׁכַת הַחֶסֶד מִלְמַעְלָה רַחֲמָנָא לִצְּלָן,
וְאִם כֵּן הָעִיקָּר הוּא שֶׁיִּהְיֶה מָסוּר לֵאלֹקוּת וּכְשֶׁנִּמְשָׁךְ בְּחִינַת הַחֶסֶד עֶלְיוֹן הִנֵּה בִּרְכַּת ה' הִיא תַעֲשִׁיר גַּם בִּמְעַט עֵסֶק.

[kitzur_begin]לְבוּשֵׁי הַטֶּבַע הֲגַם שֶׁהֵם עַל פִּי סֵדֶר שֶׁלְּמַעְלָה הִנֵּה הֵמָּה מַעְלִימִים בִּמְאֹד, וּבֶאֱמֶת הֵם רַק לְבוּשִׁים בִּלְבָד שֶׁאֵין לְהַאֲרִיךְ בָּהֶם וְהָעִיקָּר הוּא אֱלֹקוּת הַשּׁוֹפֵעַ וְצָרִיךְ לִהְיוֹת זָהִיר בִּתְפִלָּה בְּצִבּוּר וּבִמְתִינוּת וְלִקְבּוֹעַ עִתִּים לַתּוֹרָה, דְּכַאֲשֶׁר חַס וְשָׁלוֹם אֵינוֹ רָאוּי לֹא יוֹעִיל גַּם רִיבּוּי הִתְעַסְּקוּת וּכְשֶׁרָאוּי הִנֵּה גַּם בִּמְעַט עֵסֶק בִּרְכַּת ה' תַּעֲשִׁיר.[kitzur_end]
