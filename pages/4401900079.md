מַאֲמָר כ.
וְהִנֵּה זֶה שֶׁהָאָדָם נִידּוֹן בְּכָל יוֹם, הוּא דַּעַת רַבִּי יוֹסֵי בִּבְרַיְיתָא דְרֹאשׁ הַשָּׁנָה (טז. א) וּלְדַעַת רַבִּי נָתָן נִידּוֹן בְּכָל שָׁעָה שֶׁנֶּאֱמַר לִרְגָעִים תִּבְחָנֶנּוּ (אִיּוֹב ז. יח) וְהִקְשׁוּ עַל זֶה דְּאִם כֵּן לְרַבִּי יוֹסֵי וּלְרַבִּי נָתָן אֵין בֵּין רֹאשׁ הַשָּׁנָה לְכָל יְמוֹת הַשָּׁנָה כְּלוּם לְעִנְיַן דִּין שֶׁל מַעְלָה וְאֵיךְ יְפָרְשׁוּ הַפָּסוּק דְּכִי חֹק לְיִשְׂרָאֵל הוּא מִשְׁפָּט לֵאלֹקֵי יַעֲקֹב (תְּהִלִּים פא. ה) דְּקָאֵי עַל רֹאשׁ הַשָּׁנָה.
[cup]וְגַם[/cup] מַה שֶּׁכָּתוּב עֵינֵי ה' אֱלֹקֶיךָ בָּהּ מֵרֵאשִׁית הַשָּׁנָה (דְּבָרִים יא. יב) שֶׁמִּזֶּה לָמְדוּ בִּגְמָרָא (שָׁם ח סוֹף עַמּוּד א) דְּרֹאשׁ הַשָּׁנָה הוּא רֹאשׁ הַשָּׁנָה לְדִין,
וְעוֹד מַהוּ עִנְיַן מַלְכִיּוֹת זִכְרוֹנוֹת וְשׁוֹפְרוֹת שֶׁאוֹמְרִים בְּרֹאשׁ הַשָּׁנָה, וּמַה יּוֹם מִיּוֹמַיִים, וְאֵין לְהַעֲלוֹת עַל הַדַּעַת כְּלָל דִּלְרַבִּי יוֹסֵי וּלְרַבִּי נָתָן אֵין חִידּוּשׁ בְּרֹאשׁ הַשָּׁנָה לְגַבֵּי שְׁאָר הַיָּמִים,
אֶלָּא וַדַּאי אֵין מַחֲלוֹקֶת בָּזֶה כְּלָל אֲשֶׁר רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים הוּא יוֹם הַדִּין, וְכֵן כָּתַב בַּעֲשָׂרָה מַאֲמָרוֹת מַאֲמַר חִקּוּר דִּין חֵלֶק ב פֶּרֶק א וּפֶרֶק כ"ו שֶׁלֹּא נֶחֱלַק אָדָם בַּדָּבָר, וְהַיְינוּ דְּהַכֹּל מוֹדִים שֶׁאָז נֶחְתָּךְ קִצְבַת הַחֶסֶד לְכָל בְּרִיָּה כו', וְלָכֵן אוֹמְרִים מַלְכִיּוֹת זִכְרוֹנוֹת וְשׁוֹפְרוֹת בְּרֹאשׁ הַשָּׁנָה בִּכְדֵי שֶׁתַּמְלִיכוּנִי עֲלֵיכֶם כו'
דְּבִנְיַן הַמַּלְכוּת דַּאֲצִילוּת הוּא מֵרֹאשׁ הַשָּׁנָה עַד יוֹם הַכִּפּוּרִים, שֶׁאָז בִּנְעִילָה מְקַבֶּלֶת הַמַּלְכוּת מֵה' גְבוּרוֹת דְּעַתִּיק, וְהוּא הַמְשָׁכַת מְקוֹר הַחַיּוּת לִבְרִיאָה-יְצִירָה-עֲשִׂיָּה לְכָל הַשָּׁנָה, וְהַיְינוּ דִּבְרֹאשׁ הַשָּׁנָה נִמְשָׁךְ הַחֶסֶד כְּפִי שֶׁנִּקְצָב לְכָל הָעוֹלָם בִּכְלָל, וּבִנְעִילָה דְיוֹם הַכִּפּוּרִים נֶחְתָּם וְנֶחְתָּךְ קִצְבַת הַחֶסֶד לְכָל אֶחָד וְאֶחָד בִּפְרָט וְזֶה לְדִבְרֵי הַכֹּל,
אֶלָּא דְּמִכָּל מָקוֹם סְבִירָא לֵיהּ לְרַבִּי יוֹסֵי וְרַבִּי נָתָן, דְּאָדָם נִידּוֹן בְּכָל יוֹם דִּין וּמִשְׁפָּט גָּמוּר עַל כָּל הָעִנְיָנִים לְחוֹלִי וְלִבְרִיאוּת וּלְחַיִּים כו', וְהַיְינוּ לְפִי שֶׁהַחֶסֶד הַקָּצוּב לוֹ בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים אַף שֶׁכְּבָר נִמְשַׁךְ לִבְחִינַת מַלְכוּת דַּאֲצִילוּת אֵין בּוֹ עֲדַיִין שׁוּם תּוֹאַר וּתְמוּנָה מֵעִנְיְנֵי עוֹלָם הַזֶּה, כִּי אֵין עֲרוֹךְ עֲשִׂיָּה לְגַבֵּי אֲצִילוּת, עַל כֵּן אַף שֶׁיּוּכַל לִהְיוֹת מִמֶּנּוּ כָּל הַשְׁפָּעוֹת עִנְיְנֵי עוֹלָם הַזֶּה, מִכָּל מָקוֹם בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים לֹא נִגְמַר הַמִּשְׁפָּט בְּפֵירוּשׁ מַה יִּהְיֶה מֵהַחֶסֶד הַזֶּה וּבַמֶּה תִּהְיֶה הַהַשְׁפָּעָה אִם בְּבָנֵי אוֹ בְּחַיֵּי אוֹ בִּמְזוֹנֵי אוֹ בְּכֻלָּם אוֹ בְּעִנְיָנִים הָרוּחָנִים וּבְחַיֵּי עוֹלָם הַבָּא, וְהַדִּין וּמִשְׁפָּט בַּמֶּה תִּהְיֶה הַהַשְׁפָּעָה הוּא בְּבֵית דִּין שֶׁלְּמַעְלָה בְּהֵיכָלוֹת דִּבְרִיאָה וִיצִירָה כו', וְזֶהוּ הַדִּין וּמִשְׁפָּט שֶׁבְּכָל יוֹם מַמָּשׁ וּכְמוֹ שֶׁנִּתְבָּאֵר.
[cup]וְהַדִּין[/cup] וּמִשְׁפָּט דְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים הוּא קִצְבַת הַחֶסֶד שֶׁיּוּמְשַׁךְ לְכָל הָעוֹלָם בִּכְלָל וּלְכָל בְּרִיָּה בִּפְרָט כַּמָּה מֵהַחֶסֶד נִקְצַב אֵלָיו, וְזֶה עִיקַּר הַדִּין וּבוֹ תָּלוּי הַכֹּל כִּי לְפִי רִיבּוּי קִצְבַת הַחֶסֶד כַּךְ תִּהְיֶה הַבְּרָכָה וְהַטּוֹב כָּל הַשָּׁנָה מַה שֶּׁאֵין כֵּן בְּמִיעוּטוֹ חַס וְשָׁלוֹם, וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל (מַאֲמָר יט פֶּרֶק ב) דְּכַאֲשֶׁר הַחֶסֶד בְּרִיבּוּי יֵשׁ בְּאֶפְשָׁרִי שֶׁתִּהְיֶה הַשְׁפָּעַת הַחֶסֶד בְּכָל הַדְּבָרִים דְּבָנֵי חַיֵּי וּמְזוֹנֵי וְלִפְעָמִים אֵין בְּאֶפְשָׁרִי שֶׁיִּהְיוּ כָּל הַדְּבָרִים כִּי אִם שְׁנַיִם מֵהֶם אוֹ אֶחָד מֵהֶם כו', וְגַם אִם אֲפִילוּ בְּכָל הַג' דְּבָרִים יֵשׁ נַפְקָא מִינָּהּ אִם בְּרִיבּוּי אוֹ בְּצִמְצוּם חַס וְשָׁלוֹם, וְזֶה תָּלוּי בִּכְלָלוּת קִצְבַת הַחֶסֶד בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים לְכָל הָעוֹלָם בִּכְלָל וְאֵלָיו בִּפְרָט אִם בְּרִיבּוּי אוֹ כו',
מִכָּל מָקוֹם יֵשׁ דִּין וּמִשְׁפָּט בִּפְרָט בְּכָל יוֹם מַה יִּצְטַיֵּיר וְיוּמְשַׁךְ מֵהַחֶסֶד הַזֶּה, וּכְמוֹ אִם יֵשׁ בְּאֶפְשָׁרוּת הַחֶסֶד שֶׁנִּקְצַב אֵלָיו בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים שֶׁיִּהְיוּ ב' דְּבָרִים מֵהַג' הַנַ"ל אֵיזֶה ב' דְבָרִים יִהְיוּ אִם בָּנֵי וְחַיֵּי אוֹ מְזוֹנֵי וְחַיֵּי וְכַדּוֹמֶה וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל. וְגַם אִם יֵשׁ בְּאֶפְשָׁרִי עַל כָּל הַג' דְּבָרִים הַנַ"ל מִכָּל מָקוֹם הַמִּשְׁפָּט הוּא אוּלַי יִהְיֶה הַחֶסֶד הַזֶּה הַשְׁפָּעוֹת עִנְיָנִים רוּחָנִים אוֹ לְחַיֵּי עוֹלָם הַבָּא וכו', מֵאַחַר שֶׁבְּהַחֶסֶד שֶׁנִּקְצַב בְּרֹאשׁ הַשָּׁנָה אֵינוֹ נִיכָּר עֲדַיִין מַה שֶּׁיִּהְיֶה מִמֶּנּוּ הַהַשְׁפָּעָה בַּעֲשִׂיָּה שֶׁאֵין שָׁם תּוֹאַר זֶה כְּלָל כִּי אִם זֶה תָּלוּי בְּבֵית דִּין שֶׁל מַעְלָה. וְזֶהוּ לְפִי מַעֲשָׂיו, כֵּן הוּא הַדִּין וּמִשְׁפָּט בְּהֵיכָלוֹת דִּבְרִיאָה.
[cup]וְנִמְצָא[/cup] שְׁנֵיהֶם אֱמֶת, דְּעִיקַּר הַדִּין עַל הַמְשָׁכַת הַחֶסֶד הוּא בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים, וּמִכָּל מָקוֹם יֵשׁ דִּין וּמִשְׁפָּט גָּמוּר בְּכָל יוֹם.

[kitzur_begin]קִצְבַת הַחֶסֶד דְּעַתִּיק בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכֵּפּוּרִים אֵין בּוֹ עֲדַיִין שׁוּם תּוֹאַר מֵעִנְיְנֵי עוֹלָם הַזֶּה, גַּם בִּלְתִּי מְחוּלָּק בְּאֵיכוּת וְכַמּוּת בְּבָנֵי חַיֵּי וּמְזוֹנֵי, וְעַל זֶה הוּא הַדִּין בְּכָל יוֹם לְדַעַת רַבִּי יוֹסֵי וּבְכָל שָׁעָה לְדַעַת רַבִּי נָתָן.[kitzur_end]
