שה"ה גילוי בעצם רק שהוא נעלם מהמקבלים שא"ז העלם באמת, וא"כ מהו"ע המכסה אני כו'. ובמד"ר פ' נשא אי' נתן האלקי' את העולם לאברהם שנא' וה' אמר המכסה אני מאברהם כו', וצ"ל היכן מרומז במ"ש וה' אמר המכסה אני שנתן את העולם לאברהם כו', וגם מהו"ע שנתן לו את העולם כו'. גם צ"ל דלכאו' מהו ההתחדשות בזה הלא כתיב והארץ נתן לבנ"א, א"כ מהו ההתחדשות שנתן האלקי' את העולם לאברהם כו'. גם צ"ל מ"ש ואברהם היו יהי' כו', דמשמעות הענין שהן שייכים זל"ז דלפי שהמכסה אני מאברהם כו' ע"כ ואברהם היו יהי' לגוי גדול, וצ"ל מהו שייכות הענינים זל"ז כו'. ולהבין כ"ז ילה"ק משנת"ל דבח"ס יש ב' מדרי' חיצוני' ופנימי', דחיצוני' ח"ס הוא בבחי' מקור לחכ' דאצי' כו', וג"ז הוא למע' מבחי' החכ' והוא בבחי' מקיף כו' (דעם היותו בחי' פנימי ה"ז מ"מ בחי' מקיף והיינו מקיף דאו"י כו'), ולכן ההמשכה היא ע"י ריח הקטרת דוקא שעי"ז נמשך בחי' שמן המנורה כו', דריח הוא בחי' מקיף כו', אמנם בחי' פנימי' ח"ס הוא בחי' מוחא דעתיקא ממש בחי' פנימי' עתיק כו'. וזהו בחי' מקיף דאו"ח שסתום ונעלם בעצם כו'. וההמשכה היא ע"י הקטרת דיוה"כ שהו"ע התשו' כו'.
[cup]רכג)[/cup] וע"פ כל הנ"ל יובן איך שעיקר הממוצע בין המאציל אל הנאצלים (וכן בין עצמות א"ס לנש"י) הוא בחי' ח"ס דוקא, וכמשנת"ל פקצ"ו דא"א לומר דמה שהכתר הוא בחי' ממוצע זהו רק בהב' מדרי' דעתיק ואריך, שהרי עתיק הוא ממש בחי' תחתונה שבעצמות (מקצת מן העצם כו') שהוא בחי' עצמות כו', וע"כ הוא בבחינת העתקה לגמרי מן הנאצלים גם מבחי' אריך כו', וכמו"כ גם בחי' אריך שהוא ראש ומקור הנאצלים אינו ממהות הנאצלים שהן בחי' חכ' וא"א הוא בחי' רצון כו' ואיך הוא בבחי' ממוצע כו', ובפרט שעיקר ענין הממוצע הוא שעי"ז יהי' חבור ושייכות להנאצלים אל העצמות כו', ואיך יכול להיות זאת ע"י ממוצע כזה דבחי' העליונה שבו היא בבחי' הבדלה לגמרי מן הנאצלים ובחי' תחתונה ג"כ אינו כמהות הנאצלים כו'. אך עיקר הממוצע הוא בחי' ח"ס שזהו בחי' חכ' כמהות הנאצלים והוא בבחי' מקור לחכ' דאצי' כו', ועם היותו בבחי' מקור מ"מ ה"ז בבחי' הבדלה מהחכ', דאבא יונק ממזלא בחי' שערות לבד שהן הארה דהארה כו', וכמשנת"ל פ' ר"ד ור"ח, וזהו בחי' חיצוני' דח"ס, אבל
פנימי' דח"ס הוא בחי' מוחא דעתיקא ממש שמלובש בזה בחי' פנימי' עתיק ממש כו' וכמשנת"ל פר"ג, וע"כ בכדי להגיע לזה הוא בכח או"ח דתשו' דוקא כו' וכמשנת"ל פרכ"ב, ובכללות הוא בחי' חכ' רק שהוא בבחי' חכ' מובדלת דלא בחכ' ידיעא כו', והיינו שאינו בחי' עצמו' שהרי העצמו' אינו בגדר חכ' כו', וגם לא בחי' הנאצלים שה"ה מובדל מן החכ' כו', ויש בו מבחי' העצמו' ממש בבחי' הפנימי' והעצמי' שלו וגם יש בו מן הנאצלים שה"ה בבחי' חכ' עכ"פ כו'. וע"כ הוא בבחי' ממוצע בין המאציל אל הנאצלי' ושעי"ז יוכל להיות חבור הנאצלים עם העצמות, וכמו"כ חבור וקישור נש"י אל העצמו' כו', והיינו דבממוצע כזה להיותו בחי' חכ' עכ"פ ביכולת לבא ולהגיע גם לבחי' העליונה שבו כו', ועי"ז מגיע גם לבחי' העצמו' שלמע' מבחי' ממוצע כו'. והענין הוא דהנה כמשנת"ל פקצ"ח בהב' מדרי' שבחכ' כח ההפלאה וכח הירידה דהא בהא תלי' דלפי אופן ההפלאה לפי אופן כזה הוא כח הירידה כו', כמו"כ הוא גם בהב' מדרי' שבח"ס דע"ה שבחי' הפנימי' מובדל בערך מן החיצוני'
