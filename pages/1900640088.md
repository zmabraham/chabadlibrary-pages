גשמי דדצח"ם דעשי' מ"מ שרשן ומקורן הוא בעולם היצירה ושרש היצירה הוא בבריאה [כנ"ל], ונמצא אותו הדין וההלכה שנאמרה כאן בעולם העשי' הרי הדין הזה נאמר ג"כ בעולם היצי' שהוא בחי' הרוחני' דעולם העשי' דרך כלל, וכמו"כ שרש אותו הדין בעולם הבריאה שהוא הרוחני' דעולם היצי' ונמצא שרש כל ההלכות שבתלמוד בעולם הבריאה דהיינו שנתלבשה חכמה שבמל' דאצי' בחכ' שבמל' דבריאה והוא ענין לימוד הנשמות בג"ע דיני ההלכות דתשבע"פ ממש כמו אם קדם שער לבן לבהרת כו' דקא מפליגי במתיבתא דרקיע כו', וכה"ג בכל הדינים כמו הלכות טריפות דין נקיבת הוושט דטריפה הנה זה הדין הוא בחי' חכ' שבתורה המתלבשת בענין גשמי דהיינו בהמה הגשמי' שאם ניקב הוושט שלה
טריפה ולא תוכשר לעלות בסט' דקדושה האלוקי'.
[cup]והנה[/cup] שרש הבהמה הגשמי' נמשך מבחי' פני שור מהשמאל שבמרכבה דעשי', וא"כ שרש זה הדין דנקיבת הוושט הוא במרכבה דעשי' בבחי' נקיבת הוושט דשור שבמרכבה כו' דטריפה בבחי' טריפו' בערך רוחניו' דמרכבה כו', ועד"ז עד רום המעלות גם ביצי' בבחי' מרכבה דיצי' כו' עד בחי' פני שור שבבריאה בחי' [נקיבת] הוושט שלו טריפה בערך רוחניו' דעולם הבריאה, ואחר שבי"ע כולן הן בבחי' הגבול והתכלית שהוא בא בבחי' השתל' עו"ע מרוחני' לגשמי' שהרי באמת אין הפרש בין בחי' עשי' לבריאה שניהם רק בחי' לבוש להשראת חכ' האלקי' דאצי' ודוקא בבחי' לבוש ולא בבחי' כלי שבטל לגבי האור מפני שיש ערך ב[י]ניהם כו', אך בחי' כל[י] חכ' שבמל' דעשי' שמלובש בדברים גשמיים הוא בחי' כלי ומעבר לבחי' חכ' שבמל' דיצי', וכמ"כ בחי' כלי חכ' שבמל' דיצי' בחי' כלי ומעבר לחכ' שבמל' דבריאה מפני שיש ערך ב[י]ניהם כו', אבל כללות חכ' דתו' שמלובש בבי"ע הוא רק בחי' לבוש נפרד שאינו מערך כלל וכלל לגבי אור האלקי' עצמו בחי' חכ' שבמל' דאצי' כמו נקיבת הוושט שהוא כלי גם להתלבשות חכ' דמל' דבריאה, אבל בחי' כלי דנקיבת הוושט דבחי' שור דבריאה אינו בחי' כלי דחכ' שבתו' שבמל' דאצי' רק בחי' לבוש אעפ"י שנתלבשה חכ' שבמל' דאצי' בחכ' דמל' דבריאה ונק' חכ' תתאה בחי' חכ' שבא בהתלבשות בעולמות בע"ג מ"מ אין לחכ' דבריאה ערך כלל אלי', יותר משאין ערך בחי' המספר לגבי למעלה מן המספר או כערך הגשמי' לגבי הרוחני' וכמו ערך היצי' לגבי הבריאה וכה"ג, לפי שבחי' בע"ג אין ערך כערך לגבי בחי' א"ס גם שהוא במל' סופא דכל דרגין וכידוע, וזהו שאמר בע"ח שהתלמוד בבריאה פי' בחי' חכ'
