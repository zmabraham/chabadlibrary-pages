# ספרי כ"ק אדמו"ר מוהרש"ב נ"ע > קונטרס ומעין > מַאֲמָר ז. > פֶּרֶק ג

וְהִנֵּה עוֹד זֹאת בְּמִצְוֹת לֹא תַעֲשֶׂה שֶׁהֵן שׁוֹמְרוֹת שֶׁלֹּא יוּמְשַׁךְ אוֹר הָאֱלֹקִי לְמָקוֹם שָׂנאוּי שֶׁנִּקְרָא תּוֹעֲבַת ה', כִּי לֹא יְגוּרְךָ רָע כְּתִיב (תְּהִלִּים ה. ה) וּבָאָדָם הוּא גַם כֵּן עַל יְדֵי הִבָּדְלוֹ מִדְּבָרִים הָאֲסוּרִים וּקְשׁוּרִים בִּידֵי הַחִיצוֹנִים שֶׁאֵין לָהֶם עֲלִיָּה לְמַעְלָה, כִּי לֹא יִשְׁכּוֹן בָּהֶם אוֹר הָאֱלֹקִי, וּכְמוֹ שֶׁכָּתוּב (יְשַׁעְיָה מב. ח) וּכְבוֹדִי לְאַחֵר לֹא אֶתֵּן, הִנֵּה הוּא נִשְׁמָר מִן הָרָע וְנַעֲשֶׂה כְּלִי אֶל הַקְּדוּשָּׁה,
אִם כֵּן הֲרֵי מוּבָן שֶׁעִיקַּר עִנְיַן חָכְמַת הַתּוֹרָה הִיא הַחָכְמָה אֵיךְ שֶׁיִּהְיֶה גִּילּוּי אוֹר אֵין סוֹף לְמַטָּה עַל יְדֵי הָרְמַ"ח מִצְוֹת עֲשֵׂה, וּלְהָסִיר כָּל מוֹנֵעַ עַל יְדֵי שְׁסָ"ה לֹא תַעֲשֶׂה, שֶׁזֶּהוּ עִנְיַן בְּחִינַת פְּנִימִית רָצוֹן הָעֶלְיוֹן בִּכְלָל שֶׁיִּהְיֶה גִּילּוּי אוֹר אֵין סוֹף לְמַטָּה עַל יְדֵי קִיּוּם הַמִּצְוֹת בִּרְמַ"ח מִצְוֹת עֲשֵׂה, וּבַהֲסָרַת הַמּוֹנְעִים עַל יְדֵי שְׁסָ"ה לֹא תַעֲשֶׂה, שֶׁבָּזֶה נִשְׁלָם הַכַּוָּונָה וְהָרָצוֹן דְּנִתְאַוָּה הַקָּדוֹשׁ בָּרוּךְ הוּא לִהְיוֹת לוֹ דִירָה בַּתַּחְתּוֹנִים, וּבְדֶרֶךְ פְּרָט הִנֵּה יֵשׁ בְּכָל מִצְוָה וּמִצְוָה הֶאָרָה פְּרָטִית מֵרָצוֹן הַכְּלָלִי הַזֶּה. וְעַל זֶה אוֹמֵר (בַּמִּדְבָּר ו. כה) יָאֵר ה' פָּנָיו אֵלֶיךָ, לְיִשְׂרָאֵל דַּוְקָא שֶׁמְּקַיְּימִים הַמִּצְוֹת שֶׁזֶּהוּ בְּחִינַת פְּנִימִית רָצוֹן הָעֶלְיוֹן, לִהְיוֹת גִּילּוּי אוֹר אֵין סוֹף לְמַטָּה.
[cup]אַךְ[/cup] כַּאֲשֶׁר נִמְשָׁךְ בְּחִינַת פְּנִימִית הָרָצוֹן לְיִשְׂרָאֵל עַל יְדֵי תּוֹרָה וּמִצְוֹת כַּנַ"ל, נִמְשָׁךְ גַּם כֵּן בְּחִינַת אֲחוֹרַיִים וְחִיצוֹנִיּוּת הָרָצוֹן הָעֶלְיוֹן שֶׁנִּקְרֵאת בְּחִינַת עוֹרֶף וְלֹא פָנִים לְעַכּוּ"ם, שֶׁשָּׁרְשָׁם מֵע' שָׂרִים הָעֶלְיוֹנִים שֶׁמְּקַבְּלִים הַשֶּׁפַע בִּבְחִינַת חִיצוֹנִיּוּת וַאֲחוֹרַיִים שֶׁעַל כֵּן נִקְרָאִים אֱלֹקִים אֲחֵרִים כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל (מַאֲמָר ב פֶּרֶק ב), וְעִנְיַן קַבָּלָתָם הוּא רַק בְּדֶרֶךְ אַגַּב לְבָד, דְּהַיְינוּ עַל דֶּרֶךְ מָשָׁל הַנַ"ל (פֶּרֶק א) בְּמִשְׁתֵּה הַשָּׂרִים, שֶׁלְּאַחַר שֶׁכְּבָר נִמְשַׁךְ הַשֶּׁפַע לְמִי שֶׁרָאוּי שֶׁבּוֹ פְּנִימִית הָרָצוֹן, אָז יְקַבְּלוּ גַּם אוֹתָם שֶׁאֵינָן רְאוּיִם מִבְּחִינַת הַשִּׁירַיִים בְּדֶרֶךְ אַגַּב, גַּם שֶׁלֹּא הָיְתָה הַכַּוָּונָה אֲלֵיהֶם כְּלָל, כַּךְ מְקַבְּלִים גַּם הַחִיצוֹנִים יְנִיקָה וְחַיּוּת מֵאוֹר הָאֱלֹקִי בִּבְחִינַת הַשִּׁירַיִים וְהַחִיצוֹנִיּוּת שֶׁבְּרָצוֹן הָעֶלְיוֹן, לְאַחַר שֶׁכְּבָר נִמְשָׁךְ הָאוֹר וְהַשֶּׁפַע הַפְּנִימִית מִפְּנִימִית רָצוֹן הָעֶלְיוֹן לְסִטְרָא דִקְדוּשָּׁה בְּהַמְשָׁכַת אוֹרוֹת בַּכֵּלִים עַל יְדֵי הַמִּצְוֹת שֶׁהוּא עִנְיַן יָאֵר ה' פָּנָיו אֵלֶיךָ כַּנַ"ל,
וְנִמְצָא שֶׁהַהַשְׁפָּעָה לַחִיצוֹנִים מִבְּחִינַת אֲחוֹרַיִים, הֲרֵי זֶה מַמָּשׁ כְּמַאן דְּשָׁדֵי בָּתַר כַּתְפּוֹי, בְּדֶרֶךְ אַגַּב וְטָפֵל בְּשָׁעָה שֶׁמַּשְׁפִּיעַ לַפְּנִימִי בִּבְחִינַת פָּנִים, וְאֵין בְּהַשְׁפָּעָה לַחִיצוֹנִים שׁוּם הֶאָרָה פְנִימִית מִפְּנִימִית רָצוֹן הָעֶלְיוֹן כְּלָל וּכְלָל, וְעַל כֵּן אֵינָם מְקַבְּלִים רַק מִבְּחִינַת הַתַּמְצִית וְהַשְּׁמָרִים הַנּוֹפְלִים כְּיָדוּעַ, וְדַוְקָא לְאַחַר שֶׁנִּתְעוֹרֵר הַשֶּׁפַע מִבְּחִינַת פָּנִים לְסִטְרָא דִקְדוּשָּׁה, אֲזַי גַּם הֵם מְקַבְּלִים בְּחִינַת הַתַּמְצִית, אֲבָל כְּשֶׁאֵין נִמְשָׁךְ מִבְּחִינַת פָּנִים לְסִטְרָא דִקְדוּשָּׁה בִּבְחִינַת גִּילּוּי (וְהַיְינוּ בְּהַזְּמַנִּים שֶׁאָז עֲלִיּוֹת הָעוֹלָמוֹת) אָז אֵין לָהֶם יְנִיקָה כְּלָל, וְנִמְצָא דְּדַוְקָא כְּשֶׁיֵּשׁ הַמְשָׁכָה בְּחִינַת פְּנִימִית בְּסִטְרָא דִקְדוּשָּׁה אָז נִמְשָׁךְ בְּחִינַת תַּמְצִית הַשְּׁמָרִים וְהַפְּסוֹלֶת לְסִטְרָא אָחֳרָא.

[kitzur_begin]מִצְוֹת לֹא תַעֲשֶׂה שׁוֹמְרוֹת שֶׁלֹּא יוּמְשַׁךְ אוֹר אֱלֹקִי בְּמָקוֹם הַשָּׂנאוּי. פְּנִימִית הָרָצוֹן תּוֹרָה מִצְוֹת עֲשֵׂה וּמִצְוֹת לֹא תַעֲשֶׂה רַק לְיִשְׂרָאֵל, וְדֶרֶךְ אַגַּב נִמְשָׁךְ הֶאָרַת חִיצוֹנִיּוּת רָצוֹן הָעֶלְיוֹן לְהָע' שָׂרִים וְעַל יָדָם הַפְּסוֹלֶת לְסִטְרָא אָחֳרָא.[kitzur_end]
