## מַאֲמָר ט.

וְהִנֵּה שֹׁרֶשׁ הָעִנְיָן הוּא, דְּהִנֵּה יָדוּעַ דְּסֵדֶר הִשְׁתַּלְשְׁלוּת הוּא בְּחִינַת אוֹר פְּנִימִי, וְהַיְינוּ הִתְלַבְּשׁוּת אוֹרוֹת בַּכֵּלִים דַּוְקָא, וְרֵאשִׁית הַכֵּלִים הוּא בִּבְחִינַת הַחָכְמָה, וּכְמוֹ עַל דֶּרֶךְ מָשָׁל בָּאָדָם לְמַטָּה, הֲרֵי כֹּחַ הַחָכְמָה הוּא הַכֹּחַ הָרִאשׁוֹן שֶׁמְּלוּבָּשׁ בַּכֵּלִים וְאֵבְרֵי הַגּוּף, דְּהָרָצוֹן שֶׁלְּמַעְלָה מֵהַחָכְמָה אֵין לוֹ כְּלִי מְיוּחֶדֶת בְּאֵבְרֵי הַגּוּף וְהוּא נִמְצָא בְּכָל הַגּוּף בְּשָׁוֶה, וְאַף שֶׁהוּא נִרְגָּשׁ יוֹתֵר בַּמּוֹחַ שֶׁבָּרֹאשׁ, הַיְינוּ מַה שֶּׁהַשֵּׂכֶל שֶׁבַּמּוֹחַ מַרְגִּישׁ אֶת הָרָצוֹן בְּהֶרְגֵּשׁ פְּנִימִי יוֹתֵר מֵהָרֶגֶל, אֲבָל הָרָצוֹן מִצַּד עַצְמוֹ נִמְצָא בָרֶגֶל כְּמוֹ בָרֹאשׁ, וְהָרְאָיָה שֶׁהֲרֵי כְּשֶׁעוֹלֶה בִּרְצוֹנוֹ לְנַעֲנֵעַ בְּרַגְלוֹ הוּא מְנַעֲנֵעַ תֵּכֶף וּמִיָּד בְּלִי שׁוּם שְׁהִיַּת זְמַן כְּלָל, וְאִם נֹאמַר שֶׁהָרָצוֹן מִשְׁכָּנוֹ בַּמּוֹחַ שֶׁבָּרֹאשׁ (כְּמוֹ הַשֵּׂכֶל שֶׁמִּשְׁכָּנוֹ בָּרֹאשׁ) הָיָה צָרִיךְ לִהְיוֹת שְׁהִיַּת זְמַן עַד שֶׁיָּבוֹא מִן הַמּוֹחַ לִפְעוֹל בְּהָרֶגֶל, וּכְמוֹ בִּפְעוּלּוֹת שֶׁמִּן כֹּחַ הַשֵּׂכֶל בָּרֶגֶל, כְּמוֹ אִימּוּן רַגְלָיו בְּהִילּוּךְ יָשָׁר אוֹ כְּמוֹ בְּרִיקּוּד עַל פִּי כַּוָּונַת הַשֵּׂכֶל לְפִי תְּנוּעַת הַנִּיגּוּן וְכַדּוֹמֶה, צָרִיךְ שְׁהִיַּת זְמַן וְאִימּוּן וְלֹא יָבֹא בְּדֶרֶךְ מִמֵּילָא, וְזֶהוּ לְפִי שֶׁבָּא דֶּרֶךְ הִשְׁתַּלְשְׁלוּת בְּהִתְלַבְּשׁוּת בְּדֶרֶךְ אוֹר פְּנִימִי, וְהַיְינוּ שֶׁבַּמּוֹחַ שֶׁבָּרֹאשׁ הוּא בְּדֶרֶךְ הִתְלַבְּשׁוּת מַמָּשׁ, שֶׁשָּׁם מִשְׁכַּן הַשֵּׂכֶל וּמִשְׁתַּלְשֵׁל מִכֹּחַ לַכֹּחַ כְּמוֹ מִן הַשֵּׂכֶל אֶל הַמִּדּוֹת עַד שֶׁבָּא אֶל כֹּחַ הַתְּנוּעָה שֶׁבָּרֶגֶל, וְהַכֹּל בְּדֶרֶךְ הִתְלַבְּשׁוּת שֶׁאוֹר הַשֵּׂכֶל מִתְלַבֵּשׁ מַמָּשׁ בְּכֹחַ הַתְּנוּעָה שֶׁבָּרֶגֶל, וְלָזֹאת צְרִיכִים לָזֶה הִשְׁתַּדְּלוּת וְהִתְעַסְּקוּת וּשְׁהִיַּת זְמַן, וְאִם הָיָה הָרָצוֹן גַּם כֵּן מְלוּבָּשׁ בְּאֵיזֶה כְלִי הָיוּ צְרִיכִים לָזֶה גַם כֵּן הִתְעַסְּקוּת וּשְׁהִיַּת זְמַן עַד שֶׁיִּפְעוֹל בָּרֶגֶל, וְאָנוּ רוֹאִין שֶׁהָרָצוֹן פּוֹעֵל הַתְּנוּעָה בְּהָרֶגֶל תֵּיכֶף בְּלִי שׁוּם שְׁהִיַּת זְמַן וּבְלִי שׁוּם הִתְעַסְּקוּת כְּלָל, וְהַיְינוּ לְפִי שֶׁהָרָצוֹן אֵין לוֹ כְּלִי מְיוּחָד בַּגּוּף וְנִמְצָא בָרֶגֶל כְּמוֹ בָרֹאשׁ, וְהַכֹּחַ הָרִאשׁוֹן שֶׁמִּתְלַבֵּשׁ בִּכְלִי הוּא כֹּחַ הַחָכְמָה שֶׁמִּתְלַבֵּשׁ בַּמּוֹחַ, וּתְלָת מוֹחִין אִינּוּן שֶׁהֵם כֵּלִים לְג' כֹּחוֹת חָכְמָה-בִּינָה-דַּעַת, וְהַמִּדּוֹת מִשְׁכָּנָם בַּלֵּב, וְכֵן כָּל הַכֹּחוֹת יֵשׁ לָהֶם כֵּלִים וְאֵבָרִים מְיוּחָדִים.

קִיצּוּר.

כֹּחוֹת פְּנִימִים מִתְלַבְּשִׁים בְּכֵלִים פְּרָטִים וּבָאִים בְּהִשְׁתַּלְשְׁלוּת וְכֹחַ הָרִאשׁוֹן הוּא כֹּחַ הַחָכְמָה, כֹּחַ הָרָצוֹן הוּא כֹּחַ מַקִּיף שֶׁנִּמְצָא בְּכָל הָאֵבָרִים וְאֵינוֹ בָּא בְּהִשְׁתַּלְשְׁלוּת, לָכֵן פְּעוּלַּת הַשֵּׂכֶל בָּאָה בְּהִתְעַסְּקוּת וְנִמְשָׁךְ שְׁהִיַּת זְמַן וּפְעוּלַּת הָרָצוֹן בְּלִי הִתְעַסְּקוּת וּבְלִי שְׁהִיַּת זְמַן.
